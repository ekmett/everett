77 // The directory must already exist durably. Catalog creation is exclusive;
78 // a competing creator can fail this call, but no existing file is formatted.
79 static persistent_engine connect(std::filesystem::path const & root, std::string_view name,
80 connection_options const & options = {}, Ids ids = {}) {
81 catalog_detail::name(name);
82 Core seed = options.schema_id.empty() ? Core{} : Core(options.schema_id);
83 auto initial = seed.snapshot();
84 auto schema = initial.metadata().schema_id;
85 auto store = open_store(root, options.create_if_missing, std::move(ids));
86 auto found = store.find(name);
87 if (!found) {
88 if (!options.create_if_missing) throw std::out_of_range("unknown named Everett session");
89 found.emplace(store.create_session(name, initial.runtime(), initial.metadata().encode()));
90 }
91 auto current = restore(std::move(*found), schema);
92 auto core = [&] {
93 if constexpr (requires { family_type::open_storage(store.root(), schema); })
94 return Core::from_snapshot(current, family_type::open_storage(store.root(), schema));
95 else if constexpr (requires { family_type::open_storage(store.root()); })
96 return Core::from_snapshot(current, family_type::open_storage(store.root()));
97 else return Core::from_snapshot(current);
98 }();
99 return persistent_engine(std::move(store), std::move(core), std::move(current), std::move(schema));
100 }

121 world_type contribute(contribution_type input) {
122 require_active();
123 // Only empty input needs this comparison. A pending Core can own a
124 // different physical layout from the last mapped durable publication.
125 std::optional<typed_world_type> before;
126 if constexpr (requires { input.records().empty(); })
127 if (input.records().empty()) before.emplace(core_.snapshot());
128 // Core's healthy failure contract means no logical update was admitted.
129 // Encoding, old-value validation and an absent deletion can reject just
130 // this ticket; publication failures may never take that path.
131 auto updated = [&] {
132 try { return core_.contribute(std::move(input)); }
133 catch (...) { remember_failure(); if (core_.failed()) poison(); throw; }
134 }();
135 try {
136 if (before && updated.runtime().same_layout(before->runtime()) && updated.metadata() == before->metadata())
137 return current_;
138 return publish(std::move(updated));
139 } catch (...) { remember_failure(); poison(); throw; }
140 }
141 std::optional<world_type> advance(std::uint64_t budget) {
142 require_active();
143 try {
144 auto updated = core_.advance(budget);
145 if (!updated) return std::nullopt;
146 return publish(std::move(*updated));
147 } catch (...) { remember_failure(); poison(); throw; }
148 }

160 static store_type open_store(std::filesystem::path const & root, bool create, Ids ids) {
161 if (!std::filesystem::is_directory(root)) throw std::invalid_argument("Everett backing directory must already exist");
162 if (std::filesystem::exists(root / "catalog.sqlite3") || !create)

166 static world_type restore(typename store_type::stored_type saved, std::string_view schema) {
167 auto world = connection_detail::restore_checkpoint<Core>(std::move(saved.snapshot), saved.semantic, schema);
168 return {std::move(world), std::move(saved.head)};
169 }
170 world_type publish(typed_world_type updated) {
171 auto saved = store_.publish(current_.head(), updated.runtime(), updated.metadata().encode());
172 auto mapped = restore(std::move(saved), schema_);
173 // Keep partial builders alive until their carry completes. Restarting the
174 // Core on every equivalent publication would discard paid private work.
175 if (!core_.pending()) {
176 if constexpr (requires { core_.rebase(mapped); }) core_.rebase(mapped);
177 else core_ = Core::from_snapshot(mapped);
178 }
179 current_ = std::move(mapped);
180 catalog_failure_ = nullptr;
181 return current_;
182 }
183 void remember_failure() noexcept {
184 // A streamed Core has its own catalog connection. Keep its exact error
185 // alive without copying the operation string while handling a failure.
186 try { throw; }
187 catch (catalog_error const &) { catalog_failure_ = std::current_exception(); }
188 catch (...) { catalog_failure_ = nullptr; }
189 }
190 void require_active() const { if (failed()) throw std::logic_error("failed persistent Everett engine; reconnect it"); }
191 void poison() noexcept {
192 failed_ = true;
193 if constexpr (requires { { core_.poison() } noexcept; }) core_.poison();
194 }
195 };
