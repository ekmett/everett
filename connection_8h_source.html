99 value_type value;

107 template <class C> requires std::constructible_from<ordinary_contribution, C &&>
108 static contribution_type borrow(C && input) {
109 if constexpr (std::is_rvalue_reference_v<C &&> && !std::is_const_v<std::remove_reference_t<C>>)
110 return contribution_type(borrowed{&input, &input});
111 else return contribution_type(borrowed{&input, nullptr});
112 }
113 private:
114 static auto take(contribution_type & other) -> value_type {
115 if (auto p = std::get_if<borrowed>(&other.value)) {
116 if (p->move) return value_type(std::in_place_index<0>, std::move(*p->move));
117 if constexpr (std::copy_constructible<ordinary_contribution>)
118 return value_type(std::in_place_index<0>, *p->copy);
119 else throw std::logic_error("move-only command requires an rvalue");
120 }
121 return std::move(other.value);
122 }

127 // The directory must already exist durably. Catalog creation is exclusive;
128 // a competing creator can fail this call, but no existing file is formatted.
129 static persistent_engine connect(std::filesystem::path const & root, std::string_view name,
130 connection_options const & options = {}, Ids ids = {}) {
131 catalog_detail::name(name);
132 Core seed = options.schema_id.empty() ? Core{} : Core(options.schema_id);
133 auto initial = seed.snapshot();
134 auto schema = initial.metadata().schema_id;
135 auto store = open_store(root, options.create_if_missing, std::move(ids));
136 auto found = store.find(name);
137 if (!found) {
138 if (!options.create_if_missing) throw std::out_of_range("unknown named Everett session");
139 found.emplace(store.create_session(name, initial.runtime(), initial.metadata().encode()));
140 }
141 auto current = restore(std::move(*found), schema);
142 auto core = [&] {
143 if constexpr (requires { family_type::open_storage(store.root(), schema); })
144 return Core::from_snapshot(current, family_type::open_storage(store.root(), schema));
145 else if constexpr (requires { family_type::open_storage(store.root()); })
146 return Core::from_snapshot(current, family_type::open_storage(store.root()));
147 else return Core::from_snapshot(current);
148 }();
149 return persistent_engine(std::move(store), std::move(core), std::move(current), std::move(schema));
150 }

183 world_type contribute(contribution_type input) {
184 require_active();
185 if (auto prepared = std::get_if<prepared_transaction>(&input.value))
186 return install(std::move(*prepared));
187 if (!std::holds_alternative<ordinary_contribution>(input.value))
188 throw std::logic_error("borrowed contribution escaped admission");
189 return contribute_ordinary(std::move(std::get<ordinary_contribution>(input.value)));
190 }
191 world_type contribute_ordinary(ordinary_contribution input) {
192 require_active();
193 // Only empty input needs this comparison. A pending Core can own a
194 // different physical layout from the last mapped durable publication.
195 std::optional<typed_world_type> before;
196 if constexpr (requires { input.records().empty(); })
197 if (input.records().empty()) before.emplace(core_->snapshot());
198 // Core's healthy failure contract means no logical update was admitted.
199 // Encoding, old-value validation and an absent deletion can reject just
200 // this ticket; publication failures may never take that path.
201 auto updated = [&] {
202 try { return core_->contribute(std::move(input)); }
203 catch (...) { remember_failure(); if (core_->failed()) poison(); throw; }
204 }();
205 try {
206 if (before && updated.runtime().same_layout(before->runtime()) && updated.metadata() == before->metadata())
207 return current_;
208 return publish(std::move(updated), true);
209 } catch (...) { remember_failure(); poison(); throw; }
210 }
211 std::optional<world_type> advance(std::uint64_t budget) {
212 require_active();
213 try {
214 auto updated = core_->advance(budget);
215 if (!updated) return std::nullopt;
216 return publish(std::move(*updated), false);
217 } catch (...) { remember_failure(); poison(); throw; }

242 world_type publish(typed_world_type updated, bool mutation) {
243 auto saved = store_.publish(current_.head(), updated.runtime(), updated.metadata().encode());
244 auto mapped = restore(std::move(saved), schema_, mutation ?
245 std::make_shared<connection_detail::logical_state>() : current_.logical_identity());
246 // Keep partial builders alive until their carry completes. Restarting the
247 // Core on every equivalent publication would discard paid private work.
248 if (!core_->pending()) {
249 if constexpr (requires { core_->rebase(mapped); }) core_->rebase(mapped);
250 else core_ = std::make_unique<Core>(Core::from_snapshot(mapped));
251 }
252 current_ = std::move(mapped);
253 catalog_failure_ = nullptr;
254 return current_;
255 }
256 world_type install(prepared_transaction input) {
257 if (input.base.logical_identity() != current_.logical_identity()) throw transaction_conflict();
258 if (!input.core) return current_;
259 try {
260 auto updated = input.core->snapshot();
261 auto saved = store_.publish(current_.head(), updated.runtime(), updated.metadata().encode());
262 auto mapped = restore(std::move(saved), schema_);
263 if (!input.core->pending()) {
264 if constexpr (requires { input.core->rebase(mapped); }) input.core->rebase(mapped);
265 else input.core = std::make_unique<Core>(Core::from_snapshot(mapped));
266 }
267 // mapped owns the acknowledged graph before either old executor or its
268 // construction lease is released. Keep the candidate's private jobs.
269 core_ = std::move(input.core);
270 core_scope_ = std::move(input.scope);
271 current_ = std::move(mapped);
272 catalog_failure_ = nullptr;
273 return current_;
274 } catch (...) { remember_failure(); poison(); throw; }
275 }
276 void remember_failure() noexcept {
277 // A streamed Core has its own catalog connection. Keep its exact error
278 // alive without copying the operation string while handling a failure.
279 try { throw; }
280 catch (catalog_error const &) { catalog_failure_ = std::current_exception(); }
281 catch (...) { catalog_failure_ = nullptr; }
282 }
283 void require_active() const { if (failed()) throw std::logic_error("failed persistent Everett engine; reconnect it"); }
284 void poison() noexcept {
285 failed_ = true;
286 if constexpr (requires { { core_->poison() } noexcept; }) core_->poison();
287 }
