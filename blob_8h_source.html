39 inline std::array<std::byte, 9> encode_blob_value(std::optional<std::uint64_t> value) noexcept {
40 std::array<std::byte, 9> bytes{};
41 bytes[0] = value ? std::byte{1} : std::byte{0};
42 auto bits = value.value_or(0);
43 for (unsigned i = 0; i != 8; ++i) bytes[i + 1] = static_cast<std::byte>(bits >> (8 * i));
44 return bytes;
45 }
46
47 inline std::optional<std::uint64_t> decode_blob_value(std::span<std::byte const, 9> bytes) {
48 if (bytes[0] == std::byte{0}) return std::nullopt;
49 if (bytes[0] != std::byte{1}) throw std::invalid_argument("invalid blob value tag");
50 std::uint64_t value = 0;
51 for (unsigned i = 0; i != 8; ++i) {
52 value |= std::uint64_t(std::to_integer<unsigned>(bytes[i + 1])) << (8 * i);
53 }
54 return value;
55 }

87 // A bounded immutable slice: separate native and borrowed front-coded byte
88 // streams, two select15 indexes and one rank15 virtual interleave. Mapping,
89 // target identity/pins and the outer redundant-level scheduler live above it.
90 struct blob {
91 static blob build(std::span<blob_record const> native,
92 std::span<std::string const> borrowed = {},
93 std::uint64_t native_restart_factor = 18,
94 std::span<std::uint64_t const> borrowed_prefix_ceilings = {},
95 borrowed_prefix_policy policy = borrowed_prefix_policy::shared_boundaries) {
96 if (native.size() > std::numeric_limits<std::uint64_t>::max() - borrowed.size()) {
97 throw std::length_error("blob virtual count overflows");
98 }
99 blob result;
100 std::vector<front_record<9>> native_records;
101 native_records.reserve(native.size());
102 for (std::size_t i = 0; i != native.size(); ++i) {
103 if (i && compare_keys(native[i - 1].key, native[i].key) >= 0) {
104 throw std::invalid_argument("blob native keys must be strictly sorted");
105 }
106 native_records.push_back({native[i].key, encode_blob_value(native[i].value)});
107 }
108 result.native_ = std::make_shared<front_array<9> const>(
109 front_array<9>::build(native_records, {}, native_restart_factor));
110 build_index(result, native, borrowed, borrowed_prefix_ceilings, policy);
111 return result;
112 }

114 // Repairing a downstream index never changes native front coding or its
115 // select15 offsets. This initial builder materializes native keys as scratch;
116 // a streaming builder can remove that peak without changing the representation.
117 blob reindex(std::span<std::string const> borrowed,
118 std::span<std::uint64_t const> borrowed_prefix_ceilings = {},
119 borrowed_prefix_policy policy = borrowed_prefix_policy::shared_boundaries) const {
120 std::vector<blob_record> native_records;
121 native_records.reserve(static_cast<std::size_t>(native_->size()));
122 native_->view().visit_all([&](front_item<9> item) {
123 native_records.push_back({std::string(item.key.prefix), decode_blob_value(item.value)});
124 return true;
125 });
126 if (native_records.size() > std::numeric_limits<std::uint64_t>::max() - borrowed.size()) {
127 throw std::length_error("blob virtual count overflows");
128 }
129 blob result;
130 result.native_ = native_;
131 build_index(result, native_records, borrowed, borrowed_prefix_ceilings, policy);
132 return result;
133 }

136 static void build_index(blob & result, std::span<blob_record const> native,
137 std::span<std::string const> borrowed,
138 std::span<std::uint64_t const> borrowed_prefix_ceilings,
139 borrowed_prefix_policy policy) {
140 if (!borrowed_prefix_ceilings.empty() && borrowed_prefix_ceilings.size() != borrowed.size()) {
141 throw std::invalid_argument("one borrowed prefix ceiling is required per record");
142 }
143 if (policy != borrowed_prefix_policy::ordinary &&
144 policy != borrowed_prefix_policy::bidirectional &&
145 policy != borrowed_prefix_policy::shared_boundaries) {
146 throw std::invalid_argument("invalid borrowed prefix policy");
147 }
148 std::vector<std::uint64_t> ceilings;
149 if (policy == borrowed_prefix_policy::shared_boundaries) {
150 ceilings.assign(borrowed.size(), std::numeric_limits<std::uint64_t>::max());
151 if (!borrowed_prefix_ceilings.empty()) {
152 std::copy(borrowed_prefix_ceilings.begin(), borrowed_prefix_ceilings.end(), ceilings.begin());
153 }
154 borrowed_prefix_ceilings = ceilings;
155 } else if (policy == borrowed_prefix_policy::bidirectional) {
156 // shared[i] <= min(LCP(S[i-1], S[i]), LCP(S[i], S[i+1]))
157 // lets S[i] decode from an anchor on either side of it. The final
158 // sample is literal because its upper interval is unbounded.
159 // Relative to ordinary FC, extra suffix bytes are the positive drops
160 // in the adjacent-LCP sequence (zero at both ends). Their sum equals
161 // its positive rises, each <= that record's ordinary suffix length.
162 // Thus suffix bytes at most double. Framing is charged separately;
163 // arbitrary caller ceilings can add further redundancy.
164 ceilings.reserve(borrowed.size());
165 for (std::size_t i = 0; i != borrowed.size(); ++i) {
166 auto ceiling = i + 1 == borrowed.size() ? 0 : common_prefix(borrowed[i], borrowed[i + 1]);
167 if (!borrowed_prefix_ceilings.empty()) {
168 ceiling = std::min<std::uint64_t>(ceiling, borrowed_prefix_ceilings[i]);
169 }
170 ceilings.push_back(ceiling);
171 }
172 borrowed_prefix_ceilings = ceilings;
173 }
174 result.borrowed_policy_ = policy;
175 std::vector<front_record<0>> borrowed_records;
176 borrowed_records.reserve(borrowed.size());
177 result.false_borrows_.resize(borrowed.size() / 8 + (borrowed.size() % 8 != 0));
178 std::size_t native_at = 0;
179 for (std::size_t i = 0; i != borrowed.size(); ++i) {
180 if (i && compare_keys(borrowed[i - 1], borrowed[i]) > 0) {
181 throw std::invalid_argument("blob borrowed keys must be sorted");
182 }
183 while (native_at != native.size() && compare_keys(native[native_at].key, borrowed[i]) < 0) {
184 ++native_at;
185 }
186 if (native_at != native.size() && compare_keys(native[native_at].key, borrowed[i]) == 0) {
187 result.false_borrows_[i / 8] |= static_cast<std::byte>(1u << (i % 8));
188 }
189 borrowed_records.push_back({borrowed[i], {}});
190 }
191 auto count = std::uint64_t(native.size()) + borrowed.size();
192 std::vector<std::uint8_t> classes(static_cast<std::size_t>(count / 15 + (count % 15 != 0)));
193 std::size_t a = 0;
194 std::size_t s = 0;
195 for (std::uint64_t i = 0; i != count; ++i) {
196 // Native precedes borrowed on equality, including multiple borrowed
197 // copies. false_borrow then identifies a possibly earlier native slot.
198 auto take_borrowed = s != borrowed.size() &&
199 (a == native.size() || compare_keys(borrowed[s], native[a].key) < 0);
200 if (policy == borrowed_prefix_policy::shared_boundaries && i % 15 == 0 && s) {
201 // rank15(i / 15) == s, so S[s-1] is the predecessor needed by
202 // this window. It must decode from the known boundary key C[i].
203 // The rightmost applicable boundary has the smallest LCP; taking
204 // the minimum here imposes precisely that constraint. If no cut
205 // uses a record, its ordinary copy count is left unchanged.
206 auto const & boundary = take_borrowed ? borrowed[s] : native[a].key;
207 ceilings[s - 1] = std::min<std::uint64_t>(
208 ceilings[s - 1], common_prefix(borrowed[s - 1], boundary));
209 }
210 if (take_borrowed) {
211 ++classes[static_cast<std::size_t>(i / 15)];
212 ++s;
213 } else {
214 ++a;
215 }
216 }
217 result.borrowed_ = front_array<0>::build(borrowed_records, borrowed_prefix_ceilings);
218 result.interleave_ = rank15_index::build(classes, count);
219 result.virtual_count_ = count;
220 }

233 bool false_borrow(std::uint64_t ordinal) const {
234 if (ordinal >= borrowed_.size()) throw std::out_of_range("blob borrowed ordinal");
235 return (std::to_integer<unsigned>(false_borrows_[ordinal / 8]) >> (ordinal % 8)) & 1;

238 blob_window project(std::uint64_t group) const {
239 if (group >= group_count()) throw std::out_of_range("blob virtual group");
240 auto first = group * 15;
241 auto last = first + std::min<std::uint64_t>(15, virtual_count_ - first);
242 auto ranks = interleave_.view();
243 auto a = ranks.rank(group);
244 // The stored population is exactly the difference of the two ranks.
245 auto b = a + ranks.class_at(group);
246 return {first - a, last - b, a, b};
247 }

249 // The caller provides a target window selected by the incoming sample and
250 // the corresponding known lower key (or empty context at ordinal zero).
251 // For full lookup this must be the window containing the last virtual key
252 // <= query. This method does not search a large first catalog unaided.
253 blob_window_result search_window(std::string_view query, std::uint64_t group,
254 front_anchor lower) const {
255 auto window = project(group);
256 blob_window_result result;
257 native_->view().visit_window(window.native_first, window.native_last, lower, query.size(),
258 [&](front_item<9> item) {
259 auto order = compare_prefix(item.key, query);
260 if (!order) result.native = blob_native_match{item.ordinal, decode_blob_value(item.value)};
261 return order < 0;
262 });
263
264 auto consider_sample = [&](front_item<0> item) {
265 auto order = compare_prefix(item.key, query);
266 if (order > 0) return false;
267 auto is_false = false_borrow(item.ordinal);
268 result.borrowed_predecessor = blob_borrowed_predecessor{
269 item.ordinal, checked_target_ordinal(item.ordinal), true,
270 std::string(item.key.prefix), item.key.full_size, is_false};
271 if (!order && is_false && !result.native) {
272 if (!window.native_first) {
273 throw std::invalid_argument("false borrow has no native predecessor");
274 }
275 // Native-before-borrowed tie order and the builder's equality bit
276 // prove this exact slot matches query, even before this window.
277 auto ordinal = window.native_first - 1;
278 result.native = blob_native_match{
279 ordinal, decode_blob_value(native_->view().encoded_at(ordinal).value)};
280 }
281 return true;
282 };
283 if (window.borrowed_first) {
284 auto ordinal = window.borrowed_first - 1;
285 if (borrowed_policy_ != borrowed_prefix_policy::ordinary) {
286 // Both conservative policies clamp S[j]'s copied prefix to one
287 // shared with this upper boundary anchor. No backward walk is needed.
288 borrowed_.view().visit_window(ordinal, ordinal + 1, lower, query.size(), consider_sample);
289 } else {
290 result.borrowed_predecessor = blob_borrowed_predecessor{
291 ordinal, checked_target_ordinal(ordinal), false, {}, 0, false_borrow(ordinal)};
292 }
293 }
294 borrowed_.view().visit_window(window.borrowed_first, window.borrowed_last, lower,
295 query.size(), consider_sample);
296 return result;
297 }
