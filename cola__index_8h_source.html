48 std::array<std::uint64_t, 3> common{};

50 // Every live head follows the same preceding occurrence. Larger LCPs
51 // sort first; equal LCPs need only a suffix comparison. Stable origin order
52 // preserves native, main-borrow, secondary-borrow ties.
53 template <class Live, class Head> frontier_selection choose_frontier(
54 std::array<std::uint64_t, 3> const & prefixes, Live live, Head head) {
55 frontier_selection result;
56 bit_view key;
57 for (unsigned origin = 0; origin < 3; ++origin) if (live(origin)) {
58 auto candidate = head(origin);
59 if (result.origin == 3) {
60 result.origin = origin; key = candidate; result.common[origin] = key.size();
61 continue;
62 }
63 auto first = prefixes[origin], second = prefixes[result.origin];
64 bit_comparison comparison;
65 if (first != second) comparison = {std::min(first, second), first > second ? -1 : 1};
66 else {
67 comparison = compare_common_bits(candidate.subview(first, candidate.size() - first),
68 key.subview(first, key.size() - first));
69 comparison.common_bits += first;
70 }
71 if (comparison.order < 0) {
72 // New winner <= old winner <= each old loser: ordered LCP minimum.
73 for (unsigned i = 0; i < origin; ++i)
74 result.common[i] = std::min(result.common[i], comparison.common_bits);
75 result.origin = origin; key = candidate;
76 result.common[origin] = key.size();
77 } else result.common[origin] = comparison.common_bits;
78 }
79 if (result.origin == 3) error_detail::raise<std::invalid_argument>("COLA frontier has no live stream");
80 return result;
81 }

177 template <class Result, class Capture> Result search_window_with(std::uint64_t group,
178 profile_query_context<P> const & lower, Capture && capture,
179 profile_comparison_work * native_work = nullptr,
180 std::array<profile_comparison_work *, 2> borrowed_work = {}) const {

183 Result result;
184 native_.compare_window(window.native_first, window.native_last, lower,
185 [&](profile_comparison_item<P> item) {

187 result.native = capture(item.ordinal, item.value);
188 return item.comparison.order() < 0;
189 }, native_work);

192 auto is_false = false_borrow(route, ordinal);
193 result.predecessors[route] = profile_blob_borrowed_predecessor<P>{ordinal,
194 cola_detail::target_ordinal(ordinal, group_size), is_false, comparison};
195 if (!comparison.order() && is_false && !result.native) {
196 if (!window.native_first) error_detail::raise<std::invalid_argument>("COLA false borrow has no native predecessor");
197 auto native_ordinal = window.native_first - 1;
198 result.native = capture(native_ordinal, native_.encoded_at(native_ordinal).value);
199 }
200 };
201 borrowed_[route].compare_window(window.borrowed_first[route], window.borrowed_last[route], lower,
202 [&](profile_comparison_item<P> item) {
203 if (item.comparison.order() > 0) return false;
204 remember(item.ordinal, item.comparison);
205 return true;
206 }, borrowed_work[route]);

396 template <bool Compare> std::optional<bit_comparison> advance_impl() {
397 if (done()) error_detail::raise<std::out_of_range>("COLA sampler at end");
398 auto selected = choose();
399 auto first_length = selected.common[selected.origin];
400 auto common = first_length;
401 auto width = std::min<std::uint64_t>(P::group_size, count_ - ordinal_);
402 for (std::uint64_t i = 0; i < width; ++i) {
403 prefixes_ = selected.common;
404 auto next = !selected.origin ? native_.advance_comparison() :
405 borrowed_[selected.origin - 1].advance_comparison();
406 if (next && (next->order > 0 || (!selected.origin && !next->order)))
407 error_detail::raise<std::invalid_argument>("COLA sample source order mismatch");
408 prefixes_[selected.origin] = next ? next->common_bits : 0;
409 ++ordinal_;
410 if (!done() && (Compare || i + 1 < width)) {
411 selected = choose();
412 if constexpr (Compare) common = std::min(common, prefixes_[selected.origin]);
413 }
414 }
415 if constexpr (Compare) if (!done())
416 return bit_comparison{common, common == first_length &&
417 selected.common[selected.origin] == first_length ? 0 : -1};
418 return std::nullopt;
419 }

462 cola_index_builder(Output output, native_pointer native, main_pointer main = {}, native_pointer secondary = {})
463 : native_(checked(std::move(native))), main_(std::move(main)), secondary_(std::move(secondary)), output_(std::move(output)) {
464 auto remaining = std::numeric_limits<std::uint64_t>::max() - native_->size();
465 auto primary_count = main_ ? main_->group_count() : 0;
466 auto secondary_count = secondary_ ? secondary_->size() / P::group_size + (secondary_->size() % P::group_size != 0) : 0;
467 if (primary_count > remaining || secondary_count > remaining - primary_count)
468 error_detail::raise<std::length_error>("COLA augmented count overflow");
469 if (main_ || secondary_) native_cursor_.emplace(native_->view());
470 if (main_) primary_cursor_.emplace(main_);
471 if (secondary_) secondary_cursor_.emplace(secondary_->view());
472 }
473 cola_index_builder(cola_index_builder const &) = delete;
474 cola_index_builder & operator=(cola_index_builder const &) = delete;
475 cola_index_builder(cola_index_builder &&) = default;
476 cola_index_builder & operator=(cola_index_builder &&) = default;
477 bool done() const noexcept {
478 return native_ && !failed_ && (native_cursor_ ? !live(0) && !live(1) && !live(2) : count_ == native_->size());
479 }
480 bool failed() const noexcept { return failed_ || output_.failed(); }
481 bool finished() const noexcept { return finished_; }
482 std::uint64_t size() const noexcept { return count_; }
483 native_pointer native_owner() const noexcept { return native_; }
484 main_pointer main_target() const noexcept { return main_; }
485 native_pointer secondary_target() const noexcept { return secondary_; }
486 std::uint64_t step(std::uint64_t budget) {
487 require_active();
488 std::uint64_t consumed = 0;
489 try {
490 // With no targets, all augmented occurrences are native. Navigation
491 // depends only on their count: do not open a payload cursor at all.
492 if (!native_cursor_) return step_terminal(budget);
493 while (consumed < budget && !done()) {
494 auto selected = cola_detail::choose_frontier(prefixes_,
495 [&](unsigned origin) { return live(origin); },
496 [&](unsigned origin) { return head(origin); });
497 auto origin = selected.origin;
498 auto key = head(origin);
499 auto common = prefixes_[origin];
500 bool equal = count_ && common == previous_length_ && key.size() == previous_length_;
501 if (!origin && equal)
502 error_detail::raise<std::invalid_argument>("COLA sources violate sorted native order");
503 for (unsigned route = 0; route < 2; ++route)
504 if (borrowed_count_[route]) cut_common_[route] = std::min(cut_common_[route], common);
505 if (!width_) for (unsigned route = 0; route < 2; ++route)
506 cuts_[route].push_back(borrowed_count_[route] ? cut_common_[route] : 0);
507 native_equal_ = !origin || (equal && native_equal_);
508 if (origin) {
509 auto route = origin - 1;
510 // Along a sorted walk the minimum adjacent LCP since the preceding
511 // borrow is its exact LCP with this key. Reuse that same frontier
512 // for FC output instead of comparing the inherited prefix again.
513 output_.append_known(route, key, borrowed_count_[route] ? cut_common_[route] : 0);
514 auto ordinal = borrowed_count_[route]++;
515 if (!(ordinal & 7)) flags_[route].push_back(std::byte{0});
516 if (native_equal_) flags_[route].back() |= std::byte(1u << (ordinal & 7));
517 ++population_[route];
518 cut_common_[route] = key.size();
519 }
520 previous_length_ = key.size();
521 prefixes_ = selected.common;
522 if (!origin) prefixes_[0] = advance_native(*native_cursor_);
523 else if (origin == 1) {
524 auto next = primary_cursor_->advance_comparison();
525 prefixes_[1] = next ? next->common_bits : 0;
526 } else {
527 auto next_common = key.size();
528 for (std::uint64_t i = 0; i < P::group_size && !secondary_cursor_->done(); ++i)
529 next_common = std::min(next_common, advance_native(*secondary_cursor_));
530 prefixes_[2] = next_common;
531 }
532 ++count_; ++consumed; ++width_;
533 if (width_ == P::group_size) flush_group();
534 }
535 } catch (...) { failed_ = true; throw; }
536 return consumed;
537 }
538 auto finish() {
539 require_active();
540 if (!done()) error_detail::raise<std::logic_error>("COLA builder has remaining input");
541 try {
542 if (width_) flush_group();
543 cola_detail::index_metadata<P> metadata{{ranks_[0].finish(), ranks_[1].finish()},
544 std::move(flags_), std::move(cuts_), count_};
545 auto result = output_.finish(native_, main_, secondary_, std::move(metadata));
546 finished_ = true;
547 return result;
548 } catch (...) { failed_ = true; throw; }
549 }
550 private:
551 native_pointer native_;
552 main_pointer main_;
553 native_pointer secondary_;
554 using native_cursor_type = decltype(std::declval<Native const &>().view().cursor());
555 std::optional<native_cursor_type> native_cursor_;
556 std::optional<cola_sample_cursor<P, target_type>> primary_cursor_;
557 std::optional<native_cursor_type> secondary_cursor_;
558 Output output_;
559 std::array<rank_groups_builder<P::group_size>, 2> ranks_;
560 std::array<std::vector<std::byte>, 2> flags_;
561 std::array<std::vector<std::uint64_t>, 2> cuts_;
562 std::array<std::uint64_t, 2> borrowed_count_{}, cut_common_{}, population_{};
563 std::array<std::uint64_t, 3> prefixes_{};
564 std::uint64_t previous_length_ = 0, count_ = 0, width_ = 0;
565 bool native_equal_ = false, failed_ = false, finished_ = false;
566 static native_pointer checked(native_pointer native) {
567 if (!native) error_detail::raise<std::invalid_argument>("null COLA native source");
568 return native;
569 }
570 void require_active() const {
571 if (!native_ || failed() || finished_) error_detail::raise<std::logic_error>("COLA builder is no longer active");
572 }
573 bool live(unsigned origin) const noexcept {
574 if (!origin) return native_cursor_ && !native_cursor_->done();
575 if (origin == 1) return primary_cursor_ && !primary_cursor_->done();
576 return secondary_cursor_ && !secondary_cursor_->done();
577 }
578 bit_view head(unsigned origin) const {
579 if (!origin) return native_cursor_->peek().key.prefix;
580 if (origin == 1) return primary_cursor_->peek().key;
581 return secondary_cursor_->peek().key.prefix;
582 }
583 static std::uint64_t advance_native(native_cursor_type & cursor) {
584 auto next = cursor.advance_comparison();
585 if (next && next->order >= 0)
586 error_detail::raise<std::invalid_argument>("COLA native source order mismatch");
587 return next ? next->common_bits : 0;
588 }
589 std::uint64_t step_terminal(std::uint64_t budget) {
590 std::uint64_t consumed = 0;
591 auto total = native_->size();
592 while (consumed < budget && count_ < total) {
593 if (!width_) for (auto & cuts : cuts_) cuts.push_back(0);
594 auto chunk = std::min({budget - consumed, total - count_, P::group_size - width_});
595 count_ += chunk; consumed += chunk; width_ += chunk;
596 if (width_ == P::group_size) flush_group();
597 }
598 return consumed;
599 }
600 void flush_group() {
601 for (unsigned route = 0; route < 2; ++route) ranks_[route].append(population_[route], width_);
602 width_ = 0; population_ = {};
603 }
