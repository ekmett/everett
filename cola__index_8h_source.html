47 std::array<std::uint64_t, 3> common{};

49 // Every live head follows the same preceding occurrence. Larger LCPs
50 // sort first; equal LCPs need only a suffix comparison. Stable origin order
51 // preserves native, main-borrow, secondary-borrow ties.
52 template <class Live, class Head> frontier_selection choose_frontier(
53 std::array<std::uint64_t, 3> const & prefixes, Live live, Head head) {
54 frontier_selection result;
55 bit_view key;
56 for (unsigned origin = 0; origin < 3; ++origin) if (live(origin)) {
57 auto candidate = head(origin);
58 if (result.origin == 3) {
59 result.origin = origin; key = candidate; result.common[origin] = key.size();
60 continue;
61 }
62 auto first = prefixes[origin], second = prefixes[result.origin];
63 bit_comparison comparison;
64 if (first != second) comparison = {std::min(first, second), first > second ? -1 : 1};
65 else {
66 comparison = compare_common_bits(candidate.subview(first, candidate.size() - first),
67 key.subview(first, key.size() - first));
68 comparison.common_bits += first;
69 }
70 if (comparison.order < 0) {
71 // New winner <= old winner <= each old loser: ordered LCP minimum.
72 for (unsigned i = 0; i < origin; ++i)
73 result.common[i] = std::min(result.common[i], comparison.common_bits);
74 result.origin = origin; key = candidate;
75 result.common[origin] = key.size();
76 } else result.common[origin] = comparison.common_bits;
77 }
78 if (result.origin == 3) error_detail::raise<std::invalid_argument>("COLA frontier has no live stream");
79 return result;
80 }

135 cola_window project(std::uint64_t group) const {
136 if (group >= group_count()) error_detail::raise<std::out_of_range>("COLA virtual group");
137 auto first = group * group_size;
138 auto width = std::min<std::uint64_t>(group_size, count_ - first);
139 std::array<std::uint64_t, 2> prefix{}, population{};
140 for (unsigned route = 0; route < 2; ++route) {
141 prefix[route] = ranks_[route].rank(group);
142 population[route] = ranks_[route].class_at(group);
143 if (prefix[route] > first || prefix[route] > borrowed_[route].size() ||
144 population[route] > width || population[route] > borrowed_[route].size() - prefix[route])
145 error_detail::raise<std::invalid_argument>("invalid COLA rank projection");
146 }
147 if (prefix[0] > first - prefix[1] || population[0] > width - population[1])
148 error_detail::raise<std::invalid_argument>("overlapping COLA population directories");
149 auto native_first = first - prefix[0] - prefix[1];
150 auto native_last = native_first + width - population[0] - population[1];
151 if (native_last > native_.size()) error_detail::raise<std::invalid_argument>("invalid COLA native projection");
152 return {native_first, native_last, prefix,
153 {prefix[0] + population[0], prefix[1] + population[1]}};

155 cola_window_result<P> search_window(std::uint64_t group, profile_query_context<P> const & lower,
156 profile_comparison_work * native_work = nullptr,
157 std::array<profile_comparison_work *, 2> borrowed_work = {}) const {
158 if (lower.order() > 0) error_detail::raise<std::invalid_argument>("query precedes COLA boundary");
159 auto window = project(group);
160 cola_window_result<P> result;
161 native_.compare_window(window.native_first, window.native_last, lower,
162 [&](profile_comparison_item<P> item) {
163 if (!item.comparison.order())
164 result.native = profile_blob_native_match<P>{item.ordinal, bit_string::copy(item.value)};
165 return item.comparison.order() < 0;
166 }, native_work);
167 for (unsigned route = 0; route < 2; ++route) {
168 auto remember = [&](std::uint64_t ordinal, profile_query_context<P> const & comparison) {
169 auto is_false = false_borrow(route, ordinal);
170 result.predecessors[route] = profile_blob_borrowed_predecessor<P>{ordinal,
171 cola_detail::target_ordinal(ordinal, group_size), is_false, comparison};
172 if (!comparison.order() && is_false && !result.native) {
173 if (!window.native_first) error_detail::raise<std::invalid_argument>("COLA false borrow has no native predecessor");
174 auto native_ordinal = window.native_first - 1;
175 result.native = profile_blob_native_match<P>{native_ordinal,
176 bit_string::copy(native_.encoded_at(native_ordinal).value)};
177 }
178 };
179 borrowed_[route].compare_window(window.borrowed_first[route], window.borrowed_last[route], lower,
180 [&](profile_comparison_item<P> item) {
181 if (item.comparison.order() > 0) return false;
182 remember(item.ordinal, item.comparison);
183 return true;
184 }, borrowed_work[route]);
185 if (!result.predecessors[route] && window.borrowed_first[route])
186 remember(window.borrowed_first[route] - 1, lower.predecessor(cuts_[route][group]));
187 }
188 return result;
189 }

378 // Each step consumes at most budget merged occurrences. Consuming a sampled
379 // head advances its source by at most K occurrences, and bytes/comparison and
380 // allocation remain additional work. Only current cursor/key contexts are
381 // reconstructed; borrowed payload and compact navigation are output storage.
382 // Without targets only native count is inspected; step charges occurrences
383 // while doing one zero-navigation update per crossed K-group, without keys.

385 // deliberately does not revalidate their framing or strictly sorted order.

414 std::uint64_t step(std::uint64_t budget) {
415 require_active();
416 std::uint64_t consumed = 0;
417 try {
418 // With no targets, all augmented occurrences are native. Navigation
419 // depends only on their count: do not open a payload cursor at all.
420 if (!native_cursor_) return step_terminal(budget);
421 while (consumed < budget && !done()) {
422 auto selected = cola_detail::choose_frontier(prefixes_,
423 [&](unsigned origin) { return live(origin); },
424 [&](unsigned origin) { return head(origin); });
425 auto origin = selected.origin;
426 auto key = head(origin);
427 auto common = prefixes_[origin];
428 bool equal = count_ && common == previous_length_ && key.size() == previous_length_;
429 if (!origin && equal)
430 error_detail::raise<std::invalid_argument>("COLA sources violate sorted native order");
431 for (unsigned route = 0; route < 2; ++route)
432 if (borrowed_count_[route]) cut_common_[route] = std::min(cut_common_[route], common);
433 if (!width_) for (unsigned route = 0; route < 2; ++route)
434 cuts_[route].push_back(borrowed_count_[route] ? cut_common_[route] : 0);
435 native_equal_ = !origin || (equal && native_equal_);
436 if (origin) {
437 auto route = origin - 1;
438 // Along a sorted walk the minimum adjacent LCP since the preceding
439 // borrow is its exact LCP with this key. Reuse that same frontier
440 // for FC output instead of comparing the inherited prefix again.
441 writers_[route].append_known(key, borrowed_count_[route] ? cut_common_[route] : 0);
442 auto ordinal = borrowed_count_[route]++;
443 if (!(ordinal & 7)) flags_[route].push_back(std::byte{0});
444 if (native_equal_) flags_[route].back() |= std::byte(1u << (ordinal & 7));
445 ++population_[route];
446 cut_common_[route] = key.size();
447 }
448 previous_length_ = key.size();
449 prefixes_ = selected.common;
450 if (!origin) prefixes_[0] = advance_native(*native_cursor_);
451 else if (origin == 1) {
452 auto next = primary_cursor_->advance_comparison();
453 prefixes_[1] = next ? next->common_bits : 0;
454 } else {
455 auto next_common = key.size();
456 for (std::uint64_t i = 0; i < P::group_size && !secondary_cursor_->done(); ++i)
457 next_common = std::min(next_common, advance_native(*secondary_cursor_));
458 prefixes_[2] = next_common;
459 }
460 ++count_; ++consumed; ++width_;
461 if (width_ == P::group_size) flush_group();
462 }
463 } catch (...) { failed_ = true; throw; }
464 return consumed;
465 }
466 index_type finish() {
467 require_active();
468 if (!done()) error_detail::raise<std::logic_error>("COLA builder has remaining input");
469 try {
470 if (width_) flush_group();
471 std::array<typename index_type::borrowed_array, 2> borrowed{writers_[0].finish(), writers_[1].finish()};
472 std::array<rank_groups<P::group_size>, 2> ranks{ranks_[0].finish(), ranks_[1].finish()};
473 index_type result(native_, main_, secondary_, std::move(borrowed), std::move(ranks),
474 std::move(flags_), std::move(cuts_), count_);
475 finished_ = true;
476 return result;
477 } catch (...) { failed_ = true; throw; }
478 }

497 }

499 if (!native_ || failed_ || finished_) error_detail::raise<std::logic_error>("COLA builder is no longer active");
500 }

503 if (origin == 1) return primary_cursor_ && !primary_cursor_->done();
504 return secondary_cursor_ && !secondary_cursor_->done();
505 }

511 static std::uint64_t advance_native(profile_cursor<P> & cursor) {
512 auto next = cursor.advance_comparison();
513 if (next && next->order >= 0)
514 error_detail::raise<std::invalid_argument>("COLA native source order mismatch");
515 return next ? next->common_bits : 0;
516 }
517 std::uint64_t step_terminal(std::uint64_t budget) {
518 std::uint64_t consumed = 0;
519 auto total = native_->size();
520 while (consumed < budget && count_ < total) {
521 if (!width_) for (auto & cuts : cuts_) cuts.push_back(0);
522 auto chunk = std::min({budget - consumed, total - count_, P::group_size - width_});
523 count_ += chunk; consumed += chunk; width_ += chunk;
524 if (width_ == P::group_size) flush_group();
525 }
526 return consumed;
527 }
528 void flush_group() {
529 for (unsigned route = 0; route < 2; ++route) ranks_[route].append(population_[route], width_);
530 width_ = 0; population_ = {};
531 }
