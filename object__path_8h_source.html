32 // Canonical representation of a physical 128-bit object identity. The
33 // allocator supplies well-distributed IDs; this type validates spelling,
34 // not entropy or uniqueness. Logical keys never enter the path builder.
35 struct object_id {
36 explicit object_id(std::string hex) : hex_(std::move(hex)) {
37 if (hex_.size() != 32) throw std::invalid_argument("object ID must contain 32 lowercase hex digits");
38 for (char c : hex_)
39 if (!((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f')))
40 throw std::invalid_argument("noncanonical object ID");
41 }
42 static object_id from_hex(std::string_view hex) { return object_id(std::string(hex)); }
43 std::string const & hex() const noexcept { return hex_; }
44 bool operator==(object_id const &) const = default;
45 private:
46 std::string hex_;
47 };

60 // Parse the canonical generic relative spelling before filesystem lexical
61 // normalization: ab/cd/<remaining 28 hex digits>.kv (or .index).
62 inline parsed_object_path parse_object_path(std::string_view path) {
63 if (path.size() < 37 || path.size() > 40 || path[2] != '/' || path[5] != '/' || path[34] != '.')
64 throw std::invalid_argument("noncanonical sharded object path");
65 auto id = object_id(std::string(path.substr(0, 2)) + std::string(path.substr(3, 2)) +
66 std::string(path.substr(6, 28)));
67 auto extension = path.substr(34);
68 for (auto kind : {file_kind::native_blob, file_kind::fractional_index})
69 if (extension == file_extension(kind)) return {std::move(id), kind};
70 throw std::invalid_argument("unsupported Diet object extension");
71 }
