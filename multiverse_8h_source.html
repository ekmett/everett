125 // Establish missing directory names with a barrier on each new directory
126 // and its parent. Existing ancestors must already be durable and trusted.
127 // A failed barrier leaves the names in place for explicit recovery.
128 static multiverse create(std::filesystem::path root) {
129 if (root.empty()) throw std::invalid_argument("multiverse root must name a directory");
130#if defined(__APPLE__) || defined(__linux__)
131 auto path = std::filesystem::weakly_canonical(std::filesystem::absolute(root));
132 auto ancestor = path;
133 std::vector<std::string> missing;
134 while (!std::filesystem::exists(ancestor)) {
135 missing.push_back(ancestor.filename().string());
136 ancestor = ancestor.parent_path();
137 }
138 ancestor = checked_root(ancestor);
139 posix_object_ops ops;
140 struct descriptor {
141 int fd;
142 explicit descriptor(int value) : fd(value) {
143 if (fd < 0) throw std::system_error(errno, std::generic_category(), "open multiverse directory");
144 }
145 descriptor(descriptor const &) = delete;
146 ~descriptor() { if (fd >= 0) ::close(fd); }
147 } parent(ops.open_root(ancestor));
148 for (auto name = missing.rbegin(); name != missing.rend(); ++name) {
149 if (ops.make_directory(parent.fd, name->c_str()) && errno != EEXIST)
150 throw std::system_error(errno, std::generic_category(), "create multiverse directory");
151 descriptor child(ops.open_directory(parent.fd, name->c_str()));
152 if (ops.sync_directory(child.fd) || ops.sync_directory(parent.fd))
153 throw std::system_error(errno, std::generic_category(), "sync multiverse directory; creation outcome unknown");
154 if (ops.close(std::exchange(parent.fd, -1)))
155 throw std::system_error(errno, std::generic_category(), "close multiverse directory; creation outcome unknown");
156 parent.fd = std::exchange(child.fd, -1);
157 }
158 if (ops.close(std::exchange(parent.fd, -1)))
159 throw std::system_error(errno, std::generic_category(), "close multiverse directory; creation outcome unknown");
160 return multiverse(path);
161#else
162 throw std::system_error(std::make_error_code(std::errc::operation_not_supported), "create multiverse directory");
163#endif
164 }
