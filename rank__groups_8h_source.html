39 // The caller supplies at most 128 classes and exactly the words containing
40 // them. Weighted bit populations preserve fields split across word cuts.
41 template <unsigned Bits> inline unsigned prefix_portable(
42 word_view words, unsigned count) noexcept {
43 static_assert(Bits == 2 || Bits == 3 || Bits == 5);
44 constexpr auto masks = [] {
45 std::array<std::array<std::uint64_t, Bits>, Bits> result{};
46 for (unsigned word = 0; word < Bits; ++word)
47 for (unsigned bit = 0; bit < 64; ++bit)
48 result[word][(word * 64 + bit) % Bits] |= std::uint64_t{1} << bit;
49 return result;
50 }();
51 unsigned result = 0;
52 std::uint64_t pairs = 0;
53 auto bits = count * Bits;
54 for (unsigned word = 0; word * 64 < bits; ++word) {
55 auto value = words[word];
56 auto remaining = bits - word * 64;
57 if (remaining < 64) value &= (std::uint64_t{1} << remaining) - 1;
58 if constexpr (Bits == 2) {
59 value = (value & 0x3333333333333333ull) + ((value >> 2) & 0x3333333333333333ull);
60 pairs += (value & 0x0f0f0f0f0f0f0f0full) + ((value >> 4) & 0x0f0f0f0f0f0f0f0full);
61 } else {
62 for (unsigned plane = 0; plane < Bits; ++plane)
63 result += unsigned(std::popcount(value & masks[word % Bits][plane])) << plane;
64 }
65 }
66 if constexpr (Bits == 2) {
67 // Four two-bit classes per byte contribute at most twelve per word;
68 // four words fit in byte lanes. Widen before summing their total.
69 pairs = (pairs & 0x00ff00ff00ff00ffull) + ((pairs >> 8) & 0x00ff00ff00ff00ffull);
70 return unsigned((pairs * 0x0001000100010001ull) >> 48);
71 }
72 return result;
73 }

81 template <unsigned Bits, unsigned Vector, unsigned Plane = 0>
82 inline uint8x16_t weighted_bits(uint8x16_t value) noexcept {
83 if constexpr (Bits == 2) {
84 auto pairs = vaddq_u8(vandq_u8(value, vdupq_n_u8(0x33)),
85 vandq_u8(vshrq_n_u8(value, 2), vdupq_n_u8(0x33)));
86 return vaddq_u8(vandq_u8(pairs, vdupq_n_u8(15)), vshrq_n_u8(pairs, 4));
87 }

89 plane_mask<Bits, 2 * Vector + 1, Plane>()};

91 if constexpr (Plane + 1 == Bits) return population;
92 else return vaddq_u8(population, weighted_bits<Bits, Vector, Plane + 1>(value));

95 template <unsigned Bits, unsigned Vector = 0>
96 inline uint16x8_t prefix_vectors(std::byte const * bytes, unsigned bits) noexcept {
97 uint64x2_t positions{2 * Vector, 2 * Vector + 1};
98 auto boundary = vdupq_n_u64(bits >> 6);
99 auto tail = vdupq_n_u64((std::uint64_t{1} << (bits & 63)) - 1);
100 auto mask = vorrq_u64(vcltq_u64(positions, boundary),
101 vandq_u64(vceqq_u64(positions, boundary), tail));
102 auto selected = vandq_u64(vreinterpretq_u64_u8(vld1q_u8(
103 reinterpret_cast<std::uint8_t const *>(bytes + 16 * Vector))), mask);
104 // Widen before adding vectors: five-bit classes can otherwise overflow
105 // an eight-bit lane. The complete checkpoint sums to at most 128*31.
106 auto counts = vpaddlq_u8(weighted_bits<Bits, Vector>(vreinterpretq_u8_u64(selected)));
107 if constexpr (Vector + 1 == Bits) return counts;
108 else return vaddq_u16(counts, prefix_vectors<Bits, Vector + 1>(bytes, bits));
109 }

120 // Population classes for virtual groups of K=2^n-1 entries. A class occupies
121 // exactly n bits, including across word boundaries. One 64-bit checkpoint
122 // every 128 classes bounds general rank queries to 127 class reads. K=3,7,31
123 // reduce packed words; K=15 shares rank15's packed-word/SIMD reduction.
124 // No origin bitmap, arbitrary within-group rank, or select is kept.
125 // Views check section shapes; a reader must validate borrowed metadata.
126 template <std::uint64_t K> struct rank_groups_view {
127 static_assert(K >= 3 && K < std::numeric_limits<std::uint64_t>::max() && std::has_single_bit(K + 1),
128 "group size must be 2^n-1 and at least three");
129 static constexpr std::uint64_t group_size = K;
130 static constexpr unsigned class_bits = unsigned(std::countr_zero(K + 1));

139 std::uint64_t virtual_count)
140 : classes_(classes), checkpoints_(checkpoints), virtual_count_(virtual_count) {

143 error_detail::raise<std::overflow_error>("rank groups packed size");

169 // Exclusive prefix at K*group for an existing group only.
170 std::uint64_t rank(std::uint64_t group) const {
171 if (group >= group_count()) error_detail::raise<std::out_of_range>("rank groups boundary");
172 auto result = checkpoints_[group >> 7];
173 auto limit = virtual_count_;
174 if constexpr (K == 3 || K == 7 || K == 31) {
175 auto count = unsigned(group & 127);
176 if (!count) return add_prefix(result, 0, limit);
177 auto word = (group >> 7) * (2 * class_bits);
178#if defined(__aarch64__) && defined(__ARM_NEON) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
179 // Two-bit broadword sums have lower dependent query latency. Wider
180 // classes favor NEON for the rank-plus-class projection operation.
181 if constexpr (K != 3)
182 if (classes_.size() - word >= 2 * class_bits)
183 return add_prefix(result, vaddvq_u16(rank_groups_detail::prefix_vectors<class_bits>(
184 classes_.bytes().data() + word * 8, count * class_bits)), limit);
185#endif
186 return add_prefix(result, rank_groups_detail::prefix_portable<class_bits>(classes_.subspan(word), count), limit);
187 }
188 for (auto i = group & ~std::uint64_t{127}; i < group; ++i) result = add_prefix(result, read_class(i), limit);
189 return add_prefix(result, 0, limit);
190 }

203 if (shift + class_bits > 64) value |= classes_[word + 1] << (64 - shift);

238 template <std::uint64_t K> struct rank_groups {
239 static_assert(K >= 3 && K < std::numeric_limits<std::uint64_t>::max() && std::has_single_bit(K + 1),
240 "group size must be 2^n-1 and at least three");
241 static constexpr std::uint64_t group_size = K;
242 static constexpr unsigned class_bits = unsigned(std::countr_zero(K + 1));
243
244 static rank_groups build(std::span<std::uint64_t const> source, std::uint64_t count) {
245 auto groups = count / K + (count % K != 0);
246 if (source.size() != groups) error_detail::raise<std::invalid_argument>("rank groups class length");
247 if (groups > (std::numeric_limits<std::uint64_t>::max() - 63) / class_bits)
248 error_detail::raise<std::overflow_error>("rank groups packed size");
249 auto bits = groups * class_bits;
250 rank_groups result;
251 result.virtual_count = count;
252 result.classes.resize((bits + 63) >> 6);
253 std::uint64_t total = 0;
254 for (std::uint64_t i = 0; i < groups; ++i) {
255 auto limit = i + 1 == groups && count % K ? count % K : K;
256 auto value = source[i];
257 if (value > limit) error_detail::raise<std::invalid_argument>("rank groups population");
258 if ((i & 127) == 0) result.checkpoints.push_back(total);
259 auto bit = i * class_bits;
260 unsigned shift = unsigned(bit & 63);
261 result.classes[bit >> 6] |= value << shift;
262 if (shift + class_bits > 64) result.classes[(bit >> 6) + 1] |= value >> (64 - shift);
263 total += value;
264 }
265 return result;
266 }

302 void append(std::uint64_t population, std::uint64_t width = K) {
303 require_active();
304 constexpr auto maximum = std::numeric_limits<std::uint64_t>::max();
305 if (partial_ || !width || width > K || population > width)
306 error_detail::raise<std::invalid_argument>("invalid incremental rank group");
307 if (width > maximum - data_.virtual_count || groups_ >= (maximum - 63) / class_bits)
308 error_detail::raise<std::length_error>("incremental rank groups are too large");
309 auto bits = (groups_ + 1) * class_bits;
310 auto words = (bits + 63) >> 6;
311 auto checkpoint = (groups_ & 127) == 0;
312 reserve(data_.classes, words);
313 if (checkpoint) reserve(data_.checkpoints, (groups_ >> 7) + 1);
314 // All potentially failing growth precedes the first logical mutation.
315 data_.classes.resize(static_cast<std::size_t>(words), 0);
316 if (checkpoint) data_.checkpoints.push_back(population_);
317 auto bit = groups_ * class_bits;
318 auto shift = unsigned(bit & 63);
319 data_.classes[bit >> 6] |= population << shift;
320 if (shift + class_bits > 64) data_.classes[(bit >> 6) + 1] |= population >> (64 - shift);
321 data_.virtual_count += width;
322 population_ += population;
323 ++groups_;
324 partial_ = width != K;
325 }

341 static void reserve(std::vector<std::uint64_t> & data, std::uint64_t count) {
342 if (count <= data.capacity()) return;

344 auto capacity = data.capacity();
345 auto grown = capacity > (data.max_size() >> 1) ? data.max_size() : capacity << 1;
