34 // The caller supplies at most 128 classes and exactly the words containing
35 // them. Weighted bit populations preserve fields split across word cuts.
36 template <unsigned Bits> inline unsigned prefix_portable(
37 word_view words, unsigned count) noexcept {
38 static_assert(Bits == 2 || Bits == 3 || Bits == 5);
39 constexpr auto masks = [] {
40 std::array<std::array<std::uint64_t, Bits>, Bits> result{};
41 for (unsigned word = 0; word < Bits; ++word)
42 for (unsigned bit = 0; bit < 64; ++bit)
43 result[word][(word * 64 + bit) % Bits] |= std::uint64_t{1} << bit;
44 return result;
45 }();
46 unsigned result = 0;
47 std::uint64_t pairs = 0;
48 auto bits = count * Bits;
49 for (unsigned word = 0; word * 64 < bits; ++word) {
50 auto value = words[word];
51 auto remaining = bits - word * 64;
52 if (remaining < 64) value &= (std::uint64_t{1} << remaining) - 1;
53 if constexpr (Bits == 2) {
54 value = (value & 0x3333333333333333ull) + ((value >> 2) & 0x3333333333333333ull);
55 pairs += (value & 0x0f0f0f0f0f0f0f0full) + ((value >> 4) & 0x0f0f0f0f0f0f0f0full);
56 } else {
57 for (unsigned plane = 0; plane < Bits; ++plane)
58 result += unsigned(std::popcount(value & masks[word % Bits][plane])) << plane;
59 }
60 }
61 if constexpr (Bits == 2) {
62 // Four two-bit classes per byte contribute at most twelve per word;
63 // four words fit in byte lanes. Widen before summing their total.
64 pairs = (pairs & 0x00ff00ff00ff00ffull) + ((pairs >> 8) & 0x00ff00ff00ff00ffull);
65 return unsigned((pairs * 0x0001000100010001ull) >> 48);
66 }
67 return result;
68 }

76 template <unsigned Bits, unsigned Vector, unsigned Plane = 0>
77 inline uint8x16_t weighted_bits(uint8x16_t value) noexcept {
78 if constexpr (Bits == 2) {
79 auto pairs = vaddq_u8(vandq_u8(value, vdupq_n_u8(0x33)),
80 vandq_u8(vshrq_n_u8(value, 2), vdupq_n_u8(0x33)));
81 return vaddq_u8(vandq_u8(pairs, vdupq_n_u8(15)), vshrq_n_u8(pairs, 4));
82 }
83 uint64x2_t mask{plane_mask<Bits, 2 * Vector, Plane>(),
84 plane_mask<Bits, 2 * Vector + 1, Plane>()};
85 auto population = vshlq_n_u8(vcntq_u8(vandq_u8(value, vreinterpretq_u8_u64(mask))), Plane);
86 if constexpr (Plane + 1 == Bits) return population;
87 else return vaddq_u8(population, weighted_bits<Bits, Vector, Plane + 1>(value));
88 }

90 template <unsigned Bits, unsigned Vector = 0>
91 inline uint16x8_t prefix_vectors(std::byte const * bytes, unsigned bits) noexcept {
92 uint64x2_t positions{2 * Vector, 2 * Vector + 1};
93 auto boundary = vdupq_n_u64(bits >> 6);
94 auto tail = vdupq_n_u64((std::uint64_t{1} << (bits & 63)) - 1);
95 auto mask = vorrq_u64(vcltq_u64(positions, boundary),
96 vandq_u64(vceqq_u64(positions, boundary), tail));
97 auto selected = vandq_u64(vreinterpretq_u64_u8(vld1q_u8(
98 reinterpret_cast<std::uint8_t const *>(bytes + 16 * Vector))), mask);
99 // Widen before adding vectors: five-bit classes can otherwise overflow
100 // an eight-bit lane. The complete checkpoint sums to at most 128*31.
101 auto counts = vpaddlq_u8(weighted_bits<Bits, Vector>(vreinterpretq_u8_u64(selected)));
102 if constexpr (Vector + 1 == Bits) return counts;
103 else return vaddq_u16(counts, prefix_vectors<Bits, Vector + 1>(bytes, bits));
104 }

115 // Population classes for virtual groups of K=2^n-1 entries. A class occupies
116 // exactly n bits, including across word boundaries. One 64-bit checkpoint
117 // every 128 classes bounds general rank queries to 127 class reads. K=3,7,31
118 // reduce packed words; K=15 shares rank15's packed-word/SIMD reduction.
119 // No origin bitmap, arbitrary within-group rank, or select is kept.
120 // Views check section shapes; a reader must validate borrowed metadata.
121 template <std::uint64_t K> struct rank_groups_view {
122 static_assert(K >= 3 && K < std::numeric_limits<std::uint64_t>::max() && std::has_single_bit(K + 1),
123 "group size must be 2^n-1 and at least three");
124 static constexpr std::uint64_t group_size = K;
125 static constexpr unsigned class_bits = unsigned(std::countr_zero(K + 1));
126
127 rank_groups_view(std::span<std::uint64_t const> classes,
128 std::span<std::uint64_t const> checkpoints,
129 std::uint64_t virtual_count)
130 : rank_groups_view(word_view(classes), word_view(checkpoints), virtual_count) {}

132 template <class Words> requires std::is_same_v<Words, word_view>
133 rank_groups_view(Words classes, Words checkpoints,
134 std::uint64_t virtual_count)
135 : classes_(classes), checkpoints_(checkpoints), virtual_count_(virtual_count) {
136 auto groups = group_count();
137 if (groups > (std::numeric_limits<std::uint64_t>::max() - 63) / class_bits)
138 error_detail::raise<std::overflow_error>("rank groups packed size");
139 auto bits = groups * class_bits;
140 if (classes.size() != ((bits + 63) >> 6) ||
141 checkpoints.size() != ((groups + 127) >> 7))
142 error_detail::raise<std::invalid_argument>("invalid rank groups spans");
143 }

145 std::uint64_t size() const noexcept { return virtual_count_; }
146 // Derived from the final real group, without an endpoint entry.
147 std::uint64_t count() const {
148 if (!group_count()) return 0;
149 auto last = group_count() - 1;
150 auto population = class_at(last);
151 if (population > virtual_count_ - last * K)
152 error_detail::raise<std::invalid_argument>("invalid rank groups final population");
153 return add_prefix(rank(last), population, virtual_count_);
154 }
155 std::uint64_t group_count() const noexcept {
156 return virtual_count_ / K + (virtual_count_ % K != 0);
157 }

160 std::uint64_t class_at(std::uint64_t group) const {
161 if (group >= group_count()) error_detail::raise<std::out_of_range>("rank groups class");
162 return read_class(group);

164 // Exclusive prefix at K*group for an existing group only.
165 std::uint64_t rank(std::uint64_t group) const {
166 if (group >= group_count()) error_detail::raise<std::out_of_range>("rank groups boundary");
167 auto result = checkpoints_[group >> 7];
168 auto limit = virtual_count_;
169 if constexpr (K == 3 || K == 7 || K == 31) {
170 auto count = unsigned(group & 127);
171 if (!count) return add_prefix(result, 0, limit);
172 auto word = (group >> 7) * (2 * class_bits);
173#if defined(__aarch64__) && defined(__ARM_NEON) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
174 // Two-bit broadword sums have lower dependent query latency. Wider
175 // classes favor NEON for the rank-plus-class projection operation.
176 if constexpr (K != 3)
177 if (classes_.size() - word >= 2 * class_bits)
178 return add_prefix(result, vaddvq_u16(rank_groups_detail::prefix_vectors<class_bits>(
179 classes_.bytes().data() + word * 8, count * class_bits)), limit);
180#endif
181 return add_prefix(result, rank_groups_detail::prefix_portable<class_bits>(classes_.subspan(word), count), limit);
182 }
183 for (auto i = group & ~std::uint64_t{127}; i < group; ++i) result = add_prefix(result, read_class(i), limit);
184 return add_prefix(result, 0, limit);
185 }

193 std::uint64_t read_class(std::uint64_t group) const noexcept {
194 auto bit = group * class_bits;
195 auto word = bit >> 6;
196 unsigned shift = unsigned(bit & 63);
197 auto value = classes_[word] >> shift;
198 if (shift + class_bits > 64) value |= classes_[word + 1] << (64 - shift);
199 return value & K;

233 template <std::uint64_t K> struct rank_groups {
234 static_assert(K >= 3 && K < std::numeric_limits<std::uint64_t>::max() && std::has_single_bit(K + 1),
235 "group size must be 2^n-1 and at least three");
236 static constexpr std::uint64_t group_size = K;
237 static constexpr unsigned class_bits = unsigned(std::countr_zero(K + 1));
238
239 static rank_groups build(std::span<std::uint64_t const> source, std::uint64_t count) {
240 auto groups = count / K + (count % K != 0);
241 if (source.size() != groups) error_detail::raise<std::invalid_argument>("rank groups class length");
242 if (groups > (std::numeric_limits<std::uint64_t>::max() - 63) / class_bits)
243 error_detail::raise<std::overflow_error>("rank groups packed size");
244 auto bits = groups * class_bits;
245 rank_groups result;
246 result.virtual_count = count;
247 result.classes.resize((bits + 63) >> 6);
248 std::uint64_t total = 0;
249 for (std::uint64_t i = 0; i < groups; ++i) {
250 auto limit = i + 1 == groups && count % K ? count % K : K;
251 auto value = source[i];
252 if (value > limit) error_detail::raise<std::invalid_argument>("rank groups population");
253 if ((i & 127) == 0) result.checkpoints.push_back(total);
254 auto bit = i * class_bits;
255 unsigned shift = unsigned(bit & 63);
256 result.classes[bit >> 6] |= value << shift;
257 if (shift + class_bits > 64) result.classes[(bit >> 6) + 1] |= value >> (64 - shift);
258 total += value;
259 }
260 return result;
261 }
