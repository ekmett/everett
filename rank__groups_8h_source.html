20 // Population classes for virtual groups of K=2^n-1 entries. A class occupies
21 // exactly n bits, including across word boundaries. One 64-bit checkpoint
22 // every 128 classes bounds rank queries to 127 class reads (two words each
23 // at most). No origin bitmap, arbitrary within-group rank, or select is kept.
24 // Views check section shapes; a reader must validate borrowed metadata.
25 template <std::uint64_t K> struct rank_groups_view {
26 static_assert(K >= 3 && K < std::numeric_limits<std::uint64_t>::max() && std::has_single_bit(K + 1),
27 "group size must be 2^n-1 and at least three");
28 static constexpr std::uint64_t group_size = K;
29 static constexpr unsigned class_bits = unsigned(std::countr_zero(K + 1));

54 std::uint64_t rank(std::uint64_t group) const {
55 if (group > group_count()) throw std::out_of_range("rank groups boundary");
56 if (group == group_count()) return total_;

63 std::uint64_t read_class(std::uint64_t group) const noexcept {
64 auto bit = group * class_bits;
65 auto word = bit / 64;
66 unsigned shift = unsigned(bit % 64);
67 auto value = classes_[word] >> shift;
68 if (shift + class_bits > 64) value |= classes_[word + 1] << (64 - shift);

83 static rank_groups build(std::span<std::uint64_t const> source, std::uint64_t count) {
84 auto groups = count / K + (count % K != 0);
85 if (source.size() != groups) throw std::invalid_argument("rank groups class length");
86 if (groups > std::numeric_limits<std::uint64_t>::max() / class_bits)
87 throw std::overflow_error("rank groups packed size");
88 auto bits = groups * class_bits;
89 rank_groups result;
90 result.virtual_count = count;
91 result.classes.resize(bits / 64 + (bits % 64 != 0));
92 for (std::uint64_t i = 0; i < groups; ++i) {
93 auto limit = i + 1 == groups && count % K ? count % K : K;
94 auto value = source[i];
95 if (value > limit) throw std::invalid_argument("rank groups population");
96 if (i % 128 == 0) result.checkpoints.push_back(result.total);
97 auto bit = i * class_bits;
98 unsigned shift = unsigned(bit % 64);
99 result.classes[bit / 64] |= value << shift;
100 if (shift + class_bits > 64) result.classes[bit / 64 + 1] |= value >> (64 - shift);
101 result.total += value;
102 }
103 return result;
104 }
