36 // The caller supplies at most 128 classes and exactly the words containing
37 // them. Weighted bit populations preserve fields split across word cuts.
38 template <unsigned Bits> inline unsigned prefix_portable(
39 word_view words, unsigned count) noexcept {
40 static_assert(Bits == 2 || Bits == 3 || Bits == 5);
41 constexpr auto masks = [] {
42 std::array<std::array<std::uint64_t, Bits>, Bits> result{};
43 for (unsigned word = 0; word < Bits; ++word)
44 for (unsigned bit = 0; bit < 64; ++bit)
45 result[word][(word * 64 + bit) % Bits] |= std::uint64_t{1} << bit;
46 return result;
47 }();
48 unsigned result = 0;
49 std::uint64_t pairs = 0;
50 auto bits = count * Bits;
51 for (unsigned word = 0; word * 64 < bits; ++word) {
52 auto value = words[word];
53 auto remaining = bits - word * 64;
54 if (remaining < 64) value &= (std::uint64_t{1} << remaining) - 1;
55 if constexpr (Bits == 2) {
56 value = (value & 0x3333333333333333ull) + ((value >> 2) & 0x3333333333333333ull);
57 pairs += (value & 0x0f0f0f0f0f0f0f0full) + ((value >> 4) & 0x0f0f0f0f0f0f0f0full);
58 } else {
59 for (unsigned plane = 0; plane < Bits; ++plane)
60 result += unsigned(std::popcount(value & masks[word % Bits][plane])) << plane;
61 }
62 }
63 if constexpr (Bits == 2) {
64 // Four two-bit classes per byte contribute at most twelve per word;
65 // four words fit in byte lanes. Widen before summing their total.
66 pairs = (pairs & 0x00ff00ff00ff00ffull) + ((pairs >> 8) & 0x00ff00ff00ff00ffull);
67 return unsigned((pairs * 0x0001000100010001ull) >> 48);
68 }
69 return result;
70 }

78 template <unsigned Bits, unsigned Vector, unsigned Plane = 0>
79 inline uint8x16_t weighted_bits(uint8x16_t value) noexcept {
80 if constexpr (Bits == 2) {
81 auto pairs = vaddq_u8(vandq_u8(value, vdupq_n_u8(0x33)),
82 vandq_u8(vshrq_n_u8(value, 2), vdupq_n_u8(0x33)));
83 return vaddq_u8(vandq_u8(pairs, vdupq_n_u8(15)), vshrq_n_u8(pairs, 4));
84 }
85 uint64x2_t mask{plane_mask<Bits, 2 * Vector, Plane>(),
86 plane_mask<Bits, 2 * Vector + 1, Plane>()};
87 auto population = vshlq_n_u8(vcntq_u8(vandq_u8(value, vreinterpretq_u8_u64(mask))), Plane);
88 if constexpr (Plane + 1 == Bits) return population;
89 else return vaddq_u8(population, weighted_bits<Bits, Vector, Plane + 1>(value));
90 }

92 template <unsigned Bits, unsigned Vector = 0>
93 inline uint16x8_t prefix_vectors(std::byte const * bytes, unsigned bits) noexcept {
94 uint64x2_t positions{2 * Vector, 2 * Vector + 1};
95 auto boundary = vdupq_n_u64(bits >> 6);
96 auto tail = vdupq_n_u64((std::uint64_t{1} << (bits & 63)) - 1);
97 auto mask = vorrq_u64(vcltq_u64(positions, boundary),
98 vandq_u64(vceqq_u64(positions, boundary), tail));
99 auto selected = vandq_u64(vreinterpretq_u64_u8(vld1q_u8(
100 reinterpret_cast<std::uint8_t const *>(bytes + 16 * Vector))), mask);
101 // Widen before adding vectors: five-bit classes can otherwise overflow
102 // an eight-bit lane. The complete checkpoint sums to at most 128*31.
103 auto counts = vpaddlq_u8(weighted_bits<Bits, Vector>(vreinterpretq_u8_u64(selected)));
104 if constexpr (Vector + 1 == Bits) return counts;
105 else return vaddq_u16(counts, prefix_vectors<Bits, Vector + 1>(bytes, bits));
106 }

117 // Population classes for virtual groups of K=2^n-1 entries. A class occupies
118 // exactly n bits, including across word boundaries. One 64-bit checkpoint
119 // every 128 classes bounds general rank queries to 127 class reads. K=3,7,31
120 // reduce packed words; K=15 shares rank15's packed-word/SIMD reduction.
121 // No origin bitmap, arbitrary within-group rank, or select is kept.
122 // Views check section shapes; a reader must validate borrowed metadata.
123 template <std::uint64_t K> struct rank_groups_view {
124 static_assert(K >= 3 && K < std::numeric_limits<std::uint64_t>::max() && std::has_single_bit(K + 1),
125 "group size must be 2^n-1 and at least three");
126 static constexpr std::uint64_t group_size = K;
127 static constexpr unsigned class_bits = unsigned(std::countr_zero(K + 1));

134 template <class Words> requires std::is_same_v<Words, word_view>
135 rank_groups_view(Words classes, Words checkpoints,
136 std::uint64_t virtual_count)
137 : classes_(classes), checkpoints_(checkpoints), virtual_count_(virtual_count) {
138 auto groups = group_count();
139 if (groups > (std::numeric_limits<std::uint64_t>::max() - 63) / class_bits)
140 error_detail::raise<std::overflow_error>("rank groups packed size");
141 auto bits = groups * class_bits;
142 if (classes.size() != ((bits + 63) >> 6) ||
143 checkpoints.size() != ((groups + 127) >> 7))
144 error_detail::raise<std::invalid_argument>("invalid rank groups spans");
145 }

148 // Derived from the final real group, without an endpoint entry.
149 std::uint64_t count() const {

166 // Exclusive prefix at K*group for an existing group only.
167 std::uint64_t rank(std::uint64_t group) const {
168 if (group >= group_count()) error_detail::raise<std::out_of_range>("rank groups boundary");
169 auto result = checkpoints_[group >> 7];
170 auto limit = virtual_count_;
171 if constexpr (K == 3 || K == 7 || K == 31) {
172 auto count = unsigned(group & 127);
173 if (!count) return add_prefix(result, 0, limit);
174 auto word = (group >> 7) * (2 * class_bits);
175#if defined(__aarch64__) && defined(__ARM_NEON) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
176 // Two-bit broadword sums have lower dependent query latency. Wider
177 // classes favor NEON for the rank-plus-class projection operation.
178 if constexpr (K != 3)
179 if (classes_.size() - word >= 2 * class_bits)
180 return add_prefix(result, vaddvq_u16(rank_groups_detail::prefix_vectors<class_bits>(
181 classes_.bytes().data() + word * 8, count * class_bits)), limit);
182#endif
183 return add_prefix(result, rank_groups_detail::prefix_portable<class_bits>(classes_.subspan(word), count), limit);
184 }
185 for (auto i = group & ~std::uint64_t{127}; i < group; ++i) result = add_prefix(result, read_class(i), limit);
186 return add_prefix(result, 0, limit);
187 }

200 if (shift + class_bits > 64) value |= classes_[word + 1] << (64 - shift);

235 template <std::uint64_t K> struct rank_groups {
236 static_assert(K >= 3 && K < std::numeric_limits<std::uint64_t>::max() && std::has_single_bit(K + 1),
237 "group size must be 2^n-1 and at least three");

241 static rank_groups build(std::span<std::uint64_t const> source, std::uint64_t count) {
242 auto groups = count / K + (count % K != 0);
243 if (source.size() != groups) error_detail::raise<std::invalid_argument>("rank groups class length");
244 if (groups > (std::numeric_limits<std::uint64_t>::max() - 63) / class_bits)
245 error_detail::raise<std::overflow_error>("rank groups packed size");
246 auto bits = groups * class_bits;
247 rank_groups result;
248 result.virtual_count = count;
249 result.classes.resize((bits + 63) >> 6);
250 std::uint64_t total = 0;
251 for (std::uint64_t i = 0; i < groups; ++i) {
252 auto limit = i + 1 == groups && count % K ? count % K : K;
253 auto value = source[i];
254 if (value > limit) error_detail::raise<std::invalid_argument>("rank groups population");
255 if ((i & 127) == 0) result.checkpoints.push_back(total);
256 auto bit = i * class_bits;
257 unsigned shift = unsigned(bit & 63);
258 result.classes[bit >> 6] |= value << shift;
259 if (shift + class_bits > 64) result.classes[(bit >> 6) + 1] |= value >> (64 - shift);
260 total += value;
261 }
262 return result;
263 }

299 void append(std::uint64_t population, std::uint64_t width = K) {
300 require_active();
301 constexpr auto maximum = std::numeric_limits<std::uint64_t>::max();
302 if (partial_ || !width || width > K || population > width)
303 error_detail::raise<std::invalid_argument>("invalid incremental rank group");
304 if (width > maximum - data_.virtual_count || groups_ >= (maximum - 63) / class_bits)
305 error_detail::raise<std::length_error>("incremental rank groups are too large");
306 auto bits = (groups_ + 1) * class_bits;
307 auto words = (bits + 63) >> 6;
308 auto checkpoint = (groups_ & 127) == 0;
309 reserve(data_.classes, words);
310 if (checkpoint) reserve(data_.checkpoints, (groups_ >> 7) + 1);
311 // All potentially failing growth precedes the first logical mutation.
312 data_.classes.resize(static_cast<std::size_t>(words), 0);
313 if (checkpoint) data_.checkpoints.push_back(population_);
314 auto bit = groups_ * class_bits;
315 auto shift = unsigned(bit & 63);
316 data_.classes[bit >> 6] |= population << shift;
317 if (shift + class_bits > 64) data_.classes[(bit >> 6) + 1] |= population >> (64 - shift);
318 data_.virtual_count += width;
319 population_ += population;
320 ++groups_;
321 partial_ = width != K;
322 }

326 finished_ = true;
327 return std::move(data_);
328 }

338 static void reserve(std::vector<std::uint64_t> & data, std::uint64_t count) {
339 if (count <= data.capacity()) return;
340 if (count > data.max_size()) error_detail::raise<std::length_error>("incremental rank groups are too large");
341 auto capacity = data.capacity();
342 auto grown = capacity > (data.max_size() >> 1) ? data.max_size() : capacity << 1;
343 data.reserve(std::max(static_cast<std::size_t>(count), grown));
344 }
