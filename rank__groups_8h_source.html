32 // The caller supplies at most 128 classes and exactly the words containing
33 // them. Weighted bit populations preserve fields split across word cuts.
34 template <unsigned Bits> inline unsigned prefix_portable(
35 std::uint64_t const * words, unsigned count) noexcept {
36 static_assert(Bits == 2 || Bits == 3 || Bits == 5);
37 constexpr auto masks = [] {
38 std::array<std::array<std::uint64_t, Bits>, Bits> result{};
39 for (unsigned word = 0; word < Bits; ++word)
40 for (unsigned bit = 0; bit < 64; ++bit)
41 result[word][(word * 64 + bit) % Bits] |= std::uint64_t{1} << bit;
42 return result;
43 }();
44 unsigned result = 0;
45 std::uint64_t pairs = 0;
46 auto bits = count * Bits;
47 for (unsigned word = 0; word * 64 < bits; ++word) {
48 auto value = words[word];
49 auto remaining = bits - word * 64;
50 if (remaining < 64) value &= (std::uint64_t{1} << remaining) - 1;
51 if constexpr (Bits == 2) {
52 value = (value & 0x3333333333333333ull) + ((value >> 2) & 0x3333333333333333ull);
53 pairs += (value & 0x0f0f0f0f0f0f0f0full) + ((value >> 4) & 0x0f0f0f0f0f0f0f0full);
54 } else {
55 for (unsigned plane = 0; plane < Bits; ++plane)
56 result += unsigned(std::popcount(value & masks[word % Bits][plane])) << plane;
57 }
58 }
59 if constexpr (Bits == 2) {
60 // Four two-bit classes per byte contribute at most twelve per word;
61 // four words fit in byte lanes. Widen before summing their total.
62 pairs = (pairs & 0x00ff00ff00ff00ffull) + ((pairs >> 8) & 0x00ff00ff00ff00ffull);
63 return unsigned((pairs * 0x0001000100010001ull) >> 48);
64 }
65 return result;
66 }

69 template <unsigned Bits, unsigned Vector, unsigned Plane = 0>
70 inline uint8x16_t weighted_bits(uint8x16_t value) noexcept {
71 if constexpr (Bits == 2) {
72 auto pairs = vaddq_u8(vandq_u8(value, vdupq_n_u8(0x33)),
73 vandq_u8(vshrq_n_u8(value, 2), vdupq_n_u8(0x33)));
74 return vaddq_u8(vandq_u8(pairs, vdupq_n_u8(15)), vshrq_n_u8(pairs, 4));
75 }
76 uint64x2_t mask{plane_mask<Bits, 2 * Vector, Plane>(),
77 plane_mask<Bits, 2 * Vector + 1, Plane>()};
78 auto population = vshlq_n_u8(vcntq_u8(vandq_u8(value, vreinterpretq_u8_u64(mask))), Plane);
79 if constexpr (Plane + 1 == Bits) return population;
80 else return vaddq_u8(population, weighted_bits<Bits, Vector, Plane + 1>(value));
81 }

83 template <unsigned Bits, unsigned Vector = 0>
84 inline uint16x8_t prefix_vectors(std::uint64_t const * words, unsigned bits) noexcept {
85 uint64x2_t positions{2 * Vector, 2 * Vector + 1};
86 auto boundary = vdupq_n_u64(bits / 64);
87 auto tail = vdupq_n_u64((std::uint64_t{1} << (bits % 64)) - 1);
88 auto mask = vorrq_u64(vcltq_u64(positions, boundary),
89 vandq_u64(vceqq_u64(positions, boundary), tail));
90 auto selected = vandq_u64(vld1q_u64(words + 2 * Vector), mask);
91 // Widen before adding vectors: five-bit classes can otherwise overflow
92 // an eight-bit lane. The complete checkpoint sums to at most 128*31.
93 auto counts = vpaddlq_u8(weighted_bits<Bits, Vector>(vreinterpretq_u8_u64(selected)));
94 if constexpr (Vector + 1 == Bits) return counts;
95 else return vaddq_u16(counts, prefix_vectors<Bits, Vector + 1>(words, bits));
96 }

107 // Population classes for virtual groups of K=2^n-1 entries. A class occupies
108 // exactly n bits, including across word boundaries. One 64-bit checkpoint
109 // every 128 classes bounds general rank queries to 127 class reads. K=3,7,31
110 // reduce packed words; K=15 shares rank15's packed-word/SIMD reduction.
111 // No origin bitmap, arbitrary within-group rank, or select is kept.
112 // Views check section shapes; a reader must validate borrowed metadata.
113 template <std::uint64_t K> struct rank_groups_view {
114 static_assert(K >= 3 && K < std::numeric_limits<std::uint64_t>::max() && std::has_single_bit(K + 1),
115 "group size must be 2^n-1 and at least three");
116 static constexpr std::uint64_t group_size = K;
117 static constexpr unsigned class_bits = unsigned(std::countr_zero(K + 1));
118
119 rank_groups_view(std::span<std::uint64_t const> classes,
120 std::span<std::uint64_t const> checkpoints,
121 std::uint64_t virtual_count, std::uint64_t total)
122 : classes_(classes), checkpoints_(checkpoints), virtual_count_(virtual_count), total_(total) {
123 auto groups = group_count();
124 if (groups > std::numeric_limits<std::uint64_t>::max() / class_bits)
125 throw std::overflow_error("rank groups packed size");
126 auto bits = groups * class_bits;
127 if (classes.size() != bits / 64 + (bits % 64 != 0) ||
128 checkpoints.size() != groups / 128 + (groups % 128 != 0) || total > virtual_count)
129 throw std::invalid_argument("invalid rank groups spans");
130 }
131
132 std::uint64_t size() const noexcept { return virtual_count_; }
133 std::uint64_t count() const noexcept { return total_; }
134 std::uint64_t group_count() const noexcept {
135 return virtual_count_ / K + (virtual_count_ % K != 0);
136 }
137 std::uint64_t class_at(std::uint64_t group) const {
138 if (group >= group_count()) throw std::out_of_range("rank groups class");
139 return read_class(group);
140 }
141 // Exclusive prefix at K*group, with the final endpoint clamped to actual N.
142 std::uint64_t rank(std::uint64_t group) const {
143 if (group > group_count()) throw std::out_of_range("rank groups boundary");
144 if (group == group_count()) return total_;
145 auto result = checkpoints_[group / 128];
146 if constexpr (K == 3 || K == 7 || K == 31) {
147 auto count = unsigned(group % 128);
148 if (!count) return result;
149 auto word = (group / 128) * (2 * class_bits);
150#if defined(__aarch64__) && defined(__ARM_NEON) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
151 // Two-bit broadword sums have lower dependent query latency. Wider
152 // classes favor NEON for the rank-plus-class projection operation.
153 if constexpr (K != 3)
154 if (classes_.size() - word >= 2 * class_bits)
155 return result + rank_groups_detail::prefix_neon<class_bits>(classes_.data() + word, count);
156#endif
157 return result + rank_groups_detail::prefix_portable<class_bits>(classes_.data() + word, count);
158 }
159 for (auto i = (group / 128) * 128; i < group; ++i) result += read_class(i);
160 return result;
161 }

164 std::uint64_t read_class(std::uint64_t group) const noexcept {
165 auto bit = group * class_bits;
166 auto word = bit / 64;
167 unsigned shift = unsigned(bit % 64);
168 auto value = classes_[word] >> shift;
169 if (shift + class_bits > 64) value |= classes_[word + 1] << (64 - shift);
170 return value & K;
171 }

197 template <std::uint64_t K> struct rank_groups {
198 static_assert(K >= 3 && K < std::numeric_limits<std::uint64_t>::max() && std::has_single_bit(K + 1),
199 "group size must be 2^n-1 and at least three");
200 static constexpr std::uint64_t group_size = K;
201 static constexpr unsigned class_bits = unsigned(std::countr_zero(K + 1));
202
203 static rank_groups build(std::span<std::uint64_t const> source, std::uint64_t count) {
204 auto groups = count / K + (count % K != 0);
205 if (source.size() != groups) throw std::invalid_argument("rank groups class length");
206 if (groups > std::numeric_limits<std::uint64_t>::max() / class_bits)
207 throw std::overflow_error("rank groups packed size");
208 auto bits = groups * class_bits;
209 rank_groups result;
210 result.virtual_count = count;
211 result.classes.resize(bits / 64 + (bits % 64 != 0));
212 for (std::uint64_t i = 0; i < groups; ++i) {
213 auto limit = i + 1 == groups && count % K ? count % K : K;
214 auto value = source[i];
215 if (value > limit) throw std::invalid_argument("rank groups population");
216 if (i % 128 == 0) result.checkpoints.push_back(result.total);
217 auto bit = i * class_bits;
218 unsigned shift = unsigned(bit % 64);
219 result.classes[bit / 64] |= value << shift;
220 if (shift + class_bits > 64) result.classes[bit / 64 + 1] |= value >> (64 - shift);
221 result.total += value;
222 }
223 return result;
224 }
