86 // Executable protocol model only: no files, flushing, checksumming, durable
87 // pin implementation, or thread synchronization occurs here. A backend
88 // owns those operations and supplies truthful completion/recovery events.
89 // One serialized publication owner uses the state machine for one recipe.

91 merge_publication(merge_identity identity, std::string old_manifest,
92 std::string output_generation)
93 : identity_(std::move(identity)), durable_manifest_(std::move(old_manifest)),
94 generation_(std::move(output_generation)) {
95 if (identity_.recipe.empty() || identity_.input_versions.empty() ||
96 durable_manifest_.empty() || generation_.empty())
97 throw std::invalid_argument("missing merge publication identity");
98 for (std::size_t i = 0; i < identity_.input_versions.size(); ++i) {
99 auto const & id = identity_.input_versions[i];
100 if (id.empty()) throw std::invalid_argument("missing merge input identity");
101 for (std::size_t j = 0; j < i; ++j)
102 if (identity_.input_versions[j] == id)
103 throw std::invalid_argument("duplicate merge input identity");
104 }
105 generations_.push_back(generation_);
106 }

116 // A failed checkpoint write never supersedes the previous durable one.
117 void complete_checkpoint(merge_checkpoint candidate, persistence_result result) {
118 require_stage(publication_stage::building);
119 validate_checkpoint(candidate);
120 if (candidate.output_generation != generation_)
121 throw std::invalid_argument("checkpoint generation mismatch");
122 if (checkpoint_) {
123 if (candidate.exact_id == checkpoint_->exact_id ||
124 candidate.output_records < checkpoint_->output_records ||
125 candidate.completed_work < checkpoint_->completed_work)
126 throw std::invalid_argument("checkpoint did not advance");
127 for (std::size_t i = 0; i < candidate.inputs.size(); ++i)
128 if (candidate.inputs[i].ordinal < checkpoint_->inputs[i].ordinal ||
129 candidate.inputs[i].byte_offset < checkpoint_->inputs[i].byte_offset)
130 throw std::invalid_argument("checkpoint cursor moved backwards");
131 }
132 if (result == persistence_result::failure) { stage_ = publication_stage::uncertain; return; }
133 checkpoint_ = std::move(candidate);
134 }

136 void complete_output(persistence_result result) {
137 require_stage(publication_stage::building);

139 publication_stage::output_durable : publication_stage::uncertain;

142 void prepare_manifest(std::string exact_id) {
143 require_stage(publication_stage::output_durable);

145 std::find(manifests_.begin(), manifests_.end(), exact_id) != manifests_.end())

159 // Includes write, flush, verification, directory or allocator failures
160 // before publication. A transient-looking error does not relax retention.
161 void report_io_failure() {
162 if (stage_ == publication_stage::published)
163 throw std::logic_error("published merge requires separate cleanup handling");
164 stage_ = publication_stage::uncertain;

167 // Fresh output storage is mandatory. Optional retained checkpoint must be
168 // the exact last successful checkpoint and must have been independently
169 // revalidated, together with its inputs and every sealed extent, by the
170 // backend. Without it, regenerate from the original pinned inputs. If a
171 // manifest was attempted, old-root selection must additionally be made
172 // durable and all candidate reachability reconciled before restarting.
173 void resume(std::string fresh_generation, recovery_evidence evidence,
174 std::optional<merge_checkpoint> retained = std::nullopt,
175 selector_recovery selector = selector_recovery::unresolved) {
176 require_stage(publication_stage::uncertain);
177 require_evidence(evidence);
178 if (!pending_manifest_.empty() && selector != selector_recovery::old_root_selected)
179 throw std::invalid_argument("uncertain manifest selector needs reconciliation");
180 if (fresh_generation.empty() ||
181 std::find(generations_.begin(), generations_.end(), fresh_generation) != generations_.end())
182 throw std::invalid_argument("merge resumption needs a fresh generation");
183 if (retained && (!checkpoint_ || *retained != *checkpoint_))
184 throw std::invalid_argument("unrecognized recovery checkpoint");
185 generations_.push_back(fresh_generation);
186 generation_ = std::move(fresh_generation);
187 checkpoint_ = std::move(retained);
188 pending_manifest_.clear();
189 stage_ = publication_stage::building;
190 }

192 // A failed manifest flush has an uncertain outcome, not a guaranteed
193 // rollback. Accept it only when recovery independently verifies the
194 // candidate root AND its complete immutable dependency graph as durable.
195 void accept_recovered_publication(recovery_evidence evidence) {
196 require_stage(publication_stage::uncertain);
197 require_evidence(evidence);
198 if (pending_manifest_.empty()) throw std::logic_error("no manifest to reconcile");
199 durable_manifest_ = pending_manifest_;
200 stage_ = publication_stage::published;
201 }

219 void validate_checkpoint(merge_checkpoint const & candidate) const {
220 if (candidate.exact_id.empty() || candidate.identity != identity_ ||
221 candidate.inputs.size() != identity_.input_versions.size())
222 throw std::invalid_argument("checkpoint identity mismatch");
223 for (std::size_t i = 0; i < candidate.inputs.size(); ++i)
224 if (candidate.inputs[i].input_version != identity_.input_versions[i])
225 throw std::invalid_argument("checkpoint input version mismatch");
226 for (std::size_t i = 0; i < candidate.sealed_output.size(); ++i) {
227 auto const & extent = candidate.sealed_output[i];
228 if (extent.exact_id.empty() || extent.checksum.empty())
229 throw std::invalid_argument("checkpoint extent identity missing");
230 for (std::size_t j = 0; j < i; ++j)
231 if (candidate.sealed_output[j].exact_id == extent.exact_id)
232 throw std::invalid_argument("duplicate checkpoint extent");
233 }
234 (void)candidate.sealed_bytes();
