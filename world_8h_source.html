66 inline void write_u64(std::ostream & out, std::uint64_t value) {
67 std::array<char, 8> bytes;
68 for (unsigned i = 0; i < 8; ++i) bytes[i] = char((value >> (8 * i)) & 255);
69 out.write(bytes.data(), bytes.size());
70 if (!out) throw std::runtime_error("world export write failed");
71 }
72
73 inline std::uint64_t read_u64(std::istream & in) {
74 std::array<unsigned char, 8> bytes;
75 in.read(reinterpret_cast<char *>(bytes.data()), bytes.size());
76 if (!in) throw std::runtime_error("truncated world export");
77 std::uint64_t value = 0;
78 for (unsigned i = 0; i < 8; ++i) value |= std::uint64_t(bytes[i]) << (8 * i);
79 return value;
80 }

100 // Semantic reference implementation. A snapshot pins immutable sorted runs,
101 // and updates copy only the pin vector, never earlier record arrays. This has
102 // no redundant COLA scheduling or front-coded disk layout yet. It is an oracle
103 // for that implementation, not a claim of its asymptotic space/time bounds.
104 template <class V = std::uint64_t, class A = wrapping_fingerprint_algebra,
105 class H = u64_table_hash>
106 struct reference_world {
107 using value_type = V;
108 using element = typename A::element;
109 using record = world_record<V>;
110 using run = world_run<V>;
111 using pin = std::shared_ptr<run const>;
112 using pin_owner = pin_set<run, A>;
113
114 explicit reference_world(H hash = {})
115 : state_(std::make_shared<state>()), hash_(std::move(hash)) {}
116
117 static reference_world from_records(std::vector<record> records, H hash = {}) {
118 std::sort(records.begin(), records.end(),
119 [](record const & a, record const & b) { return a.key < b.key; });
120 auto result = reference_world(std::move(hash));
121 auto next = std::make_shared<state>();
122 auto signature = A::zero();
123 for (std::size_t i = 0; i < records.size(); ++i) {
124 if (!records[i].value) throw std::invalid_argument("initial record is a tombstone");
125 if (i && records[i - 1].key == records[i].key)
126 throw std::invalid_argument("duplicate initial world key");
127 signature = A::add(signature,
128 table_fingerprint<A>::binding(records[i].key, records[i].value, result.hash_));
129 }
130 next->live_size = records.size();
131 if (!records.empty()) {
132 auto object = std::make_shared<run>(std::move(records));
133 next->owner = next->owner.add({object->object_id(), object, signature, signature});
134 }
135 result.state_ = std::move(next);
136 return result;
137 }
138
139 std::optional<V> get(std::string_view key) const {
140 auto objects = state_->owner.pins();
141 for (auto it = objects.rbegin(); it != objects.rend(); ++it) {
142 auto entries = (*it)->records();
143 auto found = std::lower_bound(entries.begin(), entries.end(), key,
144 [](record const & entry, std::string_view k) { return entry.key < k; });
145 if (found != entries.end() && found->key == key) return found->value;
146 }
147 return std::nullopt;
148 }
149
150 std::size_t live_size() const noexcept { return state_->live_size; }
151 element signature() const { return state_->owner.signature(); }
152 std::span<pin const> pins() const noexcept { return state_->owner.pins(); }
153 pin_owner const & owner() const noexcept { return state_->owner; }
154
155 // A save/fork is just another owner of this exact immutable pin set.
156 reference_world snapshot() const { return *this; }
157
158 std::vector<record> resolved() const {
159 std::map<std::string, std::optional<V>, std::less<>> table;
160 for (auto const & run_pin : state_->owner.pins())
161 for (auto const & entry : run_pin->records()) table[entry.key] = entry.value;
162 std::vector<record> result;
163 result.reserve(state_->live_size);
164 for (auto const & [key, value] : table)
165 if (value) result.push_back({key, value});
166 return result;
167 }
168
169 element recompute_signature() const {
170 auto result = A::zero();
171 for (auto const & entry : resolved())
172 result = A::add(result, table_fingerprint<A>::binding(entry.key, entry.value, hash_));
173 return result;
174 }
175
176 // Eager reference compaction. Old snapshots retain their original pins.
177 reference_world compact() const {
178 auto result = from_records(resolved(), hash_);
179 std::vector<std::string> inputs;
180 for (auto const & item : state_->owner.entries()) inputs.push_back(item.object_id);
181 auto replacements = result.owner().entries();
182 auto next = std::make_shared<state>();
183 next->owner = state_->owner.replace(inputs,
184 std::vector<typename pin_owner::entry>(replacements.begin(), replacements.end()));
185 next->live_size = state_->live_size;
186 result.state_ = std::move(next);
187 return result;
188 }

190 // Portable resolved-table export for initial save/restore tests. This is
191 // deliberately distinct from the planned small manifest of pinned blobs.
192 // Callers own atomic file replacement, durability and codec/hash agreement.
193 template <class C = u64_world_codec>
194 void save(std::ostream & out, C codec = {}) const {
195 constexpr std::string_view magic = "EVRTREF1";
196 out.write(magic.data(), magic.size());
197 world_detail::write_u64(out, C::format_tag);
198 world_detail::write_u64(out, live_size());
199 for (auto const & entry : resolved()) {
200 world_detail::write_u64(out, entry.key.size());
201 out.write(entry.key.data(), static_cast<std::streamsize>(entry.key.size()));
202 codec.write(out, *entry.value);
203 }
204 if (!out) throw std::runtime_error("world export write failed");
205 }

207 template <class C = u64_world_codec>
208 static reference_world restore(std::istream & in, C codec = {}, H hash = {},
209 world_import_limits limits = {}) {
210 std::array<char, 8> magic;
211 in.read(magic.data(), magic.size());
212 if (!in || std::string_view(magic.data(), magic.size()) != "EVRTREF1")
213 throw std::runtime_error("invalid world export header");
214 if (world_detail::read_u64(in) != C::format_tag)
215 throw std::runtime_error("world export value codec mismatch");
216 auto count = world_detail::read_u64(in);
217 if (count > limits.max_records || count > std::numeric_limits<std::size_t>::max())
218 throw std::runtime_error("world export record limit");
219 std::vector<record> entries;
220 entries.reserve(static_cast<std::size_t>(count));
221 std::uint64_t remaining = limits.max_key_bytes;
222 for (std::uint64_t i = 0; i < count; ++i) {
223 auto size = world_detail::read_u64(in);
224 if (size > remaining || size > std::numeric_limits<std::size_t>::max() ||
225 size > std::uint64_t(std::numeric_limits<std::streamsize>::max()))
226 throw std::runtime_error("world export key byte limit");
227 remaining -= size;
228 std::string key(static_cast<std::size_t>(size), '\0');
229 in.read(key.data(), static_cast<std::streamsize>(size));
230 if (!in) throw std::runtime_error("truncated world export key");
231 if (!entries.empty() && !(entries.back().key < key))
232 throw std::runtime_error("world export keys are not strictly sorted");
233 entries.push_back({std::move(key), codec.read(in)});
234 }
235 return from_records(std::move(entries), std::move(hash));
236 }

244 reference_world append(std::vector<world_edit<V>> const & edits) const {
245 auto next = std::make_shared<state>(*state_);
246 std::vector<record> records;
247 records.reserve(edits.size());
248 auto contribution = A::zero();
249 auto native_signature = A::zero();
250 for (auto const & edit : edits) {
251 contribution = A::add(contribution,
252 table_fingerprint<A>::delta(edit.key, edit.before, edit.after, hash_));
253 native_signature = A::add(native_signature,
254 table_fingerprint<A>::binding(edit.key, edit.after, hash_));
255 if (!edit.before && edit.after) ++next->live_size;
256 if (edit.before && !edit.after) --next->live_size;
257 records.push_back({edit.key, edit.after});
258 }
259 if (!records.empty()) {
260 auto object = std::make_shared<run>(std::move(records));
261 next->owner = next->owner.add({object->object_id(), object, contribution, native_signature});
262 }
263 reference_world result(*this);
264 result.state_ = std::move(next);
265 return result;
266 }

289 // P is a pure, deterministic function string_view -> uint64_t. Partitions
290 // need not be key ranges. Builders may read the same immutable base in
291 // parallel; a receiver serializes apply() calls on its own round accumulator.
292 template <class V, class A, class H, class P> struct partition_round {
293 using world = reference_world<V, A, H>;
294 using batch = world_batch<V, A>;

296 partition_round(world base, std::string round_id, P partition)
297 : base_(std::move(base)), current_(base_), round_id_(std::move(round_id)),
298 partition_(std::move(partition)) {
299 if (round_id_.empty()) throw std::invalid_argument("empty world round identity");
300 }
301
302 batch make_batch(std::string batch_id, std::uint64_t owner,
303 std::vector<world_record<V>> writes) const {
304 if (batch_id.empty()) throw std::invalid_argument("empty world batch identity");
305 std::sort(writes.begin(), writes.end(),
306 [](auto const & a, auto const & b) { return a.key < b.key; });
307 batch result{round_id_, std::move(batch_id), owner, base_.signature(), A::zero(), {}};
308 for (std::size_t i = 0; i < writes.size(); ++i) {
309 auto const & entry = writes[i];
310 if (i && writes[i - 1].key == entry.key)
311 throw std::invalid_argument("duplicate world write key");
312 if (partition_(std::string_view(entry.key)) != owner)
313 throw std::invalid_argument("world write outside assigned partition");
314 auto old = base_.get(entry.key);
315 if (!old && !entry.value) throw std::invalid_argument("deleting absent world key");
316 result.delta = A::add(result.delta,
317 table_fingerprint<A>::delta(entry.key, old, entry.value, base_.hash_));
318 result.edits.push_back({entry.key, std::move(old), entry.value});
319 }
320 return result;
321 }
322
323 world_apply_result apply(batch const & update) {
324 if (update.round_id != round_id_ || update.base_signature != base_.signature())
325 throw std::invalid_argument("world batch does not name this round base");
326 if (update.batch_id.empty()) throw std::invalid_argument("empty world batch identity");
327 auto prior = accepted_.find(update.batch_id);
328 if (prior != accepted_.end()) {
329 if (prior->second != update) throw std::invalid_argument("world batch identity reused");
330 return world_apply_result::replay;
331 }
332 auto delta = A::zero();
333 for (std::size_t i = 0; i < update.edits.size(); ++i) {
334 auto const & edit = update.edits[i];
335 if (i && !(update.edits[i - 1].key < edit.key))
336 throw std::invalid_argument("world batch keys must be strictly sorted");
337 if (partition_(std::string_view(edit.key)) != update.owner)
338 throw std::invalid_argument("world edit outside assigned partition");
339 if (claimed_.contains(edit.key)) throw std::invalid_argument("overlapping world updates");
340 if (base_.get(edit.key) != edit.before)
341 throw std::invalid_argument("world edit old value mismatch");
342 if (!edit.before && !edit.after)
343 throw std::invalid_argument("deleting absent world key");
344 delta = A::add(delta,
345 table_fingerprint<A>::delta(edit.key, edit.before, edit.after, base_.hash_));
346 }
347 if (delta != update.delta)
348 throw std::invalid_argument("world batch contribution mismatch");
349
350 // Stage all allocations before publication. A rejected or failed batch
351 // leaves the current snapshot and its replay/overlap bookkeeping intact.
352 auto next = current_.append(update.edits);
353 auto accepted = accepted_;
354 auto claimed = claimed_;
355 accepted.emplace(update.batch_id, update);
356 for (auto const & edit : update.edits) claimed.emplace(edit.key, update.batch_id);
357 accepted_.swap(accepted);
358 claimed_.swap(claimed);
359 current_ = std::move(next);
360 return world_apply_result::applied;
361 }

363 world const & base() const noexcept { return base_; }
364 world snapshot() const { return current_; }
365 std::size_t accepted_batches() const noexcept { return accepted_.size(); }
