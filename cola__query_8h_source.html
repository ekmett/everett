43 static cola_query_root build(pair_type main = {}, typename Blob::native_pointer secondary = {})
44 requires std::is_same_v<Blob, cola_index<P>> {
45 return adopt_prepared(Blob::prepare_root(std::move(main), std::move(secondary)));

47 static cola_query_root adopt_prepared(pair_type source) {
48 if (!source || source->virtual_size() > P::group_size)
49 error_detail::raise<std::invalid_argument>("COLA query root needs one bounded head group");
50 std::unordered_set<Blob const *> seen;
51 for (auto current = source; current; current = current->main_target()) {
52 if (!seen.insert(current.get()).second) error_detail::raise<std::invalid_argument>("cyclic COLA main chain");
53 auto view = current->view();
54 auto main = current->main_target();
55 auto secondary = current->secondary_target();
56 auto secondary_count = secondary ? secondary->view().size() : 0;
57 if (view.borrowed(0).size() != (main ? main->group_count() : 0) ||
58 view.borrowed(1).size() != secondary_count / P::group_size + (secondary_count % P::group_size != 0))
59 error_detail::raise<std::invalid_argument>("COLA query target sample count mismatch");
60 }
61 return cola_query_root(std::move(source));
62 }
63 pair_type head() const noexcept { return head_; }
64 cola_query_cursor<P, Blob> cursor(bit_view query) const { return cola_query_cursor<P, Blob>(*this, query); }
65 cola_query_cursor<P, Blob> cursor_owned(bit_string query) const {
66 return cola_query_cursor<P, Blob>::from_owned(*this, std::move(query));
67 }
68 private:
69 explicit cola_query_root(pair_type head) : head_(std::move(head)) {}
70 pair_type head_;
71 };

107 // Synchronous replacement reads decode under the current node's owner.
108 // The callback finishes before that pin is released; public matches stay owned.
109 template <class P, class Blob, class Query, class Decode>
110 requires std::same_as<std::remove_cvref_t<Query>, bit_string> &&
111 (!std::is_reference_v<std::invoke_result_t<Decode &, bit_view>>) &&
112 requires(Blob const & blob, profile_query_context<P> const & context) {
113 query_access::search(blob.view(), 0, context, query_access::match_probe{});
114 }
115 auto first_value(cola_query_root<P, Blob> const & root, Query && query, Decode && decode)
116 -> std::optional<std::invoke_result_t<Decode &, bit_view>> {
117 using value_type = std::invoke_result_t<Decode &, bit_view>;
118 auto current = root.head();
119 // Built-in views release all context copies before this synchronous
120 // call returns, including on decoder exceptions and nested reads. A
121 // custom view may retain a copy, so copy borrowed queries into shared
122 // ownership and move rvalue queries.
123 auto context = [&] {
124 if constexpr (query_access::scoped_views<Blob>) return query_access::borrow_query<P>(query);
125 else return profile_query_context<P>::from_owned(std::forward<Query>(query));
126 }();
127 if (!current) error_detail::raise<std::invalid_argument>("COLA query root has no head");
128 if (!current->virtual_size()) return std::nullopt;
129 std::uint64_t group = 0;
130 while (current) {
131 std::optional<value_type> value;
132 auto result = query_access::search(current->view(), group, context,
133 [&](std::uint64_t, bit_view encoded) {
134 value.emplace(std::invoke(decode, encoded));
135 return true;
136 });
137 auto main = current->main_target();
138 auto secondary = current->secondary_target();
139 if (auto const & next = result.predecessors[0])
140 if (!main || next->target_ordinal % P::group_size || next->target_ordinal >= main->virtual_size())
141 error_detail::raise<std::invalid_argument>("COLA query main route has no target");
142 if (auto const & next = result.predecessors[1]) {
143 if (!secondary) error_detail::raise<std::invalid_argument>("COLA query secondary route has no target");
144 visit_secondary<P>(secondary->view(), *next, [&](std::uint64_t, bit_view encoded) {
145 if (!value) value.emplace(std::invoke(decode, encoded));
146 });
147 }
148 if (value) return value;
149 if (auto & next = result.predecessors[0]) {
150 group = next->target_ordinal / P::group_size;
151 context = std::move(next->comparison);
152 current = std::move(main);
153 } else current.reset();
154 }
155 return std::nullopt;
156 }
157 }

159 // Each charged main-node visit searches at most one augmented K-window and
160 // one terminal native K-window. Two equal native matches may result: neither

183 bool has_match() const noexcept { return pending_[0].has_value() || pending_[1].has_value(); }
184 bool done() const noexcept { return !current_ && !has_match(); }
185 bool failed() const noexcept { return failed_; }
186 std::uint64_t step(std::uint64_t main_budget = 1) {
187 if (failed_) error_detail::raise<std::logic_error>("COLA query cursor has failed");
188 if (has_match()) return 0;
189 std::uint64_t visited = 0;
190 try {
191 while (current_ && visited < main_budget) {
192 auto result = current_->view().search_window(group_, context_);
193 auto main = current_->main_target();
194 auto secondary = current_->secondary_target();
195 if (auto const & next = result.predecessors[0])
196 if (!main || next->target_ordinal % P::group_size || next->target_ordinal >= main->virtual_size())
197 error_detail::raise<std::invalid_argument>("COLA query main route has no target");
198 std::optional<profile_blob_native_match<P>> side;
199 if (auto const & next = result.predecessors[1]) {
200 if (!secondary) error_detail::raise<std::invalid_argument>("COLA query secondary route has no target");
201 side = cola_search_secondary<P>(secondary->view(), *next);
202 }
203 if (result.native) pending_[0].emplace(match_type{current_, false,
204 result.native->ordinal, std::move(result.native->value)});
205 if (side) pending_[1].emplace(match_type{current_, true, side->ordinal, std::move(side->value)});
206 if (auto & next = result.predecessors[0]) {
207 group_ = next->target_ordinal / P::group_size;
208 context_ = std::move(next->comparison);
209 current_ = std::move(main);
210 } else current_.reset();
211 ++visited;
212 if (has_match()) break;
213 }
214 } catch (...) { failed_ = true; throw; }
215 return visited;
216 }
217 match_type take_match() {
218 if (failed_) error_detail::raise<std::logic_error>("COLA query cursor has failed");
219 auto & slot = pending_[pending_[0] ? 0 : 1];
220 if (!slot) error_detail::raise<std::logic_error>("COLA query cursor has no match");
221 auto result = std::move(*slot);
222 slot.reset();
223 return result;
224 }
