27 // Keys are byte strings ordered lexicographically by unsigned byte value.
28 inline int compare_keys(std::string_view a, std::string_view b) noexcept {
29 auto n = std::min(a.size(), b.size());
30 for (std::size_t i = 0; i != n; ++i) {
31 auto x = static_cast<unsigned char>(a[i]);
32 auto y = static_cast<unsigned char>(b[i]);
33 if (x != y) return x < y ? -1 : 1;
34 }
35 return a.size() < b.size() ? -1 : a.size() > b.size() ? 1 : 0;
36 }

38 inline std::size_t common_prefix(std::string_view a, std::string_view b) noexcept {
39 std::size_t n = 0;
40 while (n < a.size() && n < b.size() && a[n] == b[n]) ++n;
41 return n;
42 }

54 inline int compare_prefix(front_anchor key, std::string_view query) {
55 auto needed = std::min<std::uint64_t>(key.full_size, query.size());
56 if (key.prefix.size() < needed) {
57 throw std::invalid_argument("front-coded comparison has an incomplete query prefix");
58 }
59 auto prefix_order = compare_keys(key.prefix.substr(0, static_cast<std::size_t>(needed)),
60 query.substr(0, static_cast<std::size_t>(needed)));
61 if (prefix_order) return prefix_order;
62 return key.full_size < query.size() ? -1 : key.full_size > query.size() ? 1 : 0;
63 }

66 inline void append_varint(std::vector<std::byte> & bytes, std::uint64_t value) {
67 while (value >= 128) {
68 bytes.push_back(static_cast<std::byte>((value & 127) | 128));
69 value >>= 7;
70 }
71 bytes.push_back(static_cast<std::byte>(value));
72 }

74 inline std::uint64_t read_varint(std::span<std::byte const> bytes, std::size_t & at) {
75 std::uint64_t value = 0;
76 for (unsigned shift = 0; shift <= 63; shift += 7) {
77 if (at == bytes.size()) throw std::invalid_argument("truncated front-coded header");
78 auto part = std::to_integer<unsigned>(bytes[at++]);
79 if (shift == 63 && part > 1) throw std::invalid_argument("overflowing front-coded header");
80 value |= std::uint64_t(part & 127) << shift;
81 if (!(part & 128)) return value;
82 }
83 throw std::invalid_argument("overflowing front-coded header");
84 }
85 }

135 // Navigation reads headers and skips suffix payloads without decoding keys.
136 front_encoded_record<N> encoded_at(std::uint64_t ordinal) const {
137 if (ordinal >= count_) throw std::out_of_range("front-coded record ordinal");
138 auto group = ordinal / 15;
139 auto at = offsets_.offset(group, N);
140 for (auto i = group * 15; i < ordinal; ++i) at = parse(at).next_offset;
141 return parse(at);
142 }

144 // Random reconstruction works for all streams but may traverse their whole
145 // prefix chain. A stream built with LPFC bounds its backwards byte span by
146 // the full target key's length, not by this requested prefix_limit.
147 front_decoded_record<N> reconstruct_at(std::uint64_t ordinal,
148 std::uint64_t prefix_limit = std::numeric_limits<std::uint64_t>::max()) const {
149 auto encoded = encoded_at(ordinal);
150 front_decoded_record<N> result;
151 result.full_size = encoded.key_size();
152 std::copy(encoded.value.begin(), encoded.value.end(), result.value.begin());
153 auto need = std::min(result.full_size, prefix_limit);
154 if (need > result.prefix.max_size()) throw std::length_error("front-coded key too large");
155 result.prefix.resize(static_cast<std::size_t>(need));
156 while (need) {
157 if (encoded.key_size() < need) {
158 throw std::invalid_argument("front-coded predecessor is shorter than copied prefix");
159 }
160 if (encoded.shared < need) {
161 auto count = need - encoded.shared;
162 std::copy_n(reinterpret_cast<char const *>(encoded.suffix.data()),
163 static_cast<std::size_t>(count), result.prefix.data() + encoded.shared);
164 need = encoded.shared;
165 }
166 if (!need) break;
167 if (!ordinal) throw std::invalid_argument("front-coded first key is not literal");
168 encoded = encoded_at(--ordinal);
169 }
170 return result;
171 }

173 // The first record's copied bytes must match the supplied anchor. A
174 // sufficient condition is actual predecessor <= anchor <= first key.
175 // Redundant coding can also permit an anchor above the first key. Only
176 // bytes through min(prefix_limit, stored_shared) must be supplied.
177 // Stored shared is a COPY count, not an exact LCP with the supplied anchor.
178 // callback's string view refers to scratch and expires on its next call.
179 template <class F>
180 void visit_window(std::uint64_t first, std::uint64_t last, front_anchor anchor,
181 std::uint64_t prefix_limit, F && callback) const {
182 if (first > last || last > count_ || last - first > 15) {
183 throw std::out_of_range("front-coded window must contain at most fifteen records");
184 }
185 if (first == last) return;
186 std::string scratch(anchor.prefix.substr(0, static_cast<std::size_t>(
187 std::min<std::uint64_t>(prefix_limit, anchor.prefix.size()))));
188 auto context_size = anchor.full_size;
189 auto encoded = encoded_at(first);
190 for (auto i = first; i != last; ++i) {
191 decode_into(encoded, prefix_limit, scratch, context_size);
192 front_item<N> item{i, {scratch, context_size}, encoded.value};
193 if (!callback(item)) break;
194 if (i + 1 != last) encoded = parse(encoded.next_offset);
195 }
196 }

198 // Sequential construction, merge input and explicit reference scans start
199 // at the literal first key. No ordinal-to-key random access is promised.
200 template <class F>
201 void visit_all(F && callback) const {
202 std::string scratch;
203 std::uint64_t context_size = 0;
204 std::uint64_t at = 0;
205 for (std::uint64_t i = 0; i != count_; ++i) {
206 auto encoded = parse(at);
207 decode_into(encoded, std::numeric_limits<std::uint64_t>::max(), scratch, context_size);
208 front_item<N> item{i, {scratch, context_size}, encoded.value};
209 if (!callback(item)) break;
210 at = encoded.next_offset;
211 }
212 }

219 front_encoded_record<N> parse(std::uint64_t offset) const {
220 if (offset > bytes_.size()) throw std::invalid_argument("front-coded offset outside stream");
221 auto at = static_cast<std::size_t>(offset);
222 auto shared = front_detail::read_varint(bytes_, at);
223 auto suffix_size = front_detail::read_varint(bytes_, at);
224 if (bytes_.size() - at < N || suffix_size > bytes_.size() - at - N) {
225 throw std::invalid_argument("truncated front-coded record");
226 }
227 auto suffix = bytes_.subspan(at, static_cast<std::size_t>(suffix_size));
228 at += static_cast<std::size_t>(suffix_size);
229 std::span<std::byte const, N> value(bytes_.data() + at, N);
230 return {shared, suffix, value, at + N};
231 }

233 static void decode_into(front_encoded_record<N> const & encoded, std::uint64_t limit,
234 std::string & scratch, std::uint64_t & context_size) {
235 if (encoded.shared > context_size) {
236 throw std::invalid_argument("front-coded copy exceeds anchor length");
237 }
238 auto keep = std::min(encoded.shared, limit);
239 if (keep > scratch.size()) {
240 throw std::invalid_argument("front-coded anchor prefix is too short");
241 }
242 auto full_size = encoded.key_size();
243 auto decoded_size = std::min(full_size, limit);
244 if (decoded_size > scratch.max_size()) throw std::length_error("front-coded key too large");
245 scratch.resize(static_cast<std::size_t>(keep));
246 auto copy = decoded_size - keep;
247 if (copy) {
248 scratch.append(reinterpret_cast<char const *>(encoded.suffix.data()),
249 static_cast<std::size_t>(copy));
250 }
251 context_size = full_size;
252 }

255 template <std::size_t N>
256 struct front_array {
257 static front_array build(std::span<front_record<N> const> records,
258 std::span<std::uint64_t const> prefix_ceilings = {},
259 std::uint64_t restart_factor = 0) {
260 if (!prefix_ceilings.empty() && prefix_ceilings.size() != records.size()) {
261 throw std::invalid_argument("one front-code prefix ceiling is required per record");
262 }
263 if (restart_factor && restart_factor < 3) {
264 throw std::invalid_argument("LPFC restart factor must be at least three");
265 }
266 front_array result;
267 std::vector<std::uint64_t> offsets;
268 std::string_view previous;
269 std::uint64_t fixed = 0;
270 std::uint64_t anchor = 0;
271 for (std::size_t i = 0; i != records.size(); ++i) {
272 auto const & record = records[i];
273 if (i && compare_keys(previous, record.key) > 0) {
274 throw std::invalid_argument("front-coded input must be sorted");
275 }
276 if (i % 15 == 0) offsets.push_back(result.bytes_.size() - fixed);
277 auto shared = common_prefix(previous, record.key);
278 if (!prefix_ceilings.empty()) shared = std::min<std::uint64_t>(shared, prefix_ceilings[i]);
279 // Count actual encoded bytes, including framing and fixed slots, to
280 // bound traversal span. The paper's key-only compression-space theorem
281 // is not claimed verbatim for this pragmatic framing convention.
282 auto start = std::uint64_t(result.bytes_.size());
283 if (shared && restart_factor &&
284 record.key.size() <= std::numeric_limits<std::uint64_t>::max() / restart_factor &&
285 start - anchor > restart_factor * record.key.size()) {
286 shared = 0;
287 }
288 if (!shared) anchor = start;
289 front_detail::append_varint(result.bytes_, shared);
290 front_detail::append_varint(result.bytes_, record.key.size() - shared);
291 auto suffix = std::as_bytes(std::span(record.key.data() + shared, record.key.size() - shared));
292 result.bytes_.insert(result.bytes_.end(), suffix.begin(), suffix.end());
293 result.bytes_.insert(result.bytes_.end(), record.value.begin(), record.value.end());
294 fixed += N;
295 previous = record.key;
296 }
297 offsets.push_back(result.bytes_.size() - fixed);
298 result.offsets_ = select15_index::build(offsets, records.size());
299 result.count_ = records.size();
300 return result;
301 }
