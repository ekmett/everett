60 // The caller splits a representable full key at its exact retained
61 // prefix and proves strict order. Bounds and value framing remain checked.
62 // Input spans need only survive this call and must not alias private output.
63 void append(std::uint64_t retained, bit_view literal, bit_view value) {
64 require_active();
65 if ((literal.size() & (P::bits_per_unit - 1)) || (value.size() & (P::bits_per_unit - 1)))
66 error_detail::raise<std::invalid_argument>("native writer record length disagrees with policy units");
67 auto value_units = value.size() >> P::unit_shift;
68 if (common_ && value_units != *common_)
69 error_detail::raise<std::invalid_argument>("native writer value disagrees with common width");
70 if (retained > previous_units_ || (count_ && literal.empty()))
71 error_detail::raise<std::invalid_argument>("native writer invalid known prefix");
72 auto literal_units = literal.size() >> P::unit_shift;
73 auto key_units = retained + literal_units;
74 auto next_count = add(count_, 1);
75 auto saved_bits = data_.bit_size;
76 auto saved_offsets = offsets_.size();
77 try {
78 if (count_ % P::codec_block_size == 0) {
79 auto stride = multiply(count_, common_.value_or(0));
80 offsets_.push_back((data_.bit_size >> P::unit_shift) - stride);
81 write_count<P>(data_, retained);
82 } else write_backspace<P>(data_, previous_units_ - retained);
83 write_count<P>(data_, literal_units);
84 if (!common_) write_count<P>(data_, value_units);
85 profile_detail::append(data_, literal);
86 profile_detail::append(data_, value);
87 } catch (...) {
88 resize(data_, saved_bits);
89 offsets_.resize(saved_offsets);
90 throw;
91 }
92 previous_units_ = key_units;
93 count_ = next_count;
94 }

96 profile_array<P> finish() {
97 require_active();
98 profile_array<P> result;
99 auto extent = data_.bit_size >> P::unit_shift;
100 auto stride = multiply(count_, common_.value_or(0));
101 offsets_.push_back(extent - stride);
102 try {
103 result.offsets_ = elias_fano::build(offsets_);
104 } catch (...) {
105 offsets_.pop_back();
106 throw;
107 }
108 result.metadata_.common_value_width = common_;
109 result.metadata_.record_count = count_;
110 result.metadata_.extent = extent;
111 result.metadata_.terminal_key_units = previous_units_;
112 result.bytes_ = std::move(data_.bytes);
113 data_.bit_size = 0;
114 offsets_ = {};
115 finished_ = true;
116 return result;
117 }

129 // Incremental ordinary-FC native output. A common value width, if any, is
130 // fixed before the first record. Otherwise every value carries its length.
131 // Keeps encoded output, one residual offset per physical block, and the last
132 // key; it does not stage a vector of full records. Output remains in memory.

154 // Keys must be strictly increasing. Input views need only survive this
155 // call. A rejected append leaves the preceding committed records intact.
156 void append(bit_view key, bit_view value) {
157 output_.require_active();
158 if ((key.size() & (P::bits_per_unit - 1)) || (value.size() & (P::bits_per_unit - 1)))
159 error_detail::raise<std::invalid_argument>("native writer record length disagrees with policy units");
160 auto common = output_.common_value_width();
161 if (common && (value.size() >> P::unit_shift) != *common)
162 error_detail::raise<std::invalid_argument>("native writer value disagrees with common width");
163 auto comparison = compare_common_bits(previous_.view(), key);
164 if (output_.size() && comparison.order >= 0)
165 error_detail::raise<std::invalid_argument>("native writer keys must be strictly increasing");
166 auto retained = comparison.common_bits >> P::unit_shift;
167 auto retained_bits = retained * P::bits_per_unit;
168 // Reserve before fallible output writes. Once the frame commits, updating
169 // the logical predecessor cannot allocate or leave a partial bit string.
170 auto next_bytes = profile_detail::byte_count(key.size());
171 if (next_bytes > previous_.bytes.max_size())
172 error_detail::raise<std::length_error>("profile bit string too large");
173 previous_.bytes.reserve(static_cast<std::size_t>(next_bytes));
174 auto literal = key.subview(retained_bits, key.size() - retained_bits);
175 output_.append(retained, literal, value);
176 profile_detail::resize(previous_, key.size());
177 profile_detail::copy_into(previous_, retained_bits, literal);
178 }
179 void append(profile_record const & record) { append(record.key.view(), record.value.view()); }

181 // Finalizing Elias–Fano visits the staged block offsets. This operation
182 // is neither byte-budgeted nor a durable checkpoint.
183 profile_array<P> finish() {
184 auto result = output_.finish();
185 previous_ = {};
186 return result;
187 }

191 bit_string previous_;
192 };
