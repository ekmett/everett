55 // The caller splits a representable full key at its exact retained
56 // prefix and proves strict order. Bounds and value framing remain checked.
57 // Input spans need only survive this call and must not alias private output.
58 void append(std::uint64_t retained, bit_view literal, bit_view value) {
59 require_active();
60 if ((literal.size() & (P::bits_per_unit - 1)) || (value.size() & (P::bits_per_unit - 1)))
61 error_detail::raise<std::invalid_argument>("native writer record length disagrees with policy units");
62 auto value_units = value.size() >> P::unit_shift;
63 if (common_ && value_units != *common_)
64 error_detail::raise<std::invalid_argument>("native writer value disagrees with common width");
65 if (retained > previous_units_ || (count_ && literal.empty()))
66 error_detail::raise<std::invalid_argument>("native writer invalid known prefix");
67 auto literal_units = literal.size() >> P::unit_shift;
68 auto key_units = retained + literal_units;
69 auto next_count = add(count_, 1);
70 auto saved_bits = data_.bit_size;
71 auto saved_offsets = offsets_.size();
72 try {
73 if (count_ % P::codec_block_size == 0) {
74 auto stride = multiply(count_, common_.value_or(0));
75 offsets_.push_back((data_.bit_size >> P::unit_shift) - stride);
76 write_count<P>(data_, previous_units_);
77 }
78 write_backspace<P>(data_, previous_units_ - retained);
79 write_count<P>(data_, literal_units);
80 if (!common_) write_count<P>(data_, value_units);
81 profile_detail::append(data_, literal);
82 profile_detail::append(data_, value);
83 } catch (...) {
84 resize(data_, saved_bits);
85 offsets_.resize(saved_offsets);
86 throw;
87 }
88 previous_units_ = key_units;
89 count_ = next_count;
90 }

92 profile_array<P> finish() {
93 require_active();
94 profile_array<P> result;
95 auto extent = data_.bit_size >> P::unit_shift;
96 auto stride = multiply(count_, common_.value_or(0));
97 offsets_.push_back(extent - stride);
98 try {
99 result.offsets_ = elias_fano::build(offsets_);
100 } catch (...) {
101 offsets_.pop_back();
102 throw;
103 }
104 result.metadata_.common_value_width = common_;
105 result.metadata_.record_count = count_;
106 result.metadata_.extent = extent;
107 result.metadata_.terminal_key_units = previous_units_;
108 result.bytes_ = std::move(data_.bytes);
109 data_.bit_size = 0;
110 offsets_ = {};
111 finished_ = true;
112 return result;
113 }

125 // Incremental ordinary-FC native output. A common value width, if any, is
126 // fixed before the first record. Otherwise every value carries its length.
127 // Keeps encoded output, one residual offset per physical block, and the last
128 // key; it does not stage a vector of full records. Output remains in memory.
129 template <class P> struct profile_native_writer {

150 // Keys must be strictly increasing. Input views need only survive this
151 // call. A rejected append leaves the preceding committed records intact.
152 void append(bit_view key, bit_view value) {
153 output_.require_active();
154 if ((key.size() & (P::bits_per_unit - 1)) || (value.size() & (P::bits_per_unit - 1)))
155 error_detail::raise<std::invalid_argument>("native writer record length disagrees with policy units");
156 auto common = output_.common_value_width();
157 if (common && (value.size() >> P::unit_shift) != *common)
158 error_detail::raise<std::invalid_argument>("native writer value disagrees with common width");
159 auto comparison = compare_common_bits(previous_.view(), key);
160 if (output_.size() && comparison.order >= 0)
161 error_detail::raise<std::invalid_argument>("native writer keys must be strictly increasing");
162 auto retained = comparison.common_bits >> P::unit_shift;
163 auto retained_bits = retained * P::bits_per_unit;
164 // Reserve before fallible output writes. Once the frame commits, updating
165 // the logical predecessor cannot allocate or leave a partial bit string.
166 auto next_bytes = profile_detail::byte_count(key.size());
167 if (next_bytes > previous_.bytes.max_size())
168 error_detail::raise<std::length_error>("profile bit string too large");
169 previous_.bytes.reserve(static_cast<std::size_t>(next_bytes));
170 auto literal = key.subview(retained_bits, key.size() - retained_bits);
171 output_.append(retained, literal, value);
172 profile_detail::resize(previous_, key.size());
173 profile_detail::copy_into(previous_, retained_bits, literal);
174 }
175 void append(profile_record const & record) { append(record.key.view(), record.value.view()); }

177 // Finalizing Elias–Fano visits the staged block offsets. This operation
178 // is neither byte-budgeted nor a durable checkpoint.
179 profile_array<P> finish() {
180 auto result = output_.finish();
181 previous_ = {};
182 return result;
183 }

185 private:
186 profile_detail::native_output<P> output_;
187 bit_string previous_;
188 };
