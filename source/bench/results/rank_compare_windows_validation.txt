CPUID/XGETBV probe source
========================
#define NOMINMAX
#include <windows.h>
#include <intrin.h>
#include <cstdint>
#include <iostream>

int main() {
  int leaves[4]{};
  __cpuidex(leaves, 1, 0);
  auto leaf1 = static_cast<unsigned>(leaves[2]);
  __cpuidex(leaves, 7, 0);
  auto leaf7 = static_cast<unsigned>(leaves[1]);
  auto xcr0 = (leaf1 & (1u << 27)) ? _xgetbv(0) : 0;
  bool avx2 = (leaf1 & (1u << 28)) && (leaf7 & (1u << 5)) && (xcr0 & 6) == 6;
  bool avx512 = avx2 && (leaf7 & (1u << 16)) && (leaf7 & (1u << 30)) && (xcr0 & 0xe6) == 0xe6;
  DWORD_PTR process_mask = 0, system_mask = 0;
  if (!GetProcessAffinityMask(GetCurrentProcess(), &process_mask, &system_mask)) return 2;
  std::cout << "msvc_full_ver=" << _MSC_FULL_VER << " cpu=" << GetCurrentProcessorNumber()
            << " affinity=0x" << std::hex << process_mask << " leaf1_ecx=0x" << leaf1
            << " leaf7_ebx=0x" << leaf7 << " xcr0=0x" << xcr0 << std::dec
            << " usable_avx2=" << avx2 << " usable_avx512f_bw=" << avx512 << '\n';
  __cpuidex(leaves, static_cast<int>(0x80000000u), 0);
  if (static_cast<unsigned>(leaves[0]) >= 0x8000001du) {
    for (int index = 0; index < 8; ++index) {
      __cpuidex(leaves, static_cast<int>(0x8000001du), index);
      auto a = static_cast<unsigned>(leaves[0]), b = static_cast<unsigned>(leaves[1]);
      if (!(a & 31)) break;
      auto bytes = std::uint64_t((b & 4095) + 1) * (((b >> 12) & 1023) + 1) *
                   (((b >> 22) & 1023) + 1) * (std::uint64_t(static_cast<unsigned>(leaves[2])) + 1);
      std::cout << "cache_level=" << ((a >> 5) & 7) << " type=" << (a & 31)
                << " bytes=" << bytes << " sharing=" << (((a >> 14) & 4095) + 1) << '\n';
    }
  }
  return avx2 && avx512 ? 0 : 3;
}
