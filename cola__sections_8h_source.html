47 inline std::span<std::byte const> section(std::span<std::byte const> body,
48 directory const & layout, std::size_t slot) {
49 if (slot >= section_count) error_detail::raise<std::out_of_range>("COLA section slot");
50 auto const & part = layout.sections[slot];
51 if (part.offset > body.size() || part.length > body.size() - part.offset)
52 error_detail::raise<std::invalid_argument>("COLA section exceeds body");
53 return body.subspan(static_cast<std::size_t>(part.offset), static_cast<std::size_t>(part.length));

55 template <class P> directory parse(file_header<P> const & header, std::span<std::byte const> body) {
56 if (header.kind != file_kind::fractional_index || body.size() < directory_bytes ||
57 body.size() > std::numeric_limits<std::uint64_t>::max() - 7)
58 error_detail::raise<std::invalid_argument>("invalid COLA index container");
59 if (header.extent != profile_detail::multiply(body.size(), 1u << (3 - P::unit_shift)))
60 error_detail::raise<std::invalid_argument>("COLA index must occupy complete bytes");

63 error_detail::raise<std::invalid_argument>("unexpected COLA index magic");

65 error_detail::raise<std::invalid_argument>("unsupported COLA index directory");
66 directory result{{}, file_detail::get(body, 8, 8), {}, {}, {}, {}, {}, section_detail::get_id(body, 88), {}, {}};
67 for (unsigned route = 0; route != 2; ++route) {
68 result.counts[route] = file_detail::get(body, 16 + 8 * route, 8);
69 result.extents[route] = file_detail::get(body, 32 + 8 * route, 8);
70 result.terminal_units[route] = file_detail::get(body, 48 + 8 * route, 8);
71 result.universes[route] = file_detail::get(body, 64 + 8 * route, 8);
72 result.low_widths[route] = unsigned(file_detail::get(body, 80 + route, 1));
73 }
74 auto main = file_detail::get(body, 82, 1), secondary = file_detail::get(body, 83, 1);
75 if (main > 1 || secondary > 1)
76 error_detail::raise<std::invalid_argument>("invalid COLA target flag");
77 section_detail::zero(body.subspan(84, 4), "nonzero COLA reserved bytes");
78 section_detail::zero(body.subspan(152, 8), "nonzero COLA reserved bytes");
79 if (main) result.main_id = blob_identity{section_detail::get_id(body, 104), section_detail::get_id(body, 120)};
80 else section_detail::zero(body.subspan(104, 32), "absent COLA main target is nonzero");
81 if (secondary) result.secondary_id = section_detail::get_id(body, 136);
82 else section_detail::zero(body.subspan(136, 16), "absent COLA secondary target is nonzero");

87 std::uint64_t end = directory_bytes;

91 if (part.offset != section_detail::align(end) || part.offset > body.size() || part.length > body.size() - part.offset)

95 }
96 if (end != body.size()) error_detail::raise<std::invalid_argument>("trailing COLA section bytes");

99 template <class P> profile_view<P, stream_role::borrowed> profile(
100 std::span<std::byte const> body, directory const & layout, unsigned route) {
101 auto metadata = profile_detail::initial_metadata<P, stream_role::borrowed>();
102 metadata.record_count = layout.counts[route];
103 metadata.extent = layout.extents[route];
104 metadata.terminal_key_units = layout.terminal_units[route];
105 metadata.common_value_width = 0;
106 auto first = profile_slot(route);
107 auto part = [&](unsigned offset) { return section(body, layout, first + offset); };
