35 // Reflected polynomial representation: bit31 is the multiplicative unit,
36 // and one rightward step multiplies by x modulo the Castagnoli polynomial.
37 constexpr std::uint32_t multiply_polynomial(std::uint32_t a, std::uint32_t b) noexcept {
38 std::uint32_t result = 0;
39 for (std::uint32_t bit = 1u << 31; bit; bit >>= 1) {
40 if (a & bit) result ^= b;
41 b = (b >> 1) ^ (0x82f63b78u & (0u - (b & 1)));
42 }
43 return result;
44 }
45 inline constexpr auto byte_powers = [] {
46 std::array<std::uint32_t, 64> result{};
47 result[0] = 1u << 23; // x^8
48 for (unsigned i = 1; i != result.size(); ++i)
49 result[i] = multiply_polynomial(result[i - 1], result[i - 1]);
50 return result;
51 }();
52 inline std::uint32_t shift(std::uint32_t crc, std::uint64_t bytes) noexcept {
53 for (unsigned i = 0; bytes; ++i, bytes >>= 1)
54 if (bytes & 1) crc = multiply_polynomial(byte_powers[i], crc);
55 return crc;
56 }

58 // Generated C kernels load scalar words through this C++ alias-safe bridge.
59 // Exactly sizeof(T) bytes are read, including unaligned and mmap-tail inputs.
60 template <class T> inline T load_little(void const * bytes) noexcept {
61 T value;
62 std::memcpy(&value, bytes, sizeof(T));
63 if constexpr (std::endian::native == std::endian::big) {
64 T reversed = 0;
65 for (unsigned i = 0; i < sizeof(T); ++i) {
66 reversed = T((reversed << 8) | (value & 255u));
67 value >>= 8;
68 }
69 return reversed;
70 } else return value;
71 }

98 // Combine finalized CRCs of A and B into CRC(A || B) without rereading either
99 // input. second_bytes is B's physical byte length. The CRC of empty B is 0.
100 // Lengths span all 64 bits; no bytes-to-bits multiplication can overflow.
101 inline std::uint32_t crc32c_combine(std::uint32_t first, std::uint32_t second,
102 std::uint64_t second_bytes) noexcept {
103 return crc32c_detail::shift(first, second_bytes) ^ second;
104 }

106 // Reflected Castagnoli CRC32C, with the conventional initial/final complement.
107 // Pinned Corsix-generated kernels are selected by the compiler target, without
108 // runtime feature probes, allocation, or instructions beyond that target.
109 // previous_crc is the finalized result of the preceding chunk, or zero for
110 // a new stream. Empty chunks preserve it; callers do not invert the state.
111 inline std::uint32_t crc32c(std::span<std::byte const> bytes,
112 std::uint32_t previous_crc = 0) noexcept {
113 auto data = reinterpret_cast<char const *>(bytes.data());
114 auto size = bytes.size();
115#if defined(__aarch64__) && defined(__ARM_FEATURE_CRC32) && (defined(__GNUC__) || defined(__clang__))
116 if (size < 128) return crc32c_detail::arm_scalar::crc32_impl(previous_crc, data, size);
117#if (defined(__ARM_FEATURE_CRYPTO) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__) && defined(__ARM_FEATURE_SHA3)
118 // The fused kernel amortizes its fold setup on larger buffers; PMULL has
119 // lower setup cost for page-sized inputs. Threshold measured on Apple M2.
120 if (size >= 65536) return crc32c_detail::arm_eor3::crc32_impl(previous_crc, data, size);
121 return crc32c_detail::arm_pmull::crc32_impl(previous_crc, data, size);
122#elif (defined(__ARM_FEATURE_CRYPTO) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__)
123 return crc32c_detail::arm_pmull::crc32_impl(previous_crc, data, size);
124#else
125 return crc32c_detail::arm_scalar::crc32_impl(previous_crc, data, size);
126#endif
127#elif defined(__x86_64__) && defined(__SSE4_2__)
128 if (size < 256) return crc32c_detail::x86_scalar::crc32_impl(previous_crc, data, size);
129#if defined(__PCLMUL__) && defined(__AVX512F__) && defined(__AVX512VL__) && defined(__VPCLMULQDQ__)
130 return crc32c_detail::x86_vpclmul::crc32_impl(previous_crc, data, size);
131#elif defined(__PCLMUL__) && defined(__AVX512F__) && defined(__AVX512VL__)
132 return crc32c_detail::x86_avx512::crc32_impl(previous_crc, data, size);
133#elif defined(__PCLMUL__)
134 return crc32c_detail::x86_pclmul::crc32_impl(previous_crc, data, size);
135#else
136 return crc32c_detail::x86_scalar::crc32_impl(previous_crc, data, size);
137#endif
138#else
139 return crc32c_detail::portable::crc32_impl(previous_crc, data, size);
140#endif
141 }
