77 struct fc_key_frame {

79 bit_view literal;
80 std::uint64_t size() const { return profile_detail::add(retained_bits, literal.size()); }
81 bit_comparison compare(bit_view query, bit_comparison previous) const {
82 if (previous.common_bits < retained_bits) return previous;
83 if (retained_bits > query.size())
84 error_detail::raise<std::invalid_argument>("FC comparison lacks inherited prefix");
85 auto result = compare_common_bits(literal, query.subview(retained_bits, query.size() - retained_bits));

88 }

91 // Strings compare as unsigned bytes. Their ordered key representation escapes
92 // zero as 00 ff and terminates with 00 00; it is distinct from FC wire framing.
93 struct ordered_string_key {
94 using value_type = std::string;
95 static bool less(value_type const & a, value_type const & b) {
96 return compare_bits(sort_codec_detail::string_bits(a), sort_codec_detail::string_bits(b)) < 0;
97 }
98 static void write_ordered(sort_bit_writer & out, value_type const & value) {
99 for (unsigned char c : value) {
100 out.write_bits(c, 8);
101 if (!c) out.write_bits(255, 8);
102 }
103 out.write_bits(0, 16);
104 }
105 static value_type read_ordered(sort_bit_reader & in) {
106 value_type result;
107 for (;;) {
108 auto c = in.read_bits(8);
109 if (!c) {
110 auto escape = in.read_bits(8);
111 if (!escape) return result;
112 if (escape != 255) error_detail::raise<std::invalid_argument>("noncanonical string escape");
113 }
114 result.push_back(static_cast<char>(c));
115 }
116 }

119 template <class CountCode = exponential_golomb<0>> struct fc_string_key : ordered_string_key {
120 static fc_key_frame read_frame(sort_bit_reader & in, std::uint64_t previous_bits) {
121 auto backspace = in.template read_count<CountCode>();
122 if (backspace > previous_bits) error_detail::raise<std::invalid_argument>("FC backspace exceeds key");
123 auto retained = previous_bits - backspace;
124 auto literal = in.take_bits(in.template read_count<CountCode>());
125 if ((profile_detail::add(retained, literal.size()) & 7) != 0)
126 error_detail::raise<std::invalid_argument>("FC string ends inside a byte");
127 return {retained, literal};
128 }
129 static void write(sort_bit_writer & out, value_type const & value, value_type const * previous = nullptr) {
130 auto key = sort_codec_detail::string_bits(value);
131 auto old = previous ? sort_codec_detail::string_bits(*previous) : bit_view{};
132 auto retained = compare_common_bits(old, key).common_bits;
133 out.template write_count<CountCode>(old.size() - retained);
134 out.template write_count<CountCode>(key.size() - retained);
135 out.append(key.subview(retained, key.size() - retained));
136 }
137 static value_type read(sort_bit_reader & in, value_type const * previous = nullptr) {
138 auto old = previous ? sort_codec_detail::string_bits(*previous) : bit_view{};
139 auto frame = read_frame(in, old.size());
140 value_type result(static_cast<std::size_t>(frame.size() >> 3), '\0');
141 auto target = reinterpret_cast<std::byte *>(result.data());
142 profile_detail::copy_bits(target, 0, old.prefix(frame.retained_bits));
143 profile_detail::copy_bits(target, frame.retained_bits, frame.literal);
144 return result;
145 }

174 static void write_ordered(sort_bit_writer & out, value_type const & value) {

352 // Replay retained discriminator bits, then consume exactly enough new bits
353 // to reach a leaf. A leaf may not terminate inside the retained path.
354 struct prefix_reader {
355 sort_bit_reader & input;
356 bit_view retained;
357 sort_bit_writer & path;
358 std::uint64_t at = 0;
359 std::uint64_t read_bits(unsigned width) {
360 std::uint64_t value = 0;
361 for (unsigned i = 0; i != width; ++i) {
362 auto bit = at < retained.size() ? profile_detail::load_bits(retained, at++, 1) : input.read_bits(1);
363 path.write_bits(bit, 1);
364 value = (value << 1) | bit;
365 }
366 return value;
367 }
368 };
369 }

441 key_codec::write(out, key, previous ? &previous->key : nullptr);

482 // Callback receives ephemeral references. Its exception consumes this record
483 // and poisons the reader; copied anchors can resume on the remaining slice.
484 template <class Visitor> bool next(Visitor && visitor) {
485 if (failed_) error_detail::raise<std::logic_error>("failed sort record reader");
486 if (empty()) return false;
487 try {
488 sort_record_control control;
489 control.start = input_.position();
490 auto backspace = input_.template read_count<TreeCode>();
491 if (backspace > state_.path.bit_size)
492 error_detail::raise<std::invalid_argument>("sort backspace exceeds path");
493 control.sort_retained = state_.path.bit_size - backspace;
494 bit_string path;
495 bool reuse = backspace == 0 && !std::holds_alternative<std::monostate>(state_.previous);
496 auto decode = [&]<class S>(std::type_identity<S> tag) {
497 sort_codec_detail::validate_value_width<S>();
498 auto path_bits = reuse ? state_.path.view() : path.view();
499 control.sort_bits = path_bits.size();
500 control.key_start = input_.position();
501 using key_codec = typename sort_codec<S>::key_codec;
502 using value_codec = std::conditional_t<Role == stream_role::native,
503 typename sort_codec<S>::value_codec, no_value>;
504 auto previous = std::get_if<sort_codec_detail::key_state<S>>(&state_.previous);
505 auto key = key_codec::read(input_, previous ? &previous->key : nullptr);
506 control.value_start = input_.position();
507 auto value = value_codec::read(input_);
508 control.end = input_.position();
509 if ((!std::holds_alternative<std::monostate>(state_.previous) &&
510 !reuse && compare_bits(state_.path.view(), path_bits) > 0) ||
511 (previous && !key_codec::less(previous->key, key)))
512 error_detail::raise<std::invalid_argument>("sort records must have unique sorted keys");
513 state_.previous = sort_codec_detail::key_state<S>{std::move(key)};
514 if (!reuse) state_.path = std::move(path);
515 auto const & stored = std::get<sort_codec_detail::key_state<S>>(state_.previous);
516 std::invoke(std::forward<Visitor>(visitor), tag, stored.key, value, control);
517 };
518 if (reuse) {
519 // Retaining the complete code already identifies the leaf. Do not
520 // walk a deep tree or copy its path again for every key of that sort.
521 std::visit([&](auto const & previous) {
522 using state = std::remove_cvref_t<decltype(previous)>;
523 if constexpr (!std::is_same_v<state, std::monostate>)
524 decode(std::type_identity<typename state::sort_type>{});
525 }, state_.previous);
526 } else {
527 sort_bit_writer path_writer(path);
528 sort_codec_detail::prefix_reader prefix{input_, state_.path.view().prefix(control.sort_retained), path_writer};
529 dispatch_sort<Registry>(prefix, [&]<class S>(std::type_identity<S> tag, auto &) {
530 if (prefix.at != control.sort_retained)
531 error_detail::raise<std::invalid_argument>("retained sort prefix crosses leaf");
532 decode(tag);
533 });
