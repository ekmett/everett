115 inline bit_comparison compare_spans(std::span<bit_view const> a, std::span<bit_view const> b,
116 std::uint64_t first_a = 0, std::uint64_t first_b = 0) {
117 std::size_t i = 0, j = 0;
118 while (i != a.size() && first_a >= a[i].size()) first_a -= a[i++].size();
119 while (j != b.size() && first_b >= b[j].size()) first_b -= b[j++].size();
120 std::uint64_t common = 0;
121 while (i != a.size() && j != b.size()) {
122 auto width = std::min(a[i].size() - first_a, b[j].size() - first_b);
123 auto cmp = compare_common_bits(a[i].subview(first_a, width), b[j].subview(first_b, width));
124 common += cmp.common_bits;
125 if (cmp.order) return {common, cmp.order};
126 first_a += width; first_b += width;
127 if (first_a == a[i].size()) { ++i; first_a = 0; }
128 if (first_b == b[j].size()) { ++j; first_b = 0; }
129 }
130 while (i != a.size() && !a[i].size()) ++i;
131 while (j != b.size() && !b[j].size()) ++j;
132 return {common, i == a.size() ? j == b.size() ? 0 : -1 : 1};
133 }

226 sort_profile_frame next(sort_profile_frame const & previous) const {
227 auto ordinal = previous.ordinal + 1;
228 if (ordinal >= size()) throw std::out_of_range("sort profile successor");
229 if (!(ordinal % P::codec_block_size)) {
230 if (block_offset(ordinal / P::codec_block_size) != previous.next_offset)
231 throw std::invalid_argument("sort profile block offset mismatch");

233 }
234 auto at = previous.next_offset;
235 auto backspace = profile_detail::read_backspace<P>(data_, at);
236 if (backspace > previous.continuation()) throw std::invalid_argument("sort profile backspace exceeds cursor");
237 auto retained = previous.continuation() - backspace;
238 sort_profile_frame result;
239 result.ordinal = ordinal;
240 result.leaf = previous.leaf;
241 result.path = previous.path;
242 result.parse = previous.parse; result.front_coded = previous.front_coded;
243 bool same = retained >= previous.path.size();

245 sort_bit_reader input(data_.subview(at, data_.size() - at));

252 at += input.position();

254 for (std::uint64_t i = 0; i != dictionary_size(); ++i) {
255 auto first = dictionary_offsets_[i], last = dictionary_offsets_[i + 1];
256 if (first > last || last > dictionary_.size()) throw std::invalid_argument("sort profile dictionary range");
257 auto candidate = dictionary_.subview(first, last - first);
258 if (!compare_bits(candidate, path.view())) { result.path = candidate; found = true; break; }
259 }
260 if (!found) throw std::invalid_argument("sort transition absent from dictionary");
261 }
262 return payload(at, retained, std::move(result), same);

264 template <class F> void compare_window(std::uint64_t first, std::uint64_t last,
265 profile_query_context<P> context, F && callback, profile_comparison_work * work = nullptr) const {
266 if (first > last || last > size() || last - first > P::group_size)
267 throw std::out_of_range("sort profile query window");
268 if (first == last) return;
269 auto record = encoded_at(first);
270 if (work) work->skipped_headers += first % P::codec_block_size;
271 for (auto i = first; i != last; ++i) {
272 auto count = context.advance_parts(record.retained, record.key_units, record.literal);
273 if (work) { ++work->visited_headers; work->compared_bits += count; }
274 if (!callback(profile_comparison_item<P>{i, context, record.value})) return;
275 if (i + 1 != last) record = next(record);
276 }

296 sort_profile_frame start_block(std::uint64_t block, sort_profile_frame const * previous = nullptr) const {
297 auto id = seed_width() ? profile_detail::load_bits(seeds_, block * seed_width(), seed_width()) : 0;
298 if (id >= dictionary_size()) throw std::invalid_argument("sort profile seed ID");
299 auto first = dictionary_offsets_[id], last = dictionary_offsets_[id + 1];
300 if (first > last || last > dictionary_.size()) throw std::invalid_argument("sort profile dictionary range");
301 auto path = dictionary_.subview(first, last - first);
302 sort_bit_reader code(path);
303 sort_profile_frame result;
304 if (previous && previous->path.storage().data() == path.storage().data() &&
305 previous->path.offset() == path.offset() && previous->path.size() == path.size()) {
306 result.leaf = previous->leaf; result.parse = previous->parse; result.front_coded = previous->front_coded;
307 } else {
308 Selector::select(code, [&]<class S>(std::type_identity<S>, auto &) { select<S>(result); });
309 if (!code.empty()) throw std::invalid_argument("trailing selector code bits");
310 }
311 result.path = path;
312 auto at = block_offset(block);
313 auto retained = profile_detail::read_count<P>(data_, at);
314 result.ordinal = block * P::codec_block_size;
315 return payload(at, retained, std::move(result), true, true);

326 sort_profile_frame payload(std::uint64_t at, std::uint64_t retained, sort_profile_frame result, bool same, bool restart = false) const {
327 sort_bit_reader in(data_.subview(at, data_.size() - at));
328 auto local = same ? retained : result.path.size();
329 if (local < result.path.size()) throw std::invalid_argument("sort profile retained position precedes leaf");
330 // The selected handler survives every same-sort frame. Selector dispatch
331 // occurs only on a sort transition or independently entered block.
332 auto parsed = result.parse(in, local - result.path.size());
333 auto frame = parsed.key;
334 bool inherit = same && (frame.retained_bits || !restart);
335 result.retained = inherit ? retained : same ? 0 : retained;
336 result.literal = {inherit ? bit_view{} :
337 result.path.subview(result.retained, result.path.size() - result.retained), frame.literal};
338 result.key_units = profile_detail::add(result.path.size(), frame.size());
339 result.value = data_.subview(at + parsed.value_start, parsed.end - parsed.value_start);
340 if (metadata_.common_value_width && result.value.size() != *metadata_.common_value_width)
341 throw std::invalid_argument("sort profile fixed value width mismatch");
342 result.next_offset = at + parsed.end;
343 return result;
344 }

347 template <class P, class Selector> struct sort_profile_cursor {
348 explicit sort_profile_cursor(sort_profile_view<P, Selector> view) : view_(view) {
349 if (view.size()) { frame_ = view_.encoded_at(0); decode(); }
350 }
351 bool done() const noexcept { return ordinal_ == view_.size(); }
352 profile_item<P> peek() const {
353 if (done()) throw std::out_of_range("sort profile cursor end");
354 return {ordinal_, profile_anchor<P>::complete(key_.view()), frame_.value};
355 }
356 void advance() { (void)advance_comparison(); }
357 std::optional<bit_comparison> advance_comparison() {
358 if (done()) throw std::out_of_range("sort profile cursor end");
359 ++ordinal_;
360 if (done()) {
361 if (frame_.next_offset != view_.data().size() || frame_.key_units != view_.metadata().terminal_key_units)
362 throw std::invalid_argument("sort profile terminal framing mismatch");
363 return std::nullopt;
364 }
365 frame_ = view_.next(frame_);
366 if (frame_.retained > key_.bit_size) throw std::invalid_argument("sort cursor missing inherited prefix");
367 std::array<bit_view, 1> before{key_.view()};
368 auto comparison = sort_profile_detail::compare_spans(before, frame_.literal, frame_.retained);
369 comparison.common_bits += frame_.retained;
370 profile_detail::resize(key_, frame_.retained);
371 for (auto part : frame_.literal) profile_detail::append(key_, part);
372 return comparison;
373 }

445 template <class S, class Output> void append(Output & out, bit_view path, std::uint64_t key_bits,
446 std::span<bit_view const> key, bit_view value, std::uint64_t common) {
447 using codec = sort_profile_key<typename sort_codec<S>::key_codec>;
448 sort_bit_reader validate(value);
449 sort_codec<S>::value_codec::skip(validate);
450 if (!validate.empty()) throw std::invalid_argument("trailing sort profile value bits");
451 auto total = std::uint64_t{0};
452 for (auto part : key) total = profile_detail::add(total, part.size());
453 if (total != profile_detail::add(path.size(), key_bits) || common > total)
454 throw std::invalid_argument("sort merge key spans");
455 constexpr auto leaf = ordinal<leaves, S>::value;
456 auto found = std::find(ids_.begin(), ids_.end(), leaf);
457 auto id = std::size_t(found - ids_.begin());
458 if (found == ids_.end()) {
459 ids_.push_back(leaf); profile_detail::append(dictionary, path);
460 dictionary_offsets.push_back(dictionary.bit_size);
461 }
462 auto same = size() && compare_bits(previous_path_.view(), path) == 0;
463 auto retained = same && codec::front_coded ? common : path.size();
464 if (retained < path.size() || retained > total) throw std::invalid_argument("sort retained prefix outside key");
465 if (!(size() % P::codec_block_size)) {
466 raw_offsets_.push_back(out.position()); block_seeds_.push_back(id);
467 out.template write_count<exponential_golomb<0>>(retained);
468 } else {
469 auto joint = same ? retained : compare_common_bits(previous_path_.view(), path).common_bits;
470 if (joint > previous_continuation_) throw std::invalid_argument("sort backspace exceeds continuation");
471 out.template write_count<typename P::backspace_encoding>(previous_continuation_ - joint);
472 if (!same) out.append(path.subview(joint, path.size() - joint));
473 }
474 auto local = retained - path.size();
475 codec::header(out, local, key_bits - local);
476 auto skip = retained;
477 for (auto part : key) {
478 auto prefix = std::min(skip, part.size()); skip -= prefix;
479 out.append(part.subview(prefix, part.size() - prefix));
480 }
481 out.append(value);
482 if (!size()) metadata.common_value_width = value.size();
483 else if (metadata.common_value_width != value.size()) metadata.common_value_width.reset();
484 ++metadata.record_count;
485 metadata.terminal_key_units = total;
486 previous_path_ = bit_string::copy(path);
487 previous_continuation_ = path.size() + (codec::front_coded ? key_bits : 0);
488 }
489 void finish(std::uint64_t extent) {
490 raw_offsets_.push_back(extent);
491 auto common = metadata.common_value_width.value_or(0);
492 for (std::size_t i = 0; i != raw_offsets_.size(); ++i) {
493 auto ordinal = i + 1 == raw_offsets_.size() ? size() : i * P::codec_block_size;
494 raw_offsets_[i] -= ordinal * common;
495 }
496 offsets = elias_fano::build(raw_offsets_);
497 auto width = ids_.size() < 2 ? 0u : std::bit_width(ids_.size() - 1);
498 sort_bit_writer out(seeds);
499 for (auto id : block_seeds_) out.write_bits(id, unsigned(width));
500 metadata.extent = extent;
501 }
502 // The same completed framing state supplies owning and streamed output.
503 // Neither adoption nor serialization repeats key/value composition.
504 sort_profile_array<P, Selector> take(bit_string data) {
505 return {std::move(data), std::move(dictionary), std::move(seeds),
506 std::move(dictionary_offsets), std::move(offsets), metadata};
507 }
508 private:
509 bit_string previous_path_;
510 std::vector<std::size_t> ids_;
511 std::vector<std::uint64_t> raw_offsets_, block_seeds_;
512 std::uint64_t previous_continuation_ = 0;
513 };
514 }

516 template <class P, class Selector> struct sort_profile_writer {
517 using policy_type = P;
518 using array_type = sort_profile_array<P, Selector>;
519 template <class S> void append(typename sort_codec<S>::key_codec::value_type const & key,
520 typename sort_codec<S>::value_codec::value_type const & value) {
521 require_active();
522 if (encoded_only_) throw std::logic_error("typed append after encoded sort frames");
523 sort_codec_detail::validate_value_width<S>();
524 bit_string path, encoded_value;
525 sort_bit_writer path_out(path), value_out(encoded_value);
526 Selector::template write<S>(path_out);
527 sort_codec<S>::value_codec::write(value_out, value);
528 auto bits = sort_profile_key<typename sort_codec<S>::key_codec>::order(key);
529 auto logical = sort_profile_detail::concatenate(path.view(), bits.view());
530 auto comparison = compare_common_bits(previous_.view(), logical.view());
531 if (size() && comparison.order >= 0) throw std::invalid_argument("sort profile requires unique sorted keys");
532 std::array<bit_view, 1> spans{logical.view()};
533 try {
534 sort_bit_writer out(data_);
535 encoder_.template append<S>(out, path.view(), bits.bit_size, spans, encoded_value.view(), comparison.common_bits);
536 previous_ = std::move(logical);
537 } catch (...) { failed_ = true; throw; }
538 }
539 // Trusted sorted merge output: spans describe the full logical key, but
540 // only the suffix after the known output LCP is copied to the file.
541 void append_frame(sort_profile_frame const & frame, std::span<bit_view const> spans,
542 std::uint64_t common, bit_view value) {
543 require_active();
544 try {
545 sort_bit_writer out(data_);
546 encoder_.append_frame(out, frame, spans, common, value);
547 encoded_only_ = true;
548 } catch (...) { failed_ = true; throw; }
549 }
550 std::uint64_t size() const noexcept { return encoder_.size(); }
551 bool failed() const noexcept { return failed_; }
552 bool finished() const noexcept { return finished_; }
553 array_type finish() {
554 require_active();
555 try {
556 encoder_.finish(data_.bit_size);
557 auto result = encoder_.take(std::move(data_));
558 finished_ = true; return result;
559 } catch (...) { failed_ = true; throw; }
560 }
561 private:
562 bit_string data_;
563 sort_profile_detail::encoder<P, Selector> encoder_;
564 bit_string previous_;
565 bool failed_ = false, finished_ = false, encoded_only_ = false;
566 void require_active() const {
567 if (failed_ || finished_) throw std::logic_error("inactive sort profile writer");
568 }
569 };

572 bit_string sort_profile_query(typename sort_codec<S>::key_codec::value_type const & key) {
573 using key_codec = sort_profile_key<typename sort_codec<S>::key_codec>;
574 auto encode = [](bit_view bits) {
575 bit_string result;
576 // Width is an optional selector hint. Selection still runs for each query.
577 if constexpr (requires { typename std::integral_constant<std::uint64_t, Selector::template code_size<S>>; }) {
578 auto size = profile_detail::add(Selector::template code_size<S>, bits.size());
579 auto bytes = profile_detail::add(size, 7) >> 3;
580 if (bytes > result.bytes.max_size()) throw std::length_error("sort query key too large");
581 result.bytes.reserve(static_cast<std::size_t>(bytes));
582 }
583 sort_bit_writer out(result);
584 Selector::template write<S>(out);
585 out.append(bits);
586 return result;
587 };
588 if constexpr (requires { key_codec::order_view(key); }) return encode(key_codec::order_view(key));
589 else {
590 auto bits = key_codec::order(key);
591 return encode(bits.view());
592 }
593 }
