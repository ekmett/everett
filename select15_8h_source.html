43 // Elias--Fano over residual byte offsets at records 0,15,30,..., then an
44 // actual-record-count sentinel. Residuals omit the fixed stride per record.
45 // Equal residual offsets are valid, including an entirely zero sequence.
46 //
47 // Select on the high bitvector is bounded: sample every 256 ones; scan at
48 // most 4096 bits (65 aligned words) for a dense group, or read an explicit
49 // exception position for a sparse group. There is no binary search. The
50 // exception array is O(n): disjoint sparse groups each span >4096 high bits,
51 // and Elias--Fano's high vector has O(n) bits. This is a practical baseline,
52 // not a tuned succinct select implementation.
53 // Views check shapes and guard navigation bounds. They do not establish
54 // semantic consistency of borrowed metadata: use a builder or validated
55 // reader. These native-endian spans are not a portable file-format parser.

57 select15_view(std::span<std::uint64_t const> low,
58 std::span<std::uint64_t const> high,
59 std::span<select15_sample const> samples,
60 std::span<std::uint64_t const> sparse,
61 std::uint64_t record_count, std::uint64_t universe,
62 unsigned low_width)
63 : low_(low), high_(high), samples_(samples), sparse_(sparse),
64 record_count_(record_count), universe_(universe), low_width_(low_width) {
65 if (low_width > 63) throw std::invalid_argument("select15 low width");
66 auto entries = group_count() + 1;
67 auto low_bits = select15_detail::multiply(entries, low_width);
68 high_bits_ = select15_detail::add(universe >> low_width, entries);
69 if (low.size() != select15_detail::words(low_bits) ||
70 high.size() != select15_detail::words(high_bits_) ||
71 samples.size() != entries / 256 + (entries % 256 != 0))

73 }

80 std::uint64_t residual(std::uint64_t group) const {
81 if (group > group_count()) throw std::out_of_range("select15 group");
82 auto position = select_high(group);
83 if (position < group) throw std::invalid_argument("invalid select15 high value");
84 auto hi = position - group;
85 if (hi > (universe_ >> low_width_)) throw std::invalid_argument("select15 high overflow");
86 std::uint64_t lo = 0;
87 if (low_width_) {
88 auto bit = group * low_width_;
89 auto word = bit / 64;
90 unsigned shift = unsigned(bit % 64);
91 lo = low_[word] >> shift;
92 if (shift + low_width_ > 64) lo |= low_[word + 1] << (64 - shift);
93 lo &= (std::uint64_t{1} << low_width_) - 1;
94 }
95 auto value = (hi << low_width_) | lo;
96 if (value > universe_) throw std::invalid_argument("select15 value exceeds universe");
97 return value;
98 }

107 std::uint64_t select_high(std::uint64_t ordinal) const {
108 auto sample = samples_[ordinal / 256];
109 unsigned remaining = unsigned(ordinal % 256);
110 if (sample.sparse != std::numeric_limits<std::uint64_t>::max()) {
111 if (sample.sparse > sparse_.size() || remaining >= sparse_.size() - sample.sparse)
112 throw std::invalid_argument("invalid select15 exception");
113 auto position = sparse_[sample.sparse + remaining];
114 if (position >= high_bits_ || !(high_[position / 64] & (std::uint64_t{1} << (position % 64))))
115 throw std::invalid_argument("invalid select15 sparse position");
116 return position;
117 }
118 if (sample.first >= high_bits_) throw std::invalid_argument("invalid select15 sample");
119 auto word = sample.first / 64;
120 auto value = high_[word] & (~std::uint64_t{0} << (sample.first % 64));
121 for (unsigned scanned = 0; scanned < 65 && word < high_.size(); ++scanned, ++word) {
122 if (scanned) value = high_[word];
123 auto population = unsigned(std::popcount(value));
124 if (remaining < population) {
125 for (; remaining; --remaining) value &= value - 1;
126 auto position = word * 64 + unsigned(std::countr_zero(value));
127 if (position >= high_bits_ || position - sample.first >= 4096)
128 throw std::invalid_argument("select15 dense span");
129 return position;
130 }
131 remaining -= population;
132 }
133 throw std::invalid_argument("select15 missing high bit");
134 }

147 static select15_index build(std::span<std::uint64_t const> residuals,
148 std::uint64_t records) {
149 auto groups = records / 15 + (records % 15 != 0);
150 if (residuals.size() != groups + 1)
151 throw std::invalid_argument("select15 residual count");
152 for (std::size_t i = 1; i < residuals.size(); ++i)
153 if (residuals[i] < residuals[i - 1])
154 throw std::invalid_argument("select15 nonmonotone offsets");
155 select15_index result;
156 result.record_count = records;
157 result.universe = residuals.back();
158 auto quotient = result.universe / residuals.size();
159 result.low_width = quotient ? unsigned(std::bit_width(quotient) - 1) : 0;
160 auto low_bits = select15_detail::multiply(residuals.size(), result.low_width);
161 auto high_bits = select15_detail::add(result.universe >> result.low_width, residuals.size());
162 result.low.assign(select15_detail::words(low_bits), 0);
163 result.high.assign(select15_detail::words(high_bits), 0);
164 result.samples.clear();
165 std::uint64_t low_mask = result.low_width ? (std::uint64_t{1} << result.low_width) - 1 : 0;
166 for (std::uint64_t i = 0; i < residuals.size(); ++i) {
167 auto value = residuals[i];
168 if (result.low_width) {
169 auto bit = i * result.low_width;
170 unsigned shift = unsigned(bit % 64);
171 result.low[bit / 64] |= (value & low_mask) << shift;
172 if (shift + result.low_width > 64)
173 result.low[bit / 64 + 1] |= (value & low_mask) >> (64 - shift);
174 }
175 auto position = (value >> result.low_width) + i;
176 result.high[position / 64] |= std::uint64_t{1} << (position % 64);
177 }
178 for (std::uint64_t begin = 0; begin < residuals.size(); begin += 256) {
179 auto end = residuals.size() - begin < 256 ? residuals.size() : begin + 256;
180 auto first = (residuals[begin] >> result.low_width) + begin;
181 auto last = (residuals[end - 1] >> result.low_width) + end - 1;
182 auto sparse = std::numeric_limits<std::uint64_t>::max();
183 if (last - first >= 4096) {
184 sparse = result.sparse.size();
185 for (auto i = begin; i < end; ++i)
186 result.sparse.push_back((residuals[i] >> result.low_width) + i);
187 }
188 result.samples.push_back({first, sparse});
189 }
190 return result;
191 }
