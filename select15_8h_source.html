45 // Elias--Fano over residual byte offsets at records 0,15,30,..., then an
46 // actual-record-count sentinel. Residuals omit the fixed stride per record.
47 // Equal residual offsets are valid, including an entirely zero sequence.
48 //
49 // Select on the high bitvector is bounded: sample every 256 ones; scan at
50 // most 4096 bits (65 aligned words) for a dense group, or read an explicit
51 // exception position for a sparse group. There is no binary search. The
52 // exception array is O(n): disjoint sparse groups each span >4096 high bits,
53 // and Elias--Fano's high vector has O(n) bits. This is a practical baseline,
54 // not a tuned succinct select implementation.
55 // Views check shapes and guard navigation bounds. They do not establish
56 // semantic consistency of borrowed metadata: use a builder or validated
57 // reader. Native spans and portable little-endian byte sections share navigation;
58 // neither constructor performs a complete semantic scan.

96 std::uint64_t residual(std::uint64_t group) const {
97 if (group > group_count()) throw std::out_of_range("select15 group");
98 auto position = select_high(group);
99 if (position < group) throw std::invalid_argument("invalid select15 high value");
100 auto hi = position - group;
101 if (hi > (universe_ >> low_width_)) throw std::invalid_argument("select15 high overflow");
102 std::uint64_t lo = 0;
103 if (low_width_) {
104 auto bit = group * low_width_;
105 auto word = bit / 64;
106 unsigned shift = unsigned(bit % 64);
107 lo = low_[word] >> shift;
108 if (shift + low_width_ > 64) lo |= low_[word + 1] << (64 - shift);
109 lo &= (std::uint64_t{1} << low_width_) - 1;
110 }
111 auto value = (hi << low_width_) | lo;
112 if (value > universe_) throw std::invalid_argument("select15 value exceeds universe");
113 return value;
114 }

123 std::uint64_t select_high(std::uint64_t ordinal) const {
124 auto sample = samples_[ordinal / 256];
125 unsigned remaining = unsigned(ordinal % 256);
126 if (sample.sparse != std::numeric_limits<std::uint64_t>::max()) {
127 if (sample.sparse > sparse_.size() || remaining >= sparse_.size() - sample.sparse)
128 throw std::invalid_argument("invalid select15 exception");
129 auto position = sparse_[sample.sparse + remaining];
130 if (position >= high_bits_ || !(high_[position / 64] & (std::uint64_t{1} << (position % 64))))
131 throw std::invalid_argument("invalid select15 sparse position");
132 return position;
133 }
134 if (sample.first >= high_bits_) throw std::invalid_argument("invalid select15 sample");
135 auto word = sample.first / 64;
136 auto value = high_[word] & (~std::uint64_t{0} << (sample.first % 64));
137 for (unsigned scanned = 0; scanned < 65 && word < high_.size(); ++scanned, ++word) {
138 if (scanned) value = high_[word];
139 auto population = unsigned(std::popcount(value));
140 if (remaining < population) {
141 auto position = word * 64 + select_groups_detail::select_word(value, remaining);
142 if (position >= high_bits_ || position - sample.first >= 4096)
143 throw std::invalid_argument("select15 dense span");
144 return position;
145 }
146 remaining -= population;
147 }
148 throw std::invalid_argument("select15 missing high bit");
149 }

161 struct select15_index {
162 static select15_index build(std::span<std::uint64_t const> residuals,
163 std::uint64_t records) {
164 auto groups = records / 15 + (records % 15 != 0);
165 if (residuals.size() != groups + 1)
166 throw std::invalid_argument("select15 residual count");
167 if (!select_groups_detail::monotone(residuals))
168 throw std::invalid_argument("select15 nonmonotone offsets");
169 select15_index result;
170 result.record_count = records;
171 result.universe = residuals.back();
172 auto quotient = result.universe / residuals.size();
173 result.low_width = quotient ? unsigned(std::bit_width(quotient) - 1) : 0;
174 auto low_bits = select15_detail::multiply(residuals.size(), result.low_width);
175 auto high_bits = select15_detail::add(result.universe >> result.low_width, residuals.size());
176 result.low.assign(select15_detail::words(low_bits), 0);
177 result.high.assign(select15_detail::words(high_bits), 0);
178 result.samples.clear();
179 select_groups_detail::pack_low(residuals, result.low, result.low_width);
180 select_groups_detail::write_high(residuals, result.high, result.low_width);
181 for (std::uint64_t begin = 0; begin < residuals.size(); begin += 256) {
182 auto end = residuals.size() - begin < 256 ? residuals.size() : begin + 256;
183 auto first = (residuals[begin] >> result.low_width) + begin;
184 auto last = (residuals[end - 1] >> result.low_width) + end - 1;
185 auto sparse = std::numeric_limits<std::uint64_t>::max();
186 if (last - first >= 4096) {
187 sparse = result.sparse.size();
188 for (auto i = begin; i < end; ++i)
189 result.sparse.push_back((residuals[i] >> result.low_width) + i);
190 }
191 result.samples.push_back({first, sparse});
192 }
193 return result;
194 }
