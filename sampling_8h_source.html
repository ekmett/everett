51 template <class P> void check_ordinal(std::uint64_t count, std::uint64_t ordinal) {
52 if (count > std::numeric_limits<std::uint64_t>::max() / P::group_size || ordinal != count * P::group_size)
53 error_detail::raise<std::invalid_argument>("sample ordinals must be consecutive policy groups");
54 }

56 // Suffix must not alias context. Allocate before changing its retained
57 // prefix, then the bounded resize/copy operations cannot allocate or fail.
58 // Preserving a prefix does not copy it unless the buffer must grow; growth
59 // is geometric so gradually lengthening keys do not reallocate each time.
60 inline void replace_suffix(bit_string & context, std::uint64_t retained, bit_view suffix) {
61 auto bits = profile_detail::add(retained, suffix.size());
62 if (bits > std::numeric_limits<std::uint64_t>::max() - 7)
63 error_detail::raise<std::length_error>("sample key is too large");
64 auto bytes = profile_detail::byte_count(bits);
65 if (bytes > context.bytes.max_size()) error_detail::raise<std::length_error>("sample key is too large");
66 if (bytes > context.bytes.capacity()) {
67 auto capacity = context.bytes.capacity();
68 auto grown = capacity > (context.bytes.max_size() >> 1) ? context.bytes.max_size() : capacity << 1;
69 context.bytes.reserve(std::max(static_cast<std::size_t>(bytes), grown));
70 }
71 profile_detail::resize(context, retained);
72 profile_detail::append(context, suffix);
73 }
74 }

76 // One reusable reconstructed key per endpoint. Validation/allocation failure
77 // leaves the preceding accepted key and ordinal intact. key() borrows that
78 // context until the next successful encode/accept, move, or destruction.
79 template <class P> struct profile_sample_encoder {
80 using policy_type = P;
81
82 profile_coded_sample<P> encode(bit_view key, std::uint64_t target_ordinal) {
83 sampling_detail::check_ordinal<P>(count_, target_ordinal);
84 if (key.size() & (P::bits_per_unit - 1)) error_detail::raise<std::invalid_argument>("sample key unit mismatch");
85 auto previous = key_.view();
86 auto comparison = compare_common_bits(previous, key);
87 if (comparison.order > 0) error_detail::raise<std::invalid_argument>("sample keys must be sorted");
88 return encode_known(key, target_ordinal, comparison.common_bits);
89 }
90
91 bit_view key() const & { return key_.view(); }
92 bit_view key() const && = delete;
93 std::uint64_t size() const noexcept { return count_; }
94
95 private:
96 template <class, class, class> friend struct index_builder;
97 // The builder supplies exact comparison state for sorted, unit-aligned
98 // keys and consecutive output ordinals; the public entry remains checked.
99 profile_coded_sample<P> encode_known(bit_view key, std::uint64_t target_ordinal,
100 std::uint64_t common_bits) {
101 auto retained = common_bits >> P::unit_shift;
102 auto retained_bits = profile_detail::multiply(retained, P::bits_per_unit);
103 // Copy the transmitted suffix before editing context. Input may alias
104 // this encoder's current key, including a subview of that key.
105 profile_coded_sample<P> result{(key_.bit_size >> P::unit_shift) - retained,
106 bit_string::copy(key.subview(retained_bits, key.size() - retained_bits)), target_ordinal};
107 sampling_detail::replace_suffix(key_, retained_bits, result.suffix.view());
108 ++count_;
109 return result;
110 }
111
112 bit_string key_;
113 std::uint64_t count_ = 0;
114 };

117 using policy_type = P;
118
119 bit_view accept(profile_coded_sample<P> const & sample) { return accept_compared(sample).first; }

121 bit_view key() const & { return key_.view(); }

126 template <class, class, class> friend struct index_builder;
127 // Only the index builder uses this full-key path. Its input cannot borrow
128 // the private decoder context; reserve before editing its retained prefix.
129 bit_comparison accept_full(bit_view key, std::uint64_t ordinal) {
130 sampling_detail::check_ordinal<P>(count_, ordinal);
131 if (key.size() & (P::bits_per_unit - 1))
132 error_detail::raise<std::invalid_argument>("sample key unit mismatch");
133 auto comparison = compare_common_bits(key_.view(), key);
134 if (comparison.order > 0)
135 error_detail::raise<std::invalid_argument>("sample keys must be sorted");
136 auto retained = comparison.common_bits & ~std::uint64_t{7};
137 sampling_detail::replace_suffix(key_, retained, key.subview(retained, key.size() - retained));
138 ++count_;
139 return comparison;
140 }
141 std::pair<bit_view, bit_comparison> accept_compared(profile_coded_sample<P> const & sample) {
142 sampling_detail::check_ordinal<P>(count_, sample.target_ordinal);
143 auto suffix = sample.suffix.view();
144 if (suffix.size() & (P::bits_per_unit - 1)) error_detail::raise<std::invalid_argument>("sample suffix unit mismatch");
145 auto previous = key_.view();
146 auto previous_units = previous.size() >> P::unit_shift;
147 if (sample.backspace > previous_units) error_detail::raise<std::invalid_argument>("sample backspace exceeds previous key");
148 auto retained = profile_detail::multiply(previous_units - sample.backspace, P::bits_per_unit);
149 // Both keys share the retained prefix. Comparing only the two remaining
150 // suffixes validates order without another full-key reconstruction.
151 auto comparison = compare_common_bits(previous.subview(retained, previous.size() - retained), suffix);
152 if (comparison.order > 0)
153 error_detail::raise<std::invalid_argument>("sample keys must be sorted");
154 comparison.common_bits += retained;
155 sampling_detail::replace_suffix(key_, retained, suffix);
156 ++count_;
157 return {key_.view(), comparison};
158 }
159
160 bit_string key_;
161 std::uint64_t count_ = 0;
162 };

173 // Sequential samples of one exact, immutable native/index pair. Both streams
174 // retain their own front-coding context. Construction decodes at most two
175 // records; advance consumes at most P::group_size merged occurrences. Native
176 // entries precede equal borrowed entries, and borrowed duplicates remain.
177 //
178 // These are record-work bounds: decoding/comparison still pays for key bits.
179 // Scratch space holds two current reconstructed keys. Values are never
180 // copied, and no array of reconstructed keys or samples is materialized.
181 // peek() borrows cursor scratch until the next advance, move, or destruction.
182 // A moved-to cursor remains valid; fresh peek() calls derive fresh key views.
183 template <class P, class Target = profile_blob<P>> struct sample_cursor {
184 using policy_type = P;
185 using target_type = Target;
186 static_assert(std::same_as<typename Target::policy_type, P>);
187 static constexpr std::uint64_t group_size = P::group_size;
188
189 explicit sample_cursor(std::shared_ptr<target_type const> target)
190 : sample_cursor(bind_target(std::move(target))) {}
191
192 bool done() const noexcept { return !target_ || (native_.done() && borrowed_.done()); }
193
194 profile_sample_view<P> peek() const & {
195 if (done()) error_detail::raise<std::out_of_range>("sample cursor at end");
196 auto item = next_borrowed_ ? borrowed_.peek() : native_.peek();
197 return {item.key.prefix, ordinal_, next_borrowed_ ? stream_role::borrowed : stream_role::native,
198 item.ordinal};
199 }
200 profile_sample_view<P> peek() const && = delete;
201
202 void advance() {
203 if (done()) error_detail::raise<std::out_of_range>("sample cursor at end");
204 auto count = std::min(group_size, target_->virtual_size() - ordinal_);
205 for (std::uint64_t i = 0; i != count; ++i) {
206 if (next_borrowed_) {
207 borrowed_.advance();
208 ++work_.borrowed_entries;
209 if (!borrowed_.done()) ++work_.decoded_entries;
210 } else {
211 native_.advance();
212 ++work_.native_entries;
213 if (!native_.done()) ++work_.decoded_entries;
214 }
215 ++ordinal_;
216 choose_next();
217 }
218 }

220 std::shared_ptr<target_type const> target() const noexcept { return target_; }
221 sampling_work const & counters() const noexcept { return work_; }

228 static binding bind_target(std::shared_ptr<target_type const> target) {
229 if (!target) error_detail::raise<std::invalid_argument>("sample cursor requires a pinned target");
230 auto view = target->view();
231 return {std::move(target), view};
232 }
233 explicit sample_cursor(binding source)
234 : target_(std::move(source.target)), native_(source.view.native()), borrowed_(source.view.borrowed()) {
235 work_.decoded_entries = std::uint64_t(!native_.done()) + std::uint64_t(!borrowed_.done());
236 choose_next();
237 }
238
239 void choose_next() {
240 if (native_.done()) next_borrowed_ = true;
241 else if (borrowed_.done()) next_borrowed_ = false;
242 else {
243 ++work_.key_comparisons;
244 next_borrowed_ = compare_bits(borrowed_.peek().key.prefix, native_.peek().key.prefix) < 0;
245 }
246 }

251 sampling_work work_;
252 std::uint64_t ordinal_ = 0;
253 bool next_borrowed_ = false;
