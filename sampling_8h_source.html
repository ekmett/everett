48 template <class P> void check_ordinal(std::uint64_t count, std::uint64_t ordinal) {
49 if (count > std::numeric_limits<std::uint64_t>::max() / P::group_size || ordinal != count * P::group_size)
50 throw std::invalid_argument("sample ordinals must be consecutive policy groups");
51 }

53 // Suffix must not alias context. Allocate before changing its retained
54 // prefix, then the bounded resize/copy operations cannot allocate or fail.
55 // Preserving a prefix does not copy it unless the buffer must grow; growth
56 // is geometric so gradually lengthening keys do not reallocate each time.
57 inline void replace_suffix(bit_string & context, std::uint64_t retained, bit_view suffix) {
58 auto bits = profile_detail::add(retained, suffix.size());
59 auto bytes = profile_detail::byte_count(bits);
60 if (bytes > context.bytes.max_size()) throw std::length_error("sample key is too large");
61 if (bytes > context.bytes.capacity()) {
62 auto capacity = context.bytes.capacity();
63 auto grown = capacity > context.bytes.max_size() / 2 ? context.bytes.max_size() : capacity * 2;
64 context.bytes.reserve(std::max(static_cast<std::size_t>(bytes), grown));
65 }
66 profile_detail::resize(context, retained);
67 profile_detail::append(context, suffix);
68 }
69 }

71 // One reusable reconstructed key per endpoint. Validation/allocation failure
72 // leaves the preceding accepted key and ordinal intact. key() borrows that
73 // context until the next successful encode/accept, move, or destruction.
74 template <class P> struct profile_sample_encoder {
75 using policy_type = P;
76
77 profile_coded_sample<P> encode(bit_view key, std::uint64_t target_ordinal) {
78 sampling_detail::check_ordinal<P>(count_, target_ordinal);
79 if (key.size() % P::bits_per_unit) throw std::invalid_argument("sample key unit mismatch");
80 auto previous = key_.view();
81 if (compare_bits(previous, key) > 0) throw std::invalid_argument("sample keys must be sorted");
82 auto retained = common_prefix_units<P>(previous, key);
83 auto retained_bits = profile_detail::multiply(retained, P::bits_per_unit);
84 // Copy the transmitted suffix before editing context. Input may alias
85 // this encoder's current key, including a subview of that key.
86 profile_coded_sample<P> result{previous.size() / P::bits_per_unit - retained,
87 bit_string::copy(key.subview(retained_bits, key.size() - retained_bits)), target_ordinal};
88 sampling_detail::replace_suffix(key_, retained_bits, result.suffix.view());
89 ++count_;
90 return result;
91 }
92
93 bit_view key() const & { return key_.view(); }
94 bit_view key() const && = delete;
95 std::uint64_t size() const noexcept { return count_; }
96
97 private:
98 bit_string key_;
99 std::uint64_t count_ = 0;
100 };

102 template <class P> struct profile_sample_decoder {
103 using policy_type = P;
104
105 bit_view accept(profile_coded_sample<P> const & sample) {
106 sampling_detail::check_ordinal<P>(count_, sample.target_ordinal);
107 auto suffix = sample.suffix.view();
108 if (suffix.size() % P::bits_per_unit) throw std::invalid_argument("sample suffix unit mismatch");
109 auto previous = key_.view();
110 auto previous_units = previous.size() / P::bits_per_unit;
111 if (sample.backspace > previous_units) throw std::invalid_argument("sample backspace exceeds previous key");
112 auto retained = profile_detail::multiply(previous_units - sample.backspace, P::bits_per_unit);
113 // Both keys share the retained prefix. Comparing only the two remaining
114 // suffixes validates order without another full-key reconstruction.
115 if (compare_bits(suffix, previous.subview(retained, previous.size() - retained)) < 0)
116 throw std::invalid_argument("sample keys must be sorted");
117 sampling_detail::replace_suffix(key_, retained, suffix);
118 ++count_;
119 return key_.view();
120 }
121
122 bit_view key() const & { return key_.view(); }
123 bit_view key() const && = delete;
124 std::uint64_t size() const noexcept { return count_; }
125
126 private:
127 bit_string key_;
128 std::uint64_t count_ = 0;
129 };

140 // Sequential samples of one exact, immutable native/index pair. Both streams
141 // retain their own front-coding context. Construction decodes at most two
142 // records; advance consumes at most P::group_size merged occurrences. Native
143 // entries precede equal borrowed entries, and borrowed duplicates remain.
144 //
145 // These are record-work bounds: decoding/comparison still pays for key bits.
146 // Scratch space holds two current reconstructed keys. Values are never
147 // copied, and no array of reconstructed keys or samples is materialized.
148 // peek() borrows cursor scratch until the next advance, move, or destruction.
149 // A moved-to cursor remains valid; fresh peek() calls derive fresh key views.
150 template <class P> struct sample_cursor {
151 using policy_type = P;
152 using target_type = profile_blob<P>;
153 static constexpr std::uint64_t group_size = P::group_size;
154
155 explicit sample_cursor(std::shared_ptr<target_type const> target)
156 : target_(checked_target(std::move(target))),
157 native_(target_->native().view()), borrowed_(target_->borrowed().view()) {
158 work_.decoded_entries = std::uint64_t(!native_.done()) + std::uint64_t(!borrowed_.done());
159 choose_next();
160 }
161
162 bool done() const noexcept { return native_.done() && borrowed_.done(); }
163
164 profile_sample_view<P> peek() const & {
165 if (done()) throw std::out_of_range("sample cursor at end");
166 auto item = next_borrowed_ ? borrowed_.peek() : native_.peek();
167 return {item.key.prefix, ordinal_, next_borrowed_ ? stream_role::borrowed : stream_role::native,
168 item.ordinal};
169 }
170 profile_sample_view<P> peek() const && = delete;
171
172 void advance() {
173 if (done()) throw std::out_of_range("sample cursor at end");
174 auto count = std::min(group_size, target_->virtual_size() - ordinal_);
175 for (std::uint64_t i = 0; i != count; ++i) {
176 if (next_borrowed_) {
177 borrowed_.advance();
178 ++work_.borrowed_entries;
179 if (!borrowed_.done()) ++work_.decoded_entries;
180 } else {
181 native_.advance();
182 ++work_.native_entries;
183 if (!native_.done()) ++work_.decoded_entries;
184 }
185 ++ordinal_;
186 choose_next();
187 }
188 }
189
190 std::shared_ptr<target_type const> target() const noexcept { return target_; }
191 sampling_work const & counters() const noexcept { return work_; }
192
193 private:
194 static std::shared_ptr<target_type const> checked_target(std::shared_ptr<target_type const> target) {
195 if (!target) throw std::invalid_argument("sample cursor requires a pinned target");
196 return target;
197 }
198
199 void choose_next() {
200 if (native_.done()) next_borrowed_ = true;
201 else if (borrowed_.done()) next_borrowed_ = false;
202 else {
203 ++work_.key_comparisons;
204 next_borrowed_ = compare_bits(borrowed_.peek().key.prefix, native_.peek().key.prefix) < 0;
205 }
206 }
207
208 std::shared_ptr<target_type const> target_;
209 profile_cursor<P, stream_role::native> native_;
210 profile_cursor<P, stream_role::borrowed> borrowed_;
211 sampling_work work_;
212 std::uint64_t ordinal_ = 0;
213 bool next_borrowed_ = false;
214 };
