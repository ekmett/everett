25 // A local destination plan pins exact existing objects. A new main retains
26 // the next level's targets; a new secondary joins an existing main and is a
27 // native-only leaf. This plan assigns no logical slots or admission mass.
28 template <class P> struct cola_destination_plan {
29 using policy_type = P;
30 using pair_type = typename cola_index<P>::pair_type;
31 using native_pointer = typename cola_index<P>::native_pointer;
32 static cola_destination_plan for_main(pair_type main = {}, native_pointer secondary = {}) {
33 if (secondary && !main)
34 error_detail::raise<std::invalid_argument>("COLA destination secondary requires a main");
35 return {cola_destination_kind::main, std::move(main), std::move(secondary)};
36 }
37 static cola_destination_plan for_secondary(pair_type main) {
38 if (!main) error_detail::raise<std::invalid_argument>("COLA secondary destination requires an existing main");
39 return {cola_destination_kind::secondary, std::move(main), {}};
40 }
41 cola_destination_kind kind() const noexcept { return kind_; }
42 pair_type main_target() const noexcept { return main_; }
43 native_pointer secondary_target() const noexcept { return secondary_; }
44 private:
45 cola_destination_kind kind_;
46 pair_type main_;
47 native_pointer secondary_;
48 cola_destination_plan(cola_destination_kind kind, pair_type main, native_pointer secondary)
49 : kind_(kind), main_(std::move(main)), secondary_(std::move(secondary)) {}
50 };

62 // One local COLA job, with explicit native/index/carrier completion. Inputs
63 // must represent adjacent chronological history in older,newer order; sorted
64 // keys cannot establish that semantic premise. No key is elided.
65 //
66 // step counts distinct native keys or augmented index occurrences in only
67 // the current stage. It never performs finish_stage implicitly. Construction
68 // and finish_stage may decode initial heads, allocate, and finalize EF/rank;
69 // their work is not bounded by the record budget. A secondary has no index
70 // stage. The ready carrier is a local empty-native routing artifact, not
71 // necessarily a prepared query root. No slots, visibility, publication,
72 // durable progress or scheduler service guarantee are implemented here.
73 //
74 // Source and plan owners remain pinned for the job's lifetime, including
75 // after failure. A failed execution/finalization poisons this job; a rejected
76 // precondition does not. Moves transfer stable builder objects without
77 // moving user callback state. Assignment releases the previous job's pins.
78 template <class P, class Compose = replace_native_value> struct cola_local_merge_job {
79 using policy_type = P;
80 using index_type = cola_index<P>;
81 using native_type = profile_array<P>;
82 using native_pointer = typename index_type::native_pointer;
83 using pair_type = typename index_type::pair_type;
84 using plan_type = cola_destination_plan<P>;
85 using result_type = cola_local_merge_result<P>;
86 using merge_type = native_merge_builder<P, native_type, Compose>;
87 using index_builder_type = cola_index_builder<P>;

98 cola_merge_phase phase() const noexcept { return older_ ? phase_ : cola_merge_phase::inactive; }
99 bool failed() const noexcept { return failed_; }
100 bool finished() const noexcept { return older_ && phase_ == cola_merge_phase::taken; }
101 bool done() const noexcept { return older_ && !failed_ && phase_ == cola_merge_phase::ready; }
102 bool stage_done() const noexcept {
103 if (!older_ || failed_ || finished()) return false;
104 if (phase_ == cola_merge_phase::native_merge) return merge_->done();
105 if (phase_ == cola_merge_phase::destination_index || phase_ == cola_merge_phase::carrier_index)
106 return index_->done();
107 return phase_ == cola_merge_phase::ready;
108 }
109 std::uint64_t step(std::uint64_t budget = 1) {
110 require_active();
111 try {
112 if (phase_ == cola_merge_phase::native_merge) return merge_->step(budget).keys;
113 if (phase_ == cola_merge_phase::destination_index || phase_ == cola_merge_phase::carrier_index)
114 return index_->step(budget);
115 return 0;
116 } catch (...) { failed_ = true; throw; }
117 }
118 void finish_stage() {
119 require_active();
120 if (phase_ == cola_merge_phase::ready || !stage_done())
121 error_detail::raise<std::logic_error>("COLA local stage is not ready for finalization");
122 try {
123 if (phase_ == cola_merge_phase::native_merge) {
124 merged_ = std::make_shared<native_type const>(merge_->finish());
125 merge_.reset();
126 if (plan_.kind() == cola_destination_kind::main) {
127 index_ = std::make_unique<index_builder_type>(merged_, plan_.main_target(), plan_.secondary_target());
128 phase_ = cola_merge_phase::destination_index;
129 } else {
130 main_ = plan_.main_target();
131 secondary_ = merged_;
132 start_carrier();
133 }
134 } else if (phase_ == cola_merge_phase::destination_index) {
135 main_ = std::make_shared<index_type const>(index_->finish());
136 index_.reset();
137 start_carrier();
138 } else {
139 carrier_ = std::make_shared<index_type const>(index_->finish());
140 index_.reset();
141 phase_ = cola_merge_phase::ready;
142 }
143 } catch (...) { failed_ = true; throw; }
144 }
145 result_type finish() {
146 require_active();
147 if (!done()) error_detail::raise<std::logic_error>("COLA local merge is not ready");
148 phase_ = cola_merge_phase::taken;
149 return {std::move(merged_), std::move(main_), std::move(secondary_), std::move(carrier_)};
150 }

155 std::unique_ptr<index_builder_type> index_;
