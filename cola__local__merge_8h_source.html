28 // A local destination plan pins exact existing objects. A new main retains
29 // the next level's targets; a new secondary joins an existing main and is a
30 // native-only leaf. This plan assigns no logical slots or admission mass.
31 template <class P> struct cola_destination_plan {
32 using policy_type = P;
33 using pair_type = typename cola_index<P>::pair_type;
34 using native_pointer = typename cola_index<P>::native_pointer;
35 static cola_destination_plan for_main(pair_type main = {}, native_pointer secondary = {}) {
36 if (secondary && !main)
37 error_detail::raise<std::invalid_argument>("COLA destination secondary requires a main");
38 return {cola_destination_kind::main, std::move(main), std::move(secondary)};
39 }
40 static cola_destination_plan for_secondary(pair_type main) {
41 if (!main) error_detail::raise<std::invalid_argument>("COLA secondary destination requires an existing main");
42 return {cola_destination_kind::secondary, std::move(main), {}};
43 }
44 cola_destination_kind kind() const noexcept { return kind_; }
45 pair_type main_target() const noexcept { return main_; }
46 native_pointer secondary_target() const noexcept { return secondary_; }
47 private:
48 cola_destination_kind kind_;
49 pair_type main_;
50 native_pointer secondary_;
51 cola_destination_plan(cola_destination_kind kind, pair_type main, native_pointer secondary)
52 : kind_(kind), main_(std::move(main)), secondary_(std::move(secondary)) {}
53 };

65 // One local COLA job, with explicit native/index/carrier completion. Inputs
66 // must represent adjacent chronological history in older,newer order; sorted
67 // keys cannot establish that semantic premise. No key is elided.
68 //
69 // step counts distinct native keys or augmented index occurrences in only
70 // the current stage. It never performs finish_stage implicitly. Construction
71 // and finish_stage may decode initial heads, allocate, and finalize EF/rank;
72 // their work is not bounded by the record budget. A secondary has no index
73 // stage. The ready carrier is a local empty-native routing artifact, not
74 // necessarily a prepared query root. No slots, visibility, publication,
75 // durable progress or scheduler service guarantee are implemented here.
76 //
77 // Source and plan owners remain pinned for the job's lifetime, including
78 // after failure. A failed execution/finalization poisons this job; a rejected
79 // precondition does not. Moves transfer stable builder objects without
80 // moving user callback state. Assignment releases the previous job's pins.

84 using native_type = profile_array<P>;
85 using native_pointer = typename index_type::native_pointer;
86 using pair_type = typename index_type::pair_type;
87 using plan_type = cola_destination_plan<P>;
88 using result_type = cola_local_merge_result<P>;
89 using merge_type = native_merge_builder<P, native_type, Compose>;
90 using index_builder_type = cola_index_builder<P>;

101 cola_merge_phase phase() const noexcept { return older_ ? phase_ : cola_merge_phase::inactive; }
102 bool failed() const noexcept { return failed_; }
103 bool finished() const noexcept { return older_ && phase_ == cola_merge_phase::taken; }
104 bool done() const noexcept { return older_ && !failed_ && phase_ == cola_merge_phase::ready; }
105 bool stage_done() const noexcept {
106 if (!older_ || failed_ || finished()) return false;
107 if (phase_ == cola_merge_phase::native_merge) return merge_->done();
108 if (phase_ == cola_merge_phase::destination_index || phase_ == cola_merge_phase::carrier_index)
109 return index_->done();
110 return phase_ == cola_merge_phase::ready;
111 }
112 std::uint64_t step(std::uint64_t budget = 1) {
113 require_active();
114 try {
115 if (phase_ == cola_merge_phase::native_merge) return merge_->step(budget).keys;
116 if (phase_ == cola_merge_phase::destination_index || phase_ == cola_merge_phase::carrier_index)
117 return index_->step(budget);
118 return 0;
119 } catch (...) { failed_ = true; throw; }
120 }
121 void finish_stage() {
122 require_active();
123 if (phase_ == cola_merge_phase::ready || !stage_done())
124 error_detail::raise<std::logic_error>("COLA local stage is not ready for finalization");
125 try {
126 if (phase_ == cola_merge_phase::native_merge) {
127 merged_ = std::make_shared<native_type const>(merge_->finish());
128 merge_.reset();
129 if (plan_.kind() == cola_destination_kind::main) {
130 index_ = std::make_unique<index_builder_type>(merged_, plan_.main_target(), plan_.secondary_target());
131 phase_ = cola_merge_phase::destination_index;
132 } else {
133 main_ = plan_.main_target();
134 secondary_ = merged_;
135 start_carrier();
136 }
137 } else if (phase_ == cola_merge_phase::destination_index) {
138 main_ = std::make_shared<index_type const>(index_->finish());
139 index_.reset();
140 start_carrier();
141 } else {
142 carrier_ = std::make_shared<index_type const>(index_->finish());
143 index_.reset();
144 phase_ = cola_merge_phase::ready;
145 }
146 } catch (...) { failed_ = true; throw; }
147 }
148 result_type finish() {
149 require_active();
150 if (!done()) error_detail::raise<std::logic_error>("COLA local merge is not ready");
151 phase_ = cola_merge_phase::taken;
152 return {std::move(merged_), std::move(main_), std::move(secondary_), std::move(carrier_)};
153 }
