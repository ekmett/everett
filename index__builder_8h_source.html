27 // Incremental ordinary-FC index construction over an unchanged native array.
28 // One incoming and one outgoing sample provide explicit backpressure. The
29 // source wrapper may expire: native_ retains its actual encoded allocation.
30 // step() budgets merged occurrences, not key bytes, allocations or final EF
31 // construction. Inputs must come from the exact target's trusted sampler;
32 // finish() checks its count but does not rescan it to authenticate every key.
33 // Coded handoff queues a backspace count, suffix and target ordinal. Backspace
34 // counts P units relative to the preceding emitted sample. Decoder, lookahead,
35 // pending-writer and outgoing-encoder contexts still retain full current keys.
36 template <class P>
37 struct index_builder {
38 using policy_type = P;
39 using blob_type = profile_blob<P>;
40 using sample_type = profile_sample<P>;
41 using coded_sample_type = profile_coded_sample<P>;

63 void push(bit_view key, std::uint64_t target_ordinal) {
64 check_active();
65 if (input_mode_ == input_mode::coded) error_detail::raise<std::logic_error>("cannot mix coded and full index inputs");
66 push_key(key, target_ordinal);

70 void push(coded_sample_type const & sample) {
71 check_active();
72 if (input_mode_ == input_mode::full) error_detail::raise<std::logic_error>("cannot mix coded and full index inputs");
73 check_input_slot(sample.target_ordinal);
74 auto key = incoming_decoder_.accept(sample);
75 try {
76 // The decoder's current key is copied once into the existing lookahead
77 // state. Failure after advancing that context makes this stage unusable.
78 push_key(key, sample.target_ordinal);
79 } catch (...) {
80 failed_ = true;
81 outgoing_.reset();
82 throw;
83 }
84 input_mode_ = input_mode::coded;
85 }

95 std::uint64_t step(std::uint64_t budget_entries) {
96 check_active();
97 std::uint64_t consumed = 0;
98 try {
99 while (consumed < budget_entries && !outgoing_) {
100 if ((!incoming_ && !input_closed_) || (!incoming_ && native_cursor_.done())) break;
101 auto take_borrowed = native_cursor_.done();
102 if (incoming_ && !native_cursor_.done()) {
103 auto order = compare_bits(incoming_->view(), native_cursor_.peek().key.prefix);
104 take_borrowed = order < 0;
105 if (!order) incoming_false_ = true;
106 }
107 auto key = take_borrowed ? incoming_->view() : native_cursor_.peek().key.prefix;
108 auto boundary = virtual_count_ % group_size == 0;
109 if (boundary) {
110 outgoing_.emplace(outgoing_encoder_.encode(key, virtual_count_));
111 classes_.push_back(0);
112 cut_lcps_.push_back(pending_ ? compare_common_bits(pending_->view(), key).common_bits : 0);
113 }
114 if (take_borrowed) {
115 flush_pending();
116 pending_ = std::move(incoming_);
117 incoming_.reset();
118 pending_false_ = incoming_false_;
119 incoming_false_ = false;
120 auto ordinal = borrowed_count_++;
121 if ((ordinal & 7) == 0) false_borrows_.push_back(std::byte{0});
122 if (pending_false_) false_borrows_.back() |= std::byte(1u << (ordinal & 7));
123 ++classes_.back();
124 } else {
125 native_cursor_.advance();
126 }
127 ++virtual_count_;
128 ++consumed;
129 }
130 } catch (...) {
131 // Partial encoding after an allocation/codec failure is private and
132 // cannot be resumed or published as though the failed step committed.
133 failed_ = true;
134 outgoing_.reset();
135 throw;
136 }
137 return consumed;
138 }

150 // through the full-key compatibility interface. Consume coded frames in order.

154 auto result = std::move(*outgoing_);
155 outgoing_.reset();
156 return result;
157 }

159 // Finalizes encoded offset/rank metadata; this can perform linear work.
160 // A nonempty sampled stream requires the exact completed target pair.
161 // Empty targets may be retained too. No durable publication is implied.
162 blob_type finish(std::shared_ptr<blob_type const> target = {}) {
163 check_active();
164 if (!done()) error_detail::raise<std::logic_error>("index builder is not drained at EOF");
165 auto target_count = target ? target->virtual_size() : 0;
166 auto expected = target_count / group_size + (target_count % group_size != 0);
167 if (expected != received_) error_detail::raise<std::invalid_argument>("index samples disagree with target extent");
168 try {
169 flush_pending();
170 blob_type result;
171 result.native_ = native_;
172 result.borrowed_ = writer_.finish();
173 result.interleave_ = rank_groups<group_size>::build(classes_, virtual_count_);
174 result.false_borrows_ = std::move(false_borrows_);
175 result.virtual_count_ = virtual_count_;
176 result.cut_lcps_ = std::move(cut_lcps_);
177 result.target_ = std::move(target);
178 finished_ = true;
179 return result;
180 } catch (...) {
181 failed_ = true;
182 throw;
183 }

213 void check_input_slot(std::uint64_t target_ordinal) const {
214 if (!needs_input()) error_detail::raise<std::logic_error>("index builder cannot accept another lookahead");
215 if (received_ > std::numeric_limits<std::uint64_t>::max() / group_size ||
216 target_ordinal != received_ * group_size)
217 error_detail::raise<std::invalid_argument>("index samples must name consecutive target groups");
218 if (received_ == std::numeric_limits<std::uint64_t>::max() - native_->size())
219 error_detail::raise<std::length_error>("index augmented count overflows");
220 }
221 void push_key(bit_view key, std::uint64_t target_ordinal) {
222 check_input_slot(target_ordinal);
223 if (key.size() & (P::bits_per_unit - 1))
224 error_detail::raise<std::invalid_argument>("index sample key disagrees with policy units");
225 auto order = pending_ ? compare_bits(pending_->view(), key) : -1;
226 if (order > 0) error_detail::raise<std::invalid_argument>("index samples must be sorted");
227 auto copy = bit_string::copy(key);
228 incoming_.emplace(std::move(copy));
229 incoming_false_ = !order && pending_false_;
230 ++received_;
231 }
232 void flush_pending() {
233 if (pending_) {
234 writer_.append(pending_->view());
235 pending_.reset();
236 }
237 }
