25 // Build indexes from the target outward. Each stage retains one input and
26 // one output sample; sample keys flow directly into the next stage without
27 // materializing an intermediate catalog or rescanning a completed index.
28 // Handoffs carry a policy-unit backspace count and suffix relative to the
29 // preceding sample from that same producer, with a literal first sample.
30 //
31 // A step quantum consumes one augmented occurrence, transfers one sample,
32 // advances the source sampler by at most K occurrences, or communicates EOF.
33 // This is an entry-work budget, not a bound on bytes or wall-clock time.
34 // Each owner separately finalizes its stages and binds exact target identities.
35 template <class P, class Target, class Stage>
36 struct pipeline_driver {
37 using policy_type = P;
38 using blob_type = Target;
39 using pair_type = std::shared_ptr<blob_type const>;

44 emitted_.resize(stages_.size());

47 }

61 std::uint64_t step(std::uint64_t quanta) {
62 if (finished_ || failed_) error_detail::raise<std::logic_error>("pipeline is no longer active");
63 std::uint64_t work = 0;
64 try {
65 while (work != quanta && !done()) {
66 if (!advance_one()) error_detail::raise<std::logic_error>("index pipeline made no progress");
67 ++work;
68 }
69 } catch (...) {
70 failed_ = true;
71 throw;
72 }
73 return work;
74 }

99 bool advance_one() {
100 // Give consumers the first opportunity to release backpressure.
101 for (std::size_t end = stages_.size(); end; --end) {
102 auto i = end - 1;
103 auto & stage = *stages_[i];
104 if (stage.has_output()) {
105 if (i + 1 == stages_.size()) {
106 auto sample = stage.take_coded_output();
107 emitted_suffix_units_[i] = profile_detail::add(
108 emitted_suffix_units_[i], (sample.suffix.bit_size >> P::unit_shift));
109 } else {
110 auto & next = *stages_[i + 1];
111 if (!next.needs_input()) continue;
112 auto sample = stage.take_coded_output();
113 next.push(sample);
114 emitted_suffix_units_[i] = profile_detail::add(
115 emitted_suffix_units_[i], (sample.suffix.bit_size >> P::unit_shift));
116 }
117 ++emitted_[i];
118 return true;
119 }
120 if (stage.done()) continue;
121 if (stage.needs_input()) {
122 if (!i) {
123 if (source_.done()) {
124 stage.close_input();
125 } else {
126 auto sample = source_.peek();
127 auto coded = source_encoder_.encode(sample.key, sample.target_ordinal);
128 stage.push(coded);
129 source_suffix_units_ = profile_detail::add(
130 source_suffix_units_, (coded.suffix.bit_size >> P::unit_shift));
131 source_.advance();
132 }
133 return true;
134 }
135 if (stages_[i - 1]->done()) {
136 stage.close_input();
137 return true;
138 }
139 continue;
140 }
141 auto consumed = stage.step(1);
142 if (consumed) {
143 consumed_[i] += consumed;
144 return true;
145 }
146 }
147 return false;
148 }
