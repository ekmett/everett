51 inline std::uint64_t reverse_bytes(std::uint64_t x) noexcept {
52 x = ((x & 0x00ff00ff00ff00ffull) << 8) | ((x >> 8) & 0x00ff00ff00ff00ffull);
53 x = ((x & 0x0000ffff0000ffffull) << 16) | ((x >> 16) & 0x0000ffff0000ffffull);

57 if constexpr (std::endian::native == std::endian::little || std::endian::native == std::endian::big) {
58 std::uint64_t value; std::memcpy(&value, source, 8);
