cmake_minimum_required(VERSION 4.4)
project(everett_byte_lookup_compare LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/everett-experiment.cmake")
include(CTest)
set(EVERETT_BENCH_HEADER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../include" CACHE PATH "Frozen header tree containing everett/")
if(NOT EXISTS "${EVERETT_BENCH_HEADER_ROOT}/everett/policy.h")
  message(FATAL_ERROR "Set EVERETT_BENCH_HEADER_ROOT to the frozen include directory")
endif()
option(EVERETT_BENCH_SANITIZE "Build correctness checks with ASan/UBSan" OFF)
foreach(name bench check)
  add_executable(${name} ${name}.cc)
  everett_experiment_target(${name})
  target_include_directories(${name} PRIVATE "${EVERETT_BENCH_HEADER_ROOT}")
  target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
  if(EVERETT_BENCH_SANITIZE)
    target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${name} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
add_test(NAME changed_values COMMAND check)
add_test(NAME byte_query COMMAND bench byte 1024 16 structured 128 1 1)
add_test(NAME bit_query COMMAND bench bit 1024 128 hash 128 1 1)
