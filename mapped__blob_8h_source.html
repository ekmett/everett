38 static pair_type bind(blob_identity identity, std::shared_ptr<native_type const> native,
39 std::shared_ptr<index_type const> index, pair_type target = {}) {
40 if (!native || !index) throw std::invalid_argument("null Diet mapped pair component");
41 if (identity.native != index->native_id()) throw std::invalid_argument("Diet index names another native object");
42 auto const & expected = index->target_id();
43 if (bool(expected) != bool(target) || (expected && *expected != target->identity()))
44 throw std::invalid_argument("Diet mapped pair target identity mismatch");
45 if (index->borrowed().size() != (target ? target->group_count() : 0))
46 throw std::invalid_argument("Diet mapped pair sample count mismatch");
47 profile_blob_view<P> view(native->view(), index->borrowed(), index->interleave(),
48 index->false_borrow_bits(), index->cut_lcps(), index->virtual_size());
49 return pair_type(new mapped_blob(std::move(identity), std::move(native), std::move(index),
50 std::move(target), std::move(view)));
51 }

78 // Explicit complete-chain scan. Reconstructs sequential keys and rebuilds
79 // compact directories as scratch; it does not materialize all full keys.
80 // Success attests current readable bytes, not recovery from a failed fsync.
81 void scan() const {
82 std::unordered_set<mapped_blob const *> seen;
83 std::unordered_set<native_type const *> native_seen;
84 for (auto current = this; current; current = current->target_.get()) {
85 if (!seen.insert(current).second) throw std::invalid_argument("cyclic Diet mapped chain");
86 if (native_seen.insert(current->native_.get()).second) current->native_->scan();
87 current->index_->scan();
88 current->scan_pair();
89 current->scan_samples();
90 }
91 }

105 struct merged_cursor {
106 profile_cursor<P, stream_role::native> native;
107 profile_cursor<P, stream_role::borrowed> borrowed;
108 explicit merged_cursor(mapped_blob const & source) : native(source.native()), borrowed(source.borrowed()) {}
109 bool done() const noexcept { return native.done() && borrowed.done(); }
110 bool is_borrowed() const {
111 if (done()) throw std::out_of_range("Diet merged cursor at end");
112 return !borrowed.done() && (native.done() ||
113 compare_bits(borrowed.peek().key.prefix, native.peek().key.prefix) < 0);
114 }
115 bit_view key() const { return is_borrowed() ? borrowed.peek().key.prefix : native.peek().key.prefix; }
116 void advance() { if (is_borrowed()) borrowed.advance(); else native.advance(); }
117 };

119 void scan_pair() const {
120 merged_cursor cursor(*this);
121 bit_string previous_native, previous_borrowed;
122 bool had_native = false;
123 std::uint64_t ordinal = 0, borrowed_count = 0, population = 0;
124 auto ranks = interleave();
125 auto cuts = cut_lcps();
126 while (!cursor.done()) {
127 auto key = cursor.key();
128 auto group = ordinal / P::group_size;
129 if (ordinal % P::group_size == 0) {
130 auto lcp = borrowed_count ? compare_common_bits(previous_borrowed.view(), key).common_bits : 0;
131 if (cuts[group] != lcp) throw std::invalid_argument("Diet cut LCP disagrees with keys");
132 if (ranks.rank(group) != borrowed_count) throw std::invalid_argument("Diet rank disagrees with interleaving");
133 population = 0;
134 }
135 if (cursor.is_borrowed()) {
136 bool expected = had_native && compare_bits(previous_native.view(), key) == 0;
137 if (false_borrow(borrowed_count) != expected)
138 throw std::invalid_argument("Diet false-borrow flag disagrees with keys");
139 previous_borrowed = bit_string::copy(key);
140 ++borrowed_count;
141 ++population;
142 } else {
143 previous_native = bit_string::copy(key);
144 had_native = true;
145 }
146 cursor.advance();
147 ++ordinal;
148 if (ordinal % P::group_size == 0 || cursor.done())
149 if (ranks.class_at(group) != population)
150 throw std::invalid_argument("Diet rank population disagrees with keys");
151 }
152 if (ordinal != virtual_size() || borrowed_count != borrowed().size())
153 throw std::invalid_argument("Diet merged stream count mismatch");
154 }
155 void scan_samples() const {
156 auto samples = borrowed().cursor();
157 if (!target_) {
158 if (!samples.done()) throw std::invalid_argument("Diet borrowed stream has no target");
159 return;
160 }
161 merged_cursor target(*target_);
162 std::uint64_t ordinal = 0;
163 while (!target.done()) {
164 if (ordinal % P::group_size == 0) {
165 if (samples.done() || compare_bits(samples.peek().key.prefix, target.key()) != 0)
166 throw std::invalid_argument("Diet borrowed key is not the exact target sample");
167 samples.advance();
168 }
169 target.advance();
170 ++ordinal;
171 }
172 if (!samples.done()) throw std::invalid_argument("Diet trailing borrowed target samples");
173 }

178 // Resolve the exact on-disk links using only fixed metadata, cache shared
179 // native mappings within this chain, and adopt its already-prepared head.
180 // This reads no FC payload or rank/EF words and never rebuilds an index.
181 template <class P> mapped_query_root<P> open_mapped_query(std::filesystem::path const & root,
182 blob_identity const & head) {
183 using native_pointer = std::shared_ptr<mapped_native<P> const>;
184 using index_pointer = std::shared_ptr<mapped_index<P> const>;
185 struct pending_pair { blob_identity identity; native_pointer native; index_pointer index; };
186 std::vector<pending_pair> pending;
187 std::unordered_set<std::string> seen;
188 std::unordered_map<std::string, native_pointer> natives;
189 std::optional<blob_identity> current = head;
190 while (current) {
191 if (!seen.insert(current->index.hex()).second) throw std::invalid_argument("cyclic Diet object identities");
192 auto index = std::make_shared<mapped_index<P> const>(
193 mapped_index<P>::open(root / object_path(current->index, file_kind::fractional_index)));
194 if (index->native_id() != current->native) throw std::invalid_argument("Diet chain native identity mismatch");
195 auto found = natives.find(current->native.hex());
196 if (found == natives.end()) {
197 auto native = std::make_shared<mapped_native<P> const>(
198 mapped_native<P>::open(root / object_path(current->native, file_kind::native_blob)));
199 found = natives.emplace(current->native.hex(), std::move(native)).first;
200 }
201 pending.push_back({*current, found->second, index});
202 current = index->target_id();
203 }
204 typename mapped_blob<P>::pair_type pair;
205 for (auto i = pending.size(); i-- > 0;)
206 pair = mapped_blob<P>::bind(std::move(pending[i].identity), std::move(pending[i].native),
207 std::move(pending[i].index), std::move(pair));
208 return mapped_query_root<P>::adopt_prepared(std::move(pair));
209 }
