41 static pair_type bind(blob_identity identity, std::shared_ptr<native_type const> native,
42 std::shared_ptr<index_type const> index, pair_type target = {}) {
43 if (!native || !index) throw std::invalid_argument("null Everett mapped pair component");
44 if (identity.native != index->native_id()) throw std::invalid_argument("Everett index names another native object");
45 auto const & expected = index->target_id();
46 if (bool(expected) != bool(target) || (expected && *expected != target->identity()))
47 throw std::invalid_argument("Everett mapped pair target identity mismatch");
48 if (index->borrowed().size() != (target ? target->group_count() : 0))
49 throw std::invalid_argument("Everett mapped pair sample count mismatch");
50 profile_blob_view<P> view(native->view(), index->borrowed(), index->interleave(),
51 index->false_borrow_bits(), index->cut_lcps(), index->virtual_size());
52 return pair_type(new mapped_blob(std::move(identity), std::move(native), std::move(index),
53 std::move(target), std::move(view)));
54 }

81 // Explicit complete-chain scan. Reconstructs sequential keys and rebuilds
82 // compact directories as scratch; it does not materialize all full keys.
83 // Success attests current readable bytes, not recovery from a failed fsync.
84 void scan() const {
85 std::unordered_set<mapped_blob const *> seen;
86 std::unordered_set<native_type const *> native_seen;
87 for (auto current = this; current; current = current->target_.get()) {
88 if (!seen.insert(current).second) throw std::invalid_argument("cyclic Everett mapped chain");
89 if (native_seen.insert(current->native_.get()).second) current->native_->scan();
90 current->index_->scan();
91 current->scan_pair();
92 current->scan_samples();
93 }
94 }

108 struct merged_cursor {
109 profile_cursor<P, stream_role::native> native;
110 profile_cursor<P, stream_role::borrowed> borrowed;
111 explicit merged_cursor(mapped_blob const & source) : native(source.native()), borrowed(source.borrowed()) {}
112 bool done() const noexcept { return native.done() && borrowed.done(); }
113 bool is_borrowed() const {
114 if (done()) throw std::out_of_range("Everett merged cursor at end");
115 return !borrowed.done() && (native.done() ||
116 compare_bits(borrowed.peek().key.prefix, native.peek().key.prefix) < 0);
117 }
118 bit_view key() const { return is_borrowed() ? borrowed.peek().key.prefix : native.peek().key.prefix; }
119 void advance() { if (is_borrowed()) borrowed.advance(); else native.advance(); }
120 };

122 void scan_pair() const {
123 merged_cursor cursor(*this);
124 bit_string previous_native, previous_borrowed;
125 bool had_native = false;
126 std::uint64_t ordinal = 0, borrowed_count = 0, population = 0;
127 auto ranks = interleave();
128 auto cuts = cut_lcps();
129 while (!cursor.done()) {
130 auto key = cursor.key();
131 auto group = ordinal / P::group_size;
132 if (ordinal % P::group_size == 0) {
133 auto lcp = borrowed_count ? compare_common_bits(previous_borrowed.view(), key).common_bits : 0;
134 if (cuts[group] != lcp) throw std::invalid_argument("Everett cut LCP disagrees with keys");
135 if (ranks.rank(group) != borrowed_count) throw std::invalid_argument("Everett rank disagrees with interleaving");
136 population = 0;
137 }
138 if (cursor.is_borrowed()) {
139 bool expected = had_native && compare_bits(previous_native.view(), key) == 0;
140 if (false_borrow(borrowed_count) != expected)
141 throw std::invalid_argument("Everett false-borrow flag disagrees with keys");
142 previous_borrowed = bit_string::copy(key);
143 ++borrowed_count;
144 ++population;
145 } else {
146 previous_native = bit_string::copy(key);
147 had_native = true;
148 }
149 cursor.advance();
150 ++ordinal;
151 if (ordinal % P::group_size == 0 || cursor.done())
152 if (ranks.class_at(group) != population)
153 throw std::invalid_argument("Everett rank population disagrees with keys");
154 }
155 if (ordinal != virtual_size() || borrowed_count != borrowed().size())
156 throw std::invalid_argument("Everett merged stream count mismatch");
157 }
158 void scan_samples() const {
159 auto samples = borrowed().cursor();
160 if (!target_) {
161 if (!samples.done()) throw std::invalid_argument("Everett borrowed stream has no target");
162 return;
163 }
164 merged_cursor target(*target_);
165 std::uint64_t ordinal = 0;
166 while (!target.done()) {
167 if (ordinal % P::group_size == 0) {
168 if (samples.done() || compare_bits(samples.peek().key.prefix, target.key()) != 0)
169 throw std::invalid_argument("Everett borrowed key is not the exact target sample");
170 samples.advance();
171 }
172 target.advance();
173 ++ordinal;
174 }
175 if (!samples.done()) throw std::invalid_argument("Everett trailing borrowed target samples");
176 }

181 // Resolve the exact on-disk links using only fixed metadata, cache shared
182 // native mappings within this chain, and adopt its already-prepared head.
183 // This reads no FC payload or rank/EF words and never rebuilds an index.
184 template <class P> mapped_query_root<P> open_mapped_query(std::filesystem::path const & root,
185 blob_identity const & head) {
186 using native_pointer = std::shared_ptr<mapped_native<P> const>;
187 using index_pointer = std::shared_ptr<mapped_index<P> const>;
188 struct pending_pair { blob_identity identity; native_pointer native; index_pointer index; };
189 std::vector<pending_pair> pending;
190 std::unordered_set<std::string> seen;
191 std::unordered_map<std::string, native_pointer> natives;
192 std::optional<blob_identity> current = head;
193 while (current) {
194 if (!seen.insert(current->index.hex()).second) throw std::invalid_argument("cyclic Everett object identities");
195 auto index = std::make_shared<mapped_index<P> const>(
196 mapped_index<P>::open(root / object_path(current->index, file_kind::fractional_index)));
197 if (index->native_id() != current->native) throw std::invalid_argument("Everett chain native identity mismatch");
198 auto found = natives.find(current->native.hex());
199 if (found == natives.end()) {
200 auto native = std::make_shared<mapped_native<P> const>(
201 mapped_native<P>::open(root / object_path(current->native, file_kind::native_blob)));
202 found = natives.emplace(current->native.hex(), std::move(native)).first;
203 }
204 pending.push_back({*current, found->second, index});
205 current = index->target_id();
206 }
207 typename mapped_blob<P>::pair_type pair;
208 for (auto i = pending.size(); i-- > 0;)
209 pair = mapped_blob<P>::bind(std::move(pending[i].identity), std::move(pending[i].native),
210 std::move(pending[i].index), std::move(pair));
211 return mapped_query_root<P>::adopt_prepared(std::move(pair));
212 }
