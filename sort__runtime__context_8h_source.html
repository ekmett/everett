118 if (!recipe) return {}; // No SQL, reservations or forced input sealing.

125 template <class Merge> native_pointer finish_merge(Merge & merge) {
126 require_active();
127 try {
128 if (merge.owner_.get() != this) throw std::invalid_argument("merge belongs to another runtime context");
129 auto completed = merge.finish();
130 if (auto owned = std::get_if<0>(&completed)) return native_type::from_owned(std::move(*owned));
131 auto receipt = std::get<1>(std::move(completed));
132 catalog_.record_sealed(ids_().hex(), receipt);
133 // Only naturally streamed completions install durable hints. Owned
134 // adaptive outputs retain their existing no-I/O path.
135 if (merge.recipe_) catalog_.record_native_merge(ids_().hex(), *merge.recipe_, receipt, ids_().hex());
136 auto native = native_type::from_sealed(root(), identity_, std::move(receipt));
137 ++sealed_outputs_; return native;
138 } catch (...) { failed_ = true; throw; }

218 template <class Compose> std::optional<catalog_native_merge> merge_recipe(
219 native_pointer const & older, native_pointer const & newer) const {
220 if (!older || !newer) throw std::invalid_argument("null native merge input");
221 // Advisory reuse owns permanent reader pins. A private transaction keeps
222 // its outputs solely under its releasable construction scope instead.
223 if (catalog_.private_construction()) return {};
224 using composer = std::remove_cvref_t<decltype(native_merge_detail::composer(std::declval<Compose &>()))>;
225 using conservative = typed_detail::replacement_compose<P, sort_key_transport<P, Selector>>;
226 // This recognizes known kernels and one registry, not arbitrary allegedly
227 // stateless user code. The persisted domain supplies stable wire identity.
228 if constexpr ((std::is_same_v<composer, replace_native_value> || std::is_same_v<composer, conservative>) &&
229 P::unit == profile_unit::bit &&
230 std::is_same_v<typename P::registry_type, string_registry> &&
231 std::is_same_v<Selector, registry_selector<string_registry>>) {
232 if (schema_.empty()) return {};
233 auto a = older->bindings_.find(identity_, root());
234 if (!a) return {};
235 auto b = newer->bindings_.find(identity_, root());
236 if (!b) return {};
237 return catalog_native_merge{schema_, a->receipt.object, b->receipt.object,
238 std::is_same_v<composer, conservative> ? catalog_native_merge_kind::conservative_tombstones :
239 catalog_native_merge_kind::replacement};
240 } else return {};
241 }
