40 static pair_type bind(blob_identity identity, native_pointer native, index_pointer index,
41 pair_type main = {}, native_pointer secondary = {}, std::optional<object_id> secondary_id = {}) {
42 if (!native || !index || identity.native != index->native_id())
43 error_detail::raise<std::invalid_argument>("invalid COLA mapped pair components");
44 if (bool(index->main_id()) != bool(main) || (main && *index->main_id() != main->identity()))
45 error_detail::raise<std::invalid_argument>("COLA main target identity mismatch");
46 if (bool(secondary) != bool(secondary_id) || secondary_id != index->secondary_id())
47 error_detail::raise<std::invalid_argument>("COLA secondary target identity mismatch");
48 auto secondary_count = secondary ? secondary->size() : 0;
49 if (index->borrowed(0).size() != (main ? main->group_count() : 0) ||
50 index->borrowed(1).size() != secondary_count / P::group_size + (secondary_count % P::group_size != 0))
51 error_detail::raise<std::invalid_argument>("COLA mapped target sample count mismatch");
52 auto view = index->view(native->view());
53 return pair_type(new mapped_cola_blob(std::move(identity), std::move(native), std::move(index),
54 std::move(main), std::move(secondary), std::move(view)));
55 }

67 // Explicit recovery/validation walk. Rebuilds compact rank/EF metadata and
68 // reconstructs sequential key frontiers, never an array of every full key.
69 void scan() const {
70 std::unordered_set<mapped_cola_blob const *> seen;
71 std::unordered_set<native_type const *> native_seen;
72 for (auto current = this; current; current = current->main_.get()) {
73 if (!seen.insert(current).second) error_detail::raise<std::invalid_argument>("cyclic COLA mapped chain");
74 if (native_seen.insert(current->native_.get()).second) current->native_->scan();
75 if (current->secondary_ && native_seen.insert(current->secondary_.get()).second) current->secondary_->scan();
76 current->index_->scan();
77 current->scan_pair();
78 current->scan_samples();
79 }
80 }

93 struct merged_cursor {
94 profile_cursor<P> native;
95 std::array<profile_cursor<P, stream_role::borrowed>, 2> borrowed;
96 explicit merged_cursor(cola_index_view<P> source)
97 : native(source.native()), borrowed{profile_cursor<P, stream_role::borrowed>(source.borrowed(0)),
98 profile_cursor<P, stream_role::borrowed>(source.borrowed(1))} {}
99 bool done() const noexcept { return native.done() && borrowed[0].done() && borrowed[1].done(); }
100 unsigned origin() const {
101 unsigned result = 3;
102 bit_view best;
103 if (!native.done()) { result = 0; best = native.peek().key.prefix; }
104 for (unsigned i = 0; i != 2; ++i)
105 if (!borrowed[i].done() && (result == 3 || compare_bits(borrowed[i].peek().key.prefix, best) < 0)) {
106 result = i + 1; best = borrowed[i].peek().key.prefix;
107 }
108 if (result == 3) error_detail::raise<std::out_of_range>("COLA merged cursor at end");
109 return result;
110 }
111 bit_view key(unsigned origin) const { return origin ? borrowed[origin - 1].peek().key.prefix : native.peek().key.prefix; }
112 void advance(unsigned origin) { if (origin) borrowed[origin - 1].advance(); else native.advance(); }
113 };
114 void scan_pair() const {
115 merged_cursor cursor(view_);
116 bit_string previous_native;
117 std::array<bit_string, 2> previous_borrowed;
118 std::array<std::uint64_t, 2> borrowed_count{}, population{};
119 bool had_native = false;
120 std::uint64_t ordinal = 0;
121 while (!cursor.done()) {
122 auto origin = cursor.origin();
123 auto key = cursor.key(origin);
124 auto group = ordinal / P::group_size;
125 if (ordinal % P::group_size == 0) {
126 for (unsigned route = 0; route != 2; ++route) {
127 auto lcp = borrowed_count[route] ? compare_common_bits(previous_borrowed[route].view(), key).common_bits : 0;
128 if (view_.cut_lcps(route)[group] != lcp || view_.interleave(route).rank(group) != borrowed_count[route])
129 error_detail::raise<std::invalid_argument>("COLA cut or rank disagrees with keys");
130 }
131 population = {};
132 }
133 if (origin) {
134 auto route = origin - 1;
135 bool expected = had_native && compare_bits(previous_native.view(), key) == 0;
136 if (view_.false_borrow(route, borrowed_count[route]) != expected)
137 error_detail::raise<std::invalid_argument>("COLA false-borrow flag disagrees with keys");
138 previous_borrowed[route] = bit_string::copy(key);
139 ++borrowed_count[route]; ++population[route];
140 } else { previous_native = bit_string::copy(key); had_native = true; }
141 cursor.advance(origin);
142 ++ordinal;
143 if (ordinal % P::group_size == 0 || cursor.done())
144 for (unsigned route = 0; route != 2; ++route)
145 if (view_.interleave(route).class_at(group) != population[route])
146 error_detail::raise<std::invalid_argument>("COLA population disagrees with keys");
147 }
148 if (ordinal != virtual_size() || borrowed_count[0] != view_.borrowed(0).size() ||
149 borrowed_count[1] != view_.borrowed(1).size())
150 error_detail::raise<std::invalid_argument>("COLA merged stream count mismatch");
151 }
152 void scan_samples() const {
153 auto samples = view_.borrowed(0).cursor();
154 if (main_) {
155 cola_sample_cursor<P, mapped_cola_blob> target(main_);
156 while (!target.done()) {
157 if (samples.done() || compare_bits(samples.peek().key.prefix, target.peek().key) != 0)
158 error_detail::raise<std::invalid_argument>("COLA main sample differs from target");
159 samples.advance(); target.advance();
160 }
161 }
162 if (!samples.done()) error_detail::raise<std::invalid_argument>("COLA trailing main samples");
163 auto side_samples = view_.borrowed(1).cursor();
164 if (secondary_) {
165 auto target = secondary_->view().cursor();
166 std::uint64_t ordinal = 0;
167 while (!target.done()) {
168 if (ordinal % P::group_size == 0) {
169 if (side_samples.done() || compare_bits(side_samples.peek().key.prefix, target.peek().key.prefix) != 0)
170 error_detail::raise<std::invalid_argument>("COLA secondary sample differs from target");
171 side_samples.advance();
172 }
173 target.advance(); ++ordinal;
174 }
175 }
176 if (!side_samples.done()) error_detail::raise<std::invalid_argument>("COLA trailing secondary samples");
177 }

188 // Resolves exact targets using fixed metadata; native mappings shared by
189 // main/secondary roles are cached. It never scans or rebuilds a payload.
190 template <class P> mapped_cola_query_root<P> open_mapped_cola_query(
191 std::filesystem::path const & root, blob_identity const & head) {
192 using blob = mapped_cola_blob<P>;
193 using native_pointer = typename blob::native_pointer;
194 struct pending { blob_identity identity; native_pointer native, secondary; typename blob::index_pointer index; };
195 std::vector<pending> chain;
196 std::unordered_set<std::string> seen;
197 std::unordered_map<std::string, native_pointer> natives;
198 auto native = [&](object_id const & id) {
199 auto found = natives.find(id.hex());
200 if (found == natives.end()) found = natives.emplace(id.hex(), std::make_shared<mapped_native<P> const>(
201 mapped_native<P>::open(root / object_path(id, file_kind::native_blob)))).first;
202 return found->second;
203 };
204 std::optional<blob_identity> current = head;
205 while (current) {
206 if (!seen.insert(current->index.hex()).second) error_detail::raise<std::invalid_argument>("cyclic COLA object identities");
207 auto index = std::make_shared<mapped_cola_index<P> const>(
208 mapped_cola_index<P>::open(root / object_path(current->index, file_kind::fractional_index)));
209 if (index->native_id() != current->native) error_detail::raise<std::invalid_argument>("COLA chain native identity mismatch");
210 chain.push_back({*current, native(current->native), index->secondary_id() ? native(*index->secondary_id()) : native_pointer{}, index});
211 current = index->main_id();
212 }
213 typename blob::pair_type pair;
214 for (auto i = chain.size(); i-- > 0;)
215 pair = blob::bind(std::move(chain[i].identity), std::move(chain[i].native), chain[i].index,
216 std::move(pair), std::move(chain[i].secondary), chain[i].index->secondary_id());
217 return mapped_cola_query_root<P>::adopt_prepared(std::move(pair));
218 }
