28 // Physical and operation identities use OS entropy; semantic fingerprints
29 // are unrelated. Catalog reservations and exclusive installation still
30 // reject a collision instead of replacing an existing object.
31 struct random_object_ids {
32 object_id operator()() const {
33#if defined(__APPLE__) || defined(__linux__)
34 std::array<unsigned char, 16> bytes{};
35 if (::getentropy(bytes.data(), bytes.size()))
36 throw std::system_error(errno, std::generic_category(), "Everett identity entropy");
37 constexpr char digits[] = "0123456789abcdef";
38 std::string text(32, '0');
39 for (std::size_t i = 0; i != bytes.size(); ++i) {
40 text[i * 2] = digits[bytes[i] >> 4]; text[i * 2 + 1] = digits[bytes[i] & 15];
41 }
42 return object_id(std::move(text));
43#else
44 throw std::system_error(std::make_error_code(std::errc::operation_not_supported), "Everett identity entropy");
45#endif
46 }
47 };

153 // Mapping one checkpoint shares physical mappings and typed facades across
154 // every visible and hidden root, preserving exact immutable dependencies.
155 struct resolver {
156 runtime_store & store;
157 mapped_resolver mapped;
158 std::unordered_map<std::string, native_pointer> natives;
159 std::unordered_map<std::string, pair_type> pairs;
160 bool restricted = false;
161 explicit resolver(runtime_store & value) : store(value), mapped(value.root()) {}
162 native_pointer native(object_id const & id) {
163 auto found = natives.find(id.hex());
164 if (found == natives.end()) {
165 if (restricted) throw std::invalid_argument("checkpoint native has no durable pin");
166 auto value = native_type::from_mapped(mapped.native(id));
167 store.sealer().bind_native(value, id);
168 found = natives.emplace(id.hex(), std::move(value)).first;
169 }
170 return found->second;
171 }
172 pair_type pair(blob_identity const & id) {
173 if (auto found = pairs.find(id.index.hex()); found != pairs.end()) {
174 if (found->second->mapped()->identity() != id)
175 throw std::invalid_argument("checkpoint index has another native identity");
176 return found->second;
177 }
178 if (restricted) {
179 throw std::invalid_argument("checkpoint pair has no durable pin");
180 }
181 auto physical = mapped.pair(id);
182 std::vector<mapped_pointer> pending;
183 pair_type result;
184 for (auto p = physical; p; p = p->main_target()) {
185 if (auto known = pairs.find(p->identity().index.hex()); known != pairs.end()) { result = known->second; break; }
186 pending.push_back(p);
187 }
188 for (auto i = pending.rbegin(); i != pending.rend(); ++i) {
189 auto const & p = *i;
190 result = node_type::from_mapped_parts(p, native(p->identity().native), result,
191 p->index_object()->secondary_id() ? native(*p->index_object()->secondary_id()) : native_pointer{});
192 pairs.emplace(p->identity().index.hex(), result);
193 store.sealer().bind_pair(result, p->identity());
194 }
195 return result;
196 }
197 bool seed_native(object_id const & id, native_pointer value) {
198 auto [found, inserted] = natives.try_emplace(id.hex(), value);
199 return inserted || found->second == value;
200 }
201 bool seed_root(pair_type value) {
202 auto const & id = value->mapped()->identity();
203 auto [found, inserted] = pairs.try_emplace(id.index.hex(), value);
204 return inserted || found->second == value;
205 }

211 template <class Resolver> static std::optional<roots> retained_roots(catalog_session_head const & head, Resolver & loaded) {
212 roots result;
213 result.pairs.reserve(head.auxiliary.pairs.size() + 1);
214 result.natives.reserve(head.auxiliary.natives.size());
215 auto pair = loaded.pair(head.timeline.head);
216 if (!pair) return std::nullopt;
217 result.pairs.push_back(std::move(pair));
218 for (auto const & id : head.auxiliary.pairs) {
219 pair = loaded.pair(id);
220 if (!pair) return std::nullopt;
221 result.pairs.push_back(std::move(pair));
222 }
223 for (auto const & id : head.auxiliary.natives) {
224 auto native = loaded.native(id);
225 if (!native) return std::nullopt;
226 result.natives.emplace_back(id, std::move(native));
227 }
228 return result;

243 stored_type decode_retained(catalog_session_head head) {
244 pinned_resolver loaded{retained_};
245 try {
246 auto restored = codec_type::decode(head.checkpoint, head.timeline.head, loaded);
247 return {std::move(head), std::move(restored.snapshot), std::move(restored.semantic)};
248 } catch (...) {
249 // A failed checkpoint never leaves an ambiguous authorization cache.
250 retained_.clear(); throw;

253 stored_type restore(catalog_session_head head, snapshot_type const * source = nullptr) {
254 if (source) {
255 resolver candidates(*this);
256 auto graph = sealer();
257 bool coherent = true;
258 codec_type::collect(*source, [&](pair_type const & pair) {
259 if (pair) coherent = candidates.seed_root(graph.mapped_pair(pair)) && coherent;
260 }, [&](native_pointer const & native) {
261 if (native) coherent = candidates.seed_native(graph.native_id(native), graph.mapped_native(native)) && coherent;
262 });
263 // Candidate roots provide owners, not authority. Only the roots in the
264 // acknowledged catalog head acquire registry references. Shared live
265 // owners stop at their local count; new edges are checked once.
266 if (coherent) {
267 candidates.restricted = true;
268 auto selected = retained_roots(head, candidates);
269 if (!selected) throw std::logic_error("published root has no prepared owner");
270 if (retained_.replace(std::move(selected->pairs), std::move(selected->natives)))
271 return decode_retained(std::move(head));
272 }
273 } else if (auto selected = retained_roots(head, retained_)) {
274 if (retained_.replace(std::move(selected->pairs), std::move(selected->natives)))
275 return decode_retained(std::move(head));
276 }
277 // An imported graph may contain distinct owners for the same physical
278 // identity, even deep in a new prefix. Intern the entire incoming graph
279 // afresh in that exceptional case. The old registry remains intact until
280 // this new closure has been validated and decoded.
281 resolver loaded(*this);
282 auto selected = retained_roots(head, loaded);
283 if (!selected) throw std::logic_error("durable root has no mapped owner");
284 loaded.restricted = true;
285 registry_type replacement;
286 if (!replacement.replace(std::move(selected->pairs), std::move(selected->natives)))
287 throw std::invalid_argument("mapped checkpoint has conflicting facade identities");
288 pinned_resolver pinned{replacement};
289 auto restored = codec_type::decode(head.checkpoint, head.timeline.head, pinned);
290 retained_ = std::move(replacement);
291 return {std::move(head), std::move(restored.snapshot), std::move(restored.semantic)};
292 }
293 struct persisted { blob_identity head; catalog_auxiliary_roots auxiliary; };
294 persisted persist(snapshot_type const & source) {
295 std::vector<pair_type> roots;
296 std::vector<native_pointer> native_roots;
297 std::unordered_set<node_type const *> seen_pairs;
298 std::unordered_set<native_type const *> seen_natives;
299 codec_type::collect(source, [&](pair_type value) {
300 if (value && seen_pairs.insert(value.get()).second) roots.push_back(std::move(value));
301 }, [&](native_pointer value) {
302 if (value && seen_natives.insert(value.get()).second) native_roots.push_back(std::move(value));
303 });
304 auto graph = sealer();
305 (void)graph.prepare_ready(roots, native_roots);
306 // Each new owner resolves its immediate dependencies once. An existing
307 // bound suffix stops this walk, irrespective of how many roots share it.
308 // Finish each remaining root before starting another. The ready batch
309 // has released every claim before this blocking dependency walk.
310 for (auto const & pair : roots) (void)graph.ensure_pair(pair);
311 for (auto const & native : native_roots) (void)graph.ensure_native(native);
312 auto primary = graph.pair_id(source.query_root().head());
313 catalog_auxiliary_roots retained;
314 for (auto const & pair : roots) retained.pairs.push_back(graph.pair_id(pair));
315 for (auto const & native : native_roots) retained.natives.push_back(graph.native_id(native));
316 return {primary, catalog_detail::canonical_auxiliary(std::move(retained), primary)};
317 }
