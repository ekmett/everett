56 // A code-stable extension can add a sort in a reserved hole. Its generic
57 // executor reads the old replacement envelope but makes no cleanup-size
58 // promise. Validate the entire old generation before projecting its table
59 // metadata; pending structural merge jobs remain in the runtime frontier.
60 static world_type restore_checkpoint(runtime_snapshot runtime, std::span<std::byte const> bytes,
61 std::string_view schema) {
62 if (schema.empty()) throw std::invalid_argument("active checkpoint requires an expected schema");
63 if constexpr (active_detail::rebuilding<P>)
64 return world_type::restore(std::move(runtime), metadata_type::decode(bytes), schema);
65 else {
66 metadata_type metadata;
67 if (bytes.size() >= 56 && bytes.size() - 56 == schema.size()) {
68 auto previous = replacement_metadata<>::decode(bytes);
69 previous.validate(runtime.admissions());
70 metadata = static_cast<metadata_type const &>(previous);
71 } else if (bytes.size() >= 16 && bytes.size() - 16 == schema.size())
72 metadata = metadata_type::decode(bytes);
73 else throw std::invalid_argument("active checkpoint does not match the expected schema layout");
74 return world_type::restore(std::move(runtime), std::move(metadata), schema);
75 }
