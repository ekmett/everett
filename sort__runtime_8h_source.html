92 // Input owns the full order keys for this synchronous call. Borrow them
93 // directly, retaining the previous view only long enough to compute LCP.
94 template <class P, class Selector, class Writer>
95 void write_sorted_native(Writer & writer, std::span<profile_record const> records) {
96 using leaves = typename registry_detail::info<typename P::registry_type>::leaves;
97 bit_view previous;
98 bool first = true;
99 for (auto const & record : records) {
100 auto key = record.key.view();
101 auto comparison = compare_common_bits(previous, key);
102 if (!first && comparison.order >= 0)
103 throw std::invalid_argument("sort native requires unique sorted keys");
104 sort_bit_reader input(key);
105 Selector::select(input, [&]<class S>(std::type_identity<S>, auto & source) {
106 sort_codec_detail::validate_value_width<S>();
107 sort_profile_frame frame;
108 frame.path = key.prefix(source.position());
109 frame.leaf = sort_profile_detail::ordinal<leaves, S>::value;
110 frame.key_units = key.size();
111 frame.front_coded = sort_profile_key<typename sort_codec<S>::key_codec>::front_coded;
112 std::array<bit_view, 1> spans{key};
113 writer.append_frame(frame, spans, comparison.common_bits, record.value.view());
114 });
115 previous = key; first = false;
116 }
117 }
118 }
