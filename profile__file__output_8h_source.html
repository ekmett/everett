80 void append(std::uint64_t retained, bit_view literal, bit_view value) {
81 require_active();
82 if ((literal.size() & (P::bits_per_unit - 1)) || (value.size() & (P::bits_per_unit - 1)))
83 error_detail::raise<std::invalid_argument>("profile file record disagrees with policy units");
84 auto value_units = value.size() >> P::unit_shift;
85 if (common_ && value_units != *common_)
86 error_detail::raise<std::invalid_argument>("profile file value disagrees with common width");
87 if (retained > previous_units_ || (!borrowed && count_ && literal.empty()))
88 error_detail::raise<std::invalid_argument>("profile file invalid known prefix");
89 auto literal_units = literal.size() >> P::unit_shift;
90 auto key_units = add(retained, literal_units);
91 (void)multiply(key_units, P::bits_per_unit);
92 auto next_count = add(count_, 1);
93 control_.bytes.clear(); control_.bit_size = 0;
94 std::uint64_t zero_bits = 0;
95 auto block = count_ % P::codec_block_size == 0;
96 if (block) write_count<P>(control_, retained);
97 else if constexpr (P::unit == profile_unit::bit && P::backspace_code == bit_backspace_code::golomb) {
98 auto backspace = previous_units_ - retained;
99 auto bits = backspace_bits<P>(backspace);
100 if (bits <= 129) write_backspace<P>(control_, backspace);
101 else {
102 // The unary quotient can be arbitrarily longer than the key's
103 // new literal. Emit it separately, never grow the control scratch.
104 constexpr auto modulus = P::backspace_parameter;
105 constexpr auto width = unsigned(std::bit_width(modulus - 1));
106 constexpr auto cutoff = golomb_cutoff<P>();
107 zero_bits = backspace / modulus;
108 auto remainder = backspace % modulus;
109 auto remainder_width = width - unsigned(remainder < cutoff);
110 resize(control_, 1 + remainder_width);
111 std::uint64_t at = 0;
112 put_fixed(control_, at, 1, 1);
113 put_fixed(control_, at, remainder < cutoff ? remainder : remainder + cutoff, remainder_width);
114 }
115 } else write_backspace<P>(control_, previous_units_ - retained);
116 write_count<P>(control_, literal_units);
117 if (!common_) write_count<P>(control_, value_units);
118 auto frame_bits = add(add(zero_bits, control_.bit_size), add(literal.size(), value.size()));
119 auto next_bits = add(fc_bits_, frame_bits);
120 if (next_bits > std::numeric_limits<std::uint64_t>::max() - 7)
121 error_detail::raise<std::length_error>("profile file FC extent is too large");
122 // Allocate the next navigation entry before any output is emitted.
123 if (block) offsets_.push_back((fc_bits_ >> P::unit_shift) - multiply(count_, common_.value_or(0)));
124 try {
125 emit_zeroes(zero_bits);
126 emit(control_.view()); emit(literal); emit(value);
127 } catch (...) { failed_ = true; throw; }
128 fc_bits_ = next_bits; previous_units_ = key_units; count_ = next_count;
129 }

137 object_seal_receipt finish_impl(index_file_sections<P> const * index) {
138 require_active();
139 if constexpr (borrowed) validate_index(*index);
140 auto extent = fc_bits_ >> P::unit_shift;
141 offsets_.push_back(extent - multiply(count_, common_.value_or(0)));
142 elias_fano offsets;
143 try { offsets = elias_fano::build<typename P::architecture>(offsets_); }
144 catch (...) { offsets_.pop_back(); throw; }
145 offsets_.pop_back();
146 std::array<std::byte, directory_bytes> directory{};
147 auto magic = borrowed ? "IX02" : "KV02";
148 for (unsigned i = 0; i != 4; ++i) directory[i] = std::byte(magic[i]);
149 file_detail::put(directory, 4, 2, section_detail::version);
150 file_detail::put(directory, 6, 2, section_count);
151 file_detail::put(directory, 8, 8, extent);
152 file_detail::put(directory, 16, 8, previous_units_);
153 file_detail::put(directory, 24, 8, offsets.universe);
154 file_detail::put(directory, 32, 1, offsets.low_width);
155 std::array<std::uint64_t, section_count> lengths{byte_count(fc_bits_), multiply(offsets.low.size(), 8),
156 multiply(offsets.high.size(), 8), multiply(offsets.samples.size(), 16), multiply(offsets.sparse.size(), 8)};
157 if constexpr (borrowed) {
158 file_detail::put(directory, 40, 8, index->virtual_count);
159 section_detail::put_id(directory, 48, index->native_id);
160 if (index->target_id) {
161 file_detail::put(directory, 33, 1, 1);
162 section_detail::put_id(directory, 64, index->target_id->native);
163 section_detail::put_id(directory, 80, index->target_id->index);
164 }
165 lengths[5] = multiply(index->interleave.classes.size(), 8);
166 lengths[6] = multiply(index->interleave.checkpoints.size(), 8);
167 lengths[7] = index->false_borrows.size();
168 lengths[8] = multiply(index->cut_lcps.size(), 8);
169 }
170 std::uint64_t end = directory.size();
171 for (std::size_t i = 0; i != lengths.size(); ++i) {
172 auto start = add(end, 7) & ~std::uint64_t{7};
173 end = add(start, lengths[i]);
174 file_detail::put(directory, descriptor_offset + (i << 4), 8, start);
175 file_detail::put(directory, descriptor_offset + (i << 4) + 8, 8, lengths[i]);
176 }
177 file_header<P> header{kind, multiply(end, 1u << (3 - P::unit_shift)), count_, common_};
178 file_detail::validate_metadata(header);
179 if (file_detail::total_bytes<P>(header.extent) > std::uint64_t(std::numeric_limits<std::int64_t>::max()))
180 error_detail::raise<std::length_error>("profile file exceeds supported file offsets");
181 // EF allocation and complete metadata admission above remain retryable.
182 // Once final bytes are emitted, no retry may duplicate the directories.
183 try {
184 flush_tail();
185 align_stream(); emit_words(offsets.low);
186 align_stream(); emit_words(offsets.high);
187 align_stream(); emit_samples(offsets.samples);
188 align_stream(); emit_words(offsets.sparse);
189 if constexpr (borrowed) {
190 align_stream(); emit_words(index->interleave.classes);
191 align_stream(); emit_words(index->interleave.checkpoints);
192 align_stream(); stream_.append(index->false_borrows);
193 align_stream(); emit_words(index->cut_lcps);
194 }
195 return stream_.finish(header, directory);
196 } catch (...) { failed_ = true; throw; }
197 }

209 void validate_index(index_file_sections<P> const & index) const {
210 auto groups = index.virtual_count / P::group_size + (index.virtual_count % P::group_size != 0);
211 if (index.virtual_count < count_ || count_ > std::numeric_limits<std::uint64_t>::max() - 7 ||
212 index.false_borrows.size() != ((count_ + 7) >> 3) || index.cut_lcps.size() != groups ||
213 index.interleave.virtual_count != index.virtual_count || (!index.target_id && count_))
214 error_detail::raise<std::invalid_argument>("streamed index metadata shape mismatch");
215 if (index.interleave.view().count() != count_)
216 error_detail::raise<std::invalid_argument>("streamed index borrowed count mismatch");
217 if ((count_ & 7) && (std::to_integer<unsigned>(index.false_borrows.back()) >> (count_ & 7)))
218 error_detail::raise<std::invalid_argument>("streamed index flag padding is nonzero");
219 }

231 void emit(bit_view source) {
232 while (!source.empty()) {
233 if (((buffered_bits_ | source.offset()) & 7) == 0 && source.size() >= buffer_bytes * 8) {
234 flush_tail();
235 auto bytes = source.size() >> 3;
236 stream_.append(source.storage().subspan(static_cast<std::size_t>(source.offset() >> 3),
237 static_cast<std::size_t>(bytes)));
238 source = source.subview(bytes << 3, source.size() - (bytes << 3));
239 continue;
240 }
241 auto count = std::min<std::uint64_t>(source.size(), buffer_bytes * 8 - buffered_bits_);
242 copy_bits(buffer_.data(), buffered_bits_, source.prefix(count));
243 buffered_bits_ += count;
244 source = source.subview(count, source.size() - count);
245 if (buffered_bits_ == buffer_bytes * 8) flush_tail();
246 }
247 }
248 void emit_zeroes(std::uint64_t bits) {
249 static constexpr std::array<std::byte, 4096> zero{};
250 while (bits) {
251 auto count = std::min<std::uint64_t>(bits, zero.size() * 8);
252 emit(bit_view(zero, count)); bits -= count;
253 }
254 }
255 void flush_tail() {
256 if (!buffered_bits_) return;
257 if (buffered_bits_ & 7)
258 buffer_[buffered_bits_ >> 3] &= std::byte(0xffu << (8 - (buffered_bits_ & 7)));
259 stream_.append(std::span(buffer_).first(static_cast<std::size_t>(byte_count(buffered_bits_))));
260 buffered_bits_ = 0;
261 }
262 void align_stream() {
263 static constexpr std::array<std::byte, 8> zero{};
264 auto gap = (0 - stream_.body_bytes()) & 7;
265 stream_.append(std::span(zero).first(static_cast<std::size_t>(gap)));
266 }
267 void emit_words(std::span<std::uint64_t const> words) {
268 if constexpr (std::endian::native == std::endian::little) stream_.append(std::as_bytes(words));
269 else {
270 while (!words.empty()) {
271 auto count = std::min(words.size(), buffer_bytes / 8);
272 for (std::size_t i = 0; i != count; ++i) file_detail::put(buffer_, i << 3, 8, words[i]);
273 stream_.append(std::span(buffer_).first(count << 3)); words = words.subspan(count);
274 }
275 }
276 }
277 void emit_samples(std::span<elias_fano_sample const> samples) {
278 static_assert(sizeof(elias_fano_sample) == 16 && offsetof(elias_fano_sample, first) == 0 &&
279 offsetof(elias_fano_sample, sparse) == 8);
280 if constexpr (std::endian::native == std::endian::little) stream_.append(std::as_bytes(samples));
281 else {
282 while (!samples.empty()) {
283 auto count = std::min(samples.size(), buffer_bytes / 16);
284 for (std::size_t i = 0; i != count; ++i) {
285 file_detail::put(buffer_, i << 4, 8, samples[i].first);
286 file_detail::put(buffer_, (i << 4) + 8, 8, samples[i].sparse);
287 }
288 stream_.append(std::span(buffer_).first(count << 4)); samples = samples.subspan(count);
289 }
290 }
