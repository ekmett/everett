77 void append(std::uint64_t retained, bit_view literal, bit_view value) {
78 require_active();
79 if ((literal.size() & (P::bits_per_unit - 1)) || (value.size() & (P::bits_per_unit - 1)))
80 error_detail::raise<std::invalid_argument>("profile file record disagrees with policy units");
81 auto value_units = value.size() >> P::unit_shift;
82 if (common_ && value_units != *common_)
83 error_detail::raise<std::invalid_argument>("profile file value disagrees with common width");
84 if (retained > previous_units_ || (!borrowed && count_ && literal.empty()))
85 error_detail::raise<std::invalid_argument>("profile file invalid known prefix");
86 auto literal_units = literal.size() >> P::unit_shift;
87 auto key_units = add(retained, literal_units);
88 (void)multiply(key_units, P::bits_per_unit);
89 auto next_count = add(count_, 1);
90 control_.bytes.clear(); control_.bit_size = 0;
91 std::uint64_t zero_bits = 0;
92 auto block = count_ % P::codec_block_size == 0;
93 if (block) write_count<P>(control_, retained);
94 else if constexpr (P::unit == profile_unit::bit && P::backspace_code == bit_backspace_code::golomb) {
95 auto backspace = previous_units_ - retained;
96 auto bits = backspace_bits<P>(backspace);
97 if (bits <= 129) write_backspace<P>(control_, backspace);
98 else {
99 // The unary quotient can be arbitrarily longer than the key's
100 // new literal. Emit it separately, never grow the control scratch.
101 constexpr auto modulus = P::backspace_parameter;
102 constexpr auto width = unsigned(std::bit_width(modulus - 1));
103 constexpr auto cutoff = golomb_cutoff<P>();
104 zero_bits = backspace / modulus;
105 auto remainder = backspace % modulus;
106 auto remainder_width = width - unsigned(remainder < cutoff);
107 resize(control_, 1 + remainder_width);
108 std::uint64_t at = 0;
109 put_fixed(control_, at, 1, 1);
110 put_fixed(control_, at, remainder < cutoff ? remainder : remainder + cutoff, remainder_width);
111 }
112 } else write_backspace<P>(control_, previous_units_ - retained);
113 write_count<P>(control_, literal_units);
114 if (!common_) write_count<P>(control_, value_units);
115 auto frame_bits = add(add(zero_bits, control_.bit_size), add(literal.size(), value.size()));
116 auto next_bits = add(fc_bits_, frame_bits);
117 if (next_bits > std::numeric_limits<std::uint64_t>::max() - 7)
118 error_detail::raise<std::length_error>("profile file FC extent is too large");
119 // Allocate the next navigation entry before any output is emitted.
120 if (block) offsets_.push_back((fc_bits_ >> P::unit_shift) - multiply(count_, common_.value_or(0)));
121 try {
122 emit_zeroes(zero_bits);
123 emit(control_.view()); emit(literal); emit(value);
124 } catch (...) { failed_ = true; throw; }
125 fc_bits_ = next_bits; previous_units_ = key_units; count_ = next_count;
126 }

134 object_seal_receipt finish_impl(index_file_sections<P> const * index) {
135 require_active();
136 if constexpr (borrowed) validate_index(*index);
137 auto extent = fc_bits_ >> P::unit_shift;
138 offsets_.push_back(extent - multiply(count_, common_.value_or(0)));
139 elias_fano offsets;
140 try { offsets = elias_fano::build(offsets_); }
141 catch (...) { offsets_.pop_back(); throw; }
142 offsets_.pop_back();
143 std::array<std::byte, directory_bytes> directory{};
144 auto magic = borrowed ? "IX02" : "KV02";
145 for (unsigned i = 0; i != 4; ++i) directory[i] = std::byte(magic[i]);
146 file_detail::put(directory, 4, 2, section_detail::version);
147 file_detail::put(directory, 6, 2, section_count);
148 file_detail::put(directory, 8, 8, extent);
149 file_detail::put(directory, 16, 8, previous_units_);
150 file_detail::put(directory, 24, 8, offsets.universe);
151 file_detail::put(directory, 32, 1, offsets.low_width);
152 std::array<std::uint64_t, section_count> lengths{byte_count(fc_bits_), multiply(offsets.low.size(), 8),
153 multiply(offsets.high.size(), 8), multiply(offsets.samples.size(), 16), multiply(offsets.sparse.size(), 8)};
154 if constexpr (borrowed) {
155 file_detail::put(directory, 40, 8, index->virtual_count);
156 section_detail::put_id(directory, 48, index->native_id);
157 if (index->target_id) {
158 file_detail::put(directory, 33, 1, 1);
159 section_detail::put_id(directory, 64, index->target_id->native);
160 section_detail::put_id(directory, 80, index->target_id->index);
161 }
162 lengths[5] = multiply(index->interleave.classes.size(), 8);
163 lengths[6] = multiply(index->interleave.checkpoints.size(), 8);
164 lengths[7] = index->false_borrows.size();
165 lengths[8] = multiply(index->cut_lcps.size(), 8);
166 }
167 std::uint64_t end = directory.size();
168 for (std::size_t i = 0; i != lengths.size(); ++i) {
169 auto start = add(end, 7) & ~std::uint64_t{7};
170 end = add(start, lengths[i]);
171 file_detail::put(directory, descriptor_offset + (i << 4), 8, start);
172 file_detail::put(directory, descriptor_offset + (i << 4) + 8, 8, lengths[i]);
173 }
174 file_header<P> header{kind, multiply(end, 1u << (3 - P::unit_shift)), count_, common_};
175 file_detail::validate_metadata(header);
176 if (file_detail::total_bytes<P>(header.extent) > std::uint64_t(std::numeric_limits<std::int64_t>::max()))
177 error_detail::raise<std::length_error>("profile file exceeds supported file offsets");
178 // EF allocation and complete metadata admission above remain retryable.
179 // Once final bytes are emitted, no retry may duplicate the directories.
180 try {
181 flush_tail();
182 align_stream(); emit_words(offsets.low);
183 align_stream(); emit_words(offsets.high);
184 align_stream(); emit_samples(offsets.samples);
185 align_stream(); emit_words(offsets.sparse);
186 if constexpr (borrowed) {
187 align_stream(); emit_words(index->interleave.classes);
188 align_stream(); emit_words(index->interleave.checkpoints);
189 align_stream(); stream_.append(index->false_borrows);
190 align_stream(); emit_words(index->cut_lcps);
191 }
192 return stream_.finish(header, directory);
193 } catch (...) { failed_ = true; throw; }
194 }

196 static std::optional<std::uint64_t> checked_width(std::optional<std::uint64_t> common) {
197 if constexpr (borrowed) {
198 if (common != std::optional<std::uint64_t>(0))

203 if (common) (void)multiply(*common, P::bits_per_unit);

206 void validate_index(index_file_sections<P> const & index) const {
207 auto groups = index.virtual_count / P::group_size + (index.virtual_count % P::group_size != 0);
208 if (index.virtual_count < count_ || count_ > std::numeric_limits<std::uint64_t>::max() - 7 ||
209 index.false_borrows.size() != ((count_ + 7) >> 3) || index.cut_lcps.size() != groups ||
210 index.interleave.virtual_count != index.virtual_count || (!index.target_id && count_))
211 error_detail::raise<std::invalid_argument>("streamed index metadata shape mismatch");
212 if (index.interleave.view().count() != count_)
213 error_detail::raise<std::invalid_argument>("streamed index borrowed count mismatch");
214 if ((count_ & 7) && (std::to_integer<unsigned>(index.false_borrows.back()) >> (count_ & 7)))
215 error_detail::raise<std::invalid_argument>("streamed index flag padding is nonzero");
216 }

228 void emit(bit_view source) {
229 while (!source.empty()) {
230 if (((buffered_bits_ | source.offset()) & 7) == 0 && source.size() >= buffer_bytes * 8) {
231 flush_tail();
232 auto bytes = source.size() >> 3;
233 stream_.append(source.storage().subspan(static_cast<std::size_t>(source.offset() >> 3),
234 static_cast<std::size_t>(bytes)));
235 source = source.subview(bytes << 3, source.size() - (bytes << 3));
236 continue;
237 }
238 auto count = std::min<std::uint64_t>(source.size(), buffer_bytes * 8 - buffered_bits_);
239 copy_bits(buffer_.data(), buffered_bits_, source.prefix(count));
240 buffered_bits_ += count;
241 source = source.subview(count, source.size() - count);
242 if (buffered_bits_ == buffer_bytes * 8) flush_tail();
243 }
244 }
245 void emit_zeroes(std::uint64_t bits) {
246 static constexpr std::array<std::byte, 4096> zero{};
247 while (bits) {
248 auto count = std::min<std::uint64_t>(bits, zero.size() * 8);
249 emit(bit_view(zero, count)); bits -= count;
250 }
251 }
252 void flush_tail() {
253 if (!buffered_bits_) return;
254 if (buffered_bits_ & 7)
255 buffer_[buffered_bits_ >> 3] &= std::byte(0xffu << (8 - (buffered_bits_ & 7)));
256 stream_.append(std::span(buffer_).first(static_cast<std::size_t>(byte_count(buffered_bits_))));
257 buffered_bits_ = 0;
258 }
259 void align_stream() {
260 static constexpr std::array<std::byte, 8> zero{};
261 auto gap = (0 - stream_.body_bytes()) & 7;
262 stream_.append(std::span(zero).first(static_cast<std::size_t>(gap)));
263 }
264 void emit_words(std::span<std::uint64_t const> words) {
265 if constexpr (std::endian::native == std::endian::little) stream_.append(std::as_bytes(words));
266 else {
267 while (!words.empty()) {
268 auto count = std::min(words.size(), buffer_bytes / 8);
269 for (std::size_t i = 0; i != count; ++i) file_detail::put(buffer_, i << 3, 8, words[i]);
270 stream_.append(std::span(buffer_).first(count << 3)); words = words.subspan(count);
271 }
272 }
273 }
274 void emit_samples(std::span<elias_fano_sample const> samples) {
275 static_assert(sizeof(elias_fano_sample) == 16 && offsetof(elias_fano_sample, first) == 0 &&
276 offsetof(elias_fano_sample, sparse) == 8);
277 if constexpr (std::endian::native == std::endian::little) stream_.append(std::as_bytes(samples));
278 else {
279 while (!samples.empty()) {
280 auto count = std::min(samples.size(), buffer_bytes / 16);
281 for (std::size_t i = 0; i != count; ++i) {
282 file_detail::put(buffer_, i << 4, 8, samples[i].first);
283 file_detail::put(buffer_, (i << 4) + 8, 8, samples[i].sparse);
284 }
285 stream_.append(std::span(buffer_).first(count << 4)); samples = samples.subspan(count);
286 }
287 }
288 }
