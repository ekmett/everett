145 // Seal one supplied object into an existing, durably established root. The
146 // caller reserves both identities and retains verified input ownership until
147 // later catalog adoption/recovery. Input spans stay readable and immutable
148 // throughout this call (including mmap-backed bodies). Only 96 header bytes
149 // are buffered; chunks need not be contiguous and empty chunks are allowed.
150 //
151 // Root and shards must share one supported local filesystem/device. Root
152 // ancestry and that filesystem are trusted: no concurrent shard renames,
153 // mount changes, in-place object mutation or hostile directory manipulation.
154 // Component handles reject shard symlinks; they are not a sandbox against
155 // mutation of the supplied root's ancestors. The filesystem must support
156 // hard links and directory fsync. Sealed means our writer stops mutating it;
157 // read-only permissions do not prevent an owner from changing them later.
158 template <class P, class Ops = posix_object_ops> struct object_writer {
159 using policy_type = P;

161 static object_seal_receipt seal(std::filesystem::path const & root,
162 object_id const & id, object_attempt_id const & attempt, file_header<P> const & header,
163 std::span<std::span<std::byte const> const> chunks, Ops & ops) {
164 if constexpr (!Ops::supported) {
165 (void)root; (void)id; (void)attempt; (void)header; (void)chunks; (void)ops;
166 throw std::system_error(std::make_error_code(std::errc::operation_not_supported),
167 "Diet object writer is unsupported on this platform");
168 } else {
169 file_detail::validate_metadata(header);
170 auto body_size = file_detail::body_bytes<P>(header.extent);
171 auto total_size = file_detail::total_bytes<P>(header.extent);
172 // POSIX file offsets are signed. The production implementations use
173 // 64-bit off_t; this stricter common bound also bounds write progress.
174 if (total_size > std::uint64_t(std::numeric_limits<std::int64_t>::max()))
175 throw std::length_error("Diet object exceeds supported file offsets");
176 std::uint64_t supplied = 0;
177 std::byte last{};
178 for (auto chunk : chunks) {
179 if (chunk.size() > body_size - supplied)
180 throw std::invalid_argument("Diet body extent mismatch");
181 supplied += chunk.size();
182 if (!chunk.empty()) last = chunk.back();
183 }
184 if (supplied != body_size) throw std::invalid_argument("Diet body extent mismatch");
185 if constexpr (P::unit == profile_unit::bit)
186 if ((header.extent & 7) && (std::to_integer<unsigned>(last) & ((1u << (8 - (header.extent & 7))) - 1)))
187 throw std::invalid_argument("noncanonical bit-profile tail padding");
188
189 operation run{root, id, attempt, header.kind, ops};
190 run.open();
191 std::array<std::byte, file_detail::header_bytes> placeholder{};
192 run.write_all(placeholder);
193 std::uint32_t crc = 0;
194 for (auto chunk : chunks) {
195 // Limit each CRC/write unit independently of the caller's span size.
196 // This is work streaming, not an allocated copy of the encoded body.
197 while (!chunk.empty()) {
198 auto part = chunk.first(std::min(chunk.size(), std::size_t{1} << 20));
199 run.write_all(part);
200 crc = crc32c(part, crc);
201 chunk = chunk.subspan(part.size());
202 }
203 }
204 auto encoded_header = encode_file_header(header, crc);
205 run.write_header(encoded_header);
206 run.stage = object_write_stage::body_written;
207 run.finish();
208 return {id, attempt, run.paths.final, total_size, crc, ops.barrier()};
209 }
210 }

242 operation(std::filesystem::path const & root, object_id const & id,
243 object_attempt_id const & attempt, file_kind kind, Ops & ops)

248 operation(operation const &) = delete;
249 operation & operator=(operation const &) = delete;
250 ~operation() {
251 // Error paths close handles once but never unlink outputs or retry any
252 // failed I/O. Preserve the first failure; close can itself lose an ack.
253 for (auto it = fds.rbegin(); it != fds.rend(); ++it)
254 if (*it >= 0) (void)ops.close(std::exchange(*it, -1));
255 }
256 [[noreturn]] void fail(char const * what, int error = errno) const {
257 throw object_write_error(error ? error : EIO, what, id, attempt, paths, stage);
258 }
259 void open() {
260 fds[0] = ops.open_root(root);
261 if (fds[0] < 0) fail("open object root");
262 for (unsigned i = 1; i <= 2; ++i) {
263 auto const & name = i == 1 ? first : second;
264 if (ops.make_directory(fds[i - 1], name.c_str()) < 0 && errno != EEXIST)
265 fail("create object shard");
266 fds[i] = ops.open_directory(fds[i - 1], name.c_str());
267 if (fds[i] < 0) fail("open object shard");
268 }
269 fds[3] = ops.create_private(fds[2], private_name.c_str());
270 if (fds[3] < 0) fail("create private object");
271 stage = object_write_stage::private_created;
272 }
273 void write_all(std::span<std::byte const> bytes) {
274 while (!bytes.empty()) {
275 auto part = bytes.first(std::min(bytes.size(), std::size_t{1} << 20));
276 auto count = ops.write(fds[3], part);
277 if (count < 0) { if (errno == EINTR) continue; fail("write object body"); }
278 if (count == 0 || std::uint64_t(count) > part.size()) fail("write object body", EIO);
279 bytes = bytes.subspan(static_cast<std::size_t>(count));
280 }
281 }
282 void write_header(std::span<std::byte const> bytes) {
283 write_at(bytes, 0, "write object header");
284 }
285 void write_at(std::span<std::byte const> bytes, std::uint64_t offset, char const * operation) {
286 while (!bytes.empty()) {
287 auto part = bytes.first(std::min(bytes.size(), std::size_t{1} << 20));
288 auto count = ops.write_at(fds[3], part, offset);
289 if (count < 0) { if (errno == EINTR) continue; fail(operation); }
290 if (count == 0 || std::uint64_t(count) > part.size()) fail(operation, EIO);
291 offset += static_cast<std::uint64_t>(count);
292 bytes = bytes.subspan(static_cast<std::size_t>(count));
293 }
294 }
295 void finish() {
296 if (ops.make_read_only(fds[3]) < 0) fail("protect object");
297 // No retry after synchronization failure, even EINTR.
298 if (ops.sync_file(fds[3]) < 0) fail("sync object contents");
299 stage = object_write_stage::content_synced;
300 if (ops.install(fds[2], private_name.c_str(), final_name.c_str()) < 0)
301 fail("install object without replacement");
302 stage = object_write_stage::name_installed;
303 // Sync existing ancestors too: another legitimate creator may have
304 // just made them. Root's own durable parent entry is a caller premise.
305 for (unsigned i = 3; i-- > 0;)
306 if (ops.sync_directory(fds[i]) < 0) fail("sync object directory");
307 stage = object_write_stage::names_synced;
308 if (ops.remove_private(fds[2], private_name.c_str()) < 0) fail("remove private object name");
309 stage = object_write_stage::private_removed;
310 if (ops.sync_directory(fds[2]) < 0) fail("sync private name removal");
311 // Also persists link-count changes; on Apple this full drive barrier
312 // follows the directory writeback requests. No unsupported fallback.
313 if (ops.sync_file(fds[3]) < 0) fail("sync installed object");
314 for (unsigned i = 4; i-- > 0;) {
315 auto fd = std::exchange(fds[i], -1);
316 if (ops.close(fd) < 0) fail("close sealed object handles");
317 }
318 stage = object_write_stage::complete;
319 }
320 };
