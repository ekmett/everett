142 // Seal one supplied object into an existing, durably established root. The
143 // caller reserves both identities and retains verified input ownership until
144 // later catalog adoption/recovery. Input spans stay readable and immutable
145 // throughout this call (including mmap-backed bodies). Only 96 header bytes
146 // are buffered; chunks need not be contiguous and empty chunks are allowed.
147 //
148 // Root and shards must share one supported local filesystem/device. Root
149 // ancestry and that filesystem are trusted: no concurrent shard renames,
150 // mount changes, in-place object mutation or hostile directory manipulation.
151 // Component handles reject shard symlinks; they are not a sandbox against
152 // mutation of the supplied root's ancestors. The filesystem must support
153 // hard links and directory fsync. Sealed means our writer stops mutating it;
154 // read-only permissions do not prevent an owner from changing them later.
155 template <class P, class Ops = posix_object_ops> struct object_writer {
156 using policy_type = P;

158 static object_seal_receipt seal(std::filesystem::path const & root,
159 object_id const & id, object_attempt_id const & attempt, file_header<P> const & header,
160 std::span<std::span<std::byte const> const> chunks, Ops & ops) {
161 if constexpr (!Ops::supported) {
162 (void)root; (void)id; (void)attempt; (void)header; (void)chunks; (void)ops;
163 throw std::system_error(std::make_error_code(std::errc::operation_not_supported),
164 "Everett object writer is unsupported on this platform");
165 } else {
166 file_detail::validate_metadata(header);
167 auto body_size = file_detail::body_bytes<P>(header.extent);
168 auto total_size = file_detail::total_bytes<P>(header.extent);
169 // POSIX file offsets are signed. The production implementations use
170 // 64-bit off_t; this stricter common bound also bounds write progress.
171 if (total_size > std::uint64_t(std::numeric_limits<std::int64_t>::max()))
172 throw std::length_error("Everett object exceeds supported file offsets");
173 std::uint64_t supplied = 0;
174 std::byte last{};
175 for (auto chunk : chunks) {
176 if (chunk.size() > body_size - supplied)
177 throw std::invalid_argument("Everett body extent mismatch");
178 supplied += chunk.size();
179 if (!chunk.empty()) last = chunk.back();
180 }
181 if (supplied != body_size) throw std::invalid_argument("Everett body extent mismatch");
182 if constexpr (P::unit == profile_unit::bit)
183 if ((header.extent & 7) && (std::to_integer<unsigned>(last) & ((1u << (8 - (header.extent & 7))) - 1)))
184 throw std::invalid_argument("noncanonical bit-profile tail padding");
185
186 operation run{root, id, attempt, header.kind, ops};
187 run.open();
188 std::array<std::byte, file_detail::header_bytes> placeholder{};
189 run.write_all(placeholder);
190 std::uint32_t crc = 0;
191 for (auto chunk : chunks) {
192 // Limit each CRC/write unit independently of the caller's span size.
193 // This is work streaming, not an allocated copy of the encoded body.
194 while (!chunk.empty()) {
195 auto part = chunk.first(std::min(chunk.size(), std::size_t{1} << 20));
196 run.write_all(part);
197 crc = crc32c(part, crc);
198 chunk = chunk.subspan(part.size());
199 }
200 }
201 auto encoded_header = encode_file_header(header, crc);
202 run.write_header(encoded_header);
203 run.stage = object_write_stage::body_written;
204 run.finish();
205 return {id, attempt, run.paths.final, total_size, crc, ops.barrier()};
206 }
207 }

238 operation(std::filesystem::path const & root, object_id const & id,
239 object_attempt_id const & attempt, file_kind kind, Ops & ops)
240 : root(root), id(id), attempt(attempt), ops(ops),
241 paths(object_output_paths(root, id, attempt, kind)),
242 first(id.hex().substr(0, 2)), second(id.hex().substr(2, 2)),
243 final_name(paths.final.filename().string()), private_name(paths.private_output.filename().string()) {}
244 operation(operation const &) = delete;
245 operation & operator=(operation const &) = delete;
246 ~operation() {
247 // Error paths close handles once but never unlink outputs or retry any
248 // failed I/O. Preserve the first failure; close can itself lose an ack.
249 for (auto it = fds.rbegin(); it != fds.rend(); ++it)
250 if (*it >= 0) (void)ops.close(std::exchange(*it, -1));
251 }
252 [[noreturn]] void fail(char const * what, int error = errno) const {
253 throw object_write_error(error ? error : EIO, what, id, attempt, paths, stage);
254 }
255 void open() {
256 fds[0] = ops.open_root(root);
257 if (fds[0] < 0) fail("open object root");
258 for (unsigned i = 1; i <= 2; ++i) {
259 auto const & name = i == 1 ? first : second;
260 if (ops.make_directory(fds[i - 1], name.c_str()) < 0 && errno != EEXIST)
261 fail("create object shard");
262 fds[i] = ops.open_directory(fds[i - 1], name.c_str());
263 if (fds[i] < 0) fail("open object shard");
264 }
265 fds[3] = ops.create_private(fds[2], private_name.c_str());
266 if (fds[3] < 0) fail("create private object");
267 stage = object_write_stage::private_created;
268 }
269 void write_all(std::span<std::byte const> bytes) {
270 while (!bytes.empty()) {
271 auto part = bytes.first(std::min(bytes.size(), std::size_t{1} << 20));
272 auto count = ops.write(fds[3], part);
273 if (count < 0) { if (errno == EINTR) continue; fail("write object body"); }
274 if (count == 0 || std::uint64_t(count) > part.size()) fail("write object body", EIO);
275 bytes = bytes.subspan(static_cast<std::size_t>(count));
276 }
277 }
278 void write_header(std::span<std::byte const> bytes) {
279 std::uint64_t offset = 0;
280 while (!bytes.empty()) {
281 auto count = ops.write_at(fds[3], bytes, offset);
282 if (count < 0) { if (errno == EINTR) continue; fail("write object header"); }
283 if (count == 0 || std::uint64_t(count) > bytes.size()) fail("write object header", EIO);
284 offset += static_cast<std::uint64_t>(count);
285 bytes = bytes.subspan(static_cast<std::size_t>(count));
286 }
287 }
288 void finish() {
289 if (ops.make_read_only(fds[3]) < 0) fail("protect object");
290 // No retry after synchronization failure, even EINTR.
291 if (ops.sync_file(fds[3]) < 0) fail("sync object contents");
292 stage = object_write_stage::content_synced;
293 if (ops.install(fds[2], private_name.c_str(), final_name.c_str()) < 0)
294 fail("install object without replacement");
295 stage = object_write_stage::name_installed;
296 // Sync existing ancestors too: another legitimate creator may have
297 // just made them. Root's own durable parent entry is a caller premise.
298 for (unsigned i = 3; i-- > 0;)
299 if (ops.sync_directory(fds[i]) < 0) fail("sync object directory");
300 stage = object_write_stage::names_synced;
301 if (ops.remove_private(fds[2], private_name.c_str()) < 0) fail("remove private object name");
302 stage = object_write_stage::private_removed;
303 if (ops.sync_directory(fds[2]) < 0) fail("sync private name removal");
304 // Also persists link-count changes; on Apple this full drive barrier
305 // follows the directory writeback requests. No unsupported fallback.
306 if (ops.sync_file(fds[3]) < 0) fail("sync installed object");
307 for (unsigned i = 4; i-- > 0;) {
308 auto fd = std::exchange(fds[i], -1);
309 if (ops.close(fd) < 0) fail("close sealed object handles");
310 }
311 stage = object_write_stage::complete;
312 }
313 };
314 };
