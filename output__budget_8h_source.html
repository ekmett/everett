22 // This accounts for caller-declared allocation capacities. It does not
23 // measure allocations or bound cursor workspace and user callbacks. Builders
24 // acquire before retaining output; finished owners keep the lease until the

53 // releasing the allowance covering the finished object's allocations.

58 }

72 auto used = state_->used.load(std::memory_order_relaxed);
73 for (;;) {
74 if (bytes > state_->limit - used) return std::nullopt;
75 if (state_->used.compare_exchange_weak(used, used + bytes, std::memory_order_relaxed))
