121 inline int compare_bits(bit_view a, bit_view b) {
122 auto count = std::min(a.size(), b.size());
123 std::uint64_t at = 0;
124 if (a.offset() % 8 == 0 && b.offset() % 8 == 0) {
125 for (; at + 8 <= count; at += 8) {
126 auto x = std::to_integer<unsigned>(a.storage()[(a.offset() + at) / 8]);
127 auto y = std::to_integer<unsigned>(b.storage()[(b.offset() + at) / 8]);
128 if (x != y) return x < y ? -1 : 1;
129 }
130 }
131 for (; at != count; ++at) if (a.at(at) != b.at(at)) return a.at(at) ? 1 : -1;
132 return a.size() < b.size() ? -1 : a.size() > b.size() ? 1 : 0;
133 }

227 value.bytes.resize(static_cast<std::size_t>(bytes));

231 inline void copy_into(bit_string & target, std::uint64_t first, bit_view source) {
232 if (first > target.bit_size || source.size() > target.bit_size - first)
233 throw std::out_of_range("profile copy exceeds destination");
234 if (!source.size()) return;
235 if (first % 8 == 0 && source.offset() % 8 == 0 && source.size() % 8 == 0) {
236 std::copy_n(source.storage().begin() + static_cast<std::ptrdiff_t>(source.offset() / 8),
237 static_cast<std::size_t>(source.size() / 8),
238 target.bytes.begin() + static_cast<std::ptrdiff_t>(first / 8));
239 } else {
240 for (std::uint64_t i = 0; i != source.size(); ++i) {
241 auto bit = first + i;
242 auto mask = static_cast<std::byte>(1u << (7 - bit % 8));
243 if (source.at(i)) target.bytes[bit / 8] |= mask;
244 else target.bytes[bit / 8] &= ~mask;
245 }
246 }

263 template <class P> void write_count(bit_string & target, std::uint64_t value) {
264 if constexpr (P::unit == profile_unit::byte) {
265 while (value >= 128) { append_byte(target, unsigned(value & 127) | 128); value >>= 7; }
266 append_byte(target, unsigned(value));
267 } else {
268 if (value == std::numeric_limits<std::uint64_t>::max()) {
269 for (unsigned i = 0; i != 64; ++i) append_bit(target, false);
270 append_bit(target, true);
271 for (unsigned i = 0; i != 64; ++i) append_bit(target, false);
272 } else {
273 auto code = value + 1;
274 auto zeros = unsigned(std::bit_width(code) - 1);
275 for (unsigned i = 0; i != zeros; ++i) append_bit(target, false);
276 for (unsigned i = zeros + 1; i != 0; --i) append_bit(target, (code >> (i - 1)) & 1);
277 }
278 }

280 template <class P> std::uint64_t read_count(bit_view data, std::uint64_t & offset) {
281 if constexpr (P::unit == profile_unit::byte) {
282 std::uint64_t value = 0;
283 for (unsigned shift = 0; shift <= 63; shift += 7) {
284 if (offset >= data.size() / 8) throw std::invalid_argument("truncated profile count");
285 auto part = std::to_integer<unsigned>(data.storage()[offset++]);
286 if (shift == 63 && part > 1) throw std::invalid_argument("overflowing profile count");
287 value |= std::uint64_t(part & 127) << shift;
288 if (!(part & 128)) {
289 if (shift && !(part & 127)) throw std::invalid_argument("noncanonical profile count");
290 return value;
291 }
292 }
293 throw std::invalid_argument("overflowing profile count");
294 } else {
295 auto next = [&] {
296 if (offset >= data.size()) throw std::invalid_argument("truncated exponential-Golomb count");
297 return data.at(offset++);
298 };
299 unsigned zeros = 0;
300 while (!next()) if (++zeros > 64) throw std::invalid_argument("overflowing exponential-Golomb count");
301 std::uint64_t suffix = 0;
302 for (unsigned i = 0; i != zeros; ++i) suffix = (suffix << 1) | unsigned(next());
303 if (zeros == 64) {
304 if (suffix) throw std::invalid_argument("overflowing exponential-Golomb count");
305 return std::numeric_limits<std::uint64_t>::max();
306 }
307 return ((std::uint64_t{1} << zeros) - 1) + suffix;
308 }
309 }
310 // Fixed-width suffixes are MSB-first. Width 64 is valid; no shift uses 64.
311 inline void put_fixed(bit_string & target, std::uint64_t & at, std::uint64_t value, unsigned width) {
312 for (unsigned remaining = width; remaining != 0; --remaining, ++at) {
313 auto mask = static_cast<std::byte>(1u << (7 - at % 8));
314 if ((value >> (remaining - 1)) & 1) target.bytes[at / 8] |= mask;
315 else target.bytes[at / 8] &= ~mask;
316 }
317 }
318 inline std::uint64_t read_fixed(bit_view data, std::uint64_t & at, unsigned width) {
319 if (at > data.size() || width > data.size() - at)
320 throw std::invalid_argument("truncated backspace remainder");
321 std::uint64_t value = 0;
322 for (unsigned i = 0; i != width; ++i) value = (value << 1) | unsigned(data.at(at++));
323 return value;
324 }
325 template <class P> constexpr std::uint64_t golomb_cutoff() {
326 constexpr auto width = std::bit_width(P::backspace_parameter - 1);
327 if constexpr (width == 64) return std::uint64_t{0} - P::backspace_parameter;
328 else return (std::uint64_t{1} << width) - P::backspace_parameter;
329 }
330 // Golomb length includes its unary quotient: a long predecessor can make
331 // a short key's backspace expensive even at an LPFC literal restart. Charge
332 // the actual codeword length as well as encoded distance from the anchor.
333 template <class P> std::uint64_t backspace_bits(std::uint64_t value) {
334 static_assert(P::unit == profile_unit::bit);
335 if constexpr (P::backspace_code == bit_backspace_code::exponential_golomb) {
336 auto quotient = value >> P::backspace_parameter;
337 auto prefix = quotient == std::numeric_limits<std::uint64_t>::max()
338 ? 129 : 2 * std::bit_width(quotient + 1) - 1;
339 return prefix + P::backspace_parameter;
340 } else {
341 constexpr auto modulus = P::backspace_parameter;
342 constexpr auto width = std::bit_width(modulus - 1);
343 auto remainder_width = width - (value % modulus < golomb_cutoff<P>());
344 return add(add(value / modulus, 1), remainder_width);
345 }
346 }
347 template <class P> void write_backspace(bit_string & target, std::uint64_t value) {
348 if constexpr (P::unit == profile_unit::byte) write_count<P>(target, value);
349 else {
350 // Check the complete length and allocate before emitting a unary run.
351 // Fresh bytes and the existing canonical tail supply its zero bits.
352 auto at = target.bit_size;
353 resize(target, add(at, backspace_bits<P>(value)));
354 if constexpr (P::backspace_code == bit_backspace_code::exponential_golomb) {
355 auto quotient = value >> P::backspace_parameter;
356 if (quotient == std::numeric_limits<std::uint64_t>::max()) {
357 at += 64;
358 put_fixed(target, at, 1, 1);
359 put_fixed(target, at, 0, 64);
360 } else {
361 auto code = quotient + 1;
362 auto width = unsigned(std::bit_width(code));
363 at += width - 1;
364 put_fixed(target, at, code, width);
365 }
366 put_fixed(target, at, value, unsigned(P::backspace_parameter));
367 } else {
368 constexpr auto modulus = P::backspace_parameter;
369 constexpr auto width = unsigned(std::bit_width(modulus - 1));
370 constexpr auto cutoff = golomb_cutoff<P>();
371 at += value / modulus;
372 put_fixed(target, at, 1, 1);
373 auto remainder = value % modulus;
374 if (remainder < cutoff) put_fixed(target, at, remainder, width - 1);
375 else put_fixed(target, at, remainder + cutoff, width);
376 }
377 }
378 }
379 template <class P> std::uint64_t read_backspace(bit_view data, std::uint64_t & offset) {
380 if constexpr (P::unit == profile_unit::byte) return read_count<P>(data, offset);
381 else if constexpr (P::backspace_code == bit_backspace_code::exponential_golomb) {
382 auto quotient = read_count<P>(data, offset);
383 if (quotient > (std::numeric_limits<std::uint64_t>::max() >> P::backspace_parameter))
384 throw std::invalid_argument("overflowing exponential-Golomb backspace");
385 auto remainder = read_fixed(data, offset, unsigned(P::backspace_parameter));
386 return (quotient << P::backspace_parameter) | remainder;
387 } else {
388 constexpr auto modulus = P::backspace_parameter;
389 constexpr auto width = unsigned(std::bit_width(modulus - 1));
390 constexpr auto cutoff = golomb_cutoff<P>();
391 constexpr auto limit = std::numeric_limits<std::uint64_t>::max() / modulus;
392 std::uint64_t quotient = 0;
393 while (!read_fixed(data, offset, 1)) {
394 if (quotient == limit) throw std::invalid_argument("overflowing Golomb quotient");
395 ++quotient;
396 }
397 std::uint64_t remainder = 0;
398 if constexpr (width != 0) {
399 remainder = read_fixed(data, offset, width - 1);
400 if (remainder >= cutoff) remainder = ((remainder << 1) | read_fixed(data, offset, 1)) - cutoff;
401 }
402 auto base = quotient * modulus;
403 if (remainder > std::numeric_limits<std::uint64_t>::max() - base)
404 throw std::invalid_argument("overflowing Golomb backspace");
405 return base + remainder;
406 }

485 // The supplied anchor must share the first record's retained prefix. An
486 // anchor between its actual predecessor and that key suffices; conservative
487 // coding can also permit an upper frontier. The copied prefix is not an
488 // exact LCP with that anchor. Callback key views expire on the next call.
489 template <class F> void visit_window(std::uint64_t first, std::uint64_t last,
490 profile_anchor<P> anchor, std::uint64_t prefix_limit, F && callback) const {
491 if (first > last || last > size() || last - first > P::group_size)
492 throw std::out_of_range("profile window exceeds the policy group size");
493 if (first == last) return;
494 if (anchor.prefix.size() % P::bits_per_unit ||
495 anchor.prefix.size() / P::bits_per_unit > anchor.full_units)
496 throw std::invalid_argument("profile anchor units");

498 std::min(prefix_limit, anchor.prefix.size() / P::bits_per_unit), P::bits_per_unit);
499 auto scratch = bit_string::copy(anchor.prefix.prefix(prefix_bits));
500 auto context = anchor.full_units;
501 auto record = encoded_at(first);
502 for (auto i = first; i != last; ++i) {
503 decode_into(record, prefix_limit, scratch, context);
504 if (!callback(profile_item<P>{i, {scratch.view(), context}, record.value})) return;
505 if (i + 1 != last) record = next_record(record, i + 1);

515 if (record.previous_units != context) throw std::invalid_argument("profile predecessor length mismatch");
516 decode_into(record, std::numeric_limits<std::uint64_t>::max(), scratch, context);
517 if (!callback(profile_item<P>{i, {scratch.view(), context}, record.value})) return;
518 if (i + 1 != size()) record = next_record(record, i + 1);
519 }
520 if (record.next_offset != metadata_.extent) throw std::invalid_argument("trailing profile data");
521 }

523 // Ordinary FC can traverse the entire prefix chain. LPFC bounds backward
524 // encoded distance by the full key length, not by prefix_limit; locating a
525 // record also scans at most K-1 headers before it. Values are copied in full.
526 profile_decoded_record<P> reconstruct_at(std::uint64_t ordinal,
527 std::uint64_t prefix_limit = std::numeric_limits<std::uint64_t>::max()) const {
528 auto record = encoded_at(ordinal);
529 profile_decoded_record<P> result;
530 result.full_units = record.key_units;
531 result.value = bit_string::copy(record.value);
532 auto need = std::min(record.key_units, prefix_limit);
533 profile_detail::resize(result.prefix, profile_detail::multiply(need, P::bits_per_unit));
534 while (need) {
535 if (record.key_units < need) throw std::invalid_argument("profile predecessor prefix is too short");
536 if (record.retained < need) {
537 auto copy = profile_detail::multiply(need - record.retained, P::bits_per_unit);
538 profile_detail::copy_into(result.prefix, profile_detail::multiply(record.retained, P::bits_per_unit),
539 record.suffix.prefix(copy));
540 need = record.retained;
541 }
542 if (!need) break;
543 if (!ordinal) throw std::invalid_argument("nonliteral first profile key");
544 auto expected = record.previous_units;
545 record = encoded_at(--ordinal);
546 if (record.key_units != expected) throw std::invalid_argument("profile predecessor length mismatch");
547 }
548 return result;
549 }

551 private:
552 friend struct profile_cursor<P, Role>;
553 std::span<std::byte const> bytes_;
554 select_groups_view<P::group_size> offsets_;
555 profile_metadata metadata_;
556 bit_view data_;

558 profile_encoded_record parse(std::uint64_t at, std::uint64_t previous) const {
559 auto backspace = profile_detail::read_backspace<P>(data_, at);
560 auto suffix = profile_detail::read_count<P>(data_, at);
561 auto value = metadata_.common_value_width ? *metadata_.common_value_width : profile_detail::read_count<P>(data_, at);
562 if (backspace > previous) throw std::invalid_argument("profile backspace exceeds predecessor");
563 auto retained = previous - backspace;
564 auto key_units = profile_detail::add(retained, suffix);
565 if (at > metadata_.extent || suffix > metadata_.extent - at || value > metadata_.extent - at - suffix)
566 throw std::invalid_argument("truncated profile payload");
567 auto suffix_bits = profile_detail::multiply(suffix, P::bits_per_unit);
568 auto value_bits = profile_detail::multiply(value, P::bits_per_unit);
569 auto key_data = data_.subview(profile_detail::multiply(at, P::bits_per_unit), suffix_bits);
570 at += suffix;
571 auto value_data = data_.subview(profile_detail::multiply(at, P::bits_per_unit), value_bits);
572 return {previous, backspace, retained, key_units, value, key_data, value_data, at + value};
573 }
574
575 profile_encoded_record next_record(profile_encoded_record const & previous, std::uint64_t ordinal) const {
576 auto at = previous.next_offset;
577 if (ordinal % P::group_size == 0) {
578 if (offsets_.offset(ordinal / P::group_size, metadata_.common_value_width.value_or(0)) != at)
579 throw std::invalid_argument("profile group offset mismatch");
580 if (profile_detail::read_count<P>(data_, at) != previous.key_units)
581 throw std::invalid_argument("profile group predecessor mismatch");
582 }
583 return parse(at, previous.key_units);
584 }

586 static void decode_into(profile_encoded_record const & record, std::uint64_t limit,
587 bit_string & scratch, std::uint64_t & context) {
588 if (record.retained > context) throw std::invalid_argument("profile anchor is too short");
589 auto keep = std::min(record.retained, limit);
590 auto keep_bits = profile_detail::multiply(keep, P::bits_per_unit);
591 if (keep_bits > scratch.bit_size) throw std::invalid_argument("profile anchor prefix is incomplete");
592 profile_detail::resize(scratch, keep_bits);
593 auto full = std::min(record.key_units, limit);
594 profile_detail::append(scratch, record.suffix.prefix(profile_detail::multiply(full - keep, P::bits_per_unit)));
595 context = record.key_units;
596 }
597 };

599 // Resumable sequential traversal. Each encoded record is parsed once and
600 // its value remains a view of the original payload. Only the current key is
601 // reconstructed. The view's sections must outlive this cursor; peek's key
602 // view additionally expires when this cursor advances or is destroyed.
603 template <class P, stream_role Role> struct profile_cursor {
604 using policy_type = P;
605 static constexpr stream_role role = Role;
606
607 explicit profile_cursor(profile_view<P, Role> view) : view_(view) {
608 if (!done()) {
609 record_ = view_.encoded_at(0);
610 profile_view<P, Role>::decode_into(record_, std::numeric_limits<std::uint64_t>::max(), scratch_, context_);
611 }
612 }
613
614 bool done() const noexcept { return ordinal_ == view_.size(); }
615 std::uint64_t ordinal() const noexcept { return ordinal_; }
616 profile_item<P> peek() const & {
617 if (done()) throw std::out_of_range("profile cursor at end");
618 return {ordinal_, {scratch_.view(), context_}, record_.value};
619 }
620 profile_item<P> peek() const && = delete;
621
622 void advance() {
623 if (done()) throw std::out_of_range("profile cursor at end");
624 if (ordinal_ + 1 == view_.size()) {
625 if (record_.next_offset != view_.metadata_.extent)
626 throw std::invalid_argument("trailing profile data");
627 ++ordinal_;
628 return;
629 }
630 auto next = view_.next_record(record_, ordinal_ + 1);
631 profile_view<P, Role>::decode_into(next, std::numeric_limits<std::uint64_t>::max(), scratch_, context_);
632 record_ = next;
633 ++ordinal_;
634 }
635
636 private:
637 profile_view<P, Role> view_;
638 bit_string scratch_;
639 std::uint64_t context_ = 0;
640 profile_encoded_record record_;
641 std::uint64_t ordinal_ = 0;
642 };

644 template <class P, stream_role Role>
645 profile_cursor<P, Role> profile_view<P, Role>::cursor() const { return profile_cursor<P, Role>(*this); }

647 template <class P, stream_role Role = stream_role::native> struct profile_array {
648 using policy_type = P;
649 static constexpr stream_role role = Role;
650
651 static profile_array build(std::span<profile_record const> records,
652 std::span<std::uint64_t const> prefix_ceilings = {}, std::uint64_t restart_factor = 0) {
653 if (!prefix_ceilings.empty() && prefix_ceilings.size() != records.size())
654 throw std::invalid_argument("one profile prefix ceiling is required per record");
655 if (restart_factor && restart_factor < 3) throw std::invalid_argument("LPFC factor must be at least three");
656 profile_array result;
657 std::optional<std::uint64_t> common;
658 if constexpr (Role == stream_role::borrowed) common = 0;
659 else if constexpr (P::fixed_width) common = P::value_width;
660 else if (records.empty()) common = 0;
661 else common = records.front().value.view().size() / P::bits_per_unit;
662 for (auto const & record : records) {
663 auto key_bits = record.key.view().size();
664 auto value_bits = record.value.view().size();
665 if (key_bits % P::bits_per_unit || value_bits % P::bits_per_unit)
666 throw std::invalid_argument("record length does not match profile unit");
667 auto width = value_bits / P::bits_per_unit;
668 if constexpr (Role == stream_role::borrowed) {
669 if (width) throw std::invalid_argument("borrowed profile cannot carry values");
670 } else if constexpr (P::fixed_width) {
671 if (width != *P::value_width) throw std::invalid_argument("value does not match fixed policy width");
672 } else {
673 if (common && *common != width) common.reset();
674 }
675 }
676 result.metadata_.common_value_width = common;
677 result.metadata_.record_count = records.size();
678 bit_string data;
679 bit_view previous;
680 std::uint64_t previous_units = 0;
681 std::uint64_t anchor_offset = 0;
682 std::vector<std::uint64_t> offsets;
683 for (std::size_t i = 0; i != records.size(); ++i) {
684 auto key = records[i].key.view();
685 auto value = records[i].value.view();
686 if (i && compare_bits(previous, key) > 0) throw std::invalid_argument("profile keys must be sorted");
687 auto position = data.bit_size / P::bits_per_unit;
688 if (i % P::group_size == 0) {
689 offsets.push_back(position - profile_detail::multiply(i, common.value_or(0)));
690 profile_detail::write_count<P>(data, previous_units);
691 }
692 auto key_units = key.size() / P::bits_per_unit;
693 auto retained = common_prefix_units<P>(previous, key);
694 if (!prefix_ceilings.empty()) retained = std::min(retained, prefix_ceilings[i]);
695 auto start = data.bit_size / P::bits_per_unit;
696 if (retained && restart_factor && key_units <= std::numeric_limits<std::uint64_t>::max() / restart_factor &&
697 start - anchor_offset > restart_factor * key_units) retained = 0;
698 if (!retained) anchor_offset = start;
699 profile_detail::write_backspace<P>(data, previous_units - retained);
700 profile_detail::write_count<P>(data, key_units - retained);
701 if (!common) profile_detail::write_count<P>(data, value.size() / P::bits_per_unit);
702 profile_detail::append(data, key.subview(profile_detail::multiply(retained, P::bits_per_unit),
703 key.size() - profile_detail::multiply(retained, P::bits_per_unit)));
704 profile_detail::append(data, value);
705 previous = key;
706 previous_units = key_units;
707 }
708 result.metadata_.extent = data.bit_size / P::bits_per_unit;
709 offsets.push_back(result.metadata_.extent - profile_detail::multiply(records.size(), common.value_or(0)));
710 result.offsets_ = select_groups<P::group_size>::build(offsets, records.size());
711 result.bytes_ = std::move(data.bytes);
712 return result;
713 }
714
715 std::uint64_t size() const noexcept { return metadata_.record_count; }
716 std::span<std::byte const> bytes() const noexcept { return bytes_; }
717 profile_metadata const & metadata() const noexcept { return metadata_; }
718 select_groups<P::group_size> const & group_offsets() const noexcept { return offsets_; }
719 profile_view<P, Role> view() const & { return {bytes_, offsets_.view(), metadata_}; }
720 profile_view<P, Role> view() const && = delete;
721
722 private:
723 friend struct profile_borrowed_writer<P>;
724 std::vector<std::byte> bytes_;
725 select_groups<P::group_size> offsets_;
726 profile_metadata metadata_ = profile_detail::initial_metadata<P, Role>();
727 };

729 // Incremental modified-FC output. The caller supplies any boundary-dependent
730 // prefix ceiling before appending that key. There is no all-keys staging:
731 // retained state is the previous key, encoded bytes and one offset per group.
732 // With the same ceilings this produces exactly the batch borrowed encoding
733 // (restart_factor == 0), including checkpoints, tail padding and EF metadata.
734 template <class P> struct profile_borrowed_writer {
735 using policy_type = P;
736 static constexpr stream_role role = stream_role::borrowed;
737
738 std::uint64_t size() const noexcept { return count_; }
739 bool finished() const noexcept { return finished_; }
740
741 void append(bit_view key, std::uint64_t prefix_ceiling = std::numeric_limits<std::uint64_t>::max()) {
742 if (finished_) throw std::logic_error("borrowed profile writer is finished");
743 if (key.size() % P::bits_per_unit) throw std::invalid_argument("key length does not match profile unit");
744 auto previous = previous_.view();
745 if (count_ && compare_bits(previous, key) > 0) throw std::invalid_argument("profile keys must be sorted");
746 auto next_count = profile_detail::add(count_, 1);
747 auto retained = std::min(common_prefix_units<P>(previous, key), prefix_ceiling);
748 auto previous_units = previous.size() / P::bits_per_unit;
749 auto key_units = key.size() / P::bits_per_unit;
750 auto saved_bits = data_.bit_size;
751 auto saved_offsets = offsets_.size();
752 // Take ownership before modifying output, including when the caller's
753 // key view points into other mutable decoding scratch.
754 auto next_previous = bit_string::copy(key);
755 try {
756 if (count_ % P::group_size == 0) {
757 offsets_.push_back(data_.bit_size / P::bits_per_unit);
758 profile_detail::write_count<P>(data_, previous_units);
759 }
760 profile_detail::write_backspace<P>(data_, previous_units - retained);
761 profile_detail::write_count<P>(data_, key_units - retained);
762 auto retained_bits = profile_detail::multiply(retained, P::bits_per_unit);
763 profile_detail::append(data_, key.subview(retained_bits, key.size() - retained_bits));
764 } catch (...) {
765 profile_detail::resize(data_, saved_bits);
766 offsets_.resize(saved_offsets);
767 throw;
768 }
769 previous_ = std::move(next_previous);
770 count_ = next_count;
771 }
772
773 // Finalizing EF takes work proportional to the staged group offsets.
774 // This is not a bounded-byte or durable checkpoint operation.
775 profile_array<P, stream_role::borrowed> finish() {
776 if (finished_) throw std::logic_error("borrowed profile writer is finished");
777 profile_array<P, stream_role::borrowed> result;
778 auto extent = data_.bit_size / P::bits_per_unit;
779 offsets_.push_back(extent);
780 try {
781 result.offsets_ = select_groups<P::group_size>::build(offsets_, count_);
782 } catch (...) {
783 offsets_.pop_back();
784 throw;
785 }
786 result.metadata_.record_count = count_;
787 result.metadata_.extent = extent;
788 result.bytes_ = std::move(data_.bytes);
789 data_.bit_size = 0;
790 previous_ = {};
791 offsets_ = std::vector<std::uint64_t>{};
792 finished_ = true;
793 return result;
794 }
795
796 private:
797 bit_string data_;
798 bit_string previous_;
799 std::vector<std::uint64_t> offsets_;
800 std::uint64_t count_ = 0;
801 bool finished_ = false;
802 };
