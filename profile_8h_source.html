89 inline std::uint64_t low_mask(unsigned width) noexcept {
90 return width == 64 ? ~std::uint64_t{0} : (std::uint64_t{1} << width) - 1;
91 }
92 // A bounded MSB-first field, returned in the low width bits. Width <= 64.
93 inline std::uint64_t load_bits(bit_view data, std::uint64_t first, unsigned width) noexcept {
94 if (!width) return 0;
95 auto offset = data.offset() + first;
96 auto source = data.storage().data() + (offset >> 3);
97 unsigned shift = unsigned(offset & 7), bytes = (shift + width + 7) >> 3;
98 if (bytes >= 8) {
99 auto value = key_detail::load_big(source);
100 if (shift) {
101 value <<= shift;
102 if (bytes == 9) value |= std::uint64_t(std::to_integer<unsigned>(source[8])) >> (8 - shift);
103 }
104 return width == 64 ? value : value >> (64 - width);
105 }
106 std::uint64_t value = 0;
107 for (unsigned i = 0; i != bytes; ++i) value = (value << 8) | std::to_integer<unsigned>(source[i]);
108 return (value >> ((bytes << 3) - shift - width)) & low_mask(width);
109 }
110 // Preserve both edge bytes outside the field. The caller provides capacity.
111 inline void store_bits(std::byte * target, std::uint64_t at, std::uint64_t value, unsigned width) noexcept {
112 if (!width) return;
113 if (width == 64 && (at & 7) == 0) { key_detail::store_big(target + (at >> 3), value); return; }
114 while (width) {
115 auto take = std::min(width, 8u - unsigned(at & 7));
116 auto shift = 8u - unsigned(at & 7) - take;
117 auto mask = unsigned(low_mask(take)) << shift;
118 auto bits = unsigned((value >> (width - take)) & low_mask(take)) << shift;
119 target[at >> 3] = std::byte((std::to_integer<unsigned>(target[at >> 3]) & ~mask) | bits);
120 at += take; width -= take;
121 }
122 }
123 inline void copy_bits(std::byte * target, std::uint64_t first, bit_view source) noexcept {
124 auto count = source.size();
125 if (!count) return;
126 if (((first | source.offset() | count) & 7) == 0) {
127 std::memmove(target + (first >> 3), source.storage().data() + (source.offset() >> 3),
128 static_cast<std::size_t>(count >> 3));
129 return;
130 }
131 // Detect overlapping storage without ordering unrelated C++ pointers.
132 auto destination = reinterpret_cast<std::uintptr_t>(target + (first >> 3));
133 auto origin = reinterpret_cast<std::uintptr_t>(source.storage().data() + (source.offset() >> 3));
134 auto source_bytes = byte_count((source.offset() & 7) + count);
135 bool backwards = (destination > origin || (destination == origin && (first & 7) > (source.offset() & 7))) &&
136 destination - origin < source_bytes;
137 if (backwards) {
138 while (count) {
139 auto width = unsigned(std::min<std::uint64_t>(count, 64));
140 count -= width;
141 store_bits(target, first + count, load_bits(source, count, width), width);
142 }
143 return;
144 }
145 std::uint64_t at = 0;
146 if (first & 7) {
147 auto width = unsigned(std::min<std::uint64_t>(count, 8 - (first & 7)));
148 store_bits(target, first, load_bits(source, 0, width), width); at += width;
149 }
150 if (((source.offset() + at) & 7) == 0) {
151 auto bytes = (count - at) >> 3;
152 if (bytes) std::memmove(target + ((first + at) >> 3),
153 source.storage().data() + ((source.offset() + at) >> 3), static_cast<std::size_t>(bytes));
154 at += bytes << 3;
155 } else {
156 for (; count - at >= 64; at += 64)
157 key_detail::store_big(target + ((first + at) >> 3), load_bits(source, at, 64));
158 }
159 while (at != count) {
160 auto width = unsigned(std::min<std::uint64_t>(count - at, 8));
161 store_bits(target, first + at, load_bits(source, at, width), width); at += width;
162 }
163 }

217 inline bit_comparison compare_common_bits(bit_view a, bit_view b) {
218 auto count = std::min(a.size(), b.size());
219 // Most unrelated keys can differ immediately; do not load a whole word
220 // just to discover a mismatch in the first meaningful bit.
221 if (count) {
222 auto x = (std::to_integer<unsigned>(a.storage()[a.offset() >> 3]) >> (7 - (a.offset() & 7))) & 1;
223 auto y = (std::to_integer<unsigned>(b.storage()[b.offset() >> 3]) >> (7 - (b.offset() & 7))) & 1;
224 if (x != y) return {0, x ? 1 : -1};
225 }
226 std::uint64_t at = 0;
227 if (count >= 8 && (a.offset() & 7) == 0 && (b.offset() & 7) == 0) {
228 at = key_detail::common_bytes(a.storage().data() + (a.offset() >> 3),
229 b.storage().data() + (b.offset() >> 3), static_cast<std::size_t>(count >> 3)) << 3;
230 if (count - at >= 8) {
231 auto x = profile_detail::load_bits(a, at, 8), y = profile_detail::load_bits(b, at, 8);
232 return {at + std::countl_zero(x ^ y) - 56, x < y ? -1 : 1};
233 }
234 }
235 while (at != count) {
236 auto width = unsigned(std::min<std::uint64_t>(count - at, 64));
237 auto x = profile_detail::load_bits(a, at, width), y = profile_detail::load_bits(b, at, width);
238 if (auto different = x ^ y)
239 return {at + std::countl_zero(different) - (64 - width), x < y ? -1 : 1};
240 at += width;
241 }
242 return {count, a.size() < b.size() ? -1 : a.size() > b.size() ? 1 : 0};
243 }

529 }
530 zeros += leading; at += leading;
531 if (leading != width) { ++at; return zeros; }
532 }
533 }
534 template <class P> void write_count(bit_string & target, std::uint64_t value) {
535 if constexpr (P::unit == profile_unit::byte) {
536 while (value >= 128) { append_byte(target, unsigned(value & 127) | 128); value >>= 7; }
537 append_byte(target, unsigned(value));
538 } else {
539 auto maximum = value == std::numeric_limits<std::uint64_t>::max();
540 auto width = maximum ? 65u : unsigned(std::bit_width(value + 1));
541 auto at = target.bit_size;
542 resize(target, add(at, 2 * width - 1));
543 if (maximum) { at += 64; put_fixed(target, at, 1, 1); put_fixed(target, at, 0, 64); }
544 else { at += width - 1; put_fixed(target, at, value + 1, width); }
545 }
546 }

548 template <class P> std::uint64_t read_count(bit_view data, std::uint64_t & offset) {
549 if constexpr (P::unit == profile_unit::byte) {
550 std::uint64_t value = 0;
551 for (unsigned shift = 0; shift <= 63; shift += 7) {
552 if (offset >= (data.size() >> 3)) error_detail::raise<std::invalid_argument>("truncated profile count");
553 auto part = std::to_integer<unsigned>(data.storage()[offset++]);
554 if (shift == 63 && part > 1) error_detail::raise<std::invalid_argument>("overflowing profile count");
555 value |= std::uint64_t(part & 127) << shift;
556 if (!(part & 128)) {
557 if (shift && !(part & 127)) error_detail::raise<std::invalid_argument>("noncanonical profile count");
558 return value;
559 }
560 }
561 error_detail::raise<std::invalid_argument>("overflowing profile count");
562 } else {
563 // A complete small code fits in the first fifteen bits. Decode its
564 // prefix and suffix from one bounded field; long codes and short tails
565 // retain the general decoder's exact offset and error behavior.
566 if (offset <= data.size() && data.size() - offset >= 16) {
567 auto word = load_bits(data, offset, 16);
568 auto zeros = unsigned(std::countl_zero(word) - 48);
569 if (zeros < 8) {
570 offset += 2 * zeros + 1;
571 return (word >> (15 - 2 * zeros)) - 1;
572 }
573 }
574 auto zeros = unsigned(read_zero_run(data, offset, 64,
575 "truncated exponential-Golomb count", "overflowing exponential-Golomb count"));
576 if (zeros > data.size() - offset) {
577 offset = data.size();
578 error_detail::raise<std::invalid_argument>("truncated exponential-Golomb count");
579 }
580 auto suffix = read_fixed(data, offset, zeros);
581 if (zeros == 64) {
582 if (suffix) error_detail::raise<std::invalid_argument>("overflowing exponential-Golomb count");
583 return std::numeric_limits<std::uint64_t>::max();
584 }
585 return ((std::uint64_t{1} << zeros) - 1) + suffix;
586 }
587 }
588 template <class P> constexpr std::uint64_t golomb_cutoff() {
589 constexpr auto width = std::bit_width(P::backspace_parameter - 1);
590 if constexpr (width == 64) return std::uint64_t{0} - P::backspace_parameter;
591 else return (std::uint64_t{1} << width) - P::backspace_parameter;
592 }
593 // Golomb length includes its unary quotient: a long predecessor can make
594 // a short key's backspace expensive even at an LPFC literal restart. Charge
595 // the actual codeword length as well as encoded distance from the anchor.
596 template <class P> std::uint64_t backspace_bits(std::uint64_t value) {
597 static_assert(P::unit == profile_unit::bit);
598 if constexpr (P::backspace_code == bit_backspace_code::exponential_golomb) {
599 auto quotient = value >> P::backspace_parameter;
600 auto prefix = quotient == std::numeric_limits<std::uint64_t>::max()
601 ? 129 : 2 * std::bit_width(quotient + 1) - 1;
602 return prefix + P::backspace_parameter;
603 } else {
604 constexpr auto modulus = P::backspace_parameter;
605 constexpr auto width = std::bit_width(modulus - 1);
606 auto remainder_width = width - (value % modulus < golomb_cutoff<P>());
607 return add(add(value / modulus, 1), remainder_width);
608 }
609 }
610 template <class P> void write_backspace(bit_string & target, std::uint64_t value) {
611 if constexpr (P::unit == profile_unit::byte) write_count<P>(target, value);
612 else {
613 // Check the complete length and allocate before emitting a unary run.
614 // Fresh bytes and the existing canonical tail supply its zero bits.
615 auto at = target.bit_size;
616 resize(target, add(at, backspace_bits<P>(value)));
617 if constexpr (P::backspace_code == bit_backspace_code::exponential_golomb) {
618 auto quotient = value >> P::backspace_parameter;
619 if (quotient == std::numeric_limits<std::uint64_t>::max()) {
620 at += 64;
621 put_fixed(target, at, 1, 1);
622 put_fixed(target, at, 0, 64);
623 } else {
624 auto code = quotient + 1;
625 auto width = unsigned(std::bit_width(code));
626 at += width - 1;
627 put_fixed(target, at, code, width);
628 }
629 put_fixed(target, at, value, unsigned(P::backspace_parameter));
630 } else {
631 constexpr auto modulus = P::backspace_parameter;
632 constexpr auto width = unsigned(std::bit_width(modulus - 1));
633 constexpr auto cutoff = golomb_cutoff<P>();
634 at += value / modulus;
635 put_fixed(target, at, 1, 1);
636 auto remainder = value % modulus;
637 if (remainder < cutoff) put_fixed(target, at, remainder, width - 1);
638 else put_fixed(target, at, remainder + cutoff, width);
639 }
640 }
641 }
642 template <class P> std::uint64_t read_backspace(bit_view data, std::uint64_t & offset) {
643 if constexpr (P::unit == profile_unit::byte) return read_count<P>(data, offset);
644 else if constexpr (P::backspace_code == bit_backspace_code::exponential_golomb) {
645 auto quotient = read_count<P>(data, offset);
646 if (quotient > (std::numeric_limits<std::uint64_t>::max() >> P::backspace_parameter))
647 error_detail::raise<std::invalid_argument>("overflowing exponential-Golomb backspace");
648 auto remainder = read_fixed(data, offset, unsigned(P::backspace_parameter));
649 return (quotient << P::backspace_parameter) | remainder;
650 } else {
651 constexpr auto modulus = P::backspace_parameter;
652 constexpr auto width = unsigned(std::bit_width(modulus - 1));
653 constexpr auto cutoff = golomb_cutoff<P>();
654 constexpr auto limit = std::numeric_limits<std::uint64_t>::max() / modulus;
655 auto quotient = read_zero_run(data, offset, limit,
656 "truncated backspace remainder", "overflowing Golomb quotient");
657 std::uint64_t remainder = 0;
658 if constexpr (width != 0) {
659 remainder = read_fixed(data, offset, width - 1);
660 if (remainder >= cutoff) remainder = ((remainder << 1) | read_fixed(data, offset, 1)) - cutoff;
661 }
662 auto base = quotient * modulus;
663 if (remainder > std::numeric_limits<std::uint64_t>::max() - base)
664 error_detail::raise<std::invalid_argument>("overflowing Golomb backspace");
665 return base + remainder;
666 }

779 // Only headers preceding the selected lane are parsed. The first candidate
780 // must share its retained prefix with the supplied query-bound context.
781 // Callback comparison references expire on the next call or return.
782 template <class F> void compare_window(std::uint64_t first, std::uint64_t last,
783 profile_query_context<P> context, F && callback, profile_comparison_work * work = nullptr,
784 std::uint64_t * predecessor = nullptr) const {
785 if (first > last || last > size() || last - first > P::group_size)
786 error_detail::raise<std::out_of_range>("profile comparison window exceeds cascade group");
787 if (first == last) {
788 if (predecessor) *predecessor = predecessor_units(first, work);
789 return;
790 }
791 auto [at, retained] = locate(first, work, predecessor);
792 auto record = parse_payload(at, retained);
793 for (auto i = first; i != last; ++i) {
794 auto compared = context.advance(record);
795 if (work) { ++work->visited_headers; work->compared_bits += compared; }
796 if (!callback(profile_comparison_item<P>{i, context, record.value})) return;
797 if (i + 1 != last) record = next_record(record, i + 1);
798 }
799 }

801 // The supplied anchor must share the first record's retained prefix. An
802 // anchor between its actual predecessor and that key suffices. The copied
803 // prefix is not an exact LCP with that anchor. Callback key views expire
804 // on the next call.
805 template <class F> void visit_window(std::uint64_t first, std::uint64_t last,
806 profile_anchor<P> anchor, std::uint64_t prefix_limit, F && callback) const {
807 if (first > last || last > size() || last - first > P::group_size)
808 error_detail::raise<std::out_of_range>("profile window exceeds the policy group size");
809 if (first == last) return;
810 if ((anchor.prefix.size() & (P::bits_per_unit - 1)) ||
811 (anchor.prefix.size() >> P::unit_shift) > anchor.full_units)
812 error_detail::raise<std::invalid_argument>("profile anchor units");
813 auto prefix_bits = profile_detail::multiply(
814 std::min(prefix_limit, (anchor.prefix.size() >> P::unit_shift)), P::bits_per_unit);
815 auto scratch = bit_string::copy(anchor.prefix.prefix(prefix_bits));
816 auto context = anchor.full_units;
817 auto record = encoded_at(first);
818 for (auto i = first; i != last; ++i) {
819 decode_into(record, prefix_limit, scratch, context);
820 if (!callback(profile_item<P>{i, {scratch.view(), context}, record.value})) return;
821 if (i + 1 != last) record = next_record(record, i + 1);
822 }
823 }

825 template <class F> void visit_all(F && callback) const {
826 if (!size()) return;
827 bit_string scratch;
828 std::uint64_t context = 0;
829 auto record = encoded_at(0);
830 for (std::uint64_t i = 0; i != size(); ++i) {
831 decode_into(record, std::numeric_limits<std::uint64_t>::max(), scratch, context);
832 if (!callback(profile_item<P>{i, {scratch.view(), context}, record.value})) return;
833 if (i + 1 != size()) record = next_record(record, i + 1);
834 }
835 if (record.next_offset != metadata_.extent) error_detail::raise<std::invalid_argument>("trailing profile data");
836 if (record.key_units != metadata_.terminal_key_units)
837 error_detail::raise<std::invalid_argument>("profile terminal length mismatch");
838 }

840 // Ordinary FC can traverse the entire prefix chain. LPFC bounds backward
841 // encoded distance by the full key length, not by prefix_limit; locating a
842 // record also scans at most W-1 headers before it. Values are copied in full.
843 profile_decoded_record<P> reconstruct_at(std::uint64_t ordinal,
844 std::uint64_t prefix_limit = std::numeric_limits<std::uint64_t>::max()) const {
845 auto record = encoded_at(ordinal);
846 profile_decoded_record<P> result;
847 result.full_units = record.key_units;
848 result.value = bit_string::copy(record.value);
849 auto need = std::min(record.key_units, prefix_limit);
850 profile_detail::resize(result.prefix, profile_detail::multiply(need, P::bits_per_unit));
851 while (need) {
852 if (record.key_units < need) error_detail::raise<std::invalid_argument>("profile predecessor prefix is too short");
853 if (record.retained < need) {
854 auto copy = profile_detail::multiply(need - record.retained, P::bits_per_unit);
855 profile_detail::copy_into(result.prefix, profile_detail::multiply(record.retained, P::bits_per_unit),
856 record.suffix.prefix(copy));
857 need = record.retained;
858 }
859 if (!need) break;
860 if (!ordinal) error_detail::raise<std::invalid_argument>("nonliteral first profile key");
861 auto retained = record.retained;
862 record = encoded_at(--ordinal);
863 if (record.key_units < retained) error_detail::raise<std::invalid_argument>("profile predecessor prefix is too short");
864 }
865 return result;
866 }

908 std::pair<std::uint64_t, std::uint64_t> locate(std::uint64_t ordinal, profile_comparison_work * work,
909 std::uint64_t * predecessor = nullptr) const {
910 auto group = ordinal / P::codec_block_size;
911 auto at = block_offset(group);
912 auto retained = read_absolute(at);
913 if (!group && retained) error_detail::raise<std::invalid_argument>("first profile key is not literal");
914 std::uint64_t previous = 0;
915 for (auto i = group * P::codec_block_size; i < ordinal; ++i) {
916 auto [key_units, next_offset] = parse_payload<false>(at, retained);
917 previous = key_units;
918 at = next_offset;
919 auto backspace = profile_detail::read_backspace<P>(data_, at);
920 if (backspace > previous) error_detail::raise<std::invalid_argument>("profile backspace exceeds predecessor");
921 retained = previous - backspace;
922 if (work) ++work->skipped_headers;
923 }
924 if (predecessor) {
925 if (ordinal && ordinal % P::codec_block_size == 0)
926 previous = predecessor_units(ordinal, work);
927 if (retained > previous)
928 error_detail::raise<std::invalid_argument>("profile retained prefix exceeds predecessor");
929 *predecessor = previous;
930 }
931 return {at, retained};
932 }

955 template <bool Views = true>
956 auto parse_payload(std::uint64_t at, std::uint64_t retained) const {
957 auto suffix = profile_detail::read_count<P>(data_, at);
958 auto value = metadata_.common_value_width ? *metadata_.common_value_width : profile_detail::read_count<P>(data_, at);
959 if (at > metadata_.extent || suffix > metadata_.extent - at || value > metadata_.extent - at - suffix)
960 error_detail::raise<std::invalid_argument>("truncated profile payload");
961 // Absolute controls establish retained <= frame start. Relative controls
962 // inherit key_units <= the previous frame's end. Thus retained <= at and
963 // the bounded suffix gives key_units <= extent, whose bit size was admitted.
964 auto key_units = retained + suffix;
965 if constexpr (Views) {
966 auto suffix_bits = profile_detail::multiply(suffix, P::bits_per_unit);
967 auto value_bits = profile_detail::multiply(value, P::bits_per_unit);
968 auto key_data = data_.subview(profile_detail::multiply(at, P::bits_per_unit), suffix_bits);
969 at += suffix;
970 auto value_data = data_.subview(profile_detail::multiply(at, P::bits_per_unit), value_bits);
971 return profile_encoded_record{retained, key_units, value, key_data, value_data, at + value};
972 } else {
973 // Skipped lanes need only framing. Do not construct literal/value views
974 // or return an encoded record when neither payload will be inspected.
975 return std::pair<std::uint64_t, std::uint64_t>{key_units, at + suffix + value};
976 }

983 error_detail::raise<std::invalid_argument>("profile group offset mismatch");
984 auto next = parse_absolute(at);
985 if (next.retained > previous.key_units)
986 error_detail::raise<std::invalid_argument>("profile retained prefix exceeds predecessor");
987 return next;
988 }

992 static void decode_into(profile_encoded_record const & record, std::uint64_t limit,
993 bit_string & scratch, std::uint64_t & context) {
994 if (record.retained > context) error_detail::raise<std::invalid_argument>("profile anchor is too short");
995 auto keep = std::min(record.retained, limit);
996 auto keep_bits = profile_detail::multiply(keep, P::bits_per_unit);
997 if (keep_bits > scratch.bit_size) error_detail::raise<std::invalid_argument>("profile anchor prefix is incomplete");
998 profile_detail::resize(scratch, keep_bits);
999 auto full = std::min(record.key_units, limit);
1000 profile_detail::append(scratch, record.suffix.prefix(profile_detail::multiply(full - keep, P::bits_per_unit)));
1001 context = record.key_units;
1002 }

1139 std::span<std::uint64_t const> prefix_ceilings = {}, std::uint64_t restart_factor = 0) {
1140 if (!prefix_ceilings.empty() && prefix_ceilings.size() != records.size())
1141 error_detail::raise<std::invalid_argument>("one profile prefix ceiling is required per record");
1142 if (restart_factor && restart_factor < 3) error_detail::raise<std::invalid_argument>("LPFC factor must be at least three");
1143 profile_array result;
1144 std::optional<std::uint64_t> common;
1145 if constexpr (Role == stream_role::borrowed) common = 0;
1146 else if constexpr (P::fixed_width) common = P::value_width;
1147 else if (records.empty()) common = 0;
1148 else common = (records.front().value.view().size() >> P::unit_shift);
1149 for (auto const & record : records) {
1150 auto key_bits = record.key.view().size();
1151 auto value_bits = record.value.view().size();
1152 if ((key_bits & (P::bits_per_unit - 1)) || (value_bits & (P::bits_per_unit - 1)))
1153 error_detail::raise<std::invalid_argument>("record length does not match profile unit");
1154 auto width = (value_bits >> P::unit_shift);
1155 if constexpr (Role == stream_role::borrowed) {
1156 if (width) error_detail::raise<std::invalid_argument>("borrowed profile cannot carry values");
1157 } else if constexpr (P::fixed_width) {
1158 if (width != *P::value_width) error_detail::raise<std::invalid_argument>("value does not match fixed policy width");
1159 } else {
1160 if (common && *common != width) common.reset();
1161 }
1162 }
1163 result.metadata_.common_value_width = common;
1164 result.metadata_.record_count = records.size();
1165 bit_string data;
1166 bit_view previous;
1167 std::uint64_t previous_units = 0;
1168 std::uint64_t anchor_offset = 0;
1169 std::vector<std::uint64_t> offsets;
1170 for (std::size_t i = 0; i != records.size(); ++i) {
1171 auto key = records[i].key.view();
1172 auto value = records[i].value.view();
1173 auto comparison = compare_common_bits(previous, key);
1174 if (i && comparison.order > 0) error_detail::raise<std::invalid_argument>("profile keys must be sorted");
1175 auto position = (data.bit_size >> P::unit_shift);
1176 if (i % P::codec_block_size == 0)
1177 offsets.push_back(position - profile_detail::multiply(i, common.value_or(0)));
1178 auto key_units = (key.size() >> P::unit_shift);
1179 auto retained = (comparison.common_bits >> P::unit_shift);
1180 if (records[i].retained_limit_bits)
1181 retained = std::min(retained, *records[i].retained_limit_bits >> P::unit_shift);
1182 if (!prefix_ceilings.empty()) retained = std::min(retained, prefix_ceilings[i]);
1183 auto start = (data.bit_size >> P::unit_shift);
1184 if (retained && restart_factor && key_units <= std::numeric_limits<std::uint64_t>::max() / restart_factor &&
1185 start - anchor_offset > restart_factor * key_units) retained = 0;
1186 if (!retained) anchor_offset = start;
1187 if (i % P::codec_block_size == 0) profile_detail::write_count<P>(data, retained);
1188 else profile_detail::write_backspace<P>(data, previous_units - retained);
1189 profile_detail::write_count<P>(data, key_units - retained);
1190 if (!common) profile_detail::write_count<P>(data, (value.size() >> P::unit_shift));
1191 profile_detail::append(data, key.subview(profile_detail::multiply(retained, P::bits_per_unit),
1192 key.size() - profile_detail::multiply(retained, P::bits_per_unit)));
1193 profile_detail::append(data, value);
1194 previous = key;
1195 previous_units = key_units;
1196 }
1197 result.metadata_.terminal_key_units = previous_units;
1198 result.metadata_.extent = (data.bit_size >> P::unit_shift);
1199 offsets.push_back(result.metadata_.extent - profile_detail::multiply(records.size(), common.value_or(0)));
1200 result.offsets_ = elias_fano::build(offsets);
1201 result.bytes_ = std::move(data.bytes);
