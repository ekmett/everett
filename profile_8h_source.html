82 inline std::uint64_t low_mask(unsigned width) noexcept {
83 return width == 64 ? ~std::uint64_t{0} : (std::uint64_t{1} << width) - 1;
84 }
85 // A bounded MSB-first field, returned in the low width bits. Width <= 64.
86 inline std::uint64_t load_bits(bit_view data, std::uint64_t first, unsigned width) noexcept {
87 if (!width) return 0;
88 auto offset = data.offset() + first;
89 auto source = data.storage().data() + (offset >> 3);
90 unsigned shift = unsigned(offset & 7), bytes = (shift + width + 7) >> 3;
91 if (bytes >= 8) {
92 auto value = key_detail::load_big(source);
93 if (shift) {
94 value <<= shift;
95 if (bytes == 9) value |= std::uint64_t(std::to_integer<unsigned>(source[8])) >> (8 - shift);
96 }
97 return width == 64 ? value : value >> (64 - width);
98 }
99 std::uint64_t value = 0;
100 for (unsigned i = 0; i != bytes; ++i) value = (value << 8) | std::to_integer<unsigned>(source[i]);
101 return (value >> ((bytes << 3) - shift - width)) & low_mask(width);
102 }
103 // Preserve both edge bytes outside the field. The caller provides capacity.
104 inline void store_bits(std::byte * target, std::uint64_t at, std::uint64_t value, unsigned width) noexcept {
105 if (!width) return;
106 if (width == 64 && (at & 7) == 0) { key_detail::store_big(target + (at >> 3), value); return; }
107 while (width) {
108 auto take = std::min(width, 8u - unsigned(at & 7));
109 auto shift = 8u - unsigned(at & 7) - take;
110 auto mask = unsigned(low_mask(take)) << shift;
111 auto bits = unsigned((value >> (width - take)) & low_mask(take)) << shift;
112 target[at >> 3] = std::byte((std::to_integer<unsigned>(target[at >> 3]) & ~mask) | bits);
113 at += take; width -= take;
114 }
115 }
116 inline void copy_bits(std::byte * target, std::uint64_t first, bit_view source) noexcept {
117 auto count = source.size();
118 if (!count) return;
119 if (((first | source.offset() | count) & 7) == 0) {
120 std::memmove(target + (first >> 3), source.storage().data() + (source.offset() >> 3),
121 static_cast<std::size_t>(count >> 3));
122 return;
123 }
124 // Detect overlapping storage without ordering unrelated C++ pointers.
125 auto destination = reinterpret_cast<std::uintptr_t>(target + (first >> 3));
126 auto origin = reinterpret_cast<std::uintptr_t>(source.storage().data() + (source.offset() >> 3));
127 auto source_bytes = byte_count((source.offset() & 7) + count);
128 bool backwards = (destination > origin || (destination == origin && (first & 7) > (source.offset() & 7))) &&
129 destination - origin < source_bytes;
130 if (backwards) {
131 while (count) {
132 auto width = unsigned(std::min<std::uint64_t>(count, 64));
133 count -= width;
134 store_bits(target, first + count, load_bits(source, count, width), width);
135 }
136 return;
137 }
138 std::uint64_t at = 0;
139 if (first & 7) {
140 auto width = unsigned(std::min<std::uint64_t>(count, 8 - (first & 7)));
141 store_bits(target, first, load_bits(source, 0, width), width); at += width;
142 }
143 if (((source.offset() + at) & 7) == 0) {
144 auto bytes = (count - at) >> 3;
145 if (bytes) std::memmove(target + ((first + at) >> 3),
146 source.storage().data() + ((source.offset() + at) >> 3), static_cast<std::size_t>(bytes));
147 at += bytes << 3;
148 } else {
149 for (; count - at >= 64; at += 64)
150 key_detail::store_big(target + ((first + at) >> 3), load_bits(source, at, 64));
151 }
152 while (at != count) {
153 auto width = unsigned(std::min<std::uint64_t>(count - at, 8));
154 store_bits(target, first + at, load_bits(source, at, width), width); at += width;
155 }
156 }

206 inline bit_comparison compare_common_bits(bit_view a, bit_view b) {
207 auto count = std::min(a.size(), b.size());
208 // Most unrelated keys can differ immediately; do not load a whole word
209 // just to discover a mismatch in the first meaningful bit.
210 if (count) {
211 auto x = (std::to_integer<unsigned>(a.storage()[a.offset() >> 3]) >> (7 - (a.offset() & 7))) & 1;
212 auto y = (std::to_integer<unsigned>(b.storage()[b.offset() >> 3]) >> (7 - (b.offset() & 7))) & 1;
213 if (x != y) return {0, x ? 1 : -1};
214 }
215 std::uint64_t at = 0;
216 if (count >= 8 && (a.offset() & 7) == 0 && (b.offset() & 7) == 0) {
217 at = key_detail::common_bytes(a.storage().data() + (a.offset() >> 3),
218 b.storage().data() + (b.offset() >> 3), static_cast<std::size_t>(count >> 3)) << 3;
219 if (count - at >= 8) {
220 auto x = profile_detail::load_bits(a, at, 8), y = profile_detail::load_bits(b, at, 8);
221 return {at + std::countl_zero(x ^ y) - 56, x < y ? -1 : 1};
222 }
223 }
224 while (at != count) {
225 auto width = unsigned(std::min<std::uint64_t>(count - at, 64));
226 auto x = profile_detail::load_bits(a, at, width), y = profile_detail::load_bits(b, at, width);
227 if (auto different = x ^ y)
228 return {at + std::countl_zero(different) - (64 - width), x < y ? -1 : 1};
229 at += width;
230 }
231 return {count, a.size() < b.size() ? -1 : a.size() > b.size() ? 1 : 0};
232 }

475 template <class P> std::uint64_t read_count(bit_view data, std::uint64_t & offset) {
476 if constexpr (P::unit == profile_unit::byte) {
477 std::uint64_t value = 0;
478 for (unsigned shift = 0; shift <= 63; shift += 7) {
479 if (offset >= (data.size() >> 3)) error_detail::raise<std::invalid_argument>("truncated profile count");
480 auto part = std::to_integer<unsigned>(data.storage()[offset++]);
481 if (shift == 63 && part > 1) error_detail::raise<std::invalid_argument>("overflowing profile count");
482 value |= std::uint64_t(part & 127) << shift;
483 if (!(part & 128)) {
484 if (shift && !(part & 127)) error_detail::raise<std::invalid_argument>("noncanonical profile count");
485 return value;
486 }
487 }
488 error_detail::raise<std::invalid_argument>("overflowing profile count");
489 } else {
490 // A complete small code fits in the first fifteen bits. Decode its
491 // prefix and suffix from one bounded field; long codes and short tails
492 // retain the general decoder's exact offset and error behavior.
493 if (offset <= data.size() && data.size() - offset >= 16) {
494 auto word = load_bits(data, offset, 16);
495 auto zeros = unsigned(std::countl_zero(word) - 48);
496 if (zeros < 8) {
497 offset += 2 * zeros + 1;
498 return (word >> (15 - 2 * zeros)) - 1;
499 }
500 }
501 auto zeros = unsigned(read_zero_run(data, offset, 64,
502 "truncated exponential-Golomb count", "overflowing exponential-Golomb count"));
503 if (zeros > data.size() - offset) {
504 offset = data.size();
505 error_detail::raise<std::invalid_argument>("truncated exponential-Golomb count");
506 }
507 auto suffix = read_fixed(data, offset, zeros);
508 if (zeros == 64) {
509 if (suffix) error_detail::raise<std::invalid_argument>("overflowing exponential-Golomb count");
510 return std::numeric_limits<std::uint64_t>::max();
511 }
512 return ((std::uint64_t{1} << zeros) - 1) + suffix;
513 }

524 static_assert(P::unit == profile_unit::bit);
525 if constexpr (P::backspace_code == bit_backspace_code::exponential_golomb) {
526 auto quotient = value >> P::backspace_parameter;
527 auto prefix = quotient == std::numeric_limits<std::uint64_t>::max()
528 ? 129 : 2 * std::bit_width(quotient + 1) - 1;
529 return prefix + P::backspace_parameter;
530 } else {
531 constexpr auto modulus = P::backspace_parameter;
532 constexpr auto width = std::bit_width(modulus - 1);
533 auto remainder_width = width - (value % modulus < golomb_cutoff<P>());
534 return add(add(value / modulus, 1), remainder_width);
535 }
536 }
537 template <class P> void write_backspace(bit_string & target, std::uint64_t value) {
538 if constexpr (P::unit == profile_unit::byte) write_count<P>(target, value);
539 else {
540 // Check the complete length and allocate before emitting a unary run.
541 // Fresh bytes and the existing canonical tail supply its zero bits.
542 auto at = target.bit_size;
543 resize(target, add(at, backspace_bits<P>(value)));
544 if constexpr (P::backspace_code == bit_backspace_code::exponential_golomb) {
545 auto quotient = value >> P::backspace_parameter;
546 if (quotient == std::numeric_limits<std::uint64_t>::max()) {
547 at += 64;
548 put_fixed(target, at, 1, 1);
549 put_fixed(target, at, 0, 64);
550 } else {
551 auto code = quotient + 1;
552 auto width = unsigned(std::bit_width(code));
553 at += width - 1;
554 put_fixed(target, at, code, width);
555 }
556 put_fixed(target, at, value, unsigned(P::backspace_parameter));
557 } else {
558 constexpr auto modulus = P::backspace_parameter;
559 constexpr auto width = unsigned(std::bit_width(modulus - 1));
560 constexpr auto cutoff = golomb_cutoff<P>();
561 at += value / modulus;
562 put_fixed(target, at, 1, 1);
563 auto remainder = value % modulus;
564 if (remainder < cutoff) put_fixed(target, at, remainder, width - 1);
565 else put_fixed(target, at, remainder + cutoff, width);
566 }
567 }
568 }
569 template <class P> std::uint64_t read_backspace(bit_view data, std::uint64_t & offset) {
570 if constexpr (P::unit == profile_unit::byte) return read_count<P>(data, offset);
571 else if constexpr (P::backspace_code == bit_backspace_code::exponential_golomb) {
572 auto quotient = read_count<P>(data, offset);
573 if (quotient > (std::numeric_limits<std::uint64_t>::max() >> P::backspace_parameter))
574 error_detail::raise<std::invalid_argument>("overflowing exponential-Golomb backspace");
575 auto remainder = read_fixed(data, offset, unsigned(P::backspace_parameter));
576 return (quotient << P::backspace_parameter) | remainder;
577 } else {
578 constexpr auto modulus = P::backspace_parameter;
579 constexpr auto width = unsigned(std::bit_width(modulus - 1));
580 constexpr auto cutoff = golomb_cutoff<P>();
581 constexpr auto limit = std::numeric_limits<std::uint64_t>::max() / modulus;
582 auto quotient = read_zero_run(data, offset, limit,
583 "truncated backspace remainder", "overflowing Golomb quotient");
584 std::uint64_t remainder = 0;
585 if constexpr (width != 0) {
586 remainder = read_fixed(data, offset, width - 1);
587 if (remainder >= cutoff) remainder = ((remainder << 1) | read_fixed(data, offset, 1)) - cutoff;
588 }
589 auto base = quotient * modulus;
590 if (remainder > std::numeric_limits<std::uint64_t>::max() - base)
591 error_detail::raise<std::invalid_argument>("overflowing Golomb backspace");
592 return base + remainder;
593 }
594 }

598 result.count_code = P::unit == profile_unit::byte ? profile_count_code::varint : profile_count_code::exp_golomb_zero;
599 result.backspace_code = P::backspace_code;

696 // Only headers preceding the selected lane are parsed. The first candidate
697 // must share its retained prefix with the supplied query-bound context.
698 // Callback comparison references expire on the next call or return.
699 template <class F> void compare_window(std::uint64_t first, std::uint64_t last,
700 profile_query_context<P> context, F && callback, profile_comparison_work * work = nullptr,
701 std::uint64_t * predecessor = nullptr) const {
702 if (first > last || last > size() || last - first > P::group_size)
703 error_detail::raise<std::out_of_range>("profile comparison window exceeds cascade group");
704 if (first == last) {
705 if (predecessor) *predecessor = predecessor_units(first, work);
706 return;
707 }
708 auto [at, previous] = locate(first, work);
709 if (predecessor) *predecessor = previous;
710 auto record = parse(at, previous);
711 for (auto i = first; i != last; ++i) {
712 auto compared = context.advance(record);
713 if (work) { ++work->visited_headers; work->compared_bits += compared; }
714 if (!callback(profile_comparison_item<P>{i, context, record.value})) return;
715 if (i + 1 != last) record = next_record(record, i + 1);
716 }
717 }

719 // The supplied anchor must share the first record's retained prefix. An
720 // anchor between its actual predecessor and that key suffices. The copied
721 // prefix is not an exact LCP with that anchor. Callback key views expire
722 // on the next call.
723 template <class F> void visit_window(std::uint64_t first, std::uint64_t last,
724 profile_anchor<P> anchor, std::uint64_t prefix_limit, F && callback) const {
725 if (first > last || last > size() || last - first > P::group_size)
726 error_detail::raise<std::out_of_range>("profile window exceeds the policy group size");
727 if (first == last) return;
728 if ((anchor.prefix.size() & (P::bits_per_unit - 1)) ||
729 (anchor.prefix.size() >> P::unit_shift) > anchor.full_units)
730 error_detail::raise<std::invalid_argument>("profile anchor units");
731 auto prefix_bits = profile_detail::multiply(
732 std::min(prefix_limit, (anchor.prefix.size() >> P::unit_shift)), P::bits_per_unit);
733 auto scratch = bit_string::copy(anchor.prefix.prefix(prefix_bits));
734 auto context = anchor.full_units;
735 auto record = encoded_at(first);
736 for (auto i = first; i != last; ++i) {
737 decode_into(record, prefix_limit, scratch, context);
738 if (!callback(profile_item<P>{i, {scratch.view(), context}, record.value})) return;
739 if (i + 1 != last) record = next_record(record, i + 1);
740 }
741 }

743 template <class F> void visit_all(F && callback) const {
744 if (!size()) return;
745 bit_string scratch;
746 std::uint64_t context = 0;
747 auto record = encoded_at(0);
748 for (std::uint64_t i = 0; i != size(); ++i) {
749 if (record.previous_units != context) error_detail::raise<std::invalid_argument>("profile predecessor length mismatch");
750 decode_into(record, std::numeric_limits<std::uint64_t>::max(), scratch, context);
751 if (!callback(profile_item<P>{i, {scratch.view(), context}, record.value})) return;
752 if (i + 1 != size()) record = next_record(record, i + 1);
753 }
754 if (record.next_offset != metadata_.extent) error_detail::raise<std::invalid_argument>("trailing profile data");
755 if (record.key_units != metadata_.terminal_key_units)
756 error_detail::raise<std::invalid_argument>("profile terminal length mismatch");
757 }

759 // Ordinary FC can traverse the entire prefix chain. LPFC bounds backward
760 // encoded distance by the full key length, not by prefix_limit; locating a
761 // record also scans at most K-1 headers before it. Values are copied in full.
762 profile_decoded_record<P> reconstruct_at(std::uint64_t ordinal,
763 std::uint64_t prefix_limit = std::numeric_limits<std::uint64_t>::max()) const {
764 auto record = encoded_at(ordinal);
765 profile_decoded_record<P> result;
766 result.full_units = record.key_units;
767 result.value = bit_string::copy(record.value);
768 auto need = std::min(record.key_units, prefix_limit);
769 profile_detail::resize(result.prefix, profile_detail::multiply(need, P::bits_per_unit));
770 while (need) {
771 if (record.key_units < need) error_detail::raise<std::invalid_argument>("profile predecessor prefix is too short");
772 if (record.retained < need) {
773 auto copy = profile_detail::multiply(need - record.retained, P::bits_per_unit);
774 profile_detail::copy_into(result.prefix, profile_detail::multiply(record.retained, P::bits_per_unit),
775 record.suffix.prefix(copy));
776 need = record.retained;
777 }
778 if (!need) break;
779 if (!ordinal) error_detail::raise<std::invalid_argument>("nonliteral first profile key");
780 auto expected = record.previous_units;
781 record = encoded_at(--ordinal);
782 if (record.key_units != expected) error_detail::raise<std::invalid_argument>("profile predecessor length mismatch");
783 }
784 return result;
785 }

825 std::pair<std::uint64_t, std::uint64_t> locate(std::uint64_t ordinal,
826 profile_comparison_work * work) const {
827 auto group = ordinal / P::codec_block_size;
828 auto at = block_offset(group);
829 auto previous = profile_detail::read_count<P>(data_, at);
830 if (!group && previous) error_detail::raise<std::invalid_argument>("first profile predecessor is not empty");
831 for (auto i = group * P::codec_block_size; i < ordinal; ++i) {
832 auto record = parse(at, previous);
833 previous = record.key_units;
834 at = record.next_offset;
835 if (work) ++work->skipped_headers;
836 }
837 return {at, previous};
838 }

840 profile_encoded_record parse(std::uint64_t at, std::uint64_t previous) const {
841 auto backspace = profile_detail::read_backspace<P>(data_, at);
842 auto suffix = profile_detail::read_count<P>(data_, at);
843 auto value = metadata_.common_value_width ? *metadata_.common_value_width : profile_detail::read_count<P>(data_, at);
844 if (backspace > previous) error_detail::raise<std::invalid_argument>("profile backspace exceeds predecessor");
845 auto retained = previous - backspace;
846 auto key_units = profile_detail::add(retained, suffix);
847 if (at > metadata_.extent || suffix > metadata_.extent - at || value > metadata_.extent - at - suffix)
848 error_detail::raise<std::invalid_argument>("truncated profile payload");
849 auto suffix_bits = profile_detail::multiply(suffix, P::bits_per_unit);
850 auto value_bits = profile_detail::multiply(value, P::bits_per_unit);
851 auto key_data = data_.subview(profile_detail::multiply(at, P::bits_per_unit), suffix_bits);
852 at += suffix;
853 auto value_data = data_.subview(profile_detail::multiply(at, P::bits_per_unit), value_bits);
854 return {previous, backspace, retained, key_units, value, key_data, value_data, at + value};
855 }

857 profile_encoded_record next_record(profile_encoded_record const & previous, std::uint64_t ordinal) const {
858 auto at = previous.next_offset;
859 if (ordinal % P::codec_block_size == 0) {
860 if (block_offset(ordinal / P::codec_block_size) != at)
861 error_detail::raise<std::invalid_argument>("profile group offset mismatch");
862 if (profile_detail::read_count<P>(data_, at) != previous.key_units)
863 error_detail::raise<std::invalid_argument>("profile group predecessor mismatch");
864 }
865 return parse(at, previous.key_units);
866 }

868 static void decode_into(profile_encoded_record const & record, std::uint64_t limit,
869 bit_string & scratch, std::uint64_t & context) {
870 if (record.retained > context) error_detail::raise<std::invalid_argument>("profile anchor is too short");
871 auto keep = std::min(record.retained, limit);
872 auto keep_bits = profile_detail::multiply(keep, P::bits_per_unit);
873 if (keep_bits > scratch.bit_size) error_detail::raise<std::invalid_argument>("profile anchor prefix is incomplete");
874 profile_detail::resize(scratch, keep_bits);
875 auto full = std::min(record.key_units, limit);
876 profile_detail::append(scratch, record.suffix.prefix(profile_detail::multiply(full - keep, P::bits_per_unit)));
877 context = record.key_units;
878 }

881 // Resumable sequential traversal. Each encoded record is parsed once and
882 // its value remains a view of the original payload. Only the current key is
883 // reconstructed. The view's sections must outlive this cursor; peek's key
884 // view additionally expires when this cursor advances or is destroyed.
885 template <class P, stream_role Role> struct profile_cursor {
886 using policy_type = P;
887 static constexpr stream_role role = Role;
888
889 explicit profile_cursor(profile_view<P, Role> view) : view_(view) {
890 if (!done()) {
891 record_ = view_.encoded_at(0);
892 profile_view<P, Role>::decode_into(record_, std::numeric_limits<std::uint64_t>::max(), scratch_, context_);
893 }
894 }
895
896 bool done() const noexcept { return ordinal_ == view_.size(); }
897 std::uint64_t ordinal() const noexcept { return ordinal_; }
898 profile_item<P> peek() const & {
899 if (done()) error_detail::raise<std::out_of_range>("profile cursor at end");
900 return {ordinal_, {scratch_.view(), context_}, record_.value};
901 }
902 profile_item<P> peek() const && = delete;
903
904 void advance() { advance_impl(nullptr); }

906 // Compare the old and new keys from the retained FC prefix onward before
907 // replacing scratch. Ordinary FC needs only its first differing unit;
908 // redundant retained-prefix encodings remain correct. No result at EOF.
909 std::optional<bit_comparison> advance_comparison() {
910 bit_comparison result;
911 advance_impl(&result);
912 return done() ? std::nullopt : std::optional<bit_comparison>{result};
913 }
914
915 private:
916 void advance_impl(bit_comparison * comparison) {
917 if (done()) error_detail::raise<std::out_of_range>("profile cursor at end");
918 if (ordinal_ + 1 == view_.size()) {
919 if (record_.next_offset != view_.metadata_.extent)
920 error_detail::raise<std::invalid_argument>("trailing profile data");
921 if (record_.key_units != view_.metadata_.terminal_key_units)
922 error_detail::raise<std::invalid_argument>("profile terminal length mismatch");
923 ++ordinal_;
924 return;
925 }
926 auto next = view_.next_record(record_, ordinal_ + 1);
927 if (comparison) {
928 auto retained_bits = profile_detail::multiply(next.retained, P::bits_per_unit);
929 auto previous = scratch_.view();
930 if (retained_bits == previous.size() || next.suffix.empty()) {
931 *comparison = {retained_bits, retained_bits != previous.size() ? 1 : next.suffix.empty() ? 0 : -1};
932 } else {
933 auto before = profile_detail::load_bits(previous, retained_bits, P::bits_per_unit);
934 auto after = profile_detail::load_bits(next.suffix, 0, P::bits_per_unit);
935 if (before != after) {
936 auto common = unsigned(std::countl_zero(before ^ after)) - (64 - P::bits_per_unit);
937 *comparison = {retained_bits + common, before < after ? -1 : 1};
938 } else {
939 auto suffix = compare_common_bits(
940 previous.subview(retained_bits, previous.size() - retained_bits), next.suffix);
941 *comparison = {retained_bits + suffix.common_bits, suffix.order};
942 }
943 }
944 }
945 profile_view<P, Role>::decode_into(next, std::numeric_limits<std::uint64_t>::max(), scratch_, context_);
946 record_ = next;
947 ++ordinal_;
948 }

951 profile_view<P, Role> view_;
952 bit_string scratch_;
953 std::uint64_t context_ = 0;
954 profile_encoded_record record_;
955 std::uint64_t ordinal_ = 0;
956 };

961 template <class P, stream_role Role = stream_role::native> struct profile_array {
962 using policy_type = P;
963 static constexpr stream_role role = Role;
964
965 static profile_array build(std::span<profile_record const> records,
966 std::span<std::uint64_t const> prefix_ceilings = {}, std::uint64_t restart_factor = 0) {
967 if (!prefix_ceilings.empty() && prefix_ceilings.size() != records.size())
968 error_detail::raise<std::invalid_argument>("one profile prefix ceiling is required per record");
969 if (restart_factor && restart_factor < 3) error_detail::raise<std::invalid_argument>("LPFC factor must be at least three");
970 profile_array result;
971 std::optional<std::uint64_t> common;
972 if constexpr (Role == stream_role::borrowed) common = 0;
973 else if constexpr (P::fixed_width) common = P::value_width;
974 else if (records.empty()) common = 0;
975 else common = (records.front().value.view().size() >> P::unit_shift);
976 for (auto const & record : records) {
977 auto key_bits = record.key.view().size();
978 auto value_bits = record.value.view().size();
979 if ((key_bits & (P::bits_per_unit - 1)) || (value_bits & (P::bits_per_unit - 1)))
980 error_detail::raise<std::invalid_argument>("record length does not match profile unit");
981 auto width = (value_bits >> P::unit_shift);
982 if constexpr (Role == stream_role::borrowed) {
983 if (width) error_detail::raise<std::invalid_argument>("borrowed profile cannot carry values");
984 } else if constexpr (P::fixed_width) {
985 if (width != *P::value_width) error_detail::raise<std::invalid_argument>("value does not match fixed policy width");
986 } else {
987 if (common && *common != width) common.reset();
988 }
989 }
990 result.metadata_.common_value_width = common;
991 result.metadata_.record_count = records.size();
992 bit_string data;
993 bit_view previous;
994 std::uint64_t previous_units = 0;
995 std::uint64_t anchor_offset = 0;
996 std::vector<std::uint64_t> offsets;
997 for (std::size_t i = 0; i != records.size(); ++i) {
998 auto key = records[i].key.view();
999 auto value = records[i].value.view();
1000 auto comparison = compare_common_bits(previous, key);
1001 if (i && comparison.order > 0) error_detail::raise<std::invalid_argument>("profile keys must be sorted");
1002 auto position = (data.bit_size >> P::unit_shift);
1003 if (i % P::codec_block_size == 0) {
1004 offsets.push_back(position - profile_detail::multiply(i, common.value_or(0)));
1005 profile_detail::write_count<P>(data, previous_units);
1006 }
1007 auto key_units = (key.size() >> P::unit_shift);
1008 auto retained = (comparison.common_bits >> P::unit_shift);
1009 if (!prefix_ceilings.empty()) retained = std::min(retained, prefix_ceilings[i]);
1010 auto start = (data.bit_size >> P::unit_shift);
1011 if (retained && restart_factor && key_units <= std::numeric_limits<std::uint64_t>::max() / restart_factor &&
1012 start - anchor_offset > restart_factor * key_units) retained = 0;
1013 if (!retained) anchor_offset = start;

1015 profile_detail::write_count<P>(data, key_units - retained);
1016 if (!common) profile_detail::write_count<P>(data, (value.size() >> P::unit_shift));
1017 profile_detail::append(data, key.subview(profile_detail::multiply(retained, P::bits_per_unit),
1018 key.size() - profile_detail::multiply(retained, P::bits_per_unit)));
1019 profile_detail::append(data, value);

1021 previous_units = key_units;
1022 }
1023 result.metadata_.terminal_key_units = previous_units;
1024 result.metadata_.extent = (data.bit_size >> P::unit_shift);
1025 offsets.push_back(result.metadata_.extent - profile_detail::multiply(records.size(), common.value_or(0)));
1026 result.offsets_ = elias_fano::build(offsets);
1027 result.bytes_ = std::move(data.bytes);
1028 return result;
1029 }

1052 // Incremental modified-FC output. The caller supplies any boundary-dependent
1053 // prefix ceiling before appending that key. There is no all-keys staging:
1054 // retained state is the previous key, encoded bytes and one offset per group.
1055 // With the same ceilings this produces exactly the batch borrowed encoding
1056 // (restart_factor == 0), including checkpoints, tail padding and EF metadata.
1057 template <class P> struct profile_borrowed_writer {
1058 using policy_type = P;
1059 static constexpr stream_role role = stream_role::borrowed;
1060
1061 std::uint64_t size() const noexcept { return count_; }
1062 bool finished() const noexcept { return finished_; }
1063
1064 void append(bit_view key, std::uint64_t prefix_ceiling = std::numeric_limits<std::uint64_t>::max()) {
1065 if (finished_) error_detail::raise<std::logic_error>("borrowed profile writer is finished");
1066 if (key.size() & (P::bits_per_unit - 1)) error_detail::raise<std::invalid_argument>("key length does not match profile unit");
1067 auto previous = previous_.view();
1068 auto comparison = compare_common_bits(previous, key);
1069 if (count_ && comparison.order > 0) error_detail::raise<std::invalid_argument>("profile keys must be sorted");
1070 auto next_count = profile_detail::add(count_, 1);
1071 auto retained = std::min((comparison.common_bits >> P::unit_shift), prefix_ceiling);
1072 auto previous_units = (previous.size() >> P::unit_shift);
1073 auto key_units = (key.size() >> P::unit_shift);
1074 auto saved_bits = data_.bit_size;
1075 auto saved_offsets = offsets_.size();
1076 // Reserve before modifying output. The private predecessor retains its
1077 // logical contents until all potentially allocating output writes finish.
1078 auto next_bytes = profile_detail::byte_count(key.size());
1079 if (next_bytes > previous_.bytes.max_size()) error_detail::raise<std::length_error>("profile bit string too large");
1080 previous_.bytes.reserve(static_cast<std::size_t>(next_bytes));
1081 try {
1082 if (count_ % P::codec_block_size == 0) {
1083 offsets_.push_back((data_.bit_size >> P::unit_shift));
1084 profile_detail::write_count<P>(data_, previous_units);
1085 }
1086 profile_detail::write_backspace<P>(data_, previous_units - retained);
1087 profile_detail::write_count<P>(data_, key_units - retained);
1088 auto retained_bits = profile_detail::multiply(retained, P::bits_per_unit);
1089 profile_detail::append(data_, key.subview(retained_bits, key.size() - retained_bits));
1090 } catch (...) {
1091 profile_detail::resize(data_, saved_bits);
1092 offsets_.resize(saved_offsets);
1093 throw;
1094 }
1095 // This resize cannot allocate after reserve. Keep the actual shared
1096 // prefix even when the encoding used a smaller boundary prefix ceiling.
1097 auto common_bits = comparison.common_bits & ~std::uint64_t(P::bits_per_unit - 1);
1098 profile_detail::resize(previous_, key.size());
1099 profile_detail::copy_into(previous_, common_bits,
1100 key.subview(common_bits, key.size() - common_bits));
1101 count_ = next_count;
1102 }
1103
1104 // Finalizing EF takes work proportional to the staged group offsets.
1105 // This is not a bounded-byte or durable checkpoint operation.
1106 profile_array<P, stream_role::borrowed> finish() {
1107 if (finished_) error_detail::raise<std::logic_error>("borrowed profile writer is finished");
1108 profile_array<P, stream_role::borrowed> result;
1109 auto extent = (data_.bit_size >> P::unit_shift);
1110 offsets_.push_back(extent);
1111 try {
1112 result.offsets_ = elias_fano::build(offsets_);
1113 } catch (...) {
1114 offsets_.pop_back();
1115 throw;
1116 }
1117 result.metadata_.record_count = count_;
1118 result.metadata_.extent = extent;
1119 result.metadata_.terminal_key_units = (previous_.bit_size >> P::unit_shift);
1120 result.bytes_ = std::move(data_.bytes);
1121 data_.bit_size = 0;
1122 previous_ = {};
1123 offsets_ = std::vector<std::uint64_t>{};
1124 finished_ = true;
1125 return result;
1126 }
1127
1128 private:
1129 bit_string data_;
1130 bit_string previous_;
1131 std::vector<std::uint64_t> offsets_;
1132 std::uint64_t count_ = 0;
1133 bool finished_ = false;
1134 };
