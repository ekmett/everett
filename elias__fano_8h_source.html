42 inline bool monotone(std::span<std::uint64_t const> source) noexcept {
43 if (source.size() < 2) return true;
44 std::size_t i = 1;
45#if defined(__aarch64__) && defined(__ARM_NEON)
46 for (; source.size() - i >= 2; i += 2)
47 if (vmaxvq_u32(vreinterpretq_u32_u64(vcltq_u64(vld1q_u64(source.data() + i),
48 vld1q_u64(source.data() + i - 1)))))
49 return false;
50#endif
51 for (; i < source.size(); ++i) if (source[i] < source[i - 1]) return false;
52 return true;
53 }

55 // Zero-based selection in a nonzero word, with ordinal < popcount(value).
56 // Byte-prefix populations fit in seven bits, so the marked subtraction
57 // finds the first byte whose cumulative population exceeds ordinal.
58 inline unsigned select_word(std::uint64_t value, unsigned ordinal) noexcept {
59#if defined(__BMI2__) && (defined(__x86_64__) || defined(_M_X64))
60 return unsigned(std::countr_zero(_pdep_u64(std::uint64_t{1} << ordinal, value)));
61#else
62 auto pairs = value - ((value >> 1) & 0x5555555555555555ull);
63 auto nibbles = (pairs & 0x3333333333333333ull) + ((pairs >> 2) & 0x3333333333333333ull);
64 auto bytes = (nibbles + (nibbles >> 4)) & 0x0f0f0f0f0f0f0f0full;
65 auto prefixes = bytes * 0x0101010101010101ull;
66 auto marked = ((prefixes | 0x8080808080808080ull) -
67 (ordinal + 1) * 0x0101010101010101ull) & 0x8080808080808080ull;
68 auto shift = unsigned(std::countr_zero(marked)) & ~7u;
69 if (shift) ordinal -= unsigned((prefixes >> (shift - 8)) & 255);
70 auto lower = unsigned((nibbles >> shift) & 15);
71 auto step = unsigned(ordinal >= lower);
72 shift += step * 4;
73 ordinal -= step * lower;
74 auto bits = unsigned(value >> shift) & 15;
75 lower = (bits & 1) + ((bits >> 1) & 1);
76 step = unsigned(ordinal >= lower);
77 shift += step * 2;
78 ordinal -= step * lower;
79 return shift + ordinal + unsigned(((value >> shift) & 1) == 0);
80#endif
81 }

105 // W-bit fields return to a word boundary after 64/gcd(W,64) values.
106 // Each tile assigns its output words once, rather than repeatedly loading
107 // and updating them for each input field. All shifts are compile-time
108 // constants below 64; the final incomplete tile uses bounded scalar work.
109 template <unsigned W>
110 inline void pack_low_fixed(std::span<std::uint64_t const> source,
111 std::span<std::uint64_t> out) noexcept {
112 static_assert(W <= 63);
113 if constexpr (W) {
114 constexpr auto divisor = std::gcd(W, 64u);
115 constexpr auto inputs = 64 / divisor;
116 constexpr auto outputs = W / divisor;
117 std::size_t at = 0;
118 for (; source.size() - at >= inputs; at += inputs) {
119 auto destination = out.data() + (at / inputs) * outputs;
120#if defined(__aarch64__) && defined(__ARM_NEON) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
121 // These complete tiles consume exactly four/eight input words and
122 // emit one word. Narrowing preserves each field's low bits.
123 if constexpr (W == 16) {
124 auto a = vmovn_u64(vld1q_u64(source.data() + at));
125 auto b = vmovn_u64(vld1q_u64(source.data() + at + 2));
126 *destination = vget_lane_u64(vreinterpret_u64_u16(vmovn_u32(vcombine_u32(a, b))), 0);
127 } else if constexpr (W == 8) {
128 auto a = vmovn_u64(vld1q_u64(source.data() + at));
129 auto b = vmovn_u64(vld1q_u64(source.data() + at + 2));
130 auto c = vmovn_u64(vld1q_u64(source.data() + at + 4));
131 auto d = vmovn_u64(vld1q_u64(source.data() + at + 6));
132 *destination = vget_lane_u64(vreinterpret_u64_u8(vmovn_u16(vcombine_u16(
133 vmovn_u32(vcombine_u32(a, b)), vmovn_u32(vcombine_u32(c, d))))), 0);
134 } else
135#endif
136 low_tile<W>(source.data() + at, destination, std::make_index_sequence<outputs>{});
137 }
138 auto tail = out.subspan((at / inputs) * outputs);
139 std::fill(tail.begin(), tail.end(), 0);
140 constexpr auto mask = (std::uint64_t{1} << W) - 1;
141 for (std::size_t i = 0; i != source.size() - at; ++i) {
142 auto bit = i * W;
143 unsigned shift = unsigned(bit & 63);
144 auto value = source[at + i] & mask;
145 tail[bit >> 6] |= value << shift;
146 if (shift + W > 64) tail[(bit >> 6) + 1] |= value >> (64 - shift);
147 }
148 }
149 }

157 // This dispatch occurs once per complete low section, not per value. The
158 // destination may contain old data: both complete and tail words are set,
159 // including zero tail padding. Source and destination must not overlap.
160 inline void pack_low(std::span<std::uint64_t const> source,
161 std::span<std::uint64_t> out, unsigned width) {
162 if (width > 63) error_detail::raise<std::invalid_argument>("elias_fano low width");
163 if (width && source.size() > (std::numeric_limits<std::uint64_t>::max() - 63) / width)
164 error_detail::raise<std::overflow_error>("elias_fano low section extent");
165 if (out.size() != words(source.size() * width))
166 error_detail::raise<std::invalid_argument>("elias_fano low output size");
167 static constexpr auto packers = low_packers(std::make_index_sequence<64>{});
168 packers[width](source, out);
169 }
170
171 // The owner has checked monotonicity, width and extent and zeroed out.
172 // (source[i] >> width) + i is strictly increasing, even for equal values.
173 // Accumulating a word locally removes repeated dependent memory updates;
174 // gaps remain zero. This preserves the exact existing high-bit ordering.
175 inline void write_high(std::span<std::uint64_t const> source,
176 std::span<std::uint64_t> out, unsigned width) noexcept {
177 std::uint64_t word = 0, value = 0;
178 for (std::uint64_t i = 0; i != source.size(); ++i) {
179 auto position = (source[i] >> width) + i;
180 auto next = position >> 6;
181 if (next != word) {
182 out[word] = value;
183 word = next;
184 value = 0;
185 }
186 value |= std::uint64_t{1} << (position & 63);
187 }
188 if (!source.empty()) out[word] = value;
189 }

199 // Elias--Fano over an arbitrary nondecreasing sequence of uint64_t values.
200 // Repeated values and the empty sequence are supported. Sampling intervals,
201 // record counts, sentinels and fixed strides belong to the caller.
202 // Select samples every 256 ones: a dense group spans at most 4096 bits
203 // (65 aligned words); a sparse group stores each one-bit position explicitly.
204 // Constructors validate section shapes without reading their contents.

253 std::uint64_t select(std::uint64_t ordinal) const {
254 if (ordinal >= entry_count_) [[unlikely]]
255 error_detail::raise<std::out_of_range>("Elias-Fano ordinal");
256 auto position = select_high(ordinal);
257 if (position < ordinal) error_detail::raise<std::invalid_argument>("invalid elias_fano high value");
258 auto hi = position - ordinal;
259 if (hi > (universe_ >> low_width_)) error_detail::raise<std::invalid_argument>("elias_fano high overflow");
260 std::uint64_t lo = 0;
261 if (low_width_) {
262 auto bit = ordinal * low_width_;
263 auto word = bit >> 6;
264 unsigned shift = unsigned(bit & 63);
265 lo = low_[word] >> shift;
266 if (shift + low_width_ > 64) lo |= low_[word + 1] << (64 - shift);
267 lo &= (std::uint64_t{1} << low_width_) - 1;
268 }
269 auto value = (hi << low_width_) | lo;
270 if (value > universe_) error_detail::raise<std::invalid_argument>("elias_fano value exceeds universe");
271 return value;
272 }

275 std::uint64_t select_high(std::uint64_t ordinal) const {
276 auto sample = samples_[ordinal >> 8];
277 unsigned remaining = unsigned(ordinal & 255);
278 if (sample.sparse != std::numeric_limits<std::uint64_t>::max()) {
279 if (sample.sparse > sparse_.size() || remaining >= sparse_.size() - sample.sparse)
280 error_detail::raise<std::invalid_argument>("invalid elias_fano exception");
281 auto position = sparse_[sample.sparse + remaining];
282 if (position >= high_bits_ || !(high_[position >> 6] & (std::uint64_t{1} << (position & 63))))
283 error_detail::raise<std::invalid_argument>("invalid elias_fano sparse position");
284 return position;
285 }
286 if (sample.first >= high_bits_) error_detail::raise<std::invalid_argument>("invalid elias_fano sample");
287 auto word = sample.first >> 6;
288 auto value = high_[word] & (~std::uint64_t{0} << (sample.first & 63));
289 for (unsigned scanned = 0; scanned < 65 && word < high_.size(); ++scanned, ++word) {
290 if (scanned) value = high_[word];
291 auto population = unsigned(std::popcount(value));
292 if (remaining < population) {
293 auto position = word * 64 + elias_fano_detail::select_word(value, remaining);
294 if (position >= high_bits_ || position - sample.first >= 4096)
295 error_detail::raise<std::invalid_argument>("elias_fano dense span");
296 return position;
297 }
298 remaining -= population;
299 }
300 error_detail::raise<std::invalid_argument>("elias_fano missing high bit");
301 }

313 struct elias_fano {
314 static elias_fano build(std::span<std::uint64_t const> residuals) {
315 if (residuals.empty()) return {};
316 if (!elias_fano_detail::monotone(residuals))
317 error_detail::raise<std::invalid_argument>("elias_fano nonmonotone offsets");
318 elias_fano result;
319 result.entry_count = residuals.size();
320 result.universe = residuals.back();
321 auto quotient = result.universe / residuals.size();
322 result.low_width = quotient ? unsigned(std::bit_width(quotient) - 1) : 0;
323 constexpr auto maximum = std::numeric_limits<std::uint64_t>::max() - 255;
324 if (residuals.size() > maximum || (result.low_width && residuals.size() > maximum / result.low_width) ||
325 (result.universe >> result.low_width) > maximum - residuals.size())
326 error_detail::raise<std::overflow_error>("elias_fano section extent");
327 auto low_bits = elias_fano_detail::multiply(residuals.size(), result.low_width);
328 auto high_bits = elias_fano_detail::add(result.universe >> result.low_width, residuals.size());
329 result.low.assign(elias_fano_detail::words(low_bits), 0);
330 result.high.assign(elias_fano_detail::words(high_bits), 0);
331 result.samples.clear();
332 elias_fano_detail::pack_low(residuals, result.low, result.low_width);
333 elias_fano_detail::write_high(residuals, result.high, result.low_width);
334 for (std::uint64_t begin = 0; begin < residuals.size(); begin += 256) {
335 auto end = residuals.size() - begin < 256 ? residuals.size() : begin + 256;
336 auto first = (residuals[begin] >> result.low_width) + begin;
337 auto last = (residuals[end - 1] >> result.low_width) + end - 1;
338 auto sparse = std::numeric_limits<std::uint64_t>::max();
339 if (last - first >= 4096) {
340 sparse = result.sparse.size();
341 for (auto i = begin; i < end; ++i)
342 result.sparse.push_back((residuals[i] >> result.low_width) + i);
343 }
344 result.samples.push_back({first, sparse});
345 }
346 return result;
347 }
