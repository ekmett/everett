120 // Read-only whole-file mapping, with native POSIX and Windows branches.
121 // The caller must keep the backing object immutable while any pin lives:
122 // external modification/truncation is not made safe by MAP_PRIVATE or by a
123 // shared_ptr. File type and size are checked on the opened handle, not via
124 // a separate path stat. This read backend makes no crash-durability claim.
125 struct mapped_file {
126 mapped_file() = default;
127
128 static mapped_file open(std::filesystem::path const & path) {
129 auto state = std::make_shared<mapped_file_detail::region>();
130#if defined(_WIN32)
131 state->file = ::CreateFileW(path.c_str(), GENERIC_READ,
132 FILE_SHARE_READ | FILE_SHARE_DELETE, nullptr,
133 OPEN_EXISTING, FILE_FLAG_BACKUP_SEMANTICS, nullptr);
134 if (state->file == INVALID_HANDLE_VALUE)
135 throw std::system_error(static_cast<int>(::GetLastError()), std::system_category(), "open mapped file");
136 BY_HANDLE_FILE_INFORMATION info{};
137 if (!::GetFileInformationByHandle(state->file, &info))
138 throw std::system_error(static_cast<int>(::GetLastError()), std::system_category(), "inspect mapped file");
139 if (::GetFileType(state->file) != FILE_TYPE_DISK || (info.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY))
140 throw std::invalid_argument("mapped file must be a regular file");
141 LARGE_INTEGER size{};
142 if (!::GetFileSizeEx(state->file, &size))
143 throw std::system_error(static_cast<int>(::GetLastError()), std::system_category(), "size mapped file");
144 if (size.QuadPart < 0) throw std::length_error("negative mapped file size");
145 state->length = static_cast<std::uint64_t>(size.QuadPart);
146#else
147 // Nonblocking open permits rejecting FIFOs rather than waiting for a
148 // writer before fstat can establish that the object is not regular.
149 int flags = O_RDONLY | O_NONBLOCK;
150#ifdef O_CLOEXEC
151 flags |= O_CLOEXEC;
152#endif
153 state->file = ::open(path.c_str(), flags);
154 if (state->file == -1) throw std::system_error(errno, std::generic_category(), "open mapped file");
155#ifndef O_CLOEXEC
156 if (::fcntl(state->file, F_SETFD, FD_CLOEXEC) == -1)
157 throw std::system_error(errno, std::generic_category(), "protect mapped descriptor");
158#endif
159 struct stat info{};
160 if (::fstat(state->file, &info) == -1)
161 throw std::system_error(errno, std::generic_category(), "inspect mapped file");
162 if (!S_ISREG(info.st_mode)) throw std::invalid_argument("mapped file must be a regular file");
163 if (info.st_size < 0) throw std::length_error("negative mapped file size");
164 state->length = static_cast<std::uint64_t>(info.st_size);
165#endif
166 if (state->length > (std::numeric_limits<std::size_t>::max)())
167 throw std::length_error("mapped file exceeds addressable span size");
168 if (state->length) {
169#if defined(_WIN32)
170 state->mapping = ::CreateFileMappingW(state->file, nullptr, PAGE_READONLY, 0, 0, nullptr);
171 if (!state->mapping)
172 throw std::system_error(static_cast<int>(::GetLastError()), std::system_category(), "create file mapping");
173 state->data = ::MapViewOfFile(state->mapping, FILE_MAP_READ, 0, 0, static_cast<SIZE_T>(state->length));
174 if (!state->data)
175 throw std::system_error(static_cast<int>(::GetLastError()), std::system_category(), "map file view");
176#else
177 auto address = ::mmap(nullptr, static_cast<std::size_t>(state->length), PROT_READ, MAP_PRIVATE, state->file, 0);
178 if (address == MAP_FAILED) throw std::system_error(errno, std::generic_category(), "map file view");
179 state->data = address;
180#endif
181 }
182 return mapped_file(std::move(state));
183 }
