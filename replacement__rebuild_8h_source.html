37 replacement_metadata() = default;
38 replacement_metadata(base_type value, std::uint64_t b, std::uint64_t u, bool active)
39 : base_type(std::move(value)), clean_base(b), mutations(u), rebuilding(active) {}
40 void validate(std::uint64_t admissions) const {
41 if (clean_base > admissions || mutations != admissions - clean_base ||
42 this->live_count > admissions ||
43 (clean_base > mutations && this->live_count < clean_base - mutations))
44 throw std::invalid_argument("invalid replacement generation mass");
45 auto trigger = clean_base / 4;
46 if (!rebuilding) {
47 if ((clean_base < 64 && (mutations || this->live_count != clean_base)) ||
48 (clean_base >= 64 && mutations >= trigger))
49 throw std::invalid_argument("inactive replacement generation passed its trigger");
50 } else {
51 // At freeze n_s <= b + floor(b/4); at most floor(n_s/8)
52 // intervening admissions fit before handoff. Divide before adding to
53 // avoid overflow in the upper bound for extreme metadata.
54 auto extra = clean_base / 8 + trigger / 8 + (clean_base % 8 + trigger % 8) / 8;
55 if (clean_base < 64 || mutations < trigger || mutations - trigger > extra)
56 throw std::invalid_argument("invalid active replacement generation");
57 }
58 }
59 std::vector<std::byte> encode() const requires std::same_as<typename A::element, std::uint64_t> {
60 auto inner = base_type::encode();
61 std::vector<std::byte> result(40 + inner.size());
62 constexpr std::array<unsigned char, 8> magic{'E', 'V', 'R', 'T','.','R','B',0};
63 for (unsigned i = 0; i != 8; ++i) result[i] = std::byte(magic[i]);
64 std::array<std::uint64_t, 4> fields{1, clean_base, mutations, rebuilding ? 1u : 0u};
65 for (unsigned n = 0; n != fields.size(); ++n)
66 for (unsigned i = 0; i != 8; ++i) result[8 + n * 8 + i] = std::byte(fields[n] >> (i << 3));
67 std::copy(inner.begin(), inner.end(), result.begin() + 40);
68 return result;

70 static replacement_metadata decode(std::span<std::byte const> data)
71 requires std::same_as<typename A::element, std::uint64_t> {
72 constexpr std::array<unsigned char, 8> magic{'E', 'V', 'R', 'T','.','R','B',0};
73 if (data.size() <= 56) throw std::invalid_argument("truncated replacement metadata");

241 static std::uint64_t reservation_work(std::uint64_t records) {
242 auto h = std::min<std::uint64_t>(64, DepthLimit - 3);
243 auto g = action_bound(h), c = runtime_type::local_charge_bound;
244 constexpr std::uint64_t runs = 128, scan = 32, setup = runs * (scan + 8) + 32;
245 auto large = add(add(mul(22, g), mul(mul(10, add(c, 32)), add(h, 1))), setup + 26 * scan + 2);
246 // Below the small threshold the valid generation has at most 256
247 // physical occurrences. L <= 64 and bit_width(L) <= 7.
248 auto small = add(setup + 321 * scan, add(mul(130, action_bound(7)), mul(512, add(c, 32))));

260 // Complete validation precedes mutation. The per-key fallback also prepares
261 // detached replay records before execution. Intermediate cuts stay private
262 // until the entire batch and its owed service succeeds.
263 world_type contribute(contribution_type input) {
264 writable();
265 if (!admission_ready()) throw std::logic_error("replacement foreground needs recovery service");
266 if (input.base() && input.base()->metadata().schema_id != published_.metadata().schema_id)
267 throw std::invalid_argument("rebuild contribution uses another schema");
268 if (published_.runtime().query_root().head()->depth() > DepthLimit ||
269 (input.base() && input.base()->runtime().query_root().head()->depth() > DepthLimit))
270 throw std::length_error("replacement query exceeds depth allowance");
271 if (input.records().empty()) return published_;
272 // Initial unique string replacements are already clean arrows. Share
273 // the typed preflight and avoid constructing detached per-key replay
274 // entries when the runtime can initialize a prefix and admit the tail
275 // before publishing the complete pristine batch.
276 if constexpr (std::is_same_v<sort_type, unsorted<std::optional<std::string>>> &&
277 requires(runtime_type & runtime, std::span<profile_record const> records) {
278 runtime.try_initialize_sorted(records, std::uint64_t{}, DepthLimit);
279 }) {
280 auto count = input.records().size();
281 if (count >= 2 && !base_ && !mutations_ &&
282 !job_ && !recovering_ && !foreground_->pending() && !mass(published_) &&
283 !work_.mutations && !work_.generations) {
284 auto metadata = foreground_->prepare(input);
285 auto accepted = add(work_.mutations, count);
286 auto prior = foreground_->work().charged;
287 try {
288 if (foreground_->initialize(input, metadata)) {
289 work_.foreground_charged = add(work_.foreground_charged, foreground_->work().charged - prior);
290 prior = foreground_->work().charged;
291 base_ = count;
292 work_.mutations = accepted;
293 published_ = publication();
294 return published_;
295 }
296 } catch (...) {
297 work_.foreground_charged = add(work_.foreground_charged, foreground_->work().charged - prior);
298 poison();
299 throw;
300 }
301 }
302 }
303 std::vector<mutation> entries; entries.reserve(input.records().size());
304 for (auto const & record : input.records()) {
305 engine_type::key_transport::dispatch(record.key.view(), [&]<class S>(std::type_identity<S>, auto const & key) {
306 static_assert(std::is_same_v<S, sort_type>);
307 auto arrow = typed_detail::value<P, S>(record.value.view());
308 auto after = semantics::apply(key, semantics::initial(key), arrow);
309 std::optional<std::uint64_t> target;
310 auto before = published_.template get_encoded<S>(key, record.key,
311 semantics::present(key, after) ? nullptr : &target);
312 if (input.base() && before != input.base()->template get_encoded<S>(key, record.key))
313 throw std::invalid_argument("stale rebuilt key value");
314 if (!semantics::present(key, before) && !semantics::present(key, after))
315 throw std::invalid_argument("deleting absent rebuilt key");
316 auto retained = record.retained_limit_bits;
317 if (target) retained = retained ? std::min(*retained, *target) : *target;
318 entries.push_back({key, std::move(before), std::move(after), std::move(arrow), 0, retained});
319 });
320 }
321 try {
322 for (auto & entry : entries) apply(std::move(entry));
323 published_ = publication();
324 return published_;
325 } catch (...) { poison(); throw; }
326 }

328 writable(); if (!budget || !pending()) return {};

412 void apply(mutation entry) {
413 require(foreground_->admission_ready(), "replacement foreground exhausted its admission service");
414 entry.ordinal = add(work_.mutations, 1);
415 // A fresh replacement in a clean string table adds one live row and
416 // one physical admission, without introducing history to collect.
417 // Extend that clean base; pending ordinary carries still receive their
418 // normal admission service. Once history exists, every write funds it.
419 bool extends_clean = false;
420 if constexpr (std::is_same_v<sort_type, unsorted<std::optional<std::string>>>)
421 extends_clean = !job_ && !mutations_ && !entry.before && bool(entry.after);
422 if (job_) {
423 require(job_->admitted < job_->horizon, "rebuild deadline exhausted before admission");
424 job_->queue.push_back(entry);
425 }
426 auto prior = foreground_->work().charged;
427 try { foreground_->contribute(command(entry)); }
428 catch (...) { work_.foreground_charged = add(work_.foreground_charged, foreground_->work().charged - prior); throw; }
429 work_.foreground_charged = add(work_.foreground_charged, foreground_->work().charged - prior);
430 work_.mutations = entry.ordinal;
431 if (extends_clean) {
432 base_ = add(base_, 1);
433 require(mass(foreground_->snapshot()) == base_, "extended clean generation mass mismatch");
434 return;
435 }
436 mutations_ = add(mutations_, 1);
437 require(mass(foreground_->snapshot()) == add(base_, mutations_), "foreground generation mass mismatch");
438 if (job_) {
439 ++job_->admitted;
440 work_.reserved = add(work_.reserved, job_->action);
441 work_.maximum_replay = std::max<std::uint64_t>(work_.maximum_replay, job_->queue.size());
442 grant(add(job_->quantum, job_->action)); service();
443 require(!job_ || job_->admitted < job_->horizon, "rebuild missed funded handoff horizon");
444 } else {
445 auto live = foreground_->snapshot().metadata().live_count;
446 bool small = base_ < small_limit || live < small_limit;
447 if (small || mutations_ >= base_ / 4) {
448 start(small);
449 if (small) { grant(job_->bound); service(); require(!job_, "small rebuild did not finish"); }
450 }
451 }

453 static std::uint64_t action_bound(std::uint64_t height) {
454 auto depth = add(height, 3), c = runtime_type::local_charge_bound;
455 auto ready = add(add(mul(2, c), mul(16, depth)), 512);
456 auto query = mul(mul(64, add(depth, 1)), add(add(P::group_size, P::codec_block_size), 16));
457 auto service = mul(mul(8, c), add(height, 2));
458 return add(add(ready, service), add(query, add(mul(8, height), 16)));
459 }
460 void start(bool small) {
461 auto source = foreground_->snapshot();
462 auto freeze = add(mul(source.runtime().runs().size(), 8), 32);
463 work_.reserved = add(work_.reserved, freeze);
464 work_.granted = add(work_.granted, freeze);
465 work_.committed = add(work_.committed, freeze);
466 auto next = std::make_unique<rebuild>(std::move(source));
467 next->live = next->frozen.metadata().live_count;
468 next->horizon = small ? 1 : next->live / 8;
469 require(next->horizon, "empty rebuilding horizon");
470 next->frozen_ordinal = work_.mutations;
471 auto limit = add(next->live, small ? 0 : next->horizon);
472 auto height = std::bit_width(limit); auto depth = add(height, 3);
473 if (depth > DepthLimit) throw std::length_error("rebuild candidate exceeds supported depth");
474 auto c = runtime_type::local_charge_bound;
475 next->action = action_bound(height);
476 std::uint64_t physical = 0;
477 auto runs = next->frozen.runtime().runs();
478 for (auto const & run : runs) physical = add(physical, run->native->size());
479 next->source_records = physical;
480 if constexpr (replacement_detail::clean_family<Family>::enabled)
481 next->tiny = small && next->live <= small_limit && physical <= tiny_record_limit;
482 next->scan_price = add(mul(2, std::bit_width(runs.size())), 16);
483 next->setup = add(mul(runs.size(), add(next->scan_price, 8)), 32);
484 auto r = add(next->setup, mul(add(add(physical, next->live), 1), next->scan_price));
485 r = add(r, mul(next->live, next->action));
486 // All maintenance through the largest possible replay cut is reserved,
487 // including a large carry first triggered by an intervening mutation.
488 r = add(r, mul(mul(add(c, 32), limit), add(height, 1)));
489 r = add(r, mul(limit, next->action)); // Nested grant fragments at settlement.
490 r = add(r, mul(next->horizon, next->action)); // Outer atomic-action fragments.
491 next->bound = add(r, next->action); // Final metadata check and ownership handoff.
492 if (next->tiny) next->bound = add(next->bound, tiny_conversion_bound());
493 next->quantum = ceil(next->bound, next->horizon);
494 work_.reserved = add(work_.reserved, next->bound);
495 job_ = std::move(next);
496 }
497 void grant(std::uint64_t amount) {
498 auto credit = add(job_->credit, amount), total = add(work_.granted, amount);
499 job_->credit = credit; work_.granted = total;
500 }
501 std::uint64_t price() const {
502 auto const & j = *job_;
503 if (j.tiny) return j.bound;
504 if (!j.candidate) return j.setup;
505 if (j.building && !j.scan->has_row() && !j.scan->done()) return j.scan_price;
506 return j.action;
507 }
508 void candidate_work(auto && operation) {
509 auto before = job_->candidate->work().charged;
510 try { operation(); }
511 catch (...) {
512 work_.candidate_charged = add(work_.candidate_charged, job_->candidate->work().charged - before);
513 throw;
514 }
515 work_.candidate_charged = add(work_.candidate_charged, job_->candidate->work().charged - before);
516 }
517 // A settled candidate with at most 64 admissions has at most seven levels.
518 // Checked restore confines every object route to the three slots per
519 // level. Their pairs, at most one carrier per level, and the prepared
520 // head chain use fewer than 8*(height+1) distinct pairs. Each pair has at most 2*64+4 occurrences:
521 // V <= 64 + ceil(V/K) + ceil(64/K), with K >= 3. Charge an index allowance
522 // including both directory streams, plus object/level admission metadata.
523 static std::uint64_t tiny_conversion_bound() {
524 auto pairs = mul(8, add(std::bit_width(small_limit), 1));
525 return add(mul(pairs, add(32, mul(add(mul(2, small_limit), 4), add(P::group_size, 8)))), 1024);
526 }
527 template <class Source> auto convert_tiny(Source const & source) {
528 using source_pair = typename Source::query_type::pair_type;
529 using source_node = std::remove_const_t<typename source_pair::element_type>;
530 using source_object = typename Source::object_pointer::element_type;
531 using target_node = typename Family::node_type;
532 using target_pair = typename target_node::pair_type;
533 using target_object = typename Family::snapshot_type::object_pointer;
534 static_assert(std::is_same_v<typename source_node::native_type, typename Family::native_type>);
535 auto const & old = source.frontier();
536 require(old.admissions <= small_limit && old.levels.size() <= std::bit_width(small_limit) && !old.service_due,
537 "tiny conversion exceeds its bounded settled frontier");
538 std::uint64_t charged = 0;
539 auto charge = [&](std::uint64_t amount) {
540 charged = add(charged, amount);
541 require(charged <= tiny_conversion_bound(), "tiny conversion exceeded its allowance");
542 work_.tiny_conversion_charged = add(work_.tiny_conversion_charged, amount);
543 work_.candidate_charged = add(work_.candidate_charged, amount);
544 };
545 charge(add(32, mul(old.levels.size(), 16)));
546 std::unordered_map<source_node const *, target_pair> pairs;
547 auto copy_pair = [&](auto && self, source_pair const & value) -> target_pair {
548 if (!value) return {};
549 if (auto found = pairs.find(value.get()); found != pairs.end()) return found->second;
550 auto main = self(self, value->main_target());
551 require(pairs.size() < 8 * (std::bit_width(small_limit) + 1), "too many tiny conversion pairs");
552 auto secondary = value->secondary_target();
553 auto a = main ? ceil(main->virtual_size(), P::group_size) : 0;
554 auto b = secondary ? ceil(secondary->size(), P::group_size) : 0;
555 auto n = add(value->native_owner()->size(), add(a, b));
556 require(n <= 2 * small_limit + 4, "tiny pair exceeds its occurrence allowance");
557 charge(add(add(16, mul(n, P::group_size + 6)), add(ceil(a, P::codec_block_size), ceil(b, P::codec_block_size))));
558 cola_index_builder<P, typename Family::native_type, target_node> builder(value->native_owner(), main, secondary);
559 while (!builder.done()) builder.step(1);
560 auto result = target_node::from_built(builder.finish());
561 work_.tiny_indexes = add(work_.tiny_indexes, 1);
562 pairs.emplace(value.get(), result); return result;
563 };
564 std::unordered_map<source_object const *, target_object> objects;
565 auto copy_object = [&](auto && self, typename Source::object_pointer const & value) -> target_object {
566 if (!value) return {};
567 if (auto found = objects.find(value.get()); found != objects.end()) return found->second;
568 typename Family::routes_type next{self(self, value->next.main), self(self, value->next.secondary)};
569 require(objects.size() < 3 * old.levels.size(), "too many tiny conversion objects");
570 charge(16);
571 auto result = std::make_shared<typename Family::object_type const>(typename Family::object_type{
572 value->identity, value->first, value->last, value->level, value->native,
573 copy_pair(copy_pair, value->pair), std::move(next)});
574 objects.emplace(value.get(), result); return result;
575 };
576 auto copy_route = [&](auto const & route) {
577 return typename Family::routes_type{copy_object(copy_object, route.main), copy_object(copy_object, route.secondary)};
578 };
579 typename Family::frontier_type out;
580 out.admissions = old.admissions; out.next_identity = old.next_identity; out.service_due = old.service_due;
581 out.levels.resize(old.levels.size());
582 for (std::size_t i = 0; i != old.levels.size(); ++i) {
583 auto const & before = old.levels[i]; auto & after = out.levels[i];
584 require(!before.job, "tiny conversion requires completed jobs");
585 unsigned active = 0;
586 for (std::size_t j = 0; j != before.slots.size(); ++j) {
587 auto const & slot = before.slots[j]; active += slot.state == redundant_slot_state::active;
588 after.slots[j] = {slot.state, copy_object(copy_object, slot.object), copy_route(slot.route),
589 copy_pair(copy_pair, slot.carrier), slot.ever_visible};
590 }
591 require(active < 2, "tiny conversion requires a settled level");
592 after.last_destination = before.last_destination; after.last_destination_visible = before.last_destination_visible;
593 }
594 out.root = copy_route(old.root);
595 auto head = copy_pair(copy_pair, source.query_root().head());
596 auto result = Family::snapshot_type::restore(std::move(out), std::move(head));
597 return result;
598 }
599 void tiny_rebuild() requires replacement_detail::clean_family<Family>::enabled {
600 using clean_family = typename replacement_detail::clean_family<Family>::type;
601 using clean_engine = typed_engine<P, A, DepthLimit, clean_family>;
602 auto & j = *job_;
603 require(j.tiny && j.live <= small_limit && j.source_records <= tiny_record_limit &&
604 j.queue.empty() && !j.admitted, "tiny rebuild escaped its bounded eager path");
605 clean_engine candidate(j.frozen.metadata().schema_id);
606 work_.candidate_charged = add(work_.candidate_charged, candidate.work().charged);
607 auto execute = [&](auto && operation) {
608 auto before = candidate.work().charged;
609 try { operation(); }
610 catch (...) { work_.candidate_charged = add(work_.candidate_charged, candidate.work().charged - before); throw; }
611 work_.candidate_charged = add(work_.candidate_charged, candidate.work().charged - before);
612 };
613 scan_type rows(j.frozen);
614 while (!rows.done()) {
615 if (!rows.has_row()) work_.scan_records = add(work_.scan_records, rows.step(1));
616 if (!rows.has_row()) continue;
617 auto row = rows.take_row();
618 auto arrow = clean(row.key, row.value);
619 require(semantics::apply(row.key, semantics::initial(row.key), arrow) == row.value, "invalid clean replacement arrow");
620 while (!candidate.admission_ready()) execute([&] { candidate.advance(j.action); });
621 execute([&] { candidate.contribute(clean_engine::template change<sort_type>(row.key, arrow)); });
622 ++j.rows; work_.clean_rows = add(work_.clean_rows, 1);
623 require(j.rows <= j.live, "tiny scan exceeds frozen live count");
624 }
625 auto scanned = candidate.snapshot();
626 require(rows.consumed() == j.source_records && j.rows == j.live &&
627 scanned.runtime().admissions() == j.live && scanned.metadata() == j.frozen.metadata(),
628 "tiny rebuild differs from frozen source");
629 while (candidate.pending()) execute([&] { candidate.advance(j.action); });
630 auto clean = candidate.snapshot();
631 auto runtime = convert_tiny(clean.runtime());
632 auto state = typed_world_type::restore(std::move(runtime), clean.metadata(), j.frozen.metadata().schema_id);
633 require(state.metadata() == foreground_->snapshot().metadata() && state.runtime().admissions() == j.live,
634 "tiny handoff differs from foreground");
635 auto replacement = engine_type::from_snapshot(std::move(state), foreground_->storage());
636 work_.candidate_charged = add(work_.candidate_charged, replacement.work().charged);
637 base_ = j.live; mutations_ = 0;
638 work_.generations = add(work_.generations, 1); work_.tiny_generations = add(work_.tiny_generations, 1);
639 foreground_ = std::make_unique<engine_type>(std::move(replacement)); job_.reset(); recovering_ = false;
640 }
641 void service() {
642 while (job_) {
643 auto cost = price(); if (job_->credit < cost) break;
644 auto limit = add(job_->bound, mul(job_->admitted, job_->action));
645 require(job_->committed <= limit && cost <= limit - job_->committed, "rebuild exceeded its reserved work bound");
646 job_->credit -= cost; job_->committed += cost; work_.committed = add(work_.committed, cost);
647 auto & j = *job_;
648 if constexpr (replacement_detail::clean_family<Family>::enabled) {
649 if (j.tiny) { tiny_rebuild(); continue; }
650 }
651 if (!j.candidate) {
652 auto seed = std::make_unique<engine_type>(j.frozen.metadata().schema_id);
653 work_.candidate_charged = add(work_.candidate_charged, seed->work().charged);
654 if constexpr (requires { foreground_->storage(); }) {
655 // The empty seed does not execute native work. Its initialization
656 // and context-aware restoration both fit the setup's 32-unit margin.
657 j.candidate = std::make_unique<engine_type>(
658 engine_type::from_snapshot(seed->snapshot(), foreground_->storage()));
659 work_.candidate_charged = add(work_.candidate_charged, j.candidate->work().charged);
660 } else j.candidate = std::move(seed);
661 j.scan = std::make_unique<scan_type>(j.frozen);
662 } else if (j.building) {
663 if (j.scan->has_row()) {
664 if (!j.candidate->admission_ready()) candidate_work([&]{ j.candidate->advance(j.action); });
665 else {
666 auto row = j.scan->take_row();
667 auto arrow = clean(row.key, row.value);
668 require(semantics::apply(row.key, semantics::initial(row.key), arrow) == row.value, "invalid clean replacement arrow");
669 candidate_work([&]{ j.candidate->contribute(engine_type::template change<sort_type>(row.key, arrow)); });
670 ++j.rows; work_.clean_rows = add(work_.clean_rows, 1);
671 }
672 } else if (!j.scan->done()) work_.scan_records = add(work_.scan_records, j.scan->step(1));
673 else {
674 require(j.scan->consumed() == j.source_records && j.rows == j.live && mass(j.candidate->snapshot()) == j.live &&
675 j.candidate->snapshot().metadata() == j.frozen.metadata(), "clean rebuild differs from frozen source");
676 j.scan.reset(); j.building = false;
677 }
678 } else if (!j.queue.empty()) {
679 if (!j.candidate->admission_ready()) candidate_work([&]{ j.candidate->advance(j.action); });
680 else {
681 auto const & entry = j.queue.front();
682 require(entry.ordinal == add(add(j.frozen_ordinal, j.replayed), 1), "rebuild replay order changed");
683 require(j.candidate->snapshot().template get<sort_type>(entry.key) == entry.before, "rebuild replay old value mismatch");
684 candidate_work([&]{ j.candidate->contribute(command(entry)); });
685 require(j.candidate->snapshot().template get<sort_type>(entry.key) == entry.after, "rebuild replay new value mismatch");
686 j.queue.pop_front(); ++j.replayed; work_.replayed = add(work_.replayed, 1);
687 }
688 } else if (j.candidate->pending()) candidate_work([&]{ j.candidate->advance(j.action); });
689 else {
690 auto state = j.candidate->snapshot();
691 require(j.replayed == j.admitted && state.runtime().admissions() == add(j.live, j.replayed) &&
692 state.metadata() == foreground_->snapshot().metadata(), "rebuild handoff differs from foreground");
693 base_ = j.live; mutations_ = j.replayed;
694 work_.maximum_handoff_mutations = std::max(work_.maximum_handoff_mutations, j.admitted);
695 work_.generations = add(work_.generations, 1);
696 foreground_ = std::move(j.candidate); job_.reset(); recovering_ = false;
697 }
698 }
699 }
