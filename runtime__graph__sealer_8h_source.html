37 // Reserve one bounded set of independent, ready publication units. Each
38 // unit retains its ordinary seal acknowledgment. Dependency discovery and
39 // slot preparation finish before any nonblocking producer claim is held.
40 // The remaining graph uses the ordinary serial walk below.
41 std::size_t prepare_ready(std::span<pair_type const> roots,
42 std::span<native_pointer const> natives) {
43 constexpr std::size_t limit = 16;
44 std::vector<pair_type> unbound;
45 std::unordered_set<node_type const *> seen;
46 std::unordered_set<native_type const *> paired;
47 auto visit = [&](auto && self, pair_type const & pair) -> void {
48 if (!pair || !seen.insert(pair.get()).second ||
49 pair->bindings_.find(catalog_.identity(), catalog_.root())) return;
50 if (!pair->built()) return; // Ordinary admission diagnoses this case.
51 paired.insert(pair->native_owner().get());
52 self(self, pair->main_target());
53 unbound.push_back(pair);
54 };
55 for (auto const & pair : roots) visit(visit, pair);
56
57 std::vector<ready_unit> units;
58 units.reserve(limit);
59 std::unordered_set<native_type const *> selected;
60 for (auto const & pair : unbound) {
61 if (units.size() == limit) break;
62 auto main = pair->main_target() ? pair->main_target()->bindings_.find(catalog_.identity(), catalog_.root()) : nullptr;
63 auto secondary = pair->secondary_target() ? pair->secondary_target()->bindings_.find(catalog_.identity(), catalog_.root()) : nullptr;
64 if ((pair->main_target() && !main) || (pair->secondary_target() && !secondary)) continue;
65 auto const & owner = pair->native_owner();
66 if (!owner) continue;
67 auto native = owner->bindings_.find(catalog_.identity(), catalog_.root());
68 if (!native && (!batch_owned(owner) || !selected.insert(owner.get()).second)) continue;
69 if (main) catalog_.verify_sealed(main->receipt, file_kind::fractional_index);
70 if (secondary) catalog_.verify_sealed(secondary->receipt, file_kind::native_blob);
71 if (native) catalog_.verify_sealed(native->receipt, file_kind::native_blob);
72 units.emplace_back(pair, owner, std::move(main), std::move(secondary), std::move(native));
73 auto & unit = units.back();
74 unit.pair_claim.emplace(pair->bindings_.prepare(catalog_.identity(), catalog_.root()));
75 if (!unit.native) unit.native_claim.emplace(owner->bindings_.prepare(catalog_.identity(), catalog_.root()));
76 }
77 for (auto const & native : natives) {
78 if (units.size() == limit) break;
79 // A native which belongs to an unbound pair must stay available for
80 // native/index fusion, even when that pair's dependencies are unready.
81 if (!native || paired.contains(native.get()) || !batch_owned(native) ||
82 !selected.insert(native.get()).second ||
83 native->bindings_.find(catalog_.identity(), catalog_.root())) continue;
84 units.emplace_back(pair_type{}, native);
85 units.back().native_claim.emplace(native->bindings_.prepare(catalog_.identity(), catalog_.root()));
86 }
87 if (units.size() < 2) return 0;
88
89 std::vector<ready_unit *> claimed;
90 claimed.reserve(units.size());
91 for (auto & unit : units) {
92 if (unit.pair_claim) {
93 if (!unit.pair_claim->try_lock()) continue;
94 if (unit.pair_claim->value()) { unit.pair_claim->release(); continue; }
95 }
96 if (unit.native_claim) {
97 if (!unit.native_claim->try_lock()) {
98 if (unit.pair_claim) unit.pair_claim->release();
99 continue;
100 }
101 if (unit.native_claim->value()) {
102 unit.native_claim->release();
103 if (unit.pair_claim) unit.pair_claim->release();
104 continue; // Re-resolve the changed plan in ordinary fallback.
105 }
106 }
107 claimed.push_back(&unit);
108 }
109 if (claimed.size() < 2) return 0; // Claims leave scope before fallback.
110
111 std::vector<catalog_object_reservation> outputs;
112 std::vector<blob_identity> inputs;
113 outputs.reserve(claimed.size() * 2);
114 for (auto * unit : claimed) {
115 if (unit->native) {
116 unit->native_id = unit->native->receipt.object;
117 } else {
118 unit->native_id = ids_();
119 outputs.push_back({*unit->native_id, file_kind::native_blob});
120 }
121 if (unit->pair) {
122 unit->index_id = ids_();
123 outputs.push_back({*unit->index_id, file_kind::fractional_index});
124 }
125 if (unit->main) inputs.push_back(unit->main->identity);
126 }
127 std::sort(inputs.begin(), inputs.end(), [](auto const & a, auto const & b) {
128 return a.native.hex() < b.native.hex() || (a.native == b.native && a.index.hex() < b.index.hex());
129 });
130 inputs.erase(std::unique(inputs.begin(), inputs.end()), inputs.end());
131 auto attempt = reserve(outputs, inputs);
132 for (auto * unit : claimed) finish_ready(*unit, attempt);
133 return claimed.size();
134 }

147 std::shared_ptr<pair_binding_type const> ensure_pair(pair_type const & pair) {
148 if (!pair) throw std::invalid_argument("null runtime pair owner");
149 auto result = pair->bindings_.get_or_create(catalog_.identity(), catalog_.root(), [&] {
150 if (!pair->built()) throw std::invalid_argument("mapped pair has no binding in this catalog");
151 // Dependencies can share this pair's native owner. Resolve them before
152 // taking its producer lock, so aliases select the acknowledged fallback.
153 auto main = pair->main_target() ? ensure_pair(pair->main_target()) : nullptr;
154 auto secondary = pair->secondary_target() ? ensure_native(pair->secondary_target()) : nullptr;
155 auto const & owner = pair->native_owner();
156 if (!owner) throw std::invalid_argument("null runtime native owner");
157 std::optional<pair_seal> fused;
158 std::shared_ptr<typename mapped_type::index_type const> fused_index;
159 auto native = owner->bindings_.get_or_create(catalog_.identity(), catalog_.root(), [&] {
160 if constexpr (requires { owner->sealed(); })
161 if (owner->sealed()) return produce_native(owner);
162 if (!owner->owned()) return produce_native(owner);
163 blob_identity id{ids_(), ids_()};
164 std::array outputs{catalog_object_reservation{id.native, file_kind::native_blob},
165 catalog_object_reservation{id.index, file_kind::fractional_index}};
166 std::array<blob_identity, 1> input{main ? main->identity : id};
167 auto attempt = reserve(outputs, std::span<blob_identity const>(input.data(), main ? 1 : 0));
168 auto native_receipt = encode_native(owner).seal(catalog_.root(), id.native, attempt);
169 auto encoded = encode_cola_sections(*pair->built(), id.native,
170 main ? std::optional<blob_identity>(main->identity) : std::nullopt,
171 secondary ? std::optional<object_id>(secondary->receipt.object) : std::nullopt);
172 auto index_receipt = encoded.seal(catalog_.root(), id.index, attempt);
173 auto [mapped_native, index] = catalog_.template seal_native_pair<mapped_type>(
174 operation(), id, native_receipt, index_receipt);
175 auto binding = std::make_shared<native_binding_type const>(
176 native_seal{catalog_.identity(), std::move(native_receipt)}, std::move(mapped_native));
177 fused.emplace(pair_seal{catalog_.identity(), id, std::move(index_receipt)});
178 fused_index = std::move(index);
179 return binding;
180 });
181 catalog_.verify_sealed(native->receipt, file_kind::native_blob);
182 if (fused) {
183 auto mapped = mapped_type::bind(fused->identity, native->mapped, std::move(fused_index),
184 main ? main->mapped : nullptr, secondary ? secondary->mapped : nullptr,
185 secondary ? std::optional<object_id>(secondary->receipt.object) : std::nullopt);
186 return std::make_shared<pair_binding_type const>(std::move(*fused), std::move(mapped));
187 }
188 blob_identity id{native->receipt.object, ids_()};
189 std::array<blob_identity, 1> input{main ? main->identity : id};
190 auto attempt = reserve(id.index, file_kind::fractional_index,
191 std::span<blob_identity const>(input.data(), main ? 1 : 0));
192 auto encoded = encode_cola_sections(*pair->built(), id.native,
193 main ? std::optional<blob_identity>(main->identity) : std::nullopt,
194 secondary ? std::optional<object_id>(secondary->receipt.object) : std::nullopt);
195 auto receipt = encoded.seal(catalog_.root(), id.index, attempt);
196 auto index = catalog_.template seal_pair<mapped_type>(operation(), id, receipt);
197 auto mapped = mapped_type::bind(id, native->mapped, std::move(index),
198 main ? main->mapped : nullptr, secondary ? secondary->mapped : nullptr,
199 secondary ? std::optional<object_id>(secondary->receipt.object) : std::nullopt);
200 return std::make_shared<pair_binding_type const>(
201 pair_seal{catalog_.identity(), std::move(id), std::move(receipt)}, std::move(mapped));
202 });
203 catalog_.verify_sealed(result->receipt, file_kind::fractional_index);
204 return result;

208 // facade owners to the completed catalog rows without retaining a second
209 // registry of weak owners. Normal reads inspect metadata only.
210 void bind_native(native_pointer const & native, object_id const & id) {
211 auto result = native->bindings_.get_or_create(catalog_.identity(), catalog_.root(), [&] {

213 catalog_.identity(), catalog_.sealed_receipt(id, file_kind::native_blob)}, native->mapped());

216 }

235 native_pointer mapped_native(native_pointer const & native) {
236 auto binding = native->bindings_.find(catalog_.identity(), catalog_.root());
237 if (!binding) throw std::logic_error("runtime native has not been sealed");
238 if (native->mapped() == binding->mapped) return native;
239 // Only a distinct facade is cached. A mapped owner returns itself above,
240 // so the memo never acquires a shared-pointer reference to its own owner.
241 return native->mapped_owners_.get_or_create(catalog_.identity(), catalog_.root(), [&] {
242 auto result = native_type::from_mapped(binding->mapped);
243 result->bindings_.get_or_create(catalog_.identity(), catalog_.root(), [&] { return binding; });
244 return result;
245 });
246 }
247 pair_type mapped_pair(pair_type const & pair) {
248 auto binding = pair->bindings_.find(catalog_.identity(), catalog_.root());
249 if (!binding) throw std::logic_error("runtime pair has not been sealed");
250 if (pair->canonical_mapped() && pair->mapped() == binding->mapped) return pair;
251 return pair->mapped_owners_.get_or_create(catalog_.identity(), catalog_.root(), [&] {
252 auto result = node_type::from_mapped_parts(binding->mapped, mapped_native(pair->native_owner()),
253 pair->main_target() ? mapped_pair(pair->main_target()) : nullptr,
254 pair->secondary_target() ? mapped_native(pair->secondary_target()) : nullptr);
255 result->bindings_.get_or_create(catalog_.identity(), catalog_.root(), [&] { return binding; });
256 return result;
257 });

279 void finish_ready(ready_unit & unit, object_attempt_id const & attempt) {
280 std::optional<object_seal_receipt> native_receipt;
281 if (!unit.native) native_receipt.emplace(encode_native(unit.owner).seal(catalog_.root(), *unit.native_id, attempt));
282 if (!unit.pair) {
283 catalog_.record_sealed(operation(), *native_receipt);
284 auto mapped = std::make_shared<mapped_native_type const>(mapped_native_type::open(native_receipt->path));
285 auto binding = std::make_shared<native_binding_type const>(
286 native_seal{catalog_.identity(), std::move(*native_receipt)}, std::move(mapped));
287 unit.native_claim->install(std::move(binding));
288 return;
289 }

291 auto encoded = encode_cola_sections(*unit.pair->built(), id.native,
292 unit.main ? std::optional<blob_identity>(unit.main->identity) : std::nullopt,
293 unit.secondary ? std::optional<object_id>(unit.secondary->receipt.object) : std::nullopt);
294 auto receipt = encoded.seal(catalog_.root(), id.index, attempt);
295 std::shared_ptr<typename mapped_type::index_type const> index;

297 auto [mapped_native, prepared_index] = catalog_.template seal_native_pair<mapped_type>(
298 operation(), id, *native_receipt, receipt);

302 index = std::move(prepared_index);
303 } else index = catalog_.template seal_pair<mapped_type>(operation(), id, receipt);
304 auto mapped = mapped_type::bind(id, unit.native->mapped, std::move(index),

307 auto binding = std::make_shared<pair_binding_type const>(
308 pair_seal{catalog_.identity(), id, std::move(receipt)}, std::move(mapped));
309 unit.pair_claim->install(std::move(binding));

324 std::shared_ptr<native_binding_type const> produce_native(native_pointer const & native) {
325 if constexpr (requires { native->sealed(); }) {
326 if (auto sealed = native->sealed()) {
327 if (sealed->catalog != catalog_.identity())
328 throw std::invalid_argument("native seal belongs to another catalog");
329 catalog_.verify_sealed(sealed->receipt, file_kind::native_blob);
330 return std::make_shared<native_binding_type const>(*sealed, native->mapped());
331 }

339 return std::make_shared<native_binding_type const>(
