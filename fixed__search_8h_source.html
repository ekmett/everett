44 inline std::uint32_t load_word(std::byte const *source) noexcept {
45 std::uint32_t value;
46 std::memcpy(&value, source, sizeof(value));
47 if constexpr (std::endian::native == std::endian::big) {
48 value = ((value & 0x00ff00ffu) << 8) | ((value >> 8) & 0x00ff00ffu);
49 value = (value << 16) | (value >> 16);
50 }
51 return value;
52 }

61 template <std::size_t Words, bool Upper>
62 inline bool precedes(std::byte const *source, std::array<std::uint32_t, Words> const &query) noexcept {
63 bool less = false, equal = true;
64 for (std::size_t i = 0; i < Words; ++i) {
65 auto value = load_word(source + (i << 2));
66 less |= equal & (value < query[i]);
67 equal &= value == query[i];
68 }
69 if constexpr (Upper) return less | equal;
70 else return less;
71 }

74 inline uint32x4_t load_partial(std::byte const *source, unsigned words) noexcept {
75 if (words >= 4) return vreinterpretq_u32_u8(vld1q_u8(reinterpret_cast<std::uint8_t const *>(source)));
76 auto result = vdupq_n_u32(0);
77 // Literal scalar loads keep partial accesses bounded without a dynamic
78 // memcpy call or a full vector read beyond the supplied byte span.
79 switch (words) {
80 case 3: result = vsetq_lane_u32(load_word(source + 8), result, 2); [[fallthrough]];
81 case 2: result = vsetq_lane_u32(load_word(source + 4), result, 1); [[fallthrough]];
82 case 1: result = vsetq_lane_u32(load_word(source), result, 0); [[fallthrough]];
83 default: return result;
84 }

87 template <bool Upper>
88 EVERETT_FIXED_SEARCH_INLINE bool precedes_four(std::byte const *source, uint32x4_t query) noexcept {
89 auto value = vreinterpretq_u32_u8(vld1q_u8(reinterpret_cast<std::uint8_t const *>(source)));
90 // A more significant unequal word outweighs every following word.
91 // Each lane contributes -1, 0, or 1, with weights 8, 4, 2, 1.
92 auto order = vreinterpretq_s32_u32(vsubq_u32(vcgtq_u32(value, query), vcltq_u32(value, query)));
93 int32x4_t shifts{3, 2, 1, 0};
94 auto comparison = vaddvq_s32(vshlq_s32(order, shifts));
95 if constexpr (Upper) return comparison >= 0;
96 else return comparison > 0;
97 }

100 // Count keys satisfying < query (or <= query), not individual word lanes.
101 // The first word of each key is the most significant comparison word.
102 // Callers supply at most four complete keys; no padding is required.
103 template <std::size_t Words, bool Upper>
104 EVERETT_FIXED_SEARCH_INLINE unsigned population(std::byte const *source, unsigned count,
105 std::array<std::uint32_t, Words> const &query) noexcept {
106 if (!count) return 0;
107 unsigned less = 0, equal = 0;
108 unsigned word_count = count * unsigned(Words);
109#if defined(__AVX512F__)
110 auto q = repeated<Words, 16>(query);
111 auto active = __mmask16((std::uint32_t{1} << word_count) - 1);
112 // Masked-off lanes do not access memory, including across a guard page.
113 auto values = _mm512_maskz_loadu_epi32(active, source);
114 auto queries = _mm512_loadu_si512(q.data());
115 less = unsigned(_mm512_cmplt_epu32_mask(values, queries));
116 equal = unsigned(_mm512_cmpeq_epu32_mask(values, queries));
117#elif defined(__AVX2__)
118 auto q = repeated<Words, 8>(query);
119 auto queries = _mm256_loadu_si256(reinterpret_cast<__m256i const *>(q.data()));
120 auto sign = _mm256_set1_epi32(std::numeric_limits<std::int32_t>::min());
121 auto positions = _mm256_setr_epi32(0, 1, 2, 3, 4, 5, 6, 7);
122 for (unsigned at = 0; at < Words * 4; at += 8) {
123 if (at >= word_count) break;
124 auto active = _mm256_cmpgt_epi32(_mm256_set1_epi32(int(word_count - at)), positions);
125 // VPMASKMOVD suppresses memory access for inactive dword lanes.
126 auto values = _mm256_maskload_epi32(reinterpret_cast<int const *>(source + (at << 2)), active);
127 auto lt = _mm256_cmpgt_epi32(_mm256_xor_si256(queries, sign), _mm256_xor_si256(values, sign));
128 auto eq = _mm256_cmpeq_epi32(values, queries);
129 less |= unsigned(_mm256_movemask_ps(_mm256_castsi256_ps(lt))) << at;
130 equal |= unsigned(_mm256_movemask_ps(_mm256_castsi256_ps(eq))) << at;
131 }
132#elif defined(__aarch64__) && defined(__ARM_NEON) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
133 if constexpr (Words == 1) {
134 auto values = load_partial(source, count);
135 auto queries = vdupq_n_u32(query[0]);
136 uint32x4_t positions{0, 1, 2, 3};
137 auto active = vcltq_u32(positions, vdupq_n_u32(count));
138 auto matches = Upper ? vcleq_u32(values, queries) : vcltq_u32(values, queries);
139 return vaddvq_u32(vshrq_n_u32(vandq_u32(matches, active), 31));
140 } else if constexpr (Words == 2) {
141 auto queries = vdupq_n_u64((std::uint64_t(query[0]) << 32) | query[1]);
142 unsigned result = 0;
143 for (unsigned at = 0; at < 4; at += 2) {
144 if (at >= count) break;
145 if (count - at == 1) {
146 result += precedes<Words, Upper>(source + (at << 3), query);
147 } else {
148 auto words = vreinterpretq_u32_u8(vld1q_u8(reinterpret_cast<std::uint8_t const *>(source + (at << 3))));
149 auto values = vreinterpretq_u64_u32(vrev64q_u32(words));
150 auto matches = Upper ? vcleq_u64(values, queries) : vcltq_u64(values, queries);
151 result += unsigned(vaddvq_u64(vshrq_n_u64(matches, 63)));
152 }
153 }
154 return result;
155 } else {
156 auto queries = vld1q_u32(query.data());
157 unsigned result = 0;
158 for (unsigned at = 0; at < 4; ++at) {
159 if (at >= count) break;
160 result += precedes_four<Upper>(source + (at << 4), queries);
161 }
162 return result;
163 }
164#elif defined(__SSE2__)
165 auto q = repeated<Words, 4>(query);
166 auto queries = _mm_loadu_si128(reinterpret_cast<__m128i const *>(q.data()));
167 auto sign = _mm_set1_epi32(std::numeric_limits<std::int32_t>::min());
168 for (unsigned at = 0; at < Words * 4; at += 4) {
169 if (at >= word_count) break;
170 std::array<std::uint32_t, 4> buffer{};
171 std::memcpy(buffer.data(), source + (at << 2), std::min(4u, word_count - at) << 2);
172 auto values = _mm_loadu_si128(reinterpret_cast<__m128i const *>(buffer.data()));
173 auto lt = _mm_cmpgt_epi32(_mm_xor_si128(queries, sign), _mm_xor_si128(values, sign));
174 auto eq = _mm_cmpeq_epi32(values, queries);
175 less |= unsigned(_mm_movemask_ps(_mm_castsi128_ps(lt))) << at;
176 equal |= unsigned(_mm_movemask_ps(_mm_castsi128_ps(eq))) << at;
177 }
178#else
179 unsigned scalar_count = 0;
180 for (unsigned i = 0; i < count; ++i) scalar_count += precedes<Words, Upper>(source + i * Words * 4, query);
181 return scalar_count;
182#endif
183 constexpr unsigned starts = Words == 1 ? 0xffffu : Words == 2 ? 0x5555u : 0x1111u;
184 unsigned prefix = starts & ((1u << word_count) - 1);
185 unsigned matches = prefix, result = 0;
186 for (unsigned word = 0; word < Words; ++word) {
187 result |= matches & (less >> word);
188 matches &= equal >> word;
189 }
190 if constexpr (Upper) result |= matches;
191 return unsigned(std::popcount(result));
192 }
193 }

195 // Borrows sorted keys encoded as one, two, or four little-endian uint32 words.
196 // Word zero is compared first; this is not a native-endian uint128 array.
197 // The owner must outlive the view. Construction checks extent, not sortedness.
198 // Bounds use local ordinals, permit duplicate keys and never allocate.
199 template <std::size_t Words> struct fixed_key_view {
200 static_assert(Words == 1 || Words == 2 || Words == 4, "fixed keys have 1, 2, or 4 words");
201 using key_type = std::array<std::uint32_t, Words>;
202 static constexpr std::size_t key_bytes = Words * 4;
203 static constexpr std::size_t explicit_simd_limit = 32;
204 fixed_key_view() = default;
205 explicit fixed_key_view(std::span<std::byte const> bytes) : bytes_(bytes) {
206 if (bytes.size() % key_bytes)
207 error_detail::raise<std::invalid_argument>("fixed-key extent is not a whole number of keys");
208 }
209 std::size_t size() const noexcept { return bytes_.size() / key_bytes; }
210 bool empty() const noexcept { return bytes_.empty(); }
211 std::span<std::byte const> bytes() const noexcept { return bytes_; }
212 key_type key_at(std::size_t ordinal) const {
213 if (ordinal >= size()) error_detail::raise<std::out_of_range>("fixed-key ordinal");
214 key_type result;
215 auto source = bytes_.data() + ordinal * key_bytes;
216 for (std::size_t i = 0; i < Words; ++i) result[i] = fixed_search_detail::load_word(source + (i << 2));
217 return result;
218 }
219 fixed_key_view subview(std::size_t first, std::size_t count) const {
220 if (first > size() || count > size() - first)
221 error_detail::raise<std::out_of_range>("fixed-key subview");
222 return fixed_key_view(bytes_.subspan(first * key_bytes, count * key_bytes));
223 }
224 std::size_t lower_bound(key_type const &query) const noexcept {
225 return prefer_simd() ? simd_bound<false>(query) : binary_bound<false>(query);
226 }
227 std::size_t upper_bound(key_type const &query) const noexcept {
228 return prefer_simd() ? simd_bound<true>(query) : binary_bound<true>(query);
229 }
230 std::size_t lower_bound_binary(key_type const &query) const noexcept { return binary_bound<false>(query); }
231 std::size_t upper_bound_binary(key_type const &query) const noexcept { return binary_bound<true>(query); }
232 std::size_t lower_bound_simd(key_type const &query) const noexcept { return simd_bound<false>(query); }
233 std::size_t upper_bound_simd(key_type const &query) const noexcept { return simd_bound<true>(query); }
234 private:
235 std::span<std::byte const> bytes_;
236 bool prefer_simd() const noexcept {
237 // Measured lower-bound choices; explicit methods remain available for
238 // other hosts/workloads. Complete 2^m-1 scalar windows are especially cheap.
239#if defined(__aarch64__) && defined(__ARM_NEON) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
240 if constexpr (Words == 4)
241 return (size() >= 2 && size() <= 4) || size() == 15 || size() == 16 || size() == 32;
242 else return size() == 4;
243#elif defined(__AVX2__) && !defined(__AVX512F__)
244 if constexpr (Words == 1) return size() == 4;
245 else if constexpr (Words == 2) return size() >= 2 && size() <= 4;
246 else return size() == 2 || size() == 4;
247#else
248 return false;
249#endif
250 }
251 template <bool Upper> std::size_t binary_bound(key_type const &query) const noexcept {
252 auto count = size();
253 std::size_t first = 0;
254 auto step = std::bit_floor(count);
255 if (std::has_single_bit(count + 1)) {
256 // Exactly log2(count+1) comparisons for a complete 2^m-1 window.
257 for (; step; step >>= 1)
258 first += step * fixed_search_detail::precedes<Words, Upper>(
259 bytes_.data() + (first + step - 1) * key_bytes, query);
260 } else {
261 for (; step; step >>= 1) {
262 auto next = first + step;
263 auto probe = std::min(next, count) - 1;
264 auto below = fixed_search_detail::precedes<Words, Upper>(bytes_.data() + probe * key_bytes, query);
265 first += step * ((next <= count) & below);
266 }
267 }
268 return first;
269 }
270 template <bool Upper> EVERETT_FIXED_SEARCH_INLINE void pivot_step(key_type const &query, std::size_t &first, std::size_t &count) const noexcept {
271 // Called only with at least three keys, giving three distinct pivots.
272 std::array<std::size_t, 3> positions{count >> 2, count >> 1, (3 * count) >> 2};
273 unsigned rank;
274#if defined(__aarch64__) && defined(__ARM_NEON) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
275 if constexpr (Words == 4) {
276 auto queries = vld1q_u32(query.data());
277 rank = 0;
278 for (unsigned i = 0; i < 3; ++i)
279 rank += fixed_search_detail::precedes_four<Upper>(bytes_.data() + (first + positions[i]) * key_bytes, queries);
280 } else
281#endif
282 {
283 std::array<std::byte, 3 * key_bytes> pivots;
284 for (unsigned i = 0; i < 3; ++i)
285 std::memcpy(pivots.data() + i * key_bytes, bytes_.data() + (first + positions[i]) * key_bytes, key_bytes);
286 rank = fixed_search_detail::population<Words, Upper>(pivots.data(), 3, query);
287 }
288 auto start = ((rank * count) >> 2) + (rank != 0);
289 auto end = ((rank + 1) * count) >> 2;
290 first += start;
291 count = end - start;
292 }
293 template <bool Upper, unsigned Bucket> EVERETT_FIXED_SEARCH_INLINE std::size_t small_bound(key_type const &query) const noexcept {
294 std::size_t first = 0, count = size();
295 if constexpr (Bucket > 4) pivot_step<Upper>(query, first, count);
296 // Original sizes 17..32 leave 3..8 keys after the first pivot step.
297 if constexpr (Bucket > 16) pivot_step<Upper>(query, first, count);
298 if (!count) return first;
299 return first + fixed_search_detail::population<Words, Upper>(bytes_.data() + first * key_bytes, unsigned(count), query);
300 }
301 template <bool Upper> EVERETT_FIXED_SEARCH_INLINE std::size_t simd_bound(key_type const &query) const noexcept {
302 if (size() <= 4) return small_bound<Upper, 4>(query);
303 if (size() <= 8) return small_bound<Upper, 8>(query);
304 if (size() <= 16) return small_bound<Upper, 16>(query);
305 if (size() <= explicit_simd_limit) return small_bound<Upper, 32>(query);
306 return binary_bound<Upper>(query);
307 }
