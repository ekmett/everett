25 // Persistent owner of an exact collection of objects and their additive
26 // contributions. A contribution is not necessarily the hash of the object's
27 // own native records: an update that deletes a key contributes its negative
28 // old binding even when its sole native record is a zero-hash tombstone.

86 // Replace a physical composition while preserving its logical contribution.
87 // All expected identities must still exist, exactly once. A new immutable
88 // owner is returned only after validating the complete replacement; old
89 // readers retain the original objects and metadata without synchronization.
90 pin_set replace(std::span<std::string const> expected_inputs,
91 std::vector<entry> replacements) const {
92 std::vector<std::string_view> inputs;
93 inputs.reserve(expected_inputs.size());
94 auto removed = A::zero();
95 for (auto const & object_id : expected_inputs) {
96 if (std::find(inputs.begin(), inputs.end(), object_id) != inputs.end())
97 throw std::invalid_argument("duplicate expected pinned input");
98 auto old = find(object_id);
99 if (!old) throw std::invalid_argument("expected pinned input is absent");
100 inputs.push_back(object_id);
101 removed = A::add(removed, old->contribution);
102 }
103
104 auto is_input = [&](std::string_view object_id) {
105 return std::find(inputs.begin(), inputs.end(), object_id) != inputs.end();
106 };
107 auto added = A::zero();
108 for (std::size_t i = 0; i < replacements.size(); ++i) {
109 auto const & item = replacements[i];
110 validate(item);
111 if (find(item.object_id))
112 throw std::invalid_argument("replacement must name a new immutable object");
113 for (std::size_t j = 0; j < i; ++j)
114 if (replacements[j].object_id == item.object_id)
115 throw std::invalid_argument("duplicate replacement object identity");
116 added = A::add(added, item.contribution);
117 }
118 if (added != removed)
119 throw std::invalid_argument("replacement changes the pinned contribution sum");
120
121 auto next = std::make_shared<state>();
122 next->signature = A::add(A::subtract(state_->signature, removed), added);
123 // Keep unrelated entries in their original order. The owner's sum is
124 // order independent; semantic recency/order belongs to its consumer.
125 for (auto const & item : state_->entries) {
126 if (is_input(item.object_id)) continue;
127 next->entries.push_back(item);
128 next->pins.push_back(item.object);
129 }
130 for (auto & item : replacements) {
131 next->pins.push_back(item.object);
132 next->entries.push_back(std::move(item));
133 }
134 return pin_set(std::move(next));
135 }
