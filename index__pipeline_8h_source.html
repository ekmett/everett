28 // Builds owning indexes through the same bounded sample queues as file output.
29 // Native stages are ordered nearest the target first, new head last. Finalizing
30 // builds compact navigation and binds each completed pair to its exact target.

37 index_pipeline(pair_type target, std::vector<pair_type> native_stages)
38 : base(std::move(target), make_stages(native_stages)), results_(this->stages_.size()) {}

40 pair_type finish() {
41 if (this->failed_) error_detail::raise<std::logic_error>("index pipeline has failed");
42 if (!this->done()) error_detail::raise<std::logic_error>("index pipeline still has input");
43 if (!this->finished_) {
44 try {
45 auto target = this->target_;
46 for (std::size_t i = 0; i != this->stages_.size(); ++i) {
47 results_[i] = std::make_shared<blob_type const>(this->stages_[i]->finish(target));
48 target = results_[i];
49 }
50 this->finished_ = true;
51 } catch (...) {
52 this->failed_ = true;
53 throw;
54 }
55 }
56 return results_.empty() ? this->target_ : results_.back();
57 }

63 static std::vector<std::unique_ptr<index_builder<P>>> make_stages(std::span<pair_type const> sources) {
64 std::vector<std::unique_ptr<index_builder<P>>> result;
65 result.reserve(sources.size());
66 for (auto const & source : sources) {
67 if (!source) error_detail::raise<std::invalid_argument>("null pipeline native source");
68 result.push_back(std::make_unique<index_builder<P>>(*source));
69 }
70 return result;
71 }
