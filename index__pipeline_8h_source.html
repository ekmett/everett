25 // Builds owning indexes through the same bounded sample queues as file output.
26 // Native stages are ordered nearest the target first, new head last. Finalizing
27 // builds compact navigation and binds each completed pair to its exact target.

34 index_pipeline(pair_type target, std::vector<pair_type> native_stages)

37 pair_type finish() {
38 if (this->failed_) error_detail::raise<std::logic_error>("index pipeline has failed");
39 if (!this->done()) error_detail::raise<std::logic_error>("index pipeline still has input");
40 if (!this->finished_) {
41 try {
42 auto target = this->target_;
43 for (std::size_t i = 0; i != this->stages_.size(); ++i) {
44 results_[i] = std::make_shared<blob_type const>(this->stages_[i]->finish(target));
45 target = results_[i];
46 }
47 this->finished_ = true;
48 } catch (...) {
49 this->failed_ = true;
50 throw;
51 }
52 }
53 return results_.empty() ? this->target_ : results_.back();
54 }

60 static std::vector<std::unique_ptr<index_builder<P>>> make_stages(std::span<pair_type const> sources) {
61 std::vector<std::unique_ptr<index_builder<P>>> result;
62 result.reserve(sources.size());
63 for (auto const & source : sources) {
64 if (!source) error_detail::raise<std::invalid_argument>("null pipeline native source");
65 result.push_back(std::make_unique<index_builder<P>>(*source));
66 }
67 return result;
68 }
