23 // Build indexes from the target outward. Each stage retains one input and
24 // one output sample; sample keys flow directly into the next stage without
25 // materializing an intermediate catalog or rescanning a completed index.
26 // Handoffs carry a policy-unit backspace count and suffix relative to the
27 // preceding sample from that same producer, with a literal first sample.
28 //
29 // A step quantum consumes one augmented occurrence, transfers one sample,
30 // advances the source sampler by at most K occurrences, or communicates EOF.
31 // This is an entry-work budget, not a bound on bytes or wall-clock time.
32 // finish() separately builds the offset/rank directories and binds each
33 // completed pair to its exact target. No durable publication is performed.
34 template <class P>
35 struct index_pipeline {
36 using policy_type = P;
37 using blob_type = profile_blob<P>;
38 using pair_type = std::shared_ptr<blob_type const>;
39
40 // Native stages are ordered nearest the target first, new head last.
41 index_pipeline(pair_type target, std::vector<pair_type> native_stages)
42 : target_(std::move(target)), source_(target_) {
43 stages_.reserve(native_stages.size());
44 for (auto const & source : native_stages) {
45 if (!source) throw std::invalid_argument("null pipeline native source");
46 stages_.push_back(std::make_unique<index_builder<P>>(*source));
47 }
48 results_.resize(stages_.size());
49 emitted_.resize(stages_.size());
50 emitted_suffix_units_.resize(stages_.size());
51 consumed_.resize(stages_.size());
52 }

66 std::uint64_t step(std::uint64_t quanta) {
67 if (finished_ || failed_) throw std::logic_error("pipeline is no longer active");
68 std::uint64_t work = 0;
69 try {
70 while (work != quanta && !done()) {
71 if (!advance_one()) throw std::logic_error("index pipeline made no progress");
72 ++work;
73 }
74 } catch (...) {
75 failed_ = true;
76 throw;
77 }
78 return work;
79 }

81 pair_type finish() {
82 if (failed_) throw std::logic_error("index pipeline has failed");
83 if (!done()) throw std::logic_error("index pipeline still has input");
84 if (!finished_) {
85 try {
86 auto target = target_;
87 for (std::size_t i = 0; i != stages_.size(); ++i) {
88 results_[i] = std::make_shared<blob_type const>(stages_[i]->finish(target));
89 target = results_[i];
90 }
91 finished_ = true;
92 } catch (...) {
93 failed_ = true;
94 throw;
95 }
96 }
97 return results_.empty() ? target_ : results_.back();
98 }

123 bool advance_one() {
124 // Give consumers the first opportunity to release backpressure.
125 for (std::size_t end = stages_.size(); end; --end) {
126 auto i = end - 1;
127 auto & stage = *stages_[i];
128 if (stage.has_output()) {
129 if (i + 1 == stages_.size()) {
130 auto sample = stage.take_coded_output();
131 emitted_suffix_units_[i] = profile_detail::add(
132 emitted_suffix_units_[i], sample.suffix.bit_size / P::bits_per_unit);
133 } else {
134 auto & next = *stages_[i + 1];
135 if (!next.needs_input()) continue;
136 auto sample = stage.take_coded_output();
137 next.push(sample);
138 emitted_suffix_units_[i] = profile_detail::add(
139 emitted_suffix_units_[i], sample.suffix.bit_size / P::bits_per_unit);
140 }
141 ++emitted_[i];
142 return true;
143 }
144 if (stage.done()) continue;
145 if (stage.needs_input()) {
146 if (!i) {
147 if (source_.done()) {
148 stage.close_input();
149 } else {
150 auto sample = source_.peek();
151 auto coded = source_encoder_.encode(sample.key, sample.target_ordinal);
152 stage.push(coded);
153 source_suffix_units_ = profile_detail::add(
154 source_suffix_units_, coded.suffix.bit_size / P::bits_per_unit);
155 source_.advance();
156 }
157 return true;
158 }
159 if (stages_[i - 1]->done()) {
160 stage.close_input();
161 return true;
162 }
163 continue;
164 }
165 auto consumed = stage.step(1);
166 if (consumed) {
167 consumed_[i] += consumed;
168 return true;
169 }
170 }
171 return false;
172 }
