169 // Insert-only durable metadata over an existing trusted local directory.
170 // Separate connections may contend; a single handle must not be used by
171 // concurrent callers. No method deletes an object, owner, operation or save.
172 // An uncertain storage/COMMIT outcome permanently disables this handle.

186 static sqlite_catalog create(std::filesystem::path const & root, object_id const & identity,
187 catalog_options options = {}, Ops ops = {}) {
188 validate_options(options);
189 catalog_detail::bytes descriptor; catalog_detail::identity(descriptor, identity);
190 auto location = std::filesystem::canonical(root);
191#if defined(__APPLE__) || defined(__linux__)
192 // Reserve this catalog name without following a pre-existing symlink.
193 auto path = location / "catalog.sqlite3";
194 int fd = ::open(path.c_str(), O_CREAT | O_EXCL | O_RDWR | O_CLOEXEC | O_NOFOLLOW, 0600);
195 if (fd < 0) throw std::system_error(errno, std::generic_category(), "create Everett catalog");
196 if (::close(fd)) throw std::system_error(errno, std::generic_category(), "close new Everett catalog");
197 auto result = connect(location, options, std::move(ops));
198 result.transaction("", "initialize", {}, [&] {
199 catalog_detail::exec(result.db_, catalog_detail::schema);
200 catalog_detail::statement insert(result.db_, "INSERT INTO catalog_info VALUES(1,1,?,?)");
201 insert.text(1, identity.hex()); insert.blob(2, policy()); insert.done();
202 // Immutable tables remain readable through ordinary SQL tooling.
203 for (auto table : {"catalog_info", "operations", "owners", "attempts", "pairs", "owner_objects", "owner_roots", "saves"}) {
204 for (auto action : {"UPDATE", "DELETE"}) {
205 std::string sql = "CREATE TRIGGER immutable_" + std::string(table) + "_" + action +
206 " BEFORE " + action + " ON " + table + " BEGIN SELECT RAISE(ABORT,'immutable catalog row'); END";
207 catalog_detail::exec(result.db_, sql.c_str());
208 }
209 }
210 catalog_detail::exec(result.db_, "CREATE TRIGGER objects_no_delete BEFORE DELETE ON objects BEGIN SELECT RAISE(ABORT,'retained object'); END");
211 return catalog_detail::bytes{};
212 }, false);
213 // SQLite's transaction is not a substitute for retaining this new name.
214 posix_object_ops barriers;
215 int directory = barriers.open_root(location);
216 if (directory < 0) throw std::system_error(errno, std::generic_category(), "open catalog directory");
217 if (barriers.sync_directory(directory)) {
218 int error = errno; barriers.close(directory);
219 throw std::system_error(error, std::generic_category(), "sync catalog directory; initialization outcome unknown");
220 }
221 if (barriers.close(directory))
222 throw std::system_error(errno, std::generic_category(), "close catalog directory; initialization outcome unknown");
223 return result;
224#else
225 (void)location; (void)options; (void)ops;
226 throw std::system_error(std::make_error_code(std::errc::operation_not_supported), "Everett catalog creation requires POSIX directory barriers");
227#endif
228 }

264 void reserve(std::string_view op, object_attempt_id const & attempt, std::string_view owner,
265 std::span<blob_identity const> inputs, std::span<catalog_object_reservation const> outputs) {
266 require_active(); catalog_detail::name(op); catalog_detail::name(owner);
267 if (outputs.empty()) throw std::invalid_argument("reservation has no outputs");
268 catalog_detail::bytes request;
269 catalog_detail::identity(request, object_id(attempt.hex())); catalog_detail::field(request, owner);
270 catalog_detail::number(request, inputs.size());
271 for (auto const & input : inputs) catalog_detail::pair(request, input);
272 catalog_detail::number(request, outputs.size());
273 for (auto const & output : outputs) {
274 catalog_detail::identity(request, output.object); (void)file_extension(output.kind);
275 catalog_detail::number(request, kind(output.kind));
276 }
277 transaction(op, "reserve", request, [&] {
278 add_owner("attempt", owner);
279 catalog_detail::statement job(db_, "INSERT INTO attempts VALUES(?,?)");
280 job.text(1, attempt.hex()); job.key(2, owner); job.done();
281 for (auto const & input : inputs) add_root("attempt", owner, input);
282 for (auto const & output : outputs) {
283 catalog_detail::statement insert(db_, "INSERT INTO objects(id,kind,attempt) VALUES(?,?,?)");
284 insert.text(1, output.object.hex()); insert.integer(2, kind(output.kind)); insert.text(3, attempt.hex()); insert.done();
285 catalog_detail::statement pin(db_, "INSERT INTO owner_objects VALUES('attempt',?,?)");
286 pin.key(1, owner); pin.text(2, output.object.hex()); pin.done();
287 }
288 return catalog_detail::bytes{};
289 });
290 }

292 void record_sealed(std::string_view op, object_seal_receipt const & receipt) {
293 require_active(); catalog_detail::name(op);
294 catalog_detail::bytes request;
295 catalog_detail::identity(request, receipt.object);
296 catalog_detail::identity(request, object_id(receipt.attempt.hex()));
297 catalog_detail::number(request, receipt.bytes); catalog_detail::number(request, receipt.body_crc32c);
298 auto barrier = static_cast<unsigned>(receipt.barrier);
299 if (barrier > 1) throw std::invalid_argument("unsupported seal barrier");
300 catalog_detail::number(request, barrier);
301 auto path = std::filesystem::canonical(receipt.path);
302 // Store a relative path in replay descriptors so relocating the complete
303 // backing directory does not turn the same receipt into a new operation.
304 catalog_detail::field(request, path.lexically_relative(root_).generic_string());
305 transaction(op, "seal", request, [&] {
306 catalog_detail::statement reserved(db_, "SELECT kind,attempt,bytes FROM objects WHERE id=?");
307 reserved.text(1, receipt.object.hex());
308 if (!reserved.row() || reserved.text(1) != receipt.attempt.hex() || !reserved.is_null(2))
309 throw std::invalid_argument("object is not this attempt's unsealed reservation");
310 auto expected = reserved.integer(0) == 0 ? file_kind::native_blob : file_kind::fractional_index;
311 if (path != root_ / object_path(receipt.object, expected))
312 throw std::invalid_argument("seal receipt names another catalog path");
313 // Header-only evidence check. A caller's successful barrier receipt is
314 // an attestation; reading bytes cannot establish or repair durability.
315 auto mapping = mapped_file::open(path);
316 auto slice = mapping.slice(0, mapping.size());
317 auto bytes = slice.bytes();
318 auto header = decode_file_header<P>(bytes);
319 if (header.kind != expected || receipt.bytes != bytes.size() ||
320 file_detail::total_bytes<P>(header.extent) != bytes.size() ||
321 file_detail::get(bytes, 64, 4) != receipt.body_crc32c)
322 throw std::invalid_argument("seal receipt disagrees with object envelope");
323 catalog_detail::statement update(db_, "UPDATE objects SET bytes=?,crc=?,barrier=? WHERE id=?");
324 update.integer(1, catalog_detail::integer(receipt.bytes)); update.integer(2, receipt.body_crc32c);
325 update.integer(3, barrier); update.text(4, receipt.object.hex()); update.done();
326 return catalog_detail::bytes{};
327 });
328 }

330 // The supplied head names catalog paths, which are reopened and pinned for
331 // this admission. Normal admission checks metadata; scan explicitly checks
332 // readable contents and exact target samples before taking the writer lock.
333 void register_chain(std::string_view op, mapped_query_root<P> const & source,
334 catalog_admission admission = catalog_admission::trusted) {
335 require_active(); catalog_detail::name(op);
336 if (admission != catalog_admission::trusted && admission != catalog_admission::scan)
337 throw std::invalid_argument("unsupported catalog admission mode");
338 auto head = source.head();
339 if (!head) throw std::invalid_argument("empty catalog query root");
340 auto opened = open_mapped_query<P>(root_, head->identity());
341 if (admission == catalog_admission::scan) opened.head()->scan();
342 std::vector<typename mapped_blob<P>::pair_type> chain;
343 catalog_detail::bytes request;
344 catalog_detail::number(request, static_cast<unsigned>(admission));
345 for (auto current = opened.head(); current; current = current->target()) {
346 chain.push_back(current);
347 catalog_detail::pair(request, current->identity());
348 catalog_detail::number(request, current->native().size());
349 catalog_detail::number(request, current->borrowed().size());
350 catalog_detail::number(request, current->virtual_size());
351 }
352 transaction(op, "register_chain", request, [&] {
353 for (auto entry = chain.rbegin(); entry != chain.rend(); ++entry) {
354 auto const & pair = **entry;
355 require_sealed(pair.identity().native, file_kind::native_blob);
356 require_sealed(pair.identity().index, file_kind::fractional_index);
357 auto target = pair.target();
358 catalog_detail::statement old(db_, "SELECT native_id,target_native,target_index,native_count,borrowed_count,virtual_count FROM pairs WHERE index_id=?");
359 old.text(1, pair.identity().index.hex());
360 if (old.row()) {
361 if (old.text(0) != pair.identity().native.hex() || old.is_null(1) != !target ||
362 (target && (old.text(1) != target->identity().native.hex() || old.text(2) != target->identity().index.hex())) ||
363 old.integer(3) != catalog_detail::integer(pair.native().size()) ||
364 old.integer(4) != catalog_detail::integer(pair.borrowed().size()) ||
365 old.integer(5) != catalog_detail::integer(pair.virtual_size()))
366 throw std::invalid_argument("registered index identity already has different contents");
367 continue;
368 }
369 catalog_detail::statement insert(db_, "INSERT INTO pairs VALUES(?,?,?,?,?,?,?)");
370 insert.text(1, pair.identity().index.hex()); insert.text(2, pair.identity().native.hex());
371 if (target) { insert.text(3, target->identity().native.hex()); insert.text(4, target->identity().index.hex()); }
372 else { insert.null(3); insert.null(4); }
373 insert.integer(5, catalog_detail::integer(pair.native().size()));
374 insert.integer(6, catalog_detail::integer(pair.borrowed().size()));
375 insert.integer(7, catalog_detail::integer(pair.virtual_size())); insert.done();
376 }
377 catalog_detail::bytes result; catalog_detail::pair(result, opened.head()->identity()); return result;
378 });
379 }

395 catalog_saved_root acquire_save(std::string_view op, std::string_view name, std::string_view reader_owner) {
396 require_active(); catalog_detail::name(op); catalog_detail::name(name); catalog_detail::name(reader_owner);
397 catalog_detail::bytes request; catalog_detail::field(request, name); catalog_detail::field(request, reader_owner);
398 transaction(op, "acquire_save", request, [&] {

400 if (!head) throw std::invalid_argument("unknown Everett save");

441 static sqlite_catalog connect(std::filesystem::path const & root, catalog_options options, Ops ops) {
442 validate_options(options);
443 sqlite3 * db = nullptr;
444 auto path = (root / "catalog.sqlite3").string();
445 auto code = sqlite3_open_v2(path.c_str(), &db, SQLITE_OPEN_READWRITE | SQLITE_OPEN_FULLMUTEX | SQLITE_OPEN_EXRESCODE | SQLITE_OPEN_NOFOLLOW, nullptr);
446 sqlite_catalog result(db, root, std::move(ops));
447 if (code != SQLITE_OK) catalog_detail::fail(db, code);
448 if (!sqlite3_db_mutex(db))
449 throw std::runtime_error("Everett requires a serialized SQLite connection");
450 sqlite3_extended_result_codes(db, 1);
451 code = sqlite3_busy_timeout(db, options.busy_timeout_ms);
452 if (code != SQLITE_OK) catalog_detail::fail(db, code);
453 int setting = 0;
454 code = sqlite3_db_config(db, SQLITE_DBCONFIG_DEFENSIVE, 1, &setting);
455 if (code != SQLITE_OK || setting != 1) catalog_detail::fail(db, code == SQLITE_OK ? SQLITE_ERROR : code);
456 catalog_detail::exec(db, "PRAGMA trusted_schema=OFF; PRAGMA foreign_keys=ON; PRAGMA synchronous=FULL; PRAGMA fullfsync=ON; PRAGMA checkpoint_fullfsync=ON;");
457 catalog_detail::statement journal(db, "PRAGMA journal_mode=WAL");
458 if (!journal.row() || journal.text(0) != "wal") throw std::runtime_error("Everett requires SQLite WAL mode");
459 for (auto sql : {"PRAGMA foreign_keys", "PRAGMA synchronous", "PRAGMA fullfsync", "PRAGMA checkpoint_fullfsync", "PRAGMA trusted_schema"}) {
460 catalog_detail::statement check(db, sql);
461 auto expected = std::string_view(sql) == "PRAGMA synchronous" ? 2 :
462 std::string_view(sql) == "PRAGMA trusted_schema" ? 0 : 1;
463 if (!check.row() || check.integer(0) != expected)
464 throw std::runtime_error("Everett SQLite persistence setting rejected");
465 }
466 return result;
467 }

474 void validate_schema() const {
475 // Validate only a bounded schema description, never external payloads or
476 // all catalog rows. SQLite preserves these canonical CREATE definitions.
477 std::string_view source = catalog_detail::schema;
478 while (true) {
479 auto begin = source.find("CREATE TABLE ");
480 if (begin == std::string_view::npos) break;
481 source.remove_prefix(begin);
482 auto end = source.find(';');
483 auto name_end = source.find('(');
484 definition("table", source.substr(13, name_end - 13), source.substr(0, end));
485 source.remove_prefix(end + 1);
486 }
487 auto first = source.find("CREATE TRIGGER sealed_immutable");
488 auto last = source.rfind("END;");
489 definition("trigger", "sealed_immutable", source.substr(first, last + 3 - first));
490 for (auto table : {"catalog_info", "operations", "owners", "attempts", "pairs", "owner_objects", "owner_roots", "saves"}) {
491 for (auto action : {"UPDATE", "DELETE"}) {
492 std::string name = "immutable_" + std::string(table) + "_" + action;
493 std::string sql = "CREATE TRIGGER " + name + " BEFORE " + action + " ON " + table +
494 " BEGIN SELECT RAISE(ABORT,'immutable catalog row'); END";
495 definition("trigger", name, sql);
496 }
497 }
498 definition("trigger", "objects_no_delete", "CREATE TRIGGER objects_no_delete BEFORE DELETE ON objects BEGIN SELECT RAISE(ABORT,'retained object'); END");
499 catalog_detail::statement triggers(db_, "SELECT count(*) FROM sqlite_schema WHERE type='trigger' AND tbl_name IN('catalog_info','operations','owners','attempts','objects','pairs','owner_objects','owner_roots','saves')");
500 if (!triggers.row() || triggers.integer(0) != 18)
501 throw std::invalid_argument("unexpected trigger on Everett catalog tables");
502 }

513 query.text(1, id.hex());
514 if (!query.row() || query.integer(0) != kind(expected) || query.is_null(1))

524 template <class F> catalog_detail::bytes transaction(std::string_view op, std::string_view kind,
525 catalog_detail::bytes const & request, F && apply, bool record = true) {
526 require_active();
527 if (record) catalog_detail::name(op);
528 bool began = false, committing = false;
529 try {
530 catalog_detail::exec(db_, "BEGIN IMMEDIATE"); began = true;
531 if (record) {
532 if (auto old = lookup_operation(op)) {
533 if (old->kind != kind || old->request != request)
534 throw std::invalid_argument("Everett operation identity reused with a different request");
535 catalog_detail::exec(db_, "ROLLBACK");
536 return std::move(old->outcome);
537 }
538 }
539 auto result = apply();
540 if (record) {
541 catalog_detail::statement insert(db_, "INSERT INTO operations VALUES(?,?,?,?)");
542 insert.key(1, op); insert.text(2, kind); insert.blob(3, request); insert.blob(4, result); insert.done();
543 }
544 committing = true;
545 auto code = ops_.commit(db_);
546 if (code != SQLITE_OK) catalog_detail::fail(db_, code);
547 if (!sqlite3_get_autocommit(db_)) catalog_detail::fail(db_, SQLITE_PROTOCOL);
548 return result;
549 } catch (catalog_error const & error) {
550 bool uncertain = committing || catalog_detail::storage_error(error.code);
551 if (began && !sqlite3_get_autocommit(db_) && sqlite3_exec(db_, "ROLLBACK", nullptr, nullptr, nullptr) != SQLITE_OK)
552 uncertain = true;
553 poisoned_ = poisoned_ || uncertain;
554 throw catalog_error(error.what(), error.code, std::string(op), uncertain);
555 } catch (...) {
556 if (committing) poisoned_ = true;
557 if (began && !sqlite3_get_autocommit(db_) && sqlite3_exec(db_, "ROLLBACK", nullptr, nullptr, nullptr) != SQLITE_OK)
558 poisoned_ = true;
559 throw;
560 }
561 }
562 };
