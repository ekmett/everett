568 // Compare the complete expected generation under the SQLite writer lock.
569 // A stale comparison is itself a committed, replay-stable outcome. Every
570 // successful publication appends a generation, even if the root is unchanged.

572 catalog_timeline_head const & expected, blob_identity const & candidate) {
573 require_timelines(); catalog_detail::name(op); validate_timeline(expected);
574 catalog_detail::bytes request; catalog_detail::timeline(request, expected); catalog_detail::pair(request, candidate);
575 auto outcome = transaction(op, "publish_timeline", request, [&] {
576 auto current = timeline_at(expected.name);
577 if (!current) throw std::invalid_argument("unknown Diet timeline");
578 bool published = *current == expected;
579 if (published) {
580 require_prepared(candidate);
581 if (expected.generation == std::uint64_t(std::numeric_limits<std::int64_t>::max()))
582 throw std::length_error("Diet timeline generation exhausted");
583 current = add_generation(expected.name, expected.generation + 1, candidate);
584 }
585 catalog_detail::bytes bytes; catalog_detail::number(bytes, published); catalog_detail::timeline(bytes, *current); return bytes;
586 });
587 return read([&] {
588 catalog_detail::outcome_reader reader{outcome};
589 auto published = reader.number();
590 if (published > 1) reader.invalid();
591 auto head = reader.timeline(); reader.end();
592 return catalog_timeline_publication{published != 0, std::move(head)};
593 });

818 template <class F> catalog_detail::bytes transaction(std::string_view op, std::string_view kind,
819 catalog_detail::bytes const & request, F && apply, bool record = true) {
820 require_active();

822 bool began = false, committing = false;
823 try {
824 catalog_detail::exec(db_, "BEGIN IMMEDIATE"); began = true;
825 if (record) {

827 if (old->kind != kind || old->request != request)

831 }
832 }
833 auto result = apply();
834 if (record) {
835 catalog_detail::statement insert(db_, "INSERT INTO operations VALUES(?,?,?,?)");
836 insert.key(1, op); insert.text(2, kind); insert.blob(3, request); insert.blob(4, result); insert.done();
837 }
838 committing = true;
839 auto code = ops_.commit(db_);
840 if (code != SQLITE_OK) catalog_detail::fail(db_, code);
841 if (!sqlite3_get_autocommit(db_)) catalog_detail::fail(db_, SQLITE_PROTOCOL);
842 return result;
843 } catch (catalog_error const & error) {

845 if (began && !sqlite3_get_autocommit(db_) && sqlite3_exec(db_, "ROLLBACK", nullptr, nullptr, nullptr) != SQLITE_OK)
846 uncertain = true;
847 poisoned_ = poisoned_ || uncertain;
848 throw catalog_error(error.what(), error.code, std::string(op), uncertain);
849 } catch (...) {
850 if (committing) poisoned_ = true;
851 if (began && !sqlite3_get_autocommit(db_) && sqlite3_exec(db_, "ROLLBACK", nullptr, nullptr, nullptr) != SQLITE_OK)
852 poisoned_ = true;

854 }
