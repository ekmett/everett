62 // Zero-based selection in a nonzero word, with ordinal < popcount(value).
63 // Byte-prefix populations fit in seven bits, so the marked subtraction
64 // finds the first byte whose cumulative population exceeds ordinal.
65 inline unsigned select_word(std::uint64_t value, unsigned ordinal) noexcept {
66#if defined(__BMI2__) && (defined(__x86_64__) || defined(_M_X64))
67 return unsigned(std::countr_zero(_pdep_u64(std::uint64_t{1} << ordinal, value)));
68#else
69 auto pairs = value - ((value >> 1) & 0x5555555555555555ull);
70 auto nibbles = (pairs & 0x3333333333333333ull) + ((pairs >> 2) & 0x3333333333333333ull);
71 auto bytes = (nibbles + (nibbles >> 4)) & 0x0f0f0f0f0f0f0f0full;
72 auto prefixes = bytes * 0x0101010101010101ull;
73 auto marked = ((prefixes | 0x8080808080808080ull) -
74 (ordinal + 1) * 0x0101010101010101ull) & 0x8080808080808080ull;
75 auto shift = unsigned(std::countr_zero(marked)) & ~7u;
76 if (shift) ordinal -= unsigned((prefixes >> (shift - 8)) & 255);
77 auto lower = unsigned((nibbles >> shift) & 15);
78 auto step = unsigned(ordinal >= lower);
79 shift += step * 4;
80 ordinal -= step * lower;
81 auto bits = unsigned(value >> shift) & 15;
82 lower = (bits & 1) + ((bits >> 1) & 1);
83 step = unsigned(ordinal >= lower);
84 shift += step * 2;
85 ordinal -= step * lower;
86 return shift + ordinal + unsigned(((value >> shift) & 1) == 0);
87#endif
88 }

112 // W-bit fields return to a word boundary after 64/gcd(W,64) values.
113 // Each tile assigns its output words once, rather than repeatedly loading
114 // and updating them for each input field. All shifts are compile-time
115 // constants below 64; the final incomplete tile uses bounded scalar work.
116 template <unsigned W>
117 inline void pack_low_fixed(std::span<std::uint64_t const> source,
118 std::span<std::uint64_t> out) noexcept {
119 static_assert(W <= 63);
120 if constexpr (W) {
121 constexpr auto divisor = std::gcd(W, 64u);
122 constexpr auto inputs = 64 / divisor;
123 constexpr auto outputs = W / divisor;
124 std::size_t at = 0;
125 for (; source.size() - at >= inputs; at += inputs) {
126 auto destination = out.data() + (at / inputs) * outputs;
127#if defined(__aarch64__) && defined(__ARM_NEON) && __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
128 // These complete tiles consume exactly four/eight input words and
129 // emit one word. Narrowing preserves each field's low bits.
130 if constexpr (W == 16) {
131 auto a = vmovn_u64(vld1q_u64(source.data() + at));
132 auto b = vmovn_u64(vld1q_u64(source.data() + at + 2));
133 *destination = vget_lane_u64(vreinterpret_u64_u16(vmovn_u32(vcombine_u32(a, b))), 0);
134 } else if constexpr (W == 8) {
135 auto a = vmovn_u64(vld1q_u64(source.data() + at));
136 auto b = vmovn_u64(vld1q_u64(source.data() + at + 2));
137 auto c = vmovn_u64(vld1q_u64(source.data() + at + 4));
138 auto d = vmovn_u64(vld1q_u64(source.data() + at + 6));
139 *destination = vget_lane_u64(vreinterpret_u64_u8(vmovn_u16(vcombine_u16(
140 vmovn_u32(vcombine_u32(a, b)), vmovn_u32(vcombine_u32(c, d))))), 0);
141 } else
142#endif
143 low_tile<W>(source.data() + at, destination, std::make_index_sequence<outputs>{});
144 }
145 auto tail = out.subspan((at / inputs) * outputs);
146 std::fill(tail.begin(), tail.end(), 0);
147 constexpr auto mask = (std::uint64_t{1} << W) - 1;
148 for (std::size_t i = 0; i != source.size() - at; ++i) {
149 auto bit = i * W;
150 unsigned shift = unsigned(bit % 64);
151 auto value = source[at + i] & mask;
152 tail[bit / 64] |= value << shift;
153 if (shift + W > 64) tail[bit / 64 + 1] |= value >> (64 - shift);
154 }
155 }
156 }

164 // This dispatch occurs once per complete low section, not per value. The
165 // destination may contain old data: both complete and tail words are set,
166 // including zero tail padding. Source and destination must not overlap.
167 inline void pack_low(std::span<std::uint64_t const> source,
168 std::span<std::uint64_t> out, unsigned width) {
169 if (width > 63) throw std::invalid_argument("select_groups low width");
170 if (out.size() != words(multiply(source.size(), width)))
171 throw std::invalid_argument("select_groups low output size");
172 static constexpr auto packers = low_packers(std::make_index_sequence<64>{});
173 packers[width](source, out);
174 }
175
176 // The owner has checked monotonicity, width and extent and zeroed out.
177 // (source[i] >> width) + i is strictly increasing, even for equal values.
178 // Accumulating a word locally removes repeated dependent memory updates;
179 // gaps remain zero. This preserves the exact existing high-bit ordering.
180 inline void write_high(std::span<std::uint64_t const> source,
181 std::span<std::uint64_t> out, unsigned width) noexcept {
182 std::uint64_t word = 0, value = 0;
183 for (std::uint64_t i = 0; i != source.size(); ++i) {
184 auto position = (source[i] >> width) + i;
185 auto next = position / 64;
186 if (next != word) {
187 out[word] = value;
188 word = next;
189 value = 0;
190 }
191 value |= std::uint64_t{1} << (position % 64);
192 }
193 if (!source.empty()) out[word] = value;
194 }

204 // Elias--Fano over residual integer addresses at records 0,K,2K,..., then an
205 // actual-record-count sentinel. Residuals omit the fixed stride per record.
206 // Addresses and fixed stride use the same units (for example bytes or bits).
207 // Equal residual offsets are valid, including an entirely zero sequence.
208 //
209 // Select on the high bitvector is bounded: sample every 256 ones; scan at
210 // most 4096 bits (65 aligned words) for a dense group, or read an explicit
211 // exception position for a sparse group. There is no binary search. The
212 // exception array is O(n): disjoint sparse groups each span >4096 high bits,
213 // and Elias--Fano's high vector has O(n) bits. This is a practical baseline,
214 // not a tuned succinct select implementation.
215 // Views check shapes and guard navigation bounds. They do not establish
216 // semantic consistency of borrowed metadata: use a builder or validated
217 // reader. Native spans and portable little-endian byte sections share navigation;
218 // neither constructor performs a complete semantic scan.
219 template <std::uint64_t K> struct select_groups_view {
220 static_assert(K >= 1 && K < std::numeric_limits<std::uint64_t>::max(),
221 "physical block size must be positive and below UINT64_MAX");
222 static constexpr std::uint64_t group_size = K;

260 std::uint64_t residual(std::uint64_t group) const {
261 if (group > group_count()) throw std::out_of_range("select_groups group");
262 auto position = select_high(group);
263 if (position < group) throw std::invalid_argument("invalid select_groups high value");
264 auto hi = position - group;
265 if (hi > (universe_ >> low_width_)) throw std::invalid_argument("select_groups high overflow");
266 std::uint64_t lo = 0;
267 if (low_width_) {
268 auto bit = group * low_width_;
269 auto word = bit / 64;
270 unsigned shift = unsigned(bit % 64);
271 lo = low_[word] >> shift;
272 if (shift + low_width_ > 64) lo |= low_[word + 1] << (64 - shift);
273 lo &= (std::uint64_t{1} << low_width_) - 1;
274 }
275 auto value = (hi << low_width_) | lo;
276 if (value > universe_) throw std::invalid_argument("select_groups value exceeds universe");
277 return value;
278 }

287 std::uint64_t select_high(std::uint64_t ordinal) const {
288 auto sample = samples_[ordinal / 256];
289 unsigned remaining = unsigned(ordinal % 256);
290 if (sample.sparse != std::numeric_limits<std::uint64_t>::max()) {
291 if (sample.sparse > sparse_.size() || remaining >= sparse_.size() - sample.sparse)
292 throw std::invalid_argument("invalid select_groups exception");
293 auto position = sparse_[sample.sparse + remaining];
294 if (position >= high_bits_ || !(high_[position / 64] & (std::uint64_t{1} << (position % 64))))
295 throw std::invalid_argument("invalid select_groups sparse position");
296 return position;
297 }
298 if (sample.first >= high_bits_) throw std::invalid_argument("invalid select_groups sample");
299 auto word = sample.first / 64;
300 auto value = high_[word] & (~std::uint64_t{0} << (sample.first % 64));
301 for (unsigned scanned = 0; scanned < 65 && word < high_.size(); ++scanned, ++word) {
302 if (scanned) value = high_[word];
303 auto population = unsigned(std::popcount(value));
304 if (remaining < population) {
305 auto position = word * 64 + select_groups_detail::select_word(value, remaining);
306 if (position >= high_bits_ || position - sample.first >= 4096)
307 throw std::invalid_argument("select_groups dense span");
308 return position;
309 }
310 remaining -= population;
311 }
312 throw std::invalid_argument("select_groups missing high bit");
313 }

330 static select_groups build(std::span<std::uint64_t const> residuals,
331 std::uint64_t records) {
332 auto groups = records / K + (records % K != 0);
333 if (residuals.size() != select_groups_detail::add(groups, 1))
334 throw std::invalid_argument("select_groups residual count");
335 if (!select_groups_detail::monotone(residuals))
336 throw std::invalid_argument("select_groups nonmonotone offsets");
337 select_groups result;
338 result.record_count = records;
339 result.universe = residuals.back();
340 auto quotient = result.universe / residuals.size();
341 result.low_width = quotient ? unsigned(std::bit_width(quotient) - 1) : 0;
342 auto low_bits = select_groups_detail::multiply(residuals.size(), result.low_width);
343 auto high_bits = select_groups_detail::add(result.universe >> result.low_width, residuals.size());
344 result.low.assign(select_groups_detail::words(low_bits), 0);
345 result.high.assign(select_groups_detail::words(high_bits), 0);
346 result.samples.clear();
347 select_groups_detail::pack_low(residuals, result.low, result.low_width);
348 select_groups_detail::write_high(residuals, result.high, result.low_width);
349 for (std::uint64_t begin = 0; begin < residuals.size(); begin += 256) {
350 auto end = residuals.size() - begin < 256 ? residuals.size() : begin + 256;
351 auto first = (residuals[begin] >> result.low_width) + begin;
352 auto last = (residuals[end - 1] >> result.low_width) + end - 1;
353 auto sparse = std::numeric_limits<std::uint64_t>::max();
354 if (last - first >= 4096) {
355 sparse = result.sparse.size();
356 for (auto i = begin; i < end; ++i)
357 result.sparse.push_back((residuals[i] >> result.low_width) + i);
358 }
359 result.samples.push_back({first, sparse});
360 }
361 return result;
362 }
