40 template <unsigned W, std::size_t O, std::size_t I>
41 inline std::uint64_t low_component(std::uint64_t const * source) noexcept {
42 constexpr auto mask = (std::uint64_t{1} << W) - 1;
43 constexpr auto bit = I * W;
44 constexpr auto first = O * 64;
45 if constexpr (bit < first) return (source[I] & mask) >> (first - bit);
46 else return (source[I] & mask) << (bit - first);
47 }

62 // W-bit fields return to a word boundary after 64/gcd(W,64) values.
63 // Each tile assigns its output words once, rather than repeatedly loading
64 // and updating them for each input field. All shifts are compile-time
65 // constants below 64; the final incomplete tile uses bounded scalar work.
66 template <unsigned W>
67 inline void pack_low_fixed(std::span<std::uint64_t const> source,
68 std::span<std::uint64_t> out) noexcept {
69 static_assert(W <= 63);
70 if constexpr (W) {
71 constexpr auto divisor = std::gcd(W, 64u);
72 constexpr auto inputs = 64 / divisor;
73 constexpr auto outputs = W / divisor;
74 std::size_t at = 0;
75 for (; source.size() - at >= inputs; at += inputs)
76 low_tile<W>(source.data() + at, out.data() + (at / inputs) * outputs,
77 std::make_index_sequence<outputs>{});
78 auto tail = out.subspan((at / inputs) * outputs);
79 std::fill(tail.begin(), tail.end(), 0);
80 constexpr auto mask = (std::uint64_t{1} << W) - 1;
81 for (std::size_t i = 0; i != source.size() - at; ++i) {
82 auto bit = i * W;
83 unsigned shift = unsigned(bit % 64);
84 auto value = source[at + i] & mask;
85 tail[bit / 64] |= value << shift;
86 if (shift + W > 64) tail[bit / 64 + 1] |= value >> (64 - shift);
87 }
88 }
89 }

92 constexpr auto low_packers(std::index_sequence<W...>) noexcept {
93 using packer = void (*)(std::span<std::uint64_t const>, std::span<std::uint64_t>) noexcept;
94 return std::array<packer, sizeof...(W)>{&pack_low_fixed<W>...};

97 // This dispatch occurs once per complete low section, not per value. The
98 // destination may contain old data: both complete and tail words are set,
99 // including zero tail padding. Source and destination must not overlap.
100 inline void pack_low(std::span<std::uint64_t const> source,
101 std::span<std::uint64_t> out, unsigned width) {
102 if (width > 63) throw std::invalid_argument("select_groups low width");
103 if (out.size() != words(multiply(source.size(), width)))
104 throw std::invalid_argument("select_groups low output size");
105 static constexpr auto packers = low_packers(std::make_index_sequence<64>{});
106 packers[width](source, out);
107 }
108
109 // The owner has checked monotonicity, width and extent and zeroed out.
110 // (source[i] >> width) + i is strictly increasing, even for equal values.
111 // Accumulating a word locally removes repeated dependent memory updates;
112 // gaps remain zero. This preserves the exact existing high-bit ordering.
113 inline void write_high(std::span<std::uint64_t const> source,
114 std::span<std::uint64_t> out, unsigned width) noexcept {
115 std::uint64_t word = 0, value = 0;
116 for (std::uint64_t i = 0; i != source.size(); ++i) {
117 auto position = (source[i] >> width) + i;
118 auto next = position / 64;
119 if (next != word) {
120 out[word] = value;
121 word = next;
122 value = 0;
123 }
124 value |= std::uint64_t{1} << (position % 64);
125 }
126 if (!source.empty()) out[word] = value;
127 }

137 // Elias--Fano over residual integer addresses at records 0,K,2K,..., then an
138 // actual-record-count sentinel. Residuals omit the fixed stride per record.
139 // Addresses and fixed stride use the same units (for example bytes or bits).
140 // Equal residual offsets are valid, including an entirely zero sequence.
141 //
142 // Select on the high bitvector is bounded: sample every 256 ones; scan at
143 // most 4096 bits (65 aligned words) for a dense group, or read an explicit
144 // exception position for a sparse group. There is no binary search. The
145 // exception array is O(n): disjoint sparse groups each span >4096 high bits,
146 // and Elias--Fano's high vector has O(n) bits. This is a practical baseline,
147 // not a tuned succinct select implementation.
148 // Views check shapes and guard navigation bounds. They do not establish
149 // semantic consistency of borrowed metadata: use a builder or validated
150 // reader. These native-endian spans are not a portable file-format parser.
151 template <std::uint64_t K> struct select_groups_view {
152 static_assert(K >= 3 && K < std::numeric_limits<std::uint64_t>::max() && std::has_single_bit(K + 1),
153 "group size must be 2^n-1 and at least three");
154 static constexpr std::uint64_t group_size = K;

179 std::uint64_t residual(std::uint64_t group) const {
180 if (group > group_count()) throw std::out_of_range("select_groups group");
181 auto position = select_high(group);
182 if (position < group) throw std::invalid_argument("invalid select_groups high value");
183 auto hi = position - group;
184 if (hi > (universe_ >> low_width_)) throw std::invalid_argument("select_groups high overflow");
185 std::uint64_t lo = 0;
186 if (low_width_) {
187 auto bit = group * low_width_;
188 auto word = bit / 64;
189 unsigned shift = unsigned(bit % 64);
190 lo = low_[word] >> shift;
191 if (shift + low_width_ > 64) lo |= low_[word + 1] << (64 - shift);
192 lo &= (std::uint64_t{1} << low_width_) - 1;
193 }
194 auto value = (hi << low_width_) | lo;
195 if (value > universe_) throw std::invalid_argument("select_groups value exceeds universe");
196 return value;
197 }

206 std::uint64_t select_high(std::uint64_t ordinal) const {
207 auto sample = samples_[ordinal / 256];
208 unsigned remaining = unsigned(ordinal % 256);
209 if (sample.sparse != std::numeric_limits<std::uint64_t>::max()) {
210 if (sample.sparse > sparse_.size() || remaining >= sparse_.size() - sample.sparse)
211 throw std::invalid_argument("invalid select_groups exception");
212 auto position = sparse_[sample.sparse + remaining];
213 if (position >= high_bits_ || !(high_[position / 64] & (std::uint64_t{1} << (position % 64))))
214 throw std::invalid_argument("invalid select_groups sparse position");
215 return position;
216 }
217 if (sample.first >= high_bits_) throw std::invalid_argument("invalid select_groups sample");
218 auto word = sample.first / 64;
219 auto value = high_[word] & (~std::uint64_t{0} << (sample.first % 64));
220 for (unsigned scanned = 0; scanned < 65 && word < high_.size(); ++scanned, ++word) {
221 if (scanned) value = high_[word];
222 auto population = unsigned(std::popcount(value));
223 if (remaining < population) {
224 for (; remaining; --remaining) value &= value - 1;
225 auto position = word * 64 + unsigned(std::countr_zero(value));
226 if (position >= high_bits_ || position - sample.first >= 4096)
227 throw std::invalid_argument("select_groups dense span");
228 return position;
229 }
230 remaining -= population;
231 }
232 throw std::invalid_argument("select_groups missing high bit");
233 }

247 "group size must be 2^n-1 and at least three");
248 static constexpr std::uint64_t group_size = K;

250 static select_groups build(std::span<std::uint64_t const> residuals,
251 std::uint64_t records) {
252 auto groups = records / K + (records % K != 0);
253 if (residuals.size() != groups + 1)
254 throw std::invalid_argument("select_groups residual count");
255 for (std::size_t i = 1; i < residuals.size(); ++i)
256 if (residuals[i] < residuals[i - 1])
257 throw std::invalid_argument("select_groups nonmonotone offsets");
258 select_groups result;
259 result.record_count = records;
260 result.universe = residuals.back();
261 auto quotient = result.universe / residuals.size();
262 result.low_width = quotient ? unsigned(std::bit_width(quotient) - 1) : 0;
263 auto low_bits = select_groups_detail::multiply(residuals.size(), result.low_width);
264 auto high_bits = select_groups_detail::add(result.universe >> result.low_width, residuals.size());
265 result.low.assign(select_groups_detail::words(low_bits), 0);
266 result.high.assign(select_groups_detail::words(high_bits), 0);
267 result.samples.clear();
268 select_groups_detail::pack_low(residuals, result.low, result.low_width);
269 select_groups_detail::write_high(residuals, result.high, result.low_width);
270 for (std::uint64_t begin = 0; begin < residuals.size(); begin += 256) {
271 auto end = residuals.size() - begin < 256 ? residuals.size() : begin + 256;
272 auto first = (residuals[begin] >> result.low_width) + begin;
273 auto last = (residuals[end - 1] >> result.low_width) + end - 1;
274 auto sparse = std::numeric_limits<std::uint64_t>::max();
275 if (last - first >= 4096) {
276 sparse = result.sparse.size();
277 for (auto i = begin; i < end; ++i)
278 result.sparse.push_back((residuals[i] >> result.low_width) + i);
279 }
280 result.samples.push_back({first, sparse});
281 }
282 return result;
283 }
