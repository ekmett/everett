80 static profile_blob build(
81 std::span<profile_record const> native,
82 std::span<bit_string const> borrowed = {},
83 std::uint64_t native_restart_factor = 18,
84 std::span<std::uint64_t const> borrowed_prefix_ceilings = {},
85 profile_borrowed_policy policy = profile_borrowed_policy::shared_boundaries) {
86 check_count(native.size(), borrowed.size());

95 build_index(result, native, borrowed, borrowed_prefix_ceilings, policy);

137 profile_blob_window project(std::uint64_t group) const {
138 if (group >= group_count()) throw std::out_of_range("profile blob virtual group");
139 auto first = group * group_size;
140 auto last = first + std::min<std::uint64_t>(group_size, virtual_count_ - first);
141 auto ranks = interleave_.view();
142 auto a = ranks.rank(group);
143 // The stored population is exactly the difference of the two ranks.
144 auto b = a + ranks.class_at(group);
145 return {first - a, last - b, a, b};
146 }

148 // The caller selects the group containing the last virtual key <= query,
149 // with group zero used when query precedes the entire catalog. lower is
150 // the group's known boundary key (empty context is sufficient at zero).
151 // The result owns its value and outgoing prefix; it survives later calls.
152 // Native equality does not suppress routing or resolve older value arrows.
153 profile_blob_window_result<P> search_window(
154 bit_view query, std::uint64_t group, profile_anchor<P> lower) const {
155 if (query.size() % P::bits_per_unit) {
156 throw std::invalid_argument("query length is not aligned to the profile unit");
157 }
158 auto query_units = query.size() / P::bits_per_unit;
159 auto window = project(group);
160 profile_blob_window_result<P> result;
161 native_->view().visit_window(window.native_first, window.native_last, lower, query_units,
162 [&](profile_item<P> item) {
163 auto order = compare_profile_prefix(item.key, query);
164 if (!order) {
165 result.native = profile_blob_native_match<P>{item.ordinal, bit_string::copy(item.value)};
166 }
167 return order < 0;
168 });
169
170 auto consider_sample = [&](profile_item<P> item) {
171 auto order = compare_profile_prefix(item.key, query);
172 if (order > 0) return false;
173 auto is_false = false_borrow(item.ordinal);
174 result.borrowed_predecessor = profile_blob_borrowed_predecessor<P>{
175 item.ordinal, checked_target_ordinal(item.ordinal), true,
176 bit_string::copy(item.key.prefix), item.key.full_units, is_false};
177 if (!order && is_false && !result.native) {
178 if (!window.native_first) {
179 throw std::invalid_argument("false borrow has no native predecessor");
180 }
181 // The native-before-borrowed tie order makes this the unique matching
182 // native slot, even when many equal borrowed samples span group cuts.
183 auto ordinal = window.native_first - 1;
184 result.native = profile_blob_native_match<P>{
185 ordinal, bit_string::copy(native_->view().encoded_at(ordinal).value)};
186 }
187 return true;
188 };
189 if (window.borrowed_first) {
190 auto ordinal = window.borrowed_first - 1;
191 if (borrowed_policy_ != profile_borrowed_policy::ordinary) {
192 borrowed_.view().visit_window(ordinal, ordinal + 1, lower, query_units, consider_sample);
193 } else {
194 result.borrowed_predecessor = profile_blob_borrowed_predecessor<P>{
195 ordinal, checked_target_ordinal(ordinal), false, {}, 0, false_borrow(ordinal)};
196 }
197 }
198 borrowed_.view().visit_window(window.borrowed_first, window.borrowed_last, lower,
199 query_units, consider_sample);
200 return result;
201 }

228 static void build_index(profile_blob & result, std::span<profile_record const> native,
229 std::span<bit_string const> borrowed,
230 std::span<std::uint64_t const> borrowed_prefix_ceilings,
231 profile_borrowed_policy policy) {
232 if (!borrowed_prefix_ceilings.empty() && borrowed_prefix_ceilings.size() != borrowed.size()) {
233 throw std::invalid_argument("one borrowed prefix ceiling is required per record");
234 }
235 if (policy != profile_borrowed_policy::ordinary &&
236 policy != profile_borrowed_policy::bidirectional &&
237 policy != profile_borrowed_policy::shared_boundaries) {
238 throw std::invalid_argument("invalid profile borrowed prefix policy");
239 }
240 std::vector<std::uint64_t> ceilings;
241 if (policy == profile_borrowed_policy::shared_boundaries) {
242 ceilings.assign(borrowed.size(), std::numeric_limits<std::uint64_t>::max());
243 if (!borrowed_prefix_ceilings.empty()) {
244 std::copy(borrowed_prefix_ceilings.begin(), borrowed_prefix_ceilings.end(), ceilings.begin());
245 }
246 borrowed_prefix_ceilings = ceilings;
247 } else if (policy == profile_borrowed_policy::bidirectional) {
248 ceilings.reserve(borrowed.size());
249 for (std::size_t i = 0; i != borrowed.size(); ++i) {
250 auto ceiling = i + 1 == borrowed.size() ? 0 :
251 common_prefix_units<P>(borrowed[i].view(), borrowed[i + 1].view());
252 if (!borrowed_prefix_ceilings.empty()) {
253 ceiling = std::min<std::uint64_t>(ceiling, borrowed_prefix_ceilings[i]);
254 }
255 ceilings.push_back(ceiling);
256 }
257 borrowed_prefix_ceilings = ceilings;
258 }
259 result.borrowed_policy_ = policy;
260 std::vector<profile_record> borrowed_records;
261 borrowed_records.reserve(borrowed.size());
262 result.false_borrows_.resize(borrowed.size() / 8 + (borrowed.size() % 8 != 0));
263 std::size_t native_at = 0;
264 for (std::size_t i = 0; i != borrowed.size(); ++i) {
265 if (i && compare_bits(borrowed[i - 1].view(), borrowed[i].view()) > 0) {
266 throw std::invalid_argument("profile blob borrowed keys must be sorted");
267 }
268 while (native_at != native.size() &&
269 compare_bits(native[native_at].key.view(), borrowed[i].view()) < 0) {
270 ++native_at;
271 }
272 if (native_at != native.size() &&
273 compare_bits(native[native_at].key.view(), borrowed[i].view()) == 0) {
274 result.false_borrows_[i / 8] |= static_cast<std::byte>(1u << (i % 8));
275 }
276 borrowed_records.push_back({borrowed[i], {}});
277 }
278 auto count = std::uint64_t(native.size()) + borrowed.size();
279 std::vector<std::uint64_t> classes(static_cast<std::size_t>(count / group_size + (count % group_size != 0)));
280 std::size_t a = 0;
281 std::size_t s = 0;
282 for (std::uint64_t i = 0; i != count; ++i) {
283 // Native precedes every borrowed occurrence of an equal key.
284 auto take_borrowed = s != borrowed.size() &&
285 (a == native.size() || compare_bits(borrowed[s].view(), native[a].key.view()) < 0);
286 if (policy == profile_borrowed_policy::shared_boundaries && i % group_size == 0 && s) {
287 auto const & boundary = take_borrowed ? borrowed[s] : native[a].key;
288 ceilings[s - 1] = std::min<std::uint64_t>(ceilings[s - 1],
289 common_prefix_units<P>(borrowed[s - 1].view(), boundary.view()));
290 }
291 if (take_borrowed) {
292 ++classes[static_cast<std::size_t>(i / group_size)];
293 ++s;
294 } else {
295 ++a;
296 }
297 }
298 result.borrowed_ = borrowed_array::build(borrowed_records, borrowed_prefix_ceilings);
299 result.interleave_ = rank_groups<group_size>::build(classes, count);
300 result.virtual_count_ = count;
301 }
