69 static profile_blob build(
70 std::span<profile_record const> native,
71 std::span<bit_string const> borrowed = {}) {
72 check_count(native.size(), borrowed.size());
73 for (std::size_t i = 1; i < native.size(); ++i) {
74 if (compare_bits(native[i - 1].key.view(), native[i].key.view()) >= 0) {
75 throw std::invalid_argument("profile blob native keys must be strictly sorted");
76 }
77 }
78 profile_blob result;
79 result.native_ = std::make_shared<native_array const>(
80 native_array::build(native));
81 build_index(result, native, borrowed);
82 return result;
83 }

86 // independent physical directory. Only index-local navigation is rebuilt.
87 profile_blob reindex(std::span<bit_string const> borrowed) const {
88 check_count(native_->size(), borrowed.size());
89 std::vector<profile_record> native_records;
90 native_records.reserve(static_cast<std::size_t>(native_->size()));
91 native_->view().visit_all([&](profile_item<P> item) {
92 native_records.push_back({bit_string::copy(item.key.prefix), {}});
93 return true;
94 });

96 result.native_ = native_;

114 bool false_borrow(std::uint64_t ordinal) const {
115 if (ordinal >= borrowed_.size()) throw std::out_of_range("profile blob borrowed ordinal");
116 return (std::to_integer<unsigned>(false_borrows_[ordinal / 8]) >> (ordinal % 8)) & 1;

119 profile_blob_window project(std::uint64_t group) const {
120 if (group >= group_count()) throw std::out_of_range("profile blob virtual group");
121 auto first = group * group_size;
122 auto last = first + std::min<std::uint64_t>(group_size, virtual_count_ - first);
123 auto ranks = interleave_.view();
124 auto a = ranks.rank(group);
125 // The stored population is exactly the difference of the two ranks.
126 auto b = a + ranks.class_at(group);
127 return {first - a, last - b, a, b};
128 }

130 // lower compares the exact sampled boundary to its owned query. At group
131 // zero an empty-key context suffices. The query must not precede lower.
132 profile_blob_window_result<P> search_window(
133 std::uint64_t group, profile_query_context<P> const & lower,
134 profile_comparison_work * native_work = nullptr,
135 profile_comparison_work * borrowed_work = nullptr) const {
136 if (lower.order() > 0) throw std::invalid_argument("query precedes its routed boundary");
137 auto window = project(group);
138 profile_blob_window_result<P> result;
139 native_->view().compare_window(window.native_first, window.native_last, lower,
140 [&](profile_comparison_item<P> item) {
141 auto order = item.comparison.order();
142 if (!order) result.native = profile_blob_native_match<P>{item.ordinal, bit_string::copy(item.value)};
143 return order < 0;
144 }, native_work);
145
146 auto remember = [&](std::uint64_t ordinal, profile_query_context<P> const & comparison) {
147 auto is_false = false_borrow(ordinal);
148 result.borrowed_predecessor = profile_blob_borrowed_predecessor<P>{
149 ordinal, checked_target_ordinal(ordinal), is_false, comparison};
150 if (!comparison.order() && is_false && !result.native) {
151 if (!window.native_first) throw std::invalid_argument("false borrow has no native predecessor");
152 auto native_ordinal = window.native_first - 1;
153 result.native = profile_blob_native_match<P>{
154 native_ordinal, bit_string::copy(native_->view().encoded_at(native_ordinal).value)};
155 }
156 };
157 std::uint64_t previous_units = 0;
158 borrowed_.view().compare_window(window.borrowed_first, window.borrowed_last, lower,
159 [&](profile_comparison_item<P> item) {
160 if (item.comparison.order() > 0) return false;
161 remember(item.ordinal, item.comparison);
162 return true;
163 }, borrowed_work, &previous_units);
164 if (!result.borrowed_predecessor && window.borrowed_first) {
165 // The cut LCP repairs comparison only. Its physical predecessor length
166 // comes from header replay or the terminal checkpoint, never key bytes.
167 auto comparison = lower.predecessor(cut_lcps_.at(static_cast<std::size_t>(group)), previous_units);
168 remember(window.borrowed_first - 1, comparison);
169 }
170 return result;
171 }

198 static void build_index(profile_blob & result, std::span<profile_record const> native,
199 std::span<bit_string const> borrowed) {
200 std::vector<profile_record> borrowed_records;
201 borrowed_records.reserve(borrowed.size());
202 result.false_borrows_.resize(borrowed.size() / 8 + (borrowed.size() % 8 != 0));
203 std::size_t native_at = 0;
204 for (std::size_t i = 0; i != borrowed.size(); ++i) {
205 if (i && compare_bits(borrowed[i - 1].view(), borrowed[i].view()) > 0) {
206 throw std::invalid_argument("profile blob borrowed keys must be sorted");
207 }
208 while (native_at != native.size() &&
209 compare_bits(native[native_at].key.view(), borrowed[i].view()) < 0) {
210 ++native_at;
211 }
212 if (native_at != native.size() &&
213 compare_bits(native[native_at].key.view(), borrowed[i].view()) == 0) {
214 result.false_borrows_[i / 8] |= static_cast<std::byte>(1u << (i % 8));
215 }
216 borrowed_records.push_back({borrowed[i], {}});
217 }
218 auto count = std::uint64_t(native.size()) + borrowed.size();
219 std::vector<std::uint64_t> classes(static_cast<std::size_t>(count / group_size + (count % group_size != 0)));
220 std::size_t a = 0;
221 std::size_t s = 0;
222 for (std::uint64_t i = 0; i != count; ++i) {
223 // Native precedes every borrowed occurrence of an equal key.
224 auto take_borrowed = s != borrowed.size() &&
225 (a == native.size() || compare_bits(borrowed[s].view(), native[a].key.view()) < 0);
226 if (i % group_size == 0) {
227 auto const & boundary = take_borrowed ? borrowed[s] : native[a].key;
228 result.cut_lcps_.push_back(s ? compare_common_bits(borrowed[s - 1].view(), boundary.view()).common_bits : 0);
229 }
230 if (take_borrowed) {
231 ++classes[static_cast<std::size_t>(i / group_size)];
232 ++s;
233 } else {
234 ++a;
235 }
236 }
237 result.borrowed_ = borrowed_array::build(borrowed_records);
238 result.interleave_ = rank_groups<group_size>::build(classes, count);
239 result.virtual_count_ = count;
240 }
