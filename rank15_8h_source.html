18 // Separate packed-class codec: 15-entry groups do not align with the
19 // 512/2048 source-bit runs of rank.h. We retain four bits per class and one
20 // 64-bit checkpoint every 128 classes, with at most eight word sums/query.
21 // There is no full origin bitvector and no arbitrary within-group rank.
22 // Views check section shapes, not the semantic consistency of borrowed
23 // classes/checkpoints; those must come from a builder or validated reader.
24 struct rank15_view {
25 rank15_view(std::span<std::uint64_t const> classes,
26 std::span<std::uint64_t const> checkpoints,
27 std::uint64_t virtual_count, std::uint64_t total)
28 : classes_(classes), checkpoints_(checkpoints),
29 virtual_count_(virtual_count), total_(total) {
30 auto groups = group_count();
31 if (classes.size() != groups / 16 + (groups % 16 != 0) ||
32 checkpoints.size() != groups / 128 + (groups % 128 != 0) ||
33 total > virtual_count)
34 throw std::invalid_argument("invalid rank15 spans");
35 }

43 unsigned class_at(std::uint64_t group) const {
44 if (group >= group_count()) throw std::out_of_range("rank15 class");
45 return unsigned((classes_[group / 16] >> (4 * (group % 16))) & 15);

48 // rank(group) counts borrowed entries before virtual position 15*group;
49 // group_count() is the actual-length endpoint of a partial final group.
50 std::uint64_t rank(std::uint64_t group) const {
51 if (group > group_count()) throw std::out_of_range("rank15 group");
52 if (group == group_count()) return total_;
53 auto result = checkpoints_[group / 128];
54 auto word = (group / 128) * 8;
55 for (; word < group / 16; ++word) result += sum_nibbles(classes_[word]);
56 auto tail = unsigned(group % 16);
57 if (tail) result += sum_nibbles(classes_[word] & ((std::uint64_t{1} << (4 * tail)) - 1));
58 return result;
59 }

61 private:
62 static unsigned sum_nibbles(std::uint64_t value) noexcept {
63 value = (value & 0x0f0f0f0f0f0f0f0full) + ((value >> 4) & 0x0f0f0f0f0f0f0f0full);
64 value = (value & 0x00ff00ff00ff00ffull) + ((value >> 8) & 0x00ff00ff00ff00ffull);
65 value = (value & 0x0000ffff0000ffffull) + ((value >> 16) & 0x0000ffff0000ffffull);
66 return unsigned((value & 0xffffffffu) + (value >> 32));
67 }

75 struct rank15_index {
76 static rank15_index build(std::span<std::uint8_t const> source, std::uint64_t count) {
77 auto groups = count / 15 + (count % 15 != 0);
78 if (source.size() != groups) throw std::invalid_argument("rank15 class length");
79 rank15_index result;
80 result.virtual_count = count;
81 result.classes.resize(groups / 16 + (groups % 16 != 0));
82 for (std::uint64_t i = 0; i < groups; ++i) {
83 auto limit = i + 1 == groups && count % 15 ? count % 15 : 15;
84 if (source[i] > limit) throw std::invalid_argument("rank15 class population");
85 if (i % 128 == 0) result.checkpoints.push_back(result.total);
86 result.classes[i / 16] |= std::uint64_t(source[i]) << (4 * (i % 16));
87 result.total += source[i];
88 }
89 return result;
90 }
