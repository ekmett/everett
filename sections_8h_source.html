70 inline unsigned hex_value(char c) noexcept { return unsigned(c <= '9' ? c - '0' : c - 'a' + 10); }
71 inline void put_id(std::span<std::byte> bytes, std::size_t at, object_id const & id) {
72 auto const & text = id.hex();
73 if (text.size() != 32) throw std::invalid_argument("invalid Everett section object identity");
74 if (at > bytes.size() || 16 > bytes.size() - at)
75 throw std::out_of_range("Everett section identity position");
76 for (std::size_t i = 0; i != 16; ++i)
77 bytes[at + i] = std::byte((hex_value(text[2 * i]) << 4) | hex_value(text[2 * i + 1]));
78 }
79 inline object_id get_id(std::span<std::byte const> bytes, std::size_t at) {
80 constexpr char hex[] = "0123456789abcdef";
81 std::string text(32, '0');
82 for (std::size_t i = 0; i != 16; ++i) {
83 auto byte = std::to_integer<unsigned>(bytes[at + i]);
84 text[2 * i] = hex[byte >> 4];
85 text[2 * i + 1] = hex[byte & 15];
86 }
87 return object_id(std::move(text));
88 }

98 template <class P> directory parse(file_header<P> const & header, std::span<std::byte const> body) {
99 directory result;
100 bool index = header.kind == file_kind::fractional_index;
101 result.count = index ? 9 : 5;
102 result.bytes = index ? index_directory_bytes : native_directory_bytes;
103 auto descriptors = index ? index_descriptor_offset : native_descriptor_offset;
104 if (body.size() > std::numeric_limits<std::uint64_t>::max() - 7)
105 throw std::invalid_argument("Everett section container extent overflows");
106 if (body.size() < result.bytes) throw std::invalid_argument("truncated Everett section directory");
107 if (header.extent != profile_detail::multiply(body.size(), 1u << (3 - P::unit_shift)))
108 throw std::invalid_argument("Everett section container must occupy complete bytes");
109 auto magic = index ? "IX01" : "KV01";
110 for (unsigned i = 0; i != 4; ++i)
111 if (std::to_integer<unsigned>(body[i]) != unsigned(magic[i]))
112 throw std::invalid_argument("unexpected Everett section magic");
113 if (file_detail::get(body, 4, 2) != 1 || file_detail::get(body, 6, 2) != result.count)
114 throw std::invalid_argument("unsupported Everett section directory");
115 result.extent = file_detail::get(body, 8, 8);
116 result.terminal_key_units = file_detail::get(body, 16, 8);
117 result.universe = file_detail::get(body, 24, 8);
118 result.low_width = unsigned(file_detail::get(body, 32, 1));
119 if (index) {
120 auto target = file_detail::get(body, 33, 1);
121 if (target > 1) throw std::invalid_argument("invalid Everett target flag");
122 zero(body.subspan(34, 6), "nonzero Everett directory reserved bytes");
123 zero(body.subspan(96, 16), "nonzero Everett directory reserved bytes");
124 result.virtual_count = file_detail::get(body, 40, 8);
125 if (result.virtual_count < header.record_count)
126 throw std::invalid_argument("borrowed records exceed virtual count");
127 result.native_id = get_id(body, 48);
128 if (target) result.target_id = blob_identity{get_id(body, 64), get_id(body, 80)};
129 else zero(body.subspan(64, 32), "absent Everett target must encode zero");
130 } else zero(body.subspan(33, 15), "nonzero Everett directory reserved bytes");
131 std::uint64_t end = result.bytes;
132 for (std::size_t i = 0; i != result.count; ++i) {
133 auto at = descriptors + i * descriptor_bytes;
134 descriptor part{file_detail::get(body, at, 8), file_detail::get(body, at + 8, 8)};
135 if (part.offset != align(end) || part.offset > body.size() || part.length > body.size() - part.offset)
136 throw std::invalid_argument("noncanonical Everett section range");
137 end = part.offset + part.length;
138 result.sections[i] = part;
139 }
140 if (end != body.size()) throw std::invalid_argument("trailing Everett section bytes");
141 return result;
142 }
143 inline void validate_gaps(std::span<std::byte const> body, directory const & layout) {
144 std::size_t end = layout.bytes;
145 for (std::size_t i = 0; i != layout.count; ++i) {
146 auto const & part = layout.sections[i];
147 zero(body.subspan(end, static_cast<std::size_t>(part.offset) - end), "nonzero Everett section alignment padding");
148 end = static_cast<std::size_t>(part.offset + part.length);
149 }

151 template <class P, stream_role Role> profile_view<P, Role> profile(
152 file_header<P> const & header, std::span<std::byte const> body, directory const & layout) {
153 auto metadata = profile_detail::initial_metadata<P, Role>();
154 metadata.record_count = header.record_count;
155 metadata.extent = layout.extent;
156 metadata.terminal_key_units = layout.terminal_key_units;
157 metadata.common_value_width = header.common_value_width;

162 profile_detail::add(header.record_count / P::codec_block_size +
163 (header.record_count % P::codec_block_size != 0), 1), layout.universe, layout.low_width};
164 auto result = profile_view<P, Role>::from_sections(part(fc), offsets, metadata);
165 result.validate_offset_metadata();

173 template <class P, stream_role Role> void scan_profile(profile_view<P, Role> const & view) {
174 view.validate_contents();
175 auto const & metadata = view.metadata();
176 auto offsets = view.group_offsets();
177 std::vector<std::uint64_t> residuals;
178 auto groups = view.block_count();
179 if (groups == std::numeric_limits<std::uint64_t>::max() || groups + 1 > residuals.max_size())
180 throw std::length_error("Everett scan offset count");
181 residuals.reserve(static_cast<std::size_t>(groups + 1));
182 bit_view data(view.bytes(), profile_detail::multiply(metadata.extent, P::bits_per_unit));
183 std::uint64_t at = 0, previous_units = 0;
184 bit_string previous;
185 for (std::uint64_t ordinal = 0; ordinal != metadata.record_count; ++ordinal) {
186 if (ordinal % P::codec_block_size == 0) {
187 auto stride = profile_detail::multiply(ordinal, metadata.common_value_width.value_or(0));
188 if (stride > at) throw std::invalid_argument("Everett fixed values exceed physical position");
189 residuals.push_back(at - stride);
190 auto checkpoint = profile_detail::read_count<P>(data, at);
191 if (checkpoint != previous_units) throw std::invalid_argument("Everett profile length checkpoint mismatch");
192 }
193 auto backspace = profile_detail::read_backspace<P>(data, at);
194 auto suffix = profile_detail::read_count<P>(data, at);
195 auto values = metadata.common_value_width ? *metadata.common_value_width : profile_detail::read_count<P>(data, at);
196 if (backspace > previous_units) throw std::invalid_argument("Everett profile backspace exceeds predecessor");
197 auto retained = previous_units - backspace;
198 auto length = profile_detail::add(retained, suffix);
199 auto start_bits = profile_detail::multiply(at, P::bits_per_unit);
200 auto suffix_bits = profile_detail::multiply(suffix, P::bits_per_unit);
201 auto value_bits = profile_detail::multiply(values, P::bits_per_unit);
202 if (start_bits > data.size() || suffix_bits > data.size() - start_bits ||
203 value_bits > data.size() - start_bits - suffix_bits)
204 throw std::invalid_argument("truncated Everett profile record");
205 auto retained_bits = profile_detail::multiply(retained, P::bits_per_unit);
206 // The retained prefix is already equal. Ordinary FC must differ at
207 // the first new unit, or extend the entire predecessor; no inherited
208 // prefix needs copying or comparison to establish maximality/order.
209 if (retained < previous_units) {
210 if (!suffix) throw std::invalid_argument("Everett profile key precedes its predecessor");
211 auto before = profile_detail::load_bits(previous.view(), retained_bits, P::bits_per_unit);
212 auto after = profile_detail::load_bits(data, start_bits, P::bits_per_unit);
213 if (after <= before)
214 throw std::invalid_argument("Everett profile is not sorted ordinary front coding");
215 } else if (ordinal && !suffix && Role == stream_role::native)
216 throw std::invalid_argument("Everett native profile keys must be unique");
217 profile_detail::resize(previous, retained_bits);
218 profile_detail::append(previous, data.subview(start_bits, suffix_bits));
219 at = profile_detail::add(at, profile_detail::add(suffix, values));
220 previous_units = length;
221 }
222 if (at != metadata.extent || previous_units != metadata.terminal_key_units)
223 throw std::invalid_argument("Everett terminal profile extent or length mismatch");
224 auto stride = profile_detail::multiply(metadata.record_count, metadata.common_value_width.value_or(0));
225 if (stride > at) throw std::invalid_argument("Everett fixed payload exceeds extent");
226 residuals.push_back(at - stride);
227 auto expected = elias_fano::build(residuals);
228 if (expected.universe != offsets.universe() || expected.low_width != offsets.low_width())
229 throw std::invalid_argument("noncanonical Everett Elias-Fano parameters");
230 equal_words(offsets.low_words(), expected.low, "Everett Elias-Fano low words mismatch");
231 equal_words(offsets.high_words(), expected.high, "Everett Elias-Fano high words mismatch");
232 equal_words(offsets.sparse_words(), expected.sparse, "Everett Elias-Fano sparse words mismatch");
233 auto actual_samples = offsets.samples();
234 if (actual_samples.size() != expected.samples.size()) throw std::invalid_argument("Everett Elias-Fano sample count mismatch");
235 for (std::size_t i = 0; i != expected.samples.size(); ++i) {
236 auto a = actual_samples[i];
237 auto b = expected.samples[i];
238 if (a.first != b.first || a.sparse != b.sparse) throw std::invalid_argument("Everett Elias-Fano sample mismatch");
239 }
240 }

265 // Returned spans borrow this encoder as well as its source arrays, and

267 std::vector<std::span<std::byte const>> chunks() const & {
268 require_active();
269 std::vector<std::span<std::byte const>> result;
270 result.reserve(2 * sections_.size() + 1);
271 result.push_back(std::span(directory_).first(directory_size_));
272 std::uint64_t end = directory_size_;
273 for (auto section : sections_) {
274 auto gap = section_detail::align(end) - end;
275 if (gap) result.push_back(std::span(zero_).first(static_cast<std::size_t>(gap)));
276 result.push_back(section);
277 end += gap + section.size();
278 }
279 return result;
280 }

283 require_active();

286 if (size > body.max_size()) throw std::length_error("Everett section body too large");
287 body.reserve(static_cast<std::size_t>(size));
288 for (auto chunk : chunks()) body.insert(body.end(), chunk.begin(), chunk.end());

327 file_detail::put(bytes, (i << 4), 8, values[i].first);
328 file_detail::put(bytes, (i << 4) + 8, 8, values[i].sparse);

333 template <stream_role Role> void profile(profile_array<P, Role> const & array) {
334 sections_.push_back(array.bytes());
335 auto const & offsets = array.group_offsets();
336 words(offsets.low); words(offsets.high); samples(offsets.samples); words(offsets.sparse);
337 auto const & metadata = array.metadata();
338 header_.record_count = metadata.record_count;
339 header_.common_value_width = metadata.common_value_width;
340 file_detail::put(directory_, 8, 8, metadata.extent);
341 file_detail::put(directory_, 16, 8, metadata.terminal_key_units);
342 file_detail::put(directory_, 24, 8, offsets.universe);
343 file_detail::put(directory_, 32, 1, offsets.low_width);
344 }
345 void finish(file_kind kind) {
346 header_.kind = kind;
347 bool index = kind == file_kind::fractional_index;
348 directory_size_ = index ? section_detail::index_directory_bytes : section_detail::native_directory_bytes;
349 auto descriptors = index ? section_detail::index_descriptor_offset : section_detail::native_descriptor_offset;
350 auto magic = index ? "IX01" : "KV01";
351 for (unsigned i = 0; i != 4; ++i) directory_[i] = std::byte(magic[i]);
352 file_detail::put(directory_, 4, 2, 1);
353 file_detail::put(directory_, 6, 2, sections_.size());
354 std::uint64_t end = directory_size_;
355 for (std::size_t i = 0; i != sections_.size(); ++i) {
356 if (end > std::numeric_limits<std::uint64_t>::max() - 7)
357 throw std::overflow_error("Everett section alignment overflows");
358 auto start = section_detail::align(end);
359 end = profile_detail::add(start, sections_[i].size());
360 file_detail::put(directory_, descriptors + (i << 4), 8, start);
361 file_detail::put(directory_, descriptors + (i << 4) + 8, 8, sections_[i].size());
362 }
363 header_.extent = profile_detail::multiply(end, 1u << (3 - P::unit_shift));
364 file_detail::validate_metadata(header_);
365 }
