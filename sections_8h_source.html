74 inline unsigned hex_value(char c) noexcept { return unsigned(c <= '9' ? c - '0' : c - 'a' + 10); }
75 inline void put_id(std::span<std::byte> bytes, std::size_t at, object_id const & id) {
76 auto const & text = id.hex();
77 if (text.size() != 32) throw std::invalid_argument("invalid Everett section object identity");
78 if (at > bytes.size() || 16 > bytes.size() - at)
79 throw std::out_of_range("Everett section identity position");
80 for (std::size_t i = 0; i != 16; ++i)
81 bytes[at + i] = std::byte((hex_value(text[2 * i]) << 4) | hex_value(text[2 * i + 1]));
82 }
83 inline object_id get_id(std::span<std::byte const> bytes, std::size_t at) {
84 constexpr char hex[] = "0123456789abcdef";
85 std::string text(32, '0');
86 for (std::size_t i = 0; i != 16; ++i) {
87 auto byte = std::to_integer<unsigned>(bytes[at + i]);
88 text[2 * i] = hex[byte >> 4];
89 text[2 * i + 1] = hex[byte & 15];
90 }
91 return object_id(std::move(text));
92 }
93 inline std::span<std::byte const> section(std::span<std::byte const> body,
94 directory const & layout, std::size_t slot) {
95 if (slot >= layout.count) throw std::out_of_range("Everett section slot");

102 template <class P> directory parse(file_header<P> const & header, std::span<std::byte const> body) {
103 directory result;
104 bool index = header.kind == file_kind::fractional_index;
105 result.count = index ? 9 : 5;
106 result.bytes = index ? index_directory_bytes : native_directory_bytes;
107 auto descriptors = index ? index_descriptor_offset : native_descriptor_offset;
108 if (body.size() > std::numeric_limits<std::uint64_t>::max() - 7)
109 throw std::invalid_argument("Everett section container extent overflows");
110 if (body.size() < result.bytes) throw std::invalid_argument("truncated Everett section directory");
111 if (header.extent != profile_detail::multiply(body.size(), 1u << (3 - P::unit_shift)))
112 throw std::invalid_argument("Everett section container must occupy complete bytes");
113 auto magic = index ? "IX02" : "KV02";
114 for (unsigned i = 0; i != 4; ++i)
115 if (std::to_integer<unsigned>(body[i]) != unsigned(magic[i]))
116 throw std::invalid_argument("unexpected Everett section magic");
117 if (file_detail::get(body, 4, 2) != version || file_detail::get(body, 6, 2) != result.count)
118 throw std::invalid_argument("unsupported Everett section directory");
119 result.extent = file_detail::get(body, 8, 8);
120 result.terminal_key_units = file_detail::get(body, 16, 8);
121 result.universe = file_detail::get(body, 24, 8);
122 result.low_width = unsigned(file_detail::get(body, 32, 1));
123 if (index) {
124 auto target = file_detail::get(body, 33, 1);
125 if (target > 1) throw std::invalid_argument("invalid Everett target flag");
126 zero(body.subspan(34, 6), "nonzero Everett directory reserved bytes");
127 zero(body.subspan(96, 16), "nonzero Everett directory reserved bytes");
128 result.virtual_count = file_detail::get(body, 40, 8);
129 if (result.virtual_count < header.record_count)
130 throw std::invalid_argument("borrowed records exceed virtual count");
131 result.native_id = get_id(body, 48);
132 if (target) result.target_id = blob_identity{get_id(body, 64), get_id(body, 80)};
133 else zero(body.subspan(64, 32), "absent Everett target must encode zero");
134 } else zero(body.subspan(33, 15), "nonzero Everett directory reserved bytes");
135 std::uint64_t end = result.bytes;
136 for (std::size_t i = 0; i != result.count; ++i) {
137 auto at = descriptors + i * descriptor_bytes;
138 descriptor part{file_detail::get(body, at, 8), file_detail::get(body, at + 8, 8)};
139 if (part.offset != align(end) || part.offset > body.size() || part.length > body.size() - part.offset)
140 throw std::invalid_argument("noncanonical Everett section range");
141 end = part.offset + part.length;
142 result.sections[i] = part;
143 }
144 if (end != body.size()) throw std::invalid_argument("trailing Everett section bytes");
145 return result;
146 }
147 inline void validate_gaps(std::span<std::byte const> body, directory const & layout) {
148 std::size_t end = layout.bytes;
149 for (std::size_t i = 0; i != layout.count; ++i) {
150 auto const & part = layout.sections[i];
151 zero(body.subspan(end, static_cast<std::size_t>(part.offset) - end), "nonzero Everett section alignment padding");
152 end = static_cast<std::size_t>(part.offset + part.length);
153 }

156 file_header<P> const & header, std::span<std::byte const> body, directory const & layout) {

158 metadata.record_count = header.record_count;
159 metadata.extent = layout.extent;
160 metadata.terminal_key_units = layout.terminal_key_units;
161 metadata.common_value_width = header.common_value_width;
162 metadata.policy_fixed_values = header.policy_value_width.has_value();
163 metadata.policy_value_width = header.policy_value_width.value_or(0);
164 auto part = [&](std::size_t slot) { return section(body, layout, slot); };
165 elias_fano_view offsets{
166 word_view::little_endian(part(low)), word_view::little_endian(part(high)),
167 sample_view::little_endian(part(samples)), word_view::little_endian(part(sparse)),
168 profile_detail::add(header.record_count / P::codec_block_size +
169 (header.record_count % P::codec_block_size != 0), 1), layout.universe, layout.low_width};
170 auto result = profile_view<P, Role>::from_sections(part(fc), offsets, metadata);
171 result.validate_offset_metadata();

179 template <class P, stream_role Role> void scan_profile(profile_view<P, Role> const & view) {
180 view.validate_contents();
181 auto const & metadata = view.metadata();
182 auto offsets = view.group_offsets();
183 std::vector<std::uint64_t> residuals;
184 auto groups = view.block_count();
185 if (groups == std::numeric_limits<std::uint64_t>::max() || groups + 1 > residuals.max_size())
186 throw std::length_error("Everett scan offset count");
187 residuals.reserve(static_cast<std::size_t>(groups + 1));
188 bit_view data(view.bytes(), profile_detail::multiply(metadata.extent, P::bits_per_unit));
189 std::uint64_t at = 0, previous_units = 0;
190 bit_string previous;
191 for (std::uint64_t ordinal = 0; ordinal != metadata.record_count; ++ordinal) {
192 std::uint64_t retained;
193 if (ordinal % P::codec_block_size == 0) {
194 auto stride = profile_detail::multiply(ordinal, metadata.common_value_width.value_or(0));
195 if (stride > at) throw std::invalid_argument("Everett fixed values exceed physical position");
196 residuals.push_back(at - stride);
197 retained = profile_detail::read_count<P>(data, at);
198 if (retained > previous_units)
199 throw std::invalid_argument("Everett profile absolute prefix exceeds predecessor");
200 } else {
201 auto backspace = profile_detail::read_backspace<P>(data, at);
202 if (backspace > previous_units) throw std::invalid_argument("Everett profile backspace exceeds predecessor");
203 retained = previous_units - backspace;
204 }
205 auto suffix = profile_detail::read_count<P>(data, at);
206 auto values = metadata.common_value_width ? *metadata.common_value_width : profile_detail::read_count<P>(data, at);
207 auto length = profile_detail::add(retained, suffix);
208 auto start_bits = profile_detail::multiply(at, P::bits_per_unit);
209 auto suffix_bits = profile_detail::multiply(suffix, P::bits_per_unit);
210 auto value_bits = profile_detail::multiply(values, P::bits_per_unit);
211 if (start_bits > data.size() || suffix_bits > data.size() - start_bits ||
212 value_bits > data.size() - start_bits - suffix_bits)
213 throw std::invalid_argument("truncated Everett profile record");
214 auto retained_bits = profile_detail::multiply(retained, P::bits_per_unit);
215 // The retained prefix is already equal. Ordinary FC must differ at
216 // the first new unit, or extend the entire predecessor; no inherited
217 // prefix needs copying or comparison to establish maximality/order.
218 if (retained < previous_units) {
219 if (!suffix) throw std::invalid_argument("Everett profile key precedes its predecessor");
220 auto before = profile_detail::load_bits(previous.view(), retained_bits, P::bits_per_unit);
221 auto after = profile_detail::load_bits(data, start_bits, P::bits_per_unit);
222 if (after <= before)
223 throw std::invalid_argument("Everett profile is not sorted ordinary front coding");
224 } else if (ordinal && !suffix && Role == stream_role::native)
225 throw std::invalid_argument("Everett native profile keys must be unique");
226 profile_detail::resize(previous, retained_bits);
227 profile_detail::append(previous, data.subview(start_bits, suffix_bits));
228 at = profile_detail::add(at, profile_detail::add(suffix, values));
229 previous_units = length;
230 }
231 if (at != metadata.extent || previous_units != metadata.terminal_key_units)
232 throw std::invalid_argument("Everett terminal profile extent or length mismatch");
233 auto stride = profile_detail::multiply(metadata.record_count, metadata.common_value_width.value_or(0));
234 if (stride > at) throw std::invalid_argument("Everett fixed payload exceeds extent");
235 residuals.push_back(at - stride);
236 auto expected = elias_fano::build(residuals);
237 if (expected.universe != offsets.universe() || expected.low_width != offsets.low_width())
238 throw std::invalid_argument("noncanonical Everett Elias-Fano parameters");
239 equal_words(offsets.low_words(), expected.low, "Everett Elias-Fano low words mismatch");
240 equal_words(offsets.high_words(), expected.high, "Everett Elias-Fano high words mismatch");
241 equal_words(offsets.sparse_words(), expected.sparse, "Everett Elias-Fano sparse words mismatch");
242 auto actual_samples = offsets.samples();
243 if (actual_samples.size() != expected.samples.size()) throw std::invalid_argument("Everett Elias-Fano sample count mismatch");
244 for (std::size_t i = 0; i != expected.samples.size(); ++i) {
245 auto a = actual_samples[i];
246 auto b = expected.samples[i];
247 if (a.first != b.first || a.sparse != b.sparse) throw std::invalid_argument("Everett Elias-Fano sample mismatch");
248 }
249 }

358 template <class Index> void index(Index const & source, object_id const & native_id,
359 std::optional<blob_identity> const & exact_target) {
360 profile(source.borrowed());
361 words(source.interleave().classes);
362 words(source.interleave().checkpoints);
363 sections_.push_back(source.false_borrow_bits());
364 words(source.cut_lcps());
365 file_detail::put(directory_, 40, 8, source.virtual_size());
366 section_detail::put_id(directory_, 48, native_id);
367 if (exact_target) {
368 file_detail::put(directory_, 33, 1, 1);
369 section_detail::put_id(directory_, 64, exact_target->native);
370 section_detail::put_id(directory_, 80, exact_target->index);
371 }
372 finish(file_kind::fractional_index);
373 }
374 void finish(file_kind kind) {
375 header_.kind = kind;
376 bool index = kind == file_kind::fractional_index;
377 directory_size_ = index ? section_detail::index_directory_bytes : section_detail::native_directory_bytes;
378 auto descriptors = index ? section_detail::index_descriptor_offset : section_detail::native_descriptor_offset;
379 auto magic = index ? "IX02" : "KV02";
380 for (unsigned i = 0; i != 4; ++i) directory_[i] = std::byte(magic[i]);
381 file_detail::put(directory_, 4, 2, section_detail::version);
382 file_detail::put(directory_, 6, 2, sections_.size());
383 std::uint64_t end = directory_size_;
384 for (std::size_t i = 0; i != sections_.size(); ++i) {
385 if (end > std::numeric_limits<std::uint64_t>::max() - 7)
386 throw std::overflow_error("Everett section alignment overflows");
387 auto start = section_detail::align(end);
388 end = profile_detail::add(start, sections_[i].size());
389 file_detail::put(directory_, descriptors + (i << 4), 8, start);
390 file_detail::put(directory_, descriptors + (i << 4) + 8, 8, sections_[i].size());
391 }
392 header_.extent = profile_detail::multiply(end, 1u << (3 - P::unit_shift));
393 file_detail::validate_metadata(header_);
394 }
