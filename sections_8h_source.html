74 inline unsigned hex_value(char c) noexcept { return unsigned(c <= '9' ? c - '0' : c - 'a' + 10); }
75 inline void put_id(std::span<std::byte> bytes, std::size_t at, object_id const & id) {
76 auto const & text = id.hex();
77 if (text.size() != 32) throw std::invalid_argument("invalid Diet section object identity");
78 if (at > bytes.size() || 16 > bytes.size() - at)
79 throw std::out_of_range("Diet section identity position");
80 for (std::size_t i = 0; i != 16; ++i)
81 bytes[at + i] = std::byte((hex_value(text[2 * i]) << 4) | hex_value(text[2 * i + 1]));
82 }
83 inline object_id get_id(std::span<std::byte const> bytes, std::size_t at) {
84 constexpr char hex[] = "0123456789abcdef";
85 std::string text(32, '0');
86 for (std::size_t i = 0; i != 16; ++i) {
87 auto byte = std::to_integer<unsigned>(bytes[at + i]);
88 text[2 * i] = hex[byte >> 4];
89 text[2 * i + 1] = hex[byte & 15];
90 }
91 return object_id(std::move(text));

102 template <class P> directory parse(file_header<P> const & header, std::span<std::byte const> body) {
103 directory result;
104 bool index = header.kind == file_kind::fractional_index;
105 result.count = index ? 9 : 5;
106 result.bytes = index ? index_directory_bytes : native_directory_bytes;
107 auto descriptors = index ? index_descriptor_offset : native_descriptor_offset;
108 if (body.size() > std::numeric_limits<std::uint64_t>::max() - 7)
109 throw std::invalid_argument("Diet section container extent overflows");
110 if (body.size() < result.bytes) throw std::invalid_argument("truncated Diet section directory");
111 if (header.extent != profile_detail::multiply(body.size(), 1u << (3 - P::unit_shift)))
112 throw std::invalid_argument("Diet section container must occupy complete bytes");
113 auto magic = index ? "IX02" : "KV02";
114 for (unsigned i = 0; i != 4; ++i)
115 if (std::to_integer<unsigned>(body[i]) != unsigned(magic[i]))
116 throw std::invalid_argument("unexpected Diet section magic");
117 if (file_detail::get(body, 4, 2) != version || file_detail::get(body, 6, 2) != result.count)
118 throw std::invalid_argument("unsupported Diet section directory");
119 result.extent = file_detail::get(body, 8, 8);
120 result.terminal_key_units = file_detail::get(body, 16, 8);
121 result.universe = file_detail::get(body, 24, 8);
122 result.low_width = unsigned(file_detail::get(body, 32, 1));
123 if (index) {
124 auto target = file_detail::get(body, 33, 1);
125 if (target > 1) throw std::invalid_argument("invalid Diet target flag");
126 zero(body.subspan(34, 6), "nonzero Diet directory reserved bytes");
127 zero(body.subspan(96, 16), "nonzero Diet directory reserved bytes");
128 result.virtual_count = file_detail::get(body, 40, 8);
129 if (result.virtual_count < header.record_count)
130 throw std::invalid_argument("borrowed records exceed virtual count");
131 result.native_id = get_id(body, 48);
132 if (target) result.target_id = blob_identity{get_id(body, 64), get_id(body, 80)};
133 else zero(body.subspan(64, 32), "absent Diet target must encode zero");
134 } else zero(body.subspan(33, 15), "nonzero Diet directory reserved bytes");
135 std::uint64_t end = result.bytes;
136 for (std::size_t i = 0; i != result.count; ++i) {
137 auto at = descriptors + i * descriptor_bytes;
138 descriptor part{file_detail::get(body, at, 8), file_detail::get(body, at + 8, 8)};
139 if (part.offset != align(end) || part.offset > body.size() || part.length > body.size() - part.offset)
140 throw std::invalid_argument("noncanonical Diet section range");
141 end = part.offset + part.length;
142 result.sections[i] = part;
143 }
144 if (end != body.size()) throw std::invalid_argument("trailing Diet section bytes");
145 return result;
146 }
147 inline void validate_gaps(std::span<std::byte const> body, directory const & layout) {
148 std::size_t end = layout.bytes;
149 for (std::size_t i = 0; i != layout.count; ++i) {
150 auto const & part = layout.sections[i];
151 zero(body.subspan(end, static_cast<std::size_t>(part.offset) - end), "nonzero Diet section alignment padding");
152 end = static_cast<std::size_t>(part.offset + part.length);
153 }

177 template <class P, stream_role Role> void scan_profile(profile_view<P, Role> const & view) {
178 view.validate_contents();
179 auto const & metadata = view.metadata();
180 auto offsets = view.group_offsets();
181 std::vector<std::uint64_t> residuals;
182 auto groups = view.block_count();
183 if (groups == std::numeric_limits<std::uint64_t>::max() || groups + 1 > residuals.max_size())
184 throw std::length_error("Diet scan offset count");
185 residuals.reserve(static_cast<std::size_t>(groups + 1));
186 bit_view data(view.bytes(), profile_detail::multiply(metadata.extent, P::bits_per_unit));
187 std::uint64_t at = 0, previous_units = 0;
188 bit_string previous;
189 for (std::uint64_t ordinal = 0; ordinal != metadata.record_count; ++ordinal) {
190 std::uint64_t retained;
191 if (ordinal % P::codec_block_size == 0) {
192 auto stride = profile_detail::multiply(ordinal, metadata.common_value_width.value_or(0));
193 if (stride > at) throw std::invalid_argument("Diet fixed values exceed physical position");
194 residuals.push_back(at - stride);
195 retained = profile_detail::read_count<P>(data, at);
196 if (retained > previous_units)
197 throw std::invalid_argument("Diet profile absolute prefix exceeds predecessor");
198 } else {
199 auto backspace = profile_detail::read_backspace<P>(data, at);
200 if (backspace > previous_units) throw std::invalid_argument("Diet profile backspace exceeds predecessor");
201 retained = previous_units - backspace;
202 }
203 auto suffix = profile_detail::read_count<P>(data, at);
204 auto values = metadata.common_value_width ? *metadata.common_value_width : profile_detail::read_count<P>(data, at);
205 auto length = profile_detail::add(retained, suffix);
206 auto start_bits = profile_detail::multiply(at, P::bits_per_unit);
207 auto suffix_bits = profile_detail::multiply(suffix, P::bits_per_unit);
208 auto value_bits = profile_detail::multiply(values, P::bits_per_unit);
209 if (start_bits > data.size() || suffix_bits > data.size() - start_bits ||
210 value_bits > data.size() - start_bits - suffix_bits)
211 throw std::invalid_argument("truncated Diet profile record");
212 auto retained_bits = profile_detail::multiply(retained, P::bits_per_unit);
213 // The retained prefix is already equal. Ordinary FC must differ at
214 // the first new unit, or extend the entire predecessor; no inherited
215 // prefix needs copying or comparison to establish maximality/order.
216 if (retained < previous_units) {
217 if (!suffix) throw std::invalid_argument("Diet profile key precedes its predecessor");
218 auto before = profile_detail::load_bits(previous.view(), retained_bits, P::bits_per_unit);
219 auto after = profile_detail::load_bits(data, start_bits, P::bits_per_unit);
220 if (after <= before)
221 throw std::invalid_argument("Diet profile is not sorted ordinary front coding");
222 } else if (ordinal && !suffix && Role == stream_role::native)
223 throw std::invalid_argument("Diet native profile keys must be unique");
224 profile_detail::resize(previous, retained_bits);
225 profile_detail::append(previous, data.subview(start_bits, suffix_bits));
226 at = profile_detail::add(at, profile_detail::add(suffix, values));
227 previous_units = length;
228 }
229 if (at != metadata.extent || previous_units != metadata.terminal_key_units)
230 throw std::invalid_argument("Diet terminal profile extent or length mismatch");
231 auto stride = profile_detail::multiply(metadata.record_count, metadata.common_value_width.value_or(0));
232 if (stride > at) throw std::invalid_argument("Diet fixed payload exceeds extent");
233 residuals.push_back(at - stride);
234 auto expected = elias_fano::build(residuals);
235 if (expected.universe != offsets.universe() || expected.low_width != offsets.low_width())
236 throw std::invalid_argument("noncanonical Diet Elias-Fano parameters");
237 equal_words(offsets.low_words(), expected.low, "Diet Elias-Fano low words mismatch");
238 equal_words(offsets.high_words(), expected.high, "Diet Elias-Fano high words mismatch");
239 equal_words(offsets.sparse_words(), expected.sparse, "Diet Elias-Fano sparse words mismatch");
240 auto actual_samples = offsets.samples();
241 if (actual_samples.size() != expected.samples.size()) throw std::invalid_argument("Diet Elias-Fano sample count mismatch");
242 for (std::size_t i = 0; i != expected.samples.size(); ++i) {
243 auto a = actual_samples[i];
244 auto b = expected.samples[i];
245 if (a.first != b.first || a.sparse != b.sparse) throw std::invalid_argument("Diet Elias-Fano sample mismatch");
246 }
247 }

354 template <class Index> void index(Index const & source, object_id const & native_id,
355 std::optional<blob_identity> const & exact_target) {
356 profile(source.borrowed());

358 words(source.interleave().checkpoints);

360 words(source.cut_lcps());

370 void finish(file_kind kind) {
371 header_.kind = kind;
372 bool index = kind == file_kind::fractional_index;
373 directory_size_ = index ? section_detail::index_directory_bytes : section_detail::native_directory_bytes;
374 auto descriptors = index ? section_detail::index_descriptor_offset : section_detail::native_descriptor_offset;

376 for (unsigned i = 0; i != 4; ++i) directory_[i] = std::byte(magic[i]);
377 file_detail::put(directory_, 4, 2, section_detail::version);
378 file_detail::put(directory_, 6, 2, sections_.size());

390 }
