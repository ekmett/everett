48 void append(bit_view key, bit_view value, std::optional<std::uint64_t> retained_limit_bits = {}) {
49 output_.require_active();
50 if ((key.size() & (P::bits_per_unit - 1)) || (value.size() & (P::bits_per_unit - 1)))
51 error_detail::raise<std::invalid_argument>("native file record disagrees with policy units");
52 auto common = output_.common_value_width();
53 if (common && (value.size() >> P::unit_shift) != *common)
54 error_detail::raise<std::invalid_argument>("native file value disagrees with common width");
55 auto comparison = compare_common_bits<typename P::architecture>(previous_.view(), key);
56 if (size() && comparison.order >= 0)
57 error_detail::raise<std::invalid_argument>("native file keys must be strictly increasing");
58 auto retained = std::min(comparison.common_bits, retained_limit_bits.value_or(comparison.common_bits)) >> P::unit_shift;
59 auto retained_bits = retained << P::unit_shift;
60 if (key.size() > std::numeric_limits<std::uint64_t>::max() - 7)
61 error_detail::raise<std::length_error>("native file key is too large");
62 auto bytes = profile_detail::byte_count(key.size());
63 if (bytes > previous_.bytes.max_size())
64 error_detail::raise<std::length_error>("native file key is too large");
65 if (bytes > previous_.bytes.capacity()) {
66 auto capacity = previous_.bytes.capacity(), maximum = previous_.bytes.max_size();
67 auto grown = capacity > maximum / 2 ? maximum : 2 * capacity;
68 previous_.bytes.reserve(std::max(static_cast<std::size_t>(bytes), grown));
69 }
70 auto literal = key.subview(retained_bits, key.size() - retained_bits);
71 output_.append(retained, literal, value);
72 profile_detail::resize(previous_, key.size());
73 profile_detail::copy_into(previous_, retained_bits, literal);
74 }
