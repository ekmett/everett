45 void append(bit_view key, bit_view value) {
46 output_.require_active();
47 if ((key.size() & (P::bits_per_unit - 1)) || (value.size() & (P::bits_per_unit - 1)))
48 error_detail::raise<std::invalid_argument>("native file record disagrees with policy units");
49 auto common = output_.common_value_width();
50 if (common && (value.size() >> P::unit_shift) != *common)
51 error_detail::raise<std::invalid_argument>("native file value disagrees with common width");
52 auto comparison = compare_common_bits(previous_.view(), key);
53 if (size() && comparison.order >= 0)
54 error_detail::raise<std::invalid_argument>("native file keys must be strictly increasing");
55 auto retained = comparison.common_bits >> P::unit_shift;
56 auto retained_bits = retained << P::unit_shift;
57 if (key.size() > std::numeric_limits<std::uint64_t>::max() - 7)
58 error_detail::raise<std::length_error>("native file key is too large");
59 auto bytes = profile_detail::byte_count(key.size());
60 if (bytes > previous_.bytes.max_size())
61 error_detail::raise<std::length_error>("native file key is too large");
62 if (bytes > previous_.bytes.capacity()) {
63 auto capacity = previous_.bytes.capacity(), maximum = previous_.bytes.max_size();
64 auto grown = capacity > maximum / 2 ? maximum : 2 * capacity;
65 previous_.bytes.reserve(std::max(static_cast<std::size_t>(bytes), grown));
66 }
67 auto literal = key.subview(retained_bits, key.size() - retained_bits);
68 output_.append(retained, literal, value);
69 profile_detail::resize(previous_, key.size());
70 profile_detail::copy_into(previous_, retained_bits, literal);
71 }
