159 // Restore the frontier, not a mutable continuation. The supplied mapped or
160 // built graph must already be trusted/admitted. Only metadata is inspected.
161 // Empty routing nodes are omitted from the oldest-first interval list.
162 static cola_runtime_snapshot restore(pair_type head, std::span<cola_runtime_interval const> intervals) {
163 auto query = query_type::adopt_prepared(std::move(head));
164 std::vector<pair_type> sources;
165 for (auto p = query.head(); p; p = p->main_target())
166 if (p->native_owner()->size()) sources.push_back(p);
167 if (sources.size() != intervals.size())
168 error_detail::raise<std::invalid_argument>("runtime interval count differs from query graph");
169 std::vector<run_type> runs;
170 std::uint64_t next = 0, previous = 0;
171 for (std::size_t i = 0; i != intervals.size(); ++i) {
172 auto span = intervals[i];
173 if (span.first != next || span.last <= span.first || !std::has_single_bit(span.mass()) ||
174 (i && span.mass() >= previous && !(i + 1 == intervals.size() && previous == 1 && span.mass() == 1)))
175 error_detail::raise<std::invalid_argument>("invalid runtime admission frontier");
176 auto source = sources[sources.size() - i - 1];
177 if (source->native_owner()->size() > span.mass())
178 error_detail::raise<std::invalid_argument>("runtime native exceeds admission mass");
179 runs.push_back({span.first, span.last, std::move(source)});
180 next = span.last; previous = span.mass();
181 }
182 return make(std::move(query), std::move(runs), next);
183 }

210 // Low-level ACTIVE encoded executor. Compose is supplied by the active
211 // registry handler; it is not the world's semantic type. Existing FC records
212 // are accepted here, not arbitrary per-sort physical grammars.
213 //
214 // An admission is immediately queryable. One private binary carry chain may
215 // remain. The next admission drains that chain first, providing conservative
216 // backpressure. Extra service changes only equivalent immutable layouts.
217 // Credits pay real operations; finalization requires its whole structural
218 // allowance and executes atomically, so single-call latency is not bounded.
219 // Exceptions during execution poison the executor while retaining its last
220 // published snapshot and private input pins. Invalid input preflight does not.

242 // Metadata-only structural cost of a ready singleton admission, excluding
243 // optional service. This does not bound key bytes, callbacks or latency.
244 std::uint64_t admission_cost() const {
245 auto const & e = active();
246 if (e.job) error_detail::raise<std::logic_error>("runtime admission requires service");
247 auto r = e.published.runs();
248 auto head = e.published.query_root().head();
249 auto borrowed = head->group_count();
250 auto n = execution::add(9, profile_detail::multiply(execution::add(borrowed, 1), P::group_size + 6));
251 n = execution::add(n, execution::add(execution::blocks(borrowed), 5));
252 n = execution::add(n, execution::add(execution::add(r.size(), head->depth()), 3));
253 if (!r.empty() && r.back().mass() == 1) n = execution::add(n, execution::add(r.size(), 2));
254 return n;
255 }

275 // A batch is nondecreasing by encoded key; equal keys are chronological
276 // separate admissions. Preflight precedes any work; a later execution
277 // failure rolls the entire batch's visible state back and poisons this handle.
278 snapshot_type contribute(std::span<profile_record const> records, std::uint64_t service_budget = 64) {
279 auto & e = writable();
280 (void)profile_detail::add(e.published.admissions(), records.size());
281 bit_view previous;
282 for (std::size_t i = 0; i != records.size(); ++i) {
283 auto key = records[i].key.view(), value = records[i].value.view();
284 if ((key.size() & (P::bits_per_unit - 1)) || (value.size() & (P::bits_per_unit - 1)) ||
285 (P::value_width && (value.size() >> P::unit_shift) != *P::value_width) ||
286 (i && compare_bits(previous, key) > 0))
287 error_detail::raise<std::invalid_argument>("invalid encoded runtime contribution");
288 previous = key;
289 }
290 auto before = e.published;
291 try {
292 for (auto const & record : records) {
293 e.drain();
294 e.admit(record);
295 if (service_budget && e.job) { e.grant(service_budget); e.serve(); }
296 }
297 } catch (...) { e.published = std::move(before); e.failed = true; throw; }
298 return e.published;
299 }

413 case phase::merge_step: return 3;

427 case phase::merge_step: next.native_allowance = add(next.native_allowance, amount); break;
428 case phase::merge_finish: next.directory_entries = add(next.directory_entries, amount - 1);
429 next.setup = add(next.setup, 1); break;
430 case phase::index_step:

440 void perform() {
441 auto & j = *job;
442 switch (j.stage) {
443 case phase::merge_start:
444 j.merge = std::make_unique<merge_type>(j.older.native_owner(), j.carry, merger());
445 j.stage = j.merge->done() ? phase::merge_finish : phase::merge_step;
446 break;
447 case phase::merge_step: {
448 auto done = j.merge->step(1);
449 work.native_input_records = add(work.native_input_records, done.input_records);
450 work.native_output_records = add(work.native_output_records, done.keys);
451 if (j.merge->done()) j.stage = phase::merge_finish;
452 break;
453 }
454 case phase::merge_finish:
455 j.carry = native_type::from_owned(j.merge->finish()); j.merge.reset();
456 work.native_merges = add(work.native_merges, 1);
457 j.first = j.older.first;
458 if (!j.prefix.empty() && j.prefix.back().mass() == j.last - j.first) {
459 j.older = j.prefix.back(); j.prefix.pop_back(); j.target = j.older.node->main_target();
460 j.stage = phase::merge_start;
461 } else {
462 j.target = near(std::move(j.target), j.last - j.first);
463 j.stage = phase::index_start;
464 }
465 break;
466 case phase::index_start:
467 j.index = std::make_unique<index_type>(j.carrier ? empty : j.carry, j.target);
468 j.stage = j.index->done() ? phase::index_finish : phase::index_step;
469 break;
470 case phase::index_step:
471 j.index->step(1);
472 if (j.index->done()) j.stage = phase::index_finish;
473 break;
474 case phase::index_finish:
475 j.head = node_type::from_built(j.index->finish()); j.index.reset();
476 work.indexes = add(work.indexes, 1);
477 if (j.carrier) work.carriers = add(work.carriers, 1);
478 else j.result_node = j.head;
479 if (j.head->virtual_size() > P::group_size) {
480 j.target = j.head; j.carrier = true; j.stage = phase::index_start;
481 } else j.stage = phase::publish;
482 break;
483 case phase::publish: {
484 j.prefix.push_back({j.first, j.last, j.result_node});
485 auto result = snapshot_type::make(query_type::adopt_prepared(j.head), std::move(j.prefix), j.last);
486 published = std::move(result); job.reset(); credit = 0;
487 break;
488 }
489 }
490 }
491 void serve() {
492 while (job) {
493 auto amount = price();
494 if (amount > credit) break;
495 charge(amount); perform();
496 }
497 }
498 void drain() {
499 while (job) {
500 auto amount = price();
501 if (credit < amount) grant(amount - credit);
502 serve();
503 }
504 }
505 };

507 explicit cola_runtime(std::unique_ptr<execution> value) : execution_(std::move(value)) {}
508 execution const & active() const {
509 if (!execution_) error_detail::raise<std::logic_error>("moved-from COLA runtime");
510 return *execution_;
511 }
512 execution & writable() {
513 (void)active();
514 if (execution_->failed) error_detail::raise<std::logic_error>("failed COLA runtime");
515 return *execution_;
516 }
517 };
