36 // Prepare every slot before acquiring a batch. Once any claim is held,
37 // try_lock is the only permitted acquisition: unrelated producers may be
38 // holding a parent while waiting for one of our dependencies.
39 // A locked producer is thread-affine: move it only within that thread,
40 // and install, release or destroy it on the thread which acquired it.
41 struct producer {
42 producer(producer const &) = delete;
43 producer & operator=(producer const &) = delete;
44 producer(producer &&) noexcept = default;
45 producer & operator=(producer &&) = delete;
46 bool try_lock() { return lock_.try_lock(); }
47 pointer value() const {
48 if (!lock_.owns_lock()) throw std::logic_error("unclaimed catalog owner");
49 return entry_->value;
50 }
51 void release() noexcept { if (lock_.owns_lock()) lock_.unlock(); }
52 void install(pointer value) {
53 if (!lock_.owns_lock() || entry_->value || !value)
54 throw std::logic_error("invalid catalog owner installation");
55 entry_->value = std::move(value);
56 lock_.unlock();
57 }

79 // A producer installs only an acknowledged immutable result. An exception
80 // leaves the slot empty: another healthy backend can make a fresh attempt.
81 // Producers may resolve immediate dependencies in an acyclic owner graph,
82 // but must not retain locks on unrelated roots. Queries never use this lock.
83 template <class F> pointer get_or_create(object_id const & catalog,
84 std::filesystem::path const & root, F && produce) const {
85 auto entry = locate(catalog, root, true);
86 std::lock_guard lock(entry->mutex);
87 if (!entry->value) {
88 pointer value = std::invoke(std::forward<F>(produce));
89 if (!value) throw std::logic_error("null catalog owner binding");
90 entry->value = std::move(value);
91 }
92 return entry->value;
93 }

99 std::shared_ptr<slot> locate(object_id const & catalog, std::filesystem::path const & root, bool create) const {
100 std::lock_guard lock(mutex_);
101 for (auto const & entry : entries_)
102 if (entry->catalog == catalog && entry->root == root) return entry;
103 if (!create) return {};
104 auto entry = std::make_shared<slot>(catalog, root);
105 entries_.push_back(entry);
106 return entry;
107 }
