22 // Native runs are oldest first. The heap orders equal keys by that age,
23 // so noncommutative arrows are applied in exactly their original order.
24 // This is a full scan, including earlier sorts before reaching S, not an
25 // indexed range seek. It retains one reconstructed key per native run and
26 // at most one resolved output row. Values continue borrowing native bytes.
27 // Each step unit consumes one physical record; key bytes, callbacks and
28 // heap comparisons are additional costs. Construction decodes each run's
29 // first key. The captured world keeps all mappings and cursors alive.
30 template <class S, class World> struct typed_scan {
31 using policy_type = typename World::policy_type;
32 using row_type = typed_row<S>;
33 using semantics = sort_semantics<S>;
34 using native_type = typename World::runtime_family::native_type;
35 using key_transport = typename World::key_transport;
36
37 explicit typed_scan(World snapshot) : snapshot_(std::move(snapshot)) {
38 prefix_ = key_transport::template prefix<S>();
39 auto runs = snapshot_.runtime().runs();
40 sources_.reserve(runs.size()); heap_.reserve(runs.size());
41 for (auto const & run : runs) {
42 auto native = [&] {
43 if constexpr (requires { run.native_owner(); }) return run.native_owner();
44 else return run->native;
45 }();
46 sources_.emplace_back(std::move(native));
47 if (!sources_.back().cursor.done()) heap_.push_back(sources_.size() - 1);
48 }
49 std::make_heap(heap_.begin(), heap_.end(), later());
50 finished_ = heap_.empty();
51 }

61 row_type take_row() {
62 if (!row_) throw std::logic_error("typed scan has no row");
63 auto result = std::move(*row_); row_.reset(); return result;
64 }
65 std::optional<row_type> next() {
66 while (!has_row() && !done()) step(256);
67 return has_row() ? std::optional<row_type>{take_row()} : std::nullopt;
68 }
69 std::uint64_t step(std::uint64_t budget) {
70 if (failed_) throw std::logic_error("failed typed scan");
71 if (!budget || done() || has_row()) return 0;
72 std::uint64_t used = 0;
73 try {
74 while (used != budget && !finished_ && !row_) {
75 if (heap_.empty()) { finish_group(); finished_ = true; break; }
76 auto current = sources_[heap_.front()].cursor.peek();
77 if (group_ && compare_bits(group_key_.view(), current.key.prefix) != 0) {
78 finish_group();
79 if (row_) break;
80 }
81 if (!group_) {
82 auto order = compare_prefix(current.key.prefix);
83 if (order > 0) { finished_ = true; break; }
84 if (order < 0) { consume(); ++used; continue; }
85 group_key_ = bit_string::copy(current.key.prefix);
86 auto key = key_transport::template decode<S>(group_key_.view().subview(prefix_.bit_size,
87 group_key_.bit_size - prefix_.bit_size));
88 auto value = semantics::initial(key);
89 group_.emplace(row_type{std::move(key), std::move(value)});
90 }
91 if constexpr (typed_detail::replacement<S>) last_value_ = current.value;
92 else group_->value = semantics::apply(group_->key, std::move(group_->value),
93 typed_detail::value<policy_type, S>(current.value));
94 consume(); ++used;
95 if (heap_.empty() || compare_bits(group_key_.view(), sources_[heap_.front()].cursor.peek().key.prefix) != 0)
96 finish_group();
97 if (heap_.empty()) finished_ = true;
98 }
99 } catch (...) { failed_ = true; throw; }
100 return used;
101 }

119 auto later() const {
120 return [this](std::size_t a, std::size_t b) {
121 auto order = compare_bits(sources_[a].cursor.peek().key.prefix, sources_[b].cursor.peek().key.prefix);
122 return order ? order > 0 : a > b;
123 };
124 }
125 int compare_prefix(bit_view key) const {
126 auto count = std::min(key.size(), prefix_.bit_size);
127 auto order = compare_bits(key.subview(0, count), prefix_.view().subview(0, count));
128 return order ? order : key.size() < prefix_.bit_size ? -1 : 0;
129 }
130 void consume() {
131 std::pop_heap(heap_.begin(), heap_.end(), later());
132 auto which = heap_.back(); heap_.pop_back();
133 auto comparison = sources_[which].cursor.advance_comparison();
134 if (comparison && comparison->order >= 0)
135 throw std::invalid_argument("scanned native keys are not unique and sorted");
136 ++consumed_;
137 if (!sources_[which].cursor.done()) {
138 heap_.push_back(which); std::push_heap(heap_.begin(), heap_.end(), later());
139 }
140 }
141 void finish_group() {
142 if (!group_) return;
143 if constexpr (typed_detail::replacement<S>)
144 group_->value = semantics::apply(group_->key, std::move(group_->value),
145 typed_detail::value<policy_type, S>(last_value_));
146 if (semantics::present(group_->key, group_->value)) row_.emplace(std::move(*group_));
147 group_.reset();
148 }
