52 std::uint64_t operator[](std::size_t index) const {
53 if (index >= size()) error_detail::raise<std::out_of_range>("word section index");
54 std::uint64_t value;
55 std::memcpy(&value, bytes_.data() + (index << 3), 8);
56 if constexpr (std::endian::native == std::endian::big) {
57 if (little_) {
58 value = ((value & 0x00ff00ff00ff00ffull) << 8) | ((value >> 8) & 0x00ff00ff00ff00ffull);
59 value = ((value & 0x0000ffff0000ffffull) << 16) | ((value >> 16) & 0x0000ffff0000ffffull);
60 value = (value << 32) | (value >> 32);
61 }
62 }

64 }
