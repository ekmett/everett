31 // Children precede their parents. Decoding can reject forward references
32 // before constructing a cycle of owning shared pointers.
33 static std::vector<object_pointer> objects(frontier_type const & frontier) {
34 std::vector<object_pointer> result;
35 std::unordered_map<std::uint64_t, object_pointer> seen;
36 std::unordered_set<std::uint64_t> active;
37 auto visit = [&](auto && self, object_pointer const & value) -> void {
38 if (!value) return;
39 if (auto found = seen.find(value->identity); found != seen.end()) {
40 if (found->second != value) throw std::invalid_argument("duplicate checkpoint object identity");
41 return;
42 }
43 if (!value->identity || !active.insert(value->identity).second)
44 throw std::invalid_argument("cyclic checkpoint object routes");
45 self(self, value->next.main); self(self, value->next.secondary);
46 active.erase(value->identity); seen.emplace(value->identity, value); result.push_back(value);
47 };
48 auto routes = [&](auto const & value) { visit(visit, value.main); visit(visit, value.secondary); };
49 routes(frontier.root);
50 for (auto const & level : frontier.levels) {
51 for (auto const & slot : level.slots) { visit(visit, slot.object); routes(slot.route); }
52 if (level.job) {
53 auto const & job = *level.job;
54 routes(job.destination_route); visit(visit, job.existing_main); visit(visit, job.output);
55 }
56 }
57 return result;
58 }
59 // A validated snapshot owns every object in exactly one level slot.
60 // Its routes point to the next level; job outputs occupy destination slots.
61 // Descending levels therefore visit children before parents without a
62 // closure allocation. Raw frontiers still use the defensive walk above.
63 template <class F> static void for_each_object(snapshot_type const & source, F && visit) {
64 auto const & levels = source.frontier().levels;
65 for (auto level = levels.rbegin(); level != levels.rend(); ++level)
66 for (auto const & slot : level->slots) if (slot.object) visit(slot.object);
67 }
68 static std::size_t object_count(snapshot_type const & source) noexcept {
69 std::size_t count = 0;
70 for_each_object(source, [&](auto const &) { ++count; });
71 return count;
72 }
