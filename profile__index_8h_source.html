36 // An owning index artifact, independent of native storage. Construction uses
37 // a pinned source and trusted samples of one exact target. The caller keeps
38 // those dependencies pinned through sealing and publication of this artifact.
39 template <class P> struct profile_index {
40 using policy_type = P;
41 using borrowed_array = profile_array<P, stream_role::borrowed>;
42 static constexpr std::uint64_t group_size = P::group_size;
43
44 // A terminal pair has no borrowed occurrences. Construct its zero
45 // directories from the admitted native count without visiting native keys.
46 static profile_index native_only(std::uint64_t count) {
47 auto groups = count / group_size + (count % group_size != 0);
48 std::vector<std::uint64_t> cuts;
49 if (groups > cuts.max_size()) error_detail::raise<std::length_error>("native index is too large");
50 cuts.resize(static_cast<std::size_t>(groups), 0);
51 auto ranks = rank_groups<group_size>::build(cuts, count);
52 return profile_index(borrowed_array::build({}), std::move(ranks), {}, std::move(cuts), count);
53 }
