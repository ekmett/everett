33 // An owning index artifact, independent of native storage. Construction uses
34 // a pinned source and trusted samples of one exact target. The caller keeps
35 // those dependencies pinned through sealing and publication of this artifact.
36 template <class P> struct profile_index {
37 using policy_type = P;
38 using borrowed_array = profile_array<P, stream_role::borrowed>;
39 static constexpr std::uint64_t group_size = P::group_size;
40
41 // A terminal pair has no borrowed occurrences. Construct its zero
42 // directories from the admitted native count without visiting native keys.
43 static profile_index native_only(std::uint64_t count) {
44 auto groups = count / group_size + (count % group_size != 0);
45 std::vector<std::uint64_t> cuts;
46 if (groups > cuts.max_size()) error_detail::raise<std::length_error>("native index is too large");
47 cuts.resize(static_cast<std::size_t>(groups), 0);
48 auto ranks = rank_groups<group_size>::build(cuts, count);
49 return profile_index(borrowed_array::build({}), std::move(ranks), {}, std::move(cuts), count);
50 }
