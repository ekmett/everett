39 // Prefix helpers take 0..512 bits. Vector variants require eight readable
40 // words; rank_view keeps shorter allocation tails on the bounded path.
41 // Read only words contributing to the prefix, including a masked tail.
42 inline unsigned prefix512_portable(std::uint64_t const * words, unsigned bits) noexcept {
43 unsigned result = 0;
44 for (unsigned word = 0; word < (bits >> 6); ++word)
45 result += unsigned(std::popcount(words[word]));
46 if (bits & 63)
47 result += unsigned(std::popcount(words[bits >> 6] & ((std::uint64_t{1} << (bits & 63)) - 1)));
48 return result;
49 }

51#if defined(__aarch64__) && defined(__ARM_NEON)
52 template <unsigned Vector> inline uint8x16_t prefix512_vector(
53 std::uint64_t const * words, unsigned bits) noexcept {
54 uint64x2_t positions{2 * Vector, 2 * Vector + 1};
55 auto boundary = vdupq_n_u64(bits >> 6);
56 auto tail = vdupq_n_u64((std::uint64_t{1} << (bits & 63)) - 1);
57 auto mask = vorrq_u64(vcltq_u64(positions, boundary),
58 vandq_u64(vceqq_u64(positions, boundary), tail));
59 return vcntq_u8(vreinterpretq_u8_u64(vandq_u64(vld1q_u64(words + 2 * Vector), mask)));
60 }

62 // Exactly eight readable words, with arbitrary uint64_t alignment.
63 inline unsigned prefix512_neon(std::uint64_t const * words, unsigned bits) noexcept {
64 auto a = prefix512_vector<0>(words, bits);
65 auto b = prefix512_vector<1>(words, bits);
66 auto c = prefix512_vector<2>(words, bits);
67 auto d = prefix512_vector<3>(words, bits);
68 return vaddlvq_u8(vaddq_u8(vaddq_u8(a, b), vaddq_u8(c, d)));
69 }

72#if defined(__AVX2__)
73 template <unsigned Vector> inline __m256i prefix512_masked_avx2(
74 std::uint64_t const * words, unsigned bits) noexcept {
75 auto positions = _mm256_setr_epi64x(4 * Vector, 4 * Vector + 1, 4 * Vector + 2, 4 * Vector + 3);
76 auto boundary = _mm256_set1_epi64x(bits >> 6);
77 auto tail = _mm256_set1_epi64x(static_cast<long long>((std::uint64_t{1} << (bits & 63)) - 1));
78 auto mask = _mm256_or_si256(_mm256_cmpgt_epi64(boundary, positions),
79 _mm256_and_si256(_mm256_cmpeq_epi64(positions, boundary), tail));
80 return _mm256_and_si256(_mm256_loadu_si256(reinterpret_cast<__m256i const *>(words + 4 * Vector)), mask);
81 }

83 // Exactly eight readable words, with no vector alignment requirement.
84 inline unsigned prefix512_avx2(std::uint64_t const * words, unsigned bits) noexcept {
85 auto lookup = _mm256_setr_epi8(0,1,1,2,1,2,2,3,1,2,2,3,2,3,3,4,
86 0,1,1,2,1,2,2,3,1,2,2,3,2,3,3,4);
87 auto nibble = _mm256_set1_epi8(15);
88 auto population = [&](auto data) {
89 auto lo = _mm256_shuffle_epi8(lookup, _mm256_and_si256(data, nibble));
90 auto hi = _mm256_shuffle_epi8(lookup, _mm256_and_si256(_mm256_srli_epi16(data, 4), nibble));
91 return _mm256_add_epi8(lo, hi);
92 };
93 // Each byte sum is at most sixteen, so both vectors can share one SAD.
94 auto counts = _mm256_add_epi8(population(prefix512_masked_avx2<0>(words, bits)),
95 population(prefix512_masked_avx2<1>(words, bits)));
96 auto sums = _mm256_sad_epu8(counts, _mm256_setzero_si256());
97 auto pair = _mm_add_epi64(_mm256_castsi256_si128(sums), _mm256_extracti128_si256(sums, 1));
98 return unsigned(_mm_cvtsi128_si32(_mm_add_epi64(pair, _mm_srli_si128(pair, 8))));
99 }
100#endif

102#if defined(__AVX512F__) && (defined(__AVX512VPOPCNTDQ__) || defined(__AVX512BW__))
103 inline __m512i prefix512_masked_avx512(std::uint64_t const * words, unsigned bits) noexcept {
104 auto lane = bits >> 6;
105 auto full = __mmask8((1u << lane) - 1);
106 auto boundary = __mmask8(1u << lane); // lane=8 gives no boundary lane.
107 auto data = _mm512_maskz_loadu_epi64(full | boundary, words);
108 auto tail = _mm512_set1_epi64(static_cast<long long>((std::uint64_t{1} << (bits & 63)) - 1));
109 return _mm512_mask_and_epi64(data, boundary, data, tail);
110 }
111#endif

119#if defined(__AVX512F__) && defined(__AVX512BW__)
120 inline unsigned prefix512_avx512bw(std::uint64_t const * words, unsigned bits) noexcept {
121 auto data = prefix512_masked_avx512(words, bits);
122 auto lookup = _mm512_broadcast_i32x4(_mm_setr_epi8(0,1,1,2,1,2,2,3,1,2,2,3,2,3,3,4));
123 auto nibble = _mm512_set1_epi8(15);
124 auto lo = _mm512_shuffle_epi8(lookup, _mm512_and_si512(data, nibble));
125 auto hi = _mm512_shuffle_epi8(lookup, _mm512_and_si512(_mm512_srli_epi16(data, 4), nibble));
126 auto counts = _mm512_add_epi8(lo, hi);
127 return unsigned(_mm512_reduce_add_epi64(_mm512_sad_epu8(counts, _mm512_setzero_si512())));

131 // Exactly eight readable words; no alignment beyond uint64_t is required.
132 inline unsigned popcount512_portable(std::uint64_t const * words) noexcept {
133 unsigned total = 0;
134 for (unsigned i = 0; i < 8; ++i) total += unsigned(std::popcount(words[i]));
135 return total;

138 inline unsigned popcount512(std::uint64_t const * words) noexcept {
139#if defined(__aarch64__) && defined(__ARM_NEON)
140 auto bytes = reinterpret_cast<std::uint8_t const *>(words);
141 auto a = vcntq_u8(vld1q_u8(bytes));
142 auto b = vcntq_u8(vld1q_u8(bytes + 16));
143 auto c = vcntq_u8(vld1q_u8(bytes + 32));
144 auto d = vcntq_u8(vld1q_u8(bytes + 48));
145 // Each byte lane sums to at most 32; the final horizontal sum needs
146 // sixteen bits because a completely full 512-bit run contains 512 ones.
147 return vaddlvq_u8(vaddq_u8(vaddq_u8(a, b), vaddq_u8(c, d)));
148#else
149 return popcount512_portable(words);
150#endif
151 }

153 // Caller supplies 0 <= run <= 3 and independent counts in [0,512].
154 // The stored spacer bits already leave eleven-bit lanes for the sum.
155 constexpr unsigned run_prefix(std::uint32_t packed, unsigned run) noexcept {
156 std::uint64_t selected = packed & ((std::uint64_t{1} << (11 * run)) - 1);
157 return unsigned(((selected * 0x400801ull) >> 22) & 2047u);
158 }

160 // The builder calls this at each 2048-bit block. Kept independent of
161 // payload allocation so counter transitions can be checked at 2^32 bits.
162 struct directory_cursor {
163 static constexpr bool starts_epoch(std::uint64_t block) noexcept {
164 return (block & ((std::uint64_t{1} << 21) - 1)) == 0;
165 }
166 std::uint32_t before(std::uint64_t block, std::uint64_t total) {
167 if (starts_epoch(block)) epoch_base = total;
168 if (total < epoch_base || total - epoch_base > std::numeric_limits<std::uint32_t>::max())
169 throw std::overflow_error("rank relative count");
170 return std::uint32_t(total - epoch_base);
171 }
172 std::uint64_t epoch_base = 0;
173 };
174 }

176 // Borrowed, native-endian/aligned spans. A future file-format reader owns
177 // endian conversion and mapping lifetime. The directory has 64 bits per
178 // 2048 source bits, plus one 64-bit absolute rank per 2^32 source bits.
179 // This structure supports rank only; it contains no select directory.
180 // Construction checks section shapes, not directory contents. Borrowed
181 // metadata must come from the builder or an independently validated reader.
182 struct rank_view {
183 rank_view(std::span<std::uint64_t const> words,
184 std::span<rank_block const> blocks,
185 std::span<std::uint64_t const> supers,
186 std::uint64_t bit_count)
187 : words_(words), blocks_(blocks), supers_(supers),
188 bit_count_(bit_count) {
189 if (bit_count > std::numeric_limits<std::uint64_t>::max() - 0xffffffffu ||
190 words.size() != ((bit_count + 63) >> 6) ||
191 blocks.size() != ((bit_count + 2047) >> 11) ||
192 supers.size() != ((bit_count + 0xffffffffu) >> 32))
193 throw std::invalid_argument("invalid rank spans");
194 }
195
196 std::uint64_t size() const noexcept { return bit_count_; }
197 // The final bit belongs to an existing word, including a partial tail.
198 std::uint64_t count() const {
199 if (!bit_count_) return 0;
200 auto last = bit_count_ - 1;
201 return rank(last) + ((words_[last >> 6] >> (last & 63)) & 1);
202 }
203
204 // Exclusive rank at an existing bit. Use count() for the total population.
205 std::uint64_t rank(std::uint64_t position) const {
206 if (position >= bit_count_) throw std::out_of_range("rank position");
207 auto block = blocks_[position >> 11];
208 unsigned run = unsigned((position >> 9) & 3);
209 std::uint64_t result = supers_[position >> 32] + block.before;
210 result += rank_detail::run_prefix(block.runs, run);
211 auto word = (position >> 9) << 3;
212 auto bits = unsigned(position & 511);
213 if (!bits) return result;
214#if defined(__AVX512F__) && defined(__AVX512VPOPCNTDQ__)
215 if (words_.size() - word >= 8)
216 return result + rank_detail::prefix512_avx512_vpopcnt(words_.data() + word, bits);
217#elif defined(__AVX512F__) && defined(__AVX512BW__)
218 if (words_.size() - word >= 8)
219 return result + rank_detail::prefix512_avx512bw(words_.data() + word, bits);
220#elif defined(__AVX2__)
221 if (words_.size() - word >= 8)
222 return result + rank_detail::prefix512_avx2(words_.data() + word, bits);
223#elif defined(__aarch64__) && defined(__ARM_NEON)
224 if (words_.size() - word >= 8)
225 return result + rank_detail::prefix512_neon(words_.data() + word, bits);
226#endif
227 return result + rank_detail::prefix512_portable(words_.data() + word, bits);
228 }

232 std::span<rank_block const> blocks_;

234 std::uint64_t bit_count_;
235 };

237 struct rank_index {
238 static rank_index build(std::span<std::uint64_t const> source, std::uint64_t bits) {
239 if (bits > std::numeric_limits<std::uint64_t>::max() - 0xffffffffu ||
240 source.size() != ((bits + 63) >> 6))
241 throw std::invalid_argument("rank source length");
242 rank_index result;
243 result.bit_count = bits;
244 result.words.assign(source.begin(), source.end());
245 if (bits & 63) result.words.back() &= (std::uint64_t{1} << (bits & 63)) - 1;
246 result.blocks.resize((bits + 2047) >> 11);
247 rank_detail::directory_cursor cursor;
248 std::uint64_t total = 0;
249 auto begin_block = [&](std::uint64_t block) -> rank_block & {
250 auto & entry = result.blocks[block];
251 entry.before = cursor.before(block, total);
252 if (cursor.starts_epoch(block)) result.supers.push_back(cursor.epoch_base);
253 return entry;
254 };
255 auto full_blocks = bits >> 11;
256 for (std::uint64_t block = 0; block < full_blocks; ++block) {
257 auto & entry = begin_block(block);
258 auto words = result.words.data() + block * 32;
259 auto a = rank_detail::popcount512(words);
260 auto b = rank_detail::popcount512(words + 8);
261 auto c = rank_detail::popcount512(words + 16);
262 auto d = rank_detail::popcount512(words + 24);
263 entry.runs = a | (b << 11) | (c << 22);
264 total += a + b + c + d;
265 }
266 if (full_blocks != result.blocks.size()) {
267 auto & entry = begin_block(full_blocks);
268 unsigned counts[4]{};
269 auto first = full_blocks * 32;
270 // Only the final partial block needs bounded word loads. The owning
271 // copy has already cleared unused bits in its final word.
272 for (auto word = first; word < result.words.size(); ++word)
273 counts[(word - first) >> 3] += unsigned(std::popcount(result.words[word]));
274 entry.runs = counts[0] | (counts[1] << 11) | (counts[2] << 22);
275 total += counts[0] + counts[1] + counts[2] + counts[3];
276 }
277 return result;
278 }
