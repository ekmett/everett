33 // Exactly eight readable words; no alignment beyond uint64_t is required.
34 inline unsigned popcount512_portable(std::uint64_t const * words) noexcept {
35 unsigned total = 0;
36 for (unsigned i = 0; i < 8; ++i) total += unsigned(std::popcount(words[i]));
37 return total;
38 }

40 inline unsigned popcount512(std::uint64_t const * words) noexcept {
41#if defined(__aarch64__) && defined(__ARM_NEON)
42 auto bytes = reinterpret_cast<std::uint8_t const *>(words);
43 auto a = vcntq_u8(vld1q_u8(bytes));
44 auto b = vcntq_u8(vld1q_u8(bytes + 16));
45 auto c = vcntq_u8(vld1q_u8(bytes + 32));
46 auto d = vcntq_u8(vld1q_u8(bytes + 48));
47 // Each byte lane sums to at most 32; the final horizontal sum needs
48 // sixteen bits because a completely full 512-bit run contains 512 ones.
49 return vaddlvq_u8(vaddq_u8(vaddq_u8(a, b), vaddq_u8(c, d)));
50#else
51 return popcount512_portable(words);
52#endif
53 }

55 // Caller supplies 0 <= run <= 3 and independent counts in [0,512].
56 // Widen to eleven-bit lanes before summing: 512+512+512 needs eleven
57 // bits, although each stored population needs only ten.
58 constexpr unsigned run_prefix(std::uint32_t packed, unsigned run) noexcept {
59 std::uint64_t selected = packed & ((std::uint64_t{1} << (10 * run)) - 1);
60 auto widened = (selected & 0x3ffu) |
61 ((selected & 0xffc00u) << 1) |
62 ((selected & 0x3ff00000u) << 2);
63 return unsigned(((widened * 0x400801ull) >> 22) & 2047u);
64 }

66 // The builder calls this at each 2048-bit block. Kept independent of
67 // payload allocation so counter transitions can be checked at 2^32 bits.
68 struct directory_cursor {
69 static constexpr bool starts_epoch(std::uint64_t block) noexcept {
70 return block % (std::uint64_t{1} << 21) == 0;
71 }
72 std::uint32_t before(std::uint64_t block, std::uint64_t total) {
73 if (starts_epoch(block)) epoch_base = total;
74 if (total < epoch_base || total - epoch_base > std::numeric_limits<std::uint32_t>::max())
75 throw std::overflow_error("rank relative count");
76 return std::uint32_t(total - epoch_base);
77 }
78 std::uint64_t epoch_base = 0;
79 };
80 }

82 // Borrowed, native-endian/aligned spans. A future file-format reader owns
83 // endian conversion and mapping lifetime. The directory has 64 bits per
84 // 2048 source bits, plus one 64-bit absolute rank per 2^32 source bits.
85 // This structure supports rank only; it contains no select directory.
86 // Construction checks section shapes, not directory contents. Borrowed
87 // metadata must come from the builder or an independently validated reader.
88 struct rank_view {
89 rank_view(std::span<std::uint64_t const> words,
90 std::span<rank_block const> blocks,
91 std::span<std::uint64_t const> supers,
92 std::uint64_t bit_count, std::uint64_t total)
93 : words_(words), blocks_(blocks), supers_(supers),
94 bit_count_(bit_count), total_(total) {
95 if (words.size() != bit_count / 64 + (bit_count % 64 != 0) ||
96 blocks.size() != bit_count / 2048 + (bit_count % 2048 != 0) ||
97 supers.size() != (bit_count >> 32) + ((bit_count & 0xffffffffu) != 0) ||
98 total > bit_count)
99 throw std::invalid_argument("invalid rank spans");
100 }

102 std::uint64_t size() const noexcept { return bit_count_; }
103 std::uint64_t count() const noexcept { return total_; }

105 // Exclusive rank. The endpoint is valid; positions beyond it are errors.
106 std::uint64_t rank(std::uint64_t position) const {
107 if (position > bit_count_) throw std::out_of_range("rank position");
108 if (position == bit_count_) return total_;
109 auto block = blocks_[position / 2048];
110 unsigned run = unsigned((position / 512) % 4);
111 std::uint64_t result = supers_[position >> 32] + block.before;
112 result += rank_detail::run_prefix(block.runs, run);
113 auto word = (position / 512) * 8;
114 for (; word < position / 64; ++word)
115 result += unsigned(std::popcount(words_[word]));
116 unsigned tail = unsigned(position % 64);
117 if (tail) result += unsigned(std::popcount(words_[word] & ((std::uint64_t{1} << tail) - 1)));
118 return result;
119 }

121 private:
122 std::span<std::uint64_t const> words_;
123 std::span<rank_block const> blocks_;
124 std::span<std::uint64_t const> supers_;
125 std::uint64_t bit_count_;
126 std::uint64_t total_;
127 };

129 struct rank_index {
130 static rank_index build(std::span<std::uint64_t const> source, std::uint64_t bits) {
131 if (source.size() != bits / 64 + (bits % 64 != 0))
132 throw std::invalid_argument("rank source length");
133 rank_index result;
134 result.bit_count = bits;
135 result.words.assign(source.begin(), source.end());
136 if (bits % 64) result.words.back() &= (std::uint64_t{1} << (bits % 64)) - 1;
137 result.blocks.resize(bits / 2048 + (bits % 2048 != 0));
138 rank_detail::directory_cursor cursor;
139 auto begin_block = [&](std::uint64_t block) -> rank_block & {
140 auto & entry = result.blocks[block];
141 entry.before = cursor.before(block, result.total);
142 if (cursor.starts_epoch(block)) result.supers.push_back(cursor.epoch_base);
143 return entry;
144 };
145 auto full_blocks = bits / 2048;
146 for (std::uint64_t block = 0; block < full_blocks; ++block) {
147 auto & entry = begin_block(block);
148 auto words = result.words.data() + block * 32;
149 auto a = rank_detail::popcount512(words);
150 auto b = rank_detail::popcount512(words + 8);
151 auto c = rank_detail::popcount512(words + 16);
152 auto d = rank_detail::popcount512(words + 24);
153 entry.runs = a | (b << 10) | (c << 20);
154 result.total += a + b + c + d;
155 }
156 if (full_blocks != result.blocks.size()) {
157 auto & entry = begin_block(full_blocks);
158 unsigned counts[4]{};
159 auto first = full_blocks * 32;
160 // Only the final partial block needs bounded word loads. The owning
161 // copy has already cleared unused bits in its final word.
162 for (auto word = first; word < result.words.size(); ++word)
163 counts[(word - first) / 8] += unsigned(std::popcount(result.words[word]));
164 entry.runs = counts[0] | (counts[1] << 10) | (counts[2] << 20);
165 result.total += counts[0] + counts[1] + counts[2] + counts[3];
166 }
167 return result;
168 }
169
170 rank_view view() const & { return {words, blocks, supers, bit_count, total}; }
