24 // The unfinished physical byte is high aligned. Its unused low bits
25 // need not be zero; bit_string::copy produces a canonical owning tail.
26 bit_view buffered_payload() const & noexcept { return {buffer_, buffered_}; }

28 // A metadata limit can force a spill between records. Do not pad a
29 // partial byte in the middle of the payload: keep its bits for append.
30 void flush_complete() {
31 auto bytes = buffered_ >> 3;
32 if (!bytes) return;
33 stream_.append(std::span(buffer_).first(static_cast<std::size_t>(bytes)));
34 buffered_ &= 7;
35 if (buffered_) buffer_[0] = buffer_[bytes];
36 }
37 void append(bit_view source) {
38 auto next = profile_detail::add(bits_, source.size());
39 if (next > std::numeric_limits<std::uint64_t>::max() - 7)
40 throw std::length_error("sort file bit extent");
41 while (!source.empty()) {
42 if (((buffered_ | source.offset()) & 7) == 0 && source.size() >= buffer_bytes * 8) {
43 flush();
44 auto bytes = source.size() >> 3;
45 stream_.append(source.storage().subspan(static_cast<std::size_t>(source.offset() >> 3),
46 static_cast<std::size_t>(bytes)));
47 source = source.subview(bytes << 3, source.size() - (bytes << 3));
48 } else {
49 auto count = std::min<std::uint64_t>(source.size(), buffer_bytes * 8 - buffered_);
50 profile_detail::copy_bits(buffer_.data(), buffered_, source.prefix(count));
51 buffered_ += count; source = source.subview(count, source.size() - count);
52 if (buffered_ == buffer_bytes * 8) flush();
53 }
54 }
55 bits_ = next;
56 }
57 void write_bits(std::uint64_t value, unsigned width) {
58 if (width > 64 || (width < 64 && (value >> width))) throw std::invalid_argument("sort file bit field");
59 std::array<std::byte, 8> bytes{};
60 if (width) {
61 auto aligned = value << (64 - width);
62 for (unsigned i = 0; i != 8; ++i) bytes[i] = std::byte(aligned >> (56 - (i << 3)));
63 }
64 append(bit_view(bytes, width));
65 }
66 template <class Code> void write_count(std::uint64_t value) {
67 using code = sort_codec_detail::count_policy<Code>;
68 (void)profile_detail::add(position(), profile_detail::backspace_bits<code>(value));
69 if constexpr (code::backspace_code == bit_backspace_code::exponential_golomb) {
70 auto quotient = value >> code::backspace_parameter;
71 if (quotient == std::numeric_limits<std::uint64_t>::max()) {
72 zeroes(64); write_bits(1, 1); write_bits(0, 64);
73 } else {
74 auto prefix = quotient + 1; auto width = unsigned(std::bit_width(prefix));
75 zeroes(width - 1); write_bits(prefix, width);
76 }
77 if constexpr (code::backspace_parameter)
78 write_bits(value & ((std::uint64_t{1} << code::backspace_parameter) - 1), unsigned(code::backspace_parameter));
79 } else {
80 constexpr auto modulus = code::backspace_parameter;
81 constexpr auto width = unsigned(std::bit_width(modulus - 1));
82 constexpr auto cutoff = profile_detail::golomb_cutoff<code>();
83 zeroes(value / modulus); write_bits(1, 1);
84 auto remainder = value % modulus;
85 if (remainder < cutoff) write_bits(remainder, width - 1);
86 else write_bits(remainder + cutoff, width);
87 }
88 }
89 // Only the final payload call may flush an incomplete physical byte.
90 void finish_payload() { flush(); }
91 void align() {
92 static constexpr std::array<std::byte, 8> zero{};
93 stream_.append(std::span(zero).first(static_cast<std::size_t>((0 - stream_.body_bytes()) & 7)));
94 }
95 void words(std::span<std::uint64_t const> input) {
96 if constexpr (std::endian::native == std::endian::little) stream_.append(std::as_bytes(input));
97 else while (!input.empty()) {
98 auto count = std::min(input.size(), buffer_bytes / 8);
99 for (std::size_t i = 0; i != count; ++i) file_detail::put(buffer_, i << 3, 8, input[i]);
100 stream_.append(std::span(buffer_).first(count << 3)); input = input.subspan(count);
101 }
102 }
103 void samples(std::span<elias_fano_sample const> input) {
104 static_assert(sizeof(elias_fano_sample) == 16 && offsetof(elias_fano_sample, first) == 0 && offsetof(elias_fano_sample, sparse) == 8);
105 if constexpr (std::endian::native == std::endian::little) stream_.append(std::as_bytes(input));
106 else while (!input.empty()) {
107 auto count = std::min(input.size(), buffer_bytes / 16);
108 for (std::size_t i = 0; i != count; ++i) {
109 file_detail::put(buffer_, i << 4, 8, input[i].first);
110 file_detail::put(buffer_, (i << 4) + 8, 8, input[i].sparse);
111 }
112 stream_.append(std::span(buffer_).first(count << 4)); input = input.subspan(count);
113 }
114 }
115 private:
116 Stream & stream_;
117 std::array<std::byte, buffer_bytes> buffer_{};
118 std::uint64_t buffered_ = 0, bits_ = 0;
119 void zeroes(std::uint64_t count) {
120 static constexpr std::array<std::byte, 4096> zero{};
121 while (count) {
122 auto part = std::min<std::uint64_t>(count, zero.size() * 8); append(bit_view(zero, part)); count -= part;
123 }
124 }
125 void flush() {
126 if (!buffered_) return;
127 if (buffered_ & 7) buffer_[buffered_ >> 3] &= std::byte(0xffu << (8 - (buffered_ & 7)));
128 stream_.append(std::span(buffer_).first(static_cast<std::size_t>(profile_detail::byte_count(buffered_))));
129 buffered_ = 0;
130 }

143 file_detail::put(directory, 40, 8, encoder.seeds.bit_size);
144 directory[48] = std::byte(ef.low_width);
145 std::array<std::uint64_t, 8> lengths{profile_detail::byte_count(sink.position()),
146 profile_detail::multiply(ef.low.size(), 8), profile_detail::multiply(ef.high.size(), 8),
147 profile_detail::multiply(ef.samples.size(), 16), profile_detail::multiply(ef.sparse.size(), 8),
148 encoder.dictionary.bytes.size(), profile_detail::multiply(encoder.dictionary_offsets.size(), 8), encoder.seeds.bytes.size()};
149 std::uint64_t end = directory.size();
150 for (std::size_t i = 0; i != lengths.size(); ++i) {

152 file_detail::put(directory, 64 + 16 * i, 8, start); file_detail::put(directory, 72 + 16 * i, 8, lengths[i]);
153 }

155 file_detail::validate_metadata(header);

158 sink.align(); sink.words(ef.high);
159 sink.align(); sink.samples(ef.samples);

161 sink.align(); stream.append(encoder.dictionary.bytes);
162 sink.align(); sink.words(encoder.dictionary_offsets);
