105 template <class S = typed_detail::default_sort_t<policy_type>> void change(
106 typed_detail::key_t<S> const & key, typed_detail::arrow_t<S> const & arrow) {
107 require_active();
108 using semantics = sort_semantics<S>;
109 auto encoded = transport::template encode<S>(key);
110 auto previous = nursery_->find(encoded);
111 auto base = [&] {
112 if constexpr (transaction_detail::state_is_arrow<S>) {
113 if (previous) return semantics::apply(key, semantics::initial(key),
114 typed_detail::value<policy_type, S>(previous->original->view()));
115 }
116 return stage_->world.template get<S>(key);
117 }();
118 auto before = previous ? semantics::apply(key, base,
119 typed_detail::value<policy_type, S>(previous->arrow.view())) : base;
120 auto after = semantics::apply(key, before, arrow);
121 if constexpr (typed_detail::replacement<S>) {
122 if (!semantics::present(key, before) && !semantics::present(key, after))
123 throw std::invalid_argument("transaction deletion targets an absent key");
124 }
125 // Encoding and composition finish before the mutable nursery is touched.
126 auto combined = [&] {
127 if constexpr (typed_detail::replacement<S>) return arrow;
128 else if (previous) return semantics::compose(key,
129 typed_detail::value<policy_type, S>(previous->arrow.view()), arrow);
130 else return arrow;
131 }();
132 auto value = typed_detail::value<policy_type, S>(combined);
133 if constexpr (typed_detail::replacement<S>) {
134 if (after == base) { nursery_->erase(encoded); return; }
135 }
136 std::optional<bit_string> original;
137 if constexpr (transaction_detail::state_is_arrow<S>) original.emplace(typed_detail::value<policy_type, S>(base));
138 nursery_->insert_or_assign(std::move(encoded), nursery_value{std::move(value), std::move(original)});
139 }

146 // Empty flushes allocate no Core, catalog scope or output file. A nonempty
147 // flush completes typed validation and seals a private checkpoint only.
148 snapshot_type flush() {
149 require_active();
150 if (nursery_->empty()) return snapshot();
151 auto batch = Core::batch();
152 nursery_->for_each([&](bit_string const & key, nursery_value const & value) {
153 transport::dispatch(key.view(), [&]<class S>(std::type_identity<S>, auto const & decoded) {
154 batch.template change<S>(decoded, typed_detail::value<policy_type, S>(value.arrow.view()));
155 });
156 });
157 auto input = std::move(batch).finish();
158 nursery_type empty;
159 try {
160 ensure_core();
161 while (!core_->admission_ready()) (void)core_->advance(origin_->maintenance_budget);
162 auto updated = core_->contribute(std::move(input));
163 store_->prepare(updated.runtime());
164 auto next = std::make_shared<stage const>(stage{std::move(updated), scope_, true});
165 // Preserve the old stage and frozen roots if any preceding operation
166 // fails; after this point replacement of both local roots is noexcept.
167 *nursery_ = std::move(empty);
168 stage_ = std::move(next);
169 } catch (...) { failed_ = true; throw; }
170 return snapshot();
171 }
172 ticket commit_async() {
173 require_active();
174 auto target = origin_->session.lock();
175 if (!target) throw session_closed();
176 (void)flush();
177 if (stage_->changed && !core_) ensure_core(); // branch of a flushed snapshot
178 typename engine_type::prepared_transaction prepared{std::move(core_), origin_->base, scope_};
179 typename engine_type::contribution_type command(std::move(prepared));
180 // Submission owns the candidate from here; every failure leaves the
181 // original named world unchanged locally, with uncertain SQL outcomes
182 // reported through the same ticket contract as ordinary contributions.
183 closed_ = true;
184 auto result = target->submit(std::move(command));
185 release();
186 return result;
187 }
188 stored_type commit() { return commit_async().get()->world; }
189 void abort() noexcept { closed_ = true; release(); }
190 bool failed() const noexcept { return failed_ || (nursery_ && nursery_->failed()); }

209 void ensure_core() {
210 if (core_) return;
211 if (!scope_) scope_ = scope_type::create(origin_->root);
212 if (!store_) store_ = std::make_unique<store_type>(store_type::open(origin_->root, {}, scope_->options()));
213 auto const & schema = stage_->world.metadata().schema_id;
214 if constexpr (requires { family_type::open_storage(origin_->root, schema, scope_->options()); })
215 core_ = std::make_unique<Core>(Core::from_snapshot(stage_->world,
216 family_type::open_storage(origin_->root, schema, scope_->options())));
217 else {
