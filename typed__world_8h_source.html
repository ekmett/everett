72 template <class P, class S> bit_string value(arrow_t<S> const & value) {
73 bit_string result;

76 if constexpr (P::unit == profile_unit::byte)
77 if (auto tail = result.bit_size & 7) out.write_bits(0, unsigned(8 - tail));
78 if (P::value_width && (result.bit_size >> P::unit_shift) != *P::value_width)
79 throw std::invalid_argument("sort arrow differs from the policy's fixed value width");
80 return result;
81 }

132 // Compact trusted checkpoint payload: signature, live count, then the
133 // nonempty schema ID. Runtime frontier intervals are stored separately.
134 std::vector<std::byte> encode() const requires std::same_as<typename A::element, std::uint64_t> {
135 if (schema_id.empty()) throw std::invalid_argument("empty typed schema identity");
136 std::vector<std::byte> result(16 + schema_id.size());
137 for (unsigned i = 0; i != 8; ++i) {
138 result[i] = std::byte(signature >> (i << 3));
139 result[8 + i] = std::byte(live_count >> (i << 3));
140 }
141 std::copy(std::as_bytes(std::span(schema_id)).begin(), std::as_bytes(std::span(schema_id)).end(), result.begin() + 16);
142 return result;
143 }
144 static typed_world_metadata decode(std::span<std::byte const> data)
145 requires std::same_as<typename A::element, std::uint64_t> {
146 if (data.size() <= 16) throw std::invalid_argument("truncated typed metadata");
147 typed_world_metadata result;
148 for (unsigned i = 0; i != 8; ++i) {
149 result.signature |= std::uint64_t(std::to_integer<unsigned char>(data[i])) << (i << 3);
150 result.live_count |= std::uint64_t(std::to_integer<unsigned char>(data[8 + i])) << (i << 3);
151 }
152 result.schema_id.assign(reinterpret_cast<char const *>(data.data() + 16), data.size() - 16);
153 return result;
154 }
155 bool operator==(typed_world_metadata const &) const = default;
156 };

200 // A contribution already owns its encoded key. The logical key remains
201 // available for sort semantics, while this synchronous query shares the
202 // same implementation as get without encoding it again. Only typed_batch
203 // creates contribution records, through this same Family's transport;
204 // preflight still dispatches and validates the encoded key first. Custom views and
205 // all escaping cursor contexts still take an owning copy.
206 template <class S, class Query> typed_detail::state_t<S>
207 get_encoded(typed_detail::key_t<S> const & key, Query && encoded) const {
208 static_assert(std::same_as<std::remove_cvref_t<Query>, bit_string>);
209 using semantics = sort_semantics<S>;
210 if constexpr (typed_detail::replacement<S>) {
211 auto decode = [&](bit_view value) -> typed_detail::state_t<S> {
212 return semantics::apply(key, semantics::initial(key), typed_detail::value<P, S>(value));
213 };
214 if constexpr (requires { cola_detail::first_value(state_->runtime.query_root(), std::forward<Query>(encoded), decode); }) {
215 auto value = cola_detail::first_value(state_->runtime.query_root(), std::forward<Query>(encoded), decode);
216 return value ? std::move(*value) : semantics::initial(key);
217 }
218 }
219 auto cursor = [&] {
220 if constexpr (requires { state_->runtime.cursor_owned(bit_string(std::forward<Query>(encoded))); })
221 return state_->runtime.cursor_owned(bit_string(std::forward<Query>(encoded)));
222 else return state_->runtime.cursor(encoded.view());
223 }();
224 if constexpr (typed_detail::replacement<S>) {
225 while (!cursor.done()) {
226 cursor.step(1);
227 if (cursor.has_match()) {
228 auto match = cursor.take_match();
229 return semantics::apply(key, semantics::initial(key), typed_detail::value<P, S>(match.value.view()));
230 }
231 }
232 return semantics::initial(key);
233 } else {
234 std::vector<typed_detail::arrow_t<S>> arrows;
235 while (!cursor.done()) {
236 cursor.step(1);
237 while (cursor.has_match()) {
238 auto match = cursor.take_match();
239 arrows.push_back(typed_detail::value<P, S>(match.value.view()));
240 }
241 }
242 auto state = semantics::initial(key);
243 for (auto i = arrows.rbegin(); i != arrows.rend(); ++i) state = semantics::apply(key, std::move(state), *i);
244 return state;
245 }

418 // The rebuild wrapper shares this complete preflight before installing a
419 // pristine batch; neither caller can publish metadata ahead of execution.
420 metadata_type prepare(contribution_type const & input) const {
421 if (input.base() && input.base()->metadata().schema_id != current_.metadata().schema_id)
422 throw std::invalid_argument("typed contribution uses another schema");
423 auto metadata = current_.metadata();
424 // Validate every old state and compute every delta before changing the
425 // executor. Disjoint batches from one immutable base therefore commute.
426 for (auto const & record : input.records()) {
427 key_transport::dispatch(record.key.view(), [&]<class S>(std::type_identity<S>, auto const & key) {
428 using semantics = sort_semantics<S>;
429 auto old = current_.template get_encoded<S>(key, record.key);
430 if (input.base() && old != input.base()->template get_encoded<S>(key, record.key))
431 throw std::invalid_argument("stale typed key value");
432 auto arrow = typed_detail::value<P, S>(record.value.view());
433 auto next = semantics::apply(key, old, std::move(arrow));
434 bool was = semantics::present(key, old), now = semantics::present(key, next);
435 if constexpr (typed_detail::replacement<S>)
436 if (!was && !now) throw std::invalid_argument("deleting absent typed key");
437 auto before_hash = was ? A::lift(semantics::hash_value(key, old)) : A::zero();
438 auto after_hash = now ? A::lift(semantics::hash_value(key, next)) : A::zero();
439 metadata.signature = A::add(metadata.signature,
440 A::multiply(A::lift(semantics::hash_key(key)), A::subtract(after_hash, before_hash)));
441 if (now && !was) metadata.live_count = profile_detail::add(metadata.live_count, 1);
442 if (was && !now) {
443 if (!metadata.live_count) throw std::invalid_argument("inconsistent typed live count");
444 --metadata.live_count;
445 }
446 });
447 }
448 return metadata;

476 auto count = input.records().size();
477 if (count >= 2 && !current_.runtime().admissions() && !runtime_.pending()) {

479 // funds initialization, and each remaining record retains its A.
480 auto prefix = std::bit_floor(count);
481 if (runtime_.try_initialize_sorted(input.records().first(prefix), reservation_work(prefix), DepthLimit)) {
482 require_depth(runtime_.snapshot());
483 complete(input.records().subspan(prefix), metadata);
484 return true;
485 }
486 }
487 }
488 return false;
489 } catch (...) { poison(); throw; }
