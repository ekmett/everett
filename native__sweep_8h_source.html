32 template <class World> struct native_sweep {

36 explicit native_sweep(World const & snapshot, bit_view lower = {}, range_positioning_work * work = nullptr) {
37 if (work) *work = {};
38 using P = typename World::policy_type;
39 std::unordered_map<native_type const *, std::uint64_t> starts;
40 auto node = snapshot.runtime().query_root().head();
41 auto context = profile_query_context<P>(lower);
42 std::uint64_t group = 0;
43 bool above = false;
44 while (node) {
45 if (work) ++work->catalogs;
46 auto view = node->view();
47 cola_lower_bound_result<P> found;
48 if (!above && view.virtual_size()) found = view.lower_bound_window(group, context, work ? &work->comparisons : nullptr);
49 starts.emplace(node->native_owner().get(), found.native_ordinal);
50 if (auto side = node->secondary_target()) {
51 if (work) ++work->catalogs;
52 std::uint64_t ordinal = 0;
53 if (auto const & predecessor = found.predecessors[1]) {
54 auto leaf = side->view();
55 auto first = predecessor->target_ordinal;
56 if (first >= leaf.size() || first % P::group_size)
57 throw std::invalid_argument("invalid range secondary route");
58 auto last = first + std::min<std::uint64_t>(P::group_size, leaf.size() - first);
59 ordinal = last;
60 leaf.compare_window(first, last, predecessor->comparison, [&](profile_comparison_item<P> item) {
61 if (item.comparison.order() < 0) return true;
62 ordinal = item.ordinal; return false;
63 }, work ? &work->comparisons : nullptr);
64 }
65 starts.emplace(side.get(), ordinal);
66 }
67 auto main = node->main_target();
68 if (auto const & predecessor = found.predecessors[0]) {
69 if (!main || predecessor->target_ordinal % P::group_size || predecessor->target_ordinal >= main->virtual_size())
70 throw std::invalid_argument("invalid range main route");
71 group = predecessor->target_ordinal / P::group_size;
72 context = predecessor->comparison;
73 } else above = true; // No sampled predecessor: all downstream keys exceed lower.
74 node = std::move(main);
75 }
76 if constexpr (requires { snapshot.runtime().runs(); }) {
77 auto runs = snapshot.runtime().runs();
78 sources_.reserve(runs.size()); heap_.reserve(runs.size());
79 for (auto const & run : runs) {
80 auto native = [&] {

82 else return run->native;
83 }();

85 if (position == starts.end()) throw std::invalid_argument("native run absent from range cascade");

87 work->seeded_headers += position->second % P::codec_block_size + 1;
88 sources_.emplace_back(std::move(native), position->second, lower);
89 if (!sources_.back().cursor.done()) heap_.push_back(sources_.size() - 1);
90 }
91 std::make_heap(heap_.begin(), heap_.end(), later());
92 } else throw std::logic_error("runtime does not expose a native sweep");
93 }
94 bool done() const noexcept { return heap_.empty(); }
95 std::uint64_t consumed() const noexcept { return consumed_; }
96 auto peek() const { return sources_[heap_.front()].cursor.peek(); }
97 std::uint64_t retained_bits() const {
98 auto const & cursor = sources_[heap_.front()].cursor;
99 if constexpr (requires { cursor.retained_bits(); }) return cursor.retained_bits();
100 else return 0; // A custom cursor can conservatively emit the entire key.
101 }
102 void consume() {
103 std::pop_heap(heap_.begin(), heap_.end(), later());
104 auto which = heap_.back(); heap_.pop_back();
105 auto comparison = sources_[which].cursor.advance_comparison();
106 if (comparison && comparison->order >= 0)
107 throw std::invalid_argument("scanned native keys are not unique and sorted");
108 ++consumed_;
109 if (!sources_[which].cursor.done()) {
110 heap_.push_back(which); std::push_heap(heap_.begin(), heap_.end(), later());
111 }
112 }
113 // Queries must be increasing. Values borrow pinned immutable natives.
114 std::optional<observation> replacement(bit_view key) {
115 while (!done() && compare_bits(peek().key.prefix, key) < 0) consume();
116 std::optional<observation> result;
117 while (!done() && compare_bits(peek().key.prefix, key) == 0) {
118 result = observation{peek().value, retained_bits()};
119 consume();
120 }
121 return result;
122 }

128 : native(std::move(value)), cursor(native->view(), ordinal, query) {}
129 };

134 return [this](std::size_t a, std::size_t b) {

137 };
