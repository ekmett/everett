43 // A prepared immutable chain whose augmented head fits in one K-entry window.
44 // Exact sample contents remain a construction precondition: metadata checks

56 // dependencies. Adoption checks chain shapes and cycles, never sample keys.

60 error_detail::raise<std::invalid_argument>("query root head exceeds one cascade group");

68 explicit query_root(pair_type head) : head_(std::move(head)) {}

71 static void validate(pair_type const & source) {
72 static_assert(P::group_size >= 3);
73 if (!source) error_detail::raise<std::invalid_argument>("null query root");
74 std::unordered_set<blob_type const *> seen;
75 for (auto current = source; current; current = current->target()) {
76 if (!seen.insert(current.get()).second) error_detail::raise<std::invalid_argument>("cyclic query chain");
77 auto native = current->native().size(), borrowed = current->borrowed().size();
78 if (native > std::numeric_limits<std::uint64_t>::max() - borrowed ||
79 native + borrowed != current->virtual_size())
80 error_detail::raise<std::invalid_argument>("query catalog size mismatch");
81 auto groups = current->virtual_size() / P::group_size +
82 (current->virtual_size() % P::group_size != 0);
83 if (current->group_count() != groups)
84 error_detail::raise<std::invalid_argument>("query catalog group count mismatch");
85 auto target = current->target();
86 if (borrowed != (target ? target->group_count() : 0))
87 error_detail::raise<std::invalid_argument>("query chain sample count or target mismatch");
88 if (current->cut_lcps().size() != current->group_count())
89 error_detail::raise<std::invalid_argument>("query chain cut LCP count mismatch");
90 }
91 }

94 // Shape validation visits the existing chain once without decoding its keys.
95 // A head already of size <= K keeps its exact identity and needs no sampler.
96 // Larger heads acquire empty-native routing catalogs through one pipeline:
97 // a source scan and successively ceil(size/K) samples, with no full-key array.
98 // step() inherits the pipeline's entry-work budget; bytes are not bounded by
99 // that budget, and finish() separately finalizes compact metadata.
100 template <class P> struct query_root_builder {
101 using policy_type = P;
102 using blob_type = profile_blob<P>;
103 using pair_type = std::shared_ptr<blob_type const>;
104
105 explicit query_root_builder(pair_type source) : head_(std::move(source)) {
106 query_root<P>::validate(head_);
107 auto count = head_->virtual_size();
108 std::size_t levels = 0;
109 while (count > P::group_size) {
110 count = count / P::group_size + (count % P::group_size != 0);
111 ++levels;
112 }
113 if (levels) {
114 auto empty = std::make_shared<blob_type const>(blob_type::build({}));
115 pipeline_.emplace(head_, std::vector<pair_type>(levels, std::move(empty)));
116 }
117 }
118 query_root_builder(query_root_builder const &) = delete;
119 query_root_builder & operator=(query_root_builder const &) = delete;
120 query_root_builder(query_root_builder &&) = default;
121 query_root_builder & operator=(query_root_builder &&) = default;
122
123 bool done() const noexcept { return !pipeline_ || pipeline_->done(); }
124 bool finished() const noexcept { return finished_; }
125 std::uint64_t step(std::uint64_t quanta) {
126 if (!head_) error_detail::raise<std::logic_error>("query root preparation has no source");
127 if (finished_) error_detail::raise<std::logic_error>("query root preparation is finished");
128 return pipeline_ ? pipeline_->step(quanta) : 0;
129 }
130 query_root<P> finish() {
131 if (!head_) error_detail::raise<std::logic_error>("query root preparation has no source");
132 if (!done()) error_detail::raise<std::logic_error>("query root preparation still has input");
133 if (!finished_) {
134 if (pipeline_) head_ = pipeline_->finish();
135 finished_ = true;
136 }
137 return query_root<P>(head_);
138 }

140 private:
141 pair_type head_;
142 std::optional<index_pipeline<P>> pipeline_;
143 bool finished_ = false;

146 // The cursor owns its query and current target pin. Each successful step
147 // visits at most catalog_budget catalogs and stops at its first native match.
148 // Take that owned match before stepping again. Equal native keys do not stop
149 // descent: the downstream route is retained for the next call. Visited pairs
150 // can be released unless a pending/returned match owns them. Copying a cursor
151 // shares its immutable query and copies its context and pending value; copies advance independently
152 // while sharing the same immutable suffix and source pins.
153 //
154 // Search costs O(K+W) entry/header work per visited catalog plus literal
155 // comparison and value copying. Root preparation adds O(log_K A) catalogs for a head
156 // of A entries. These bounds do not cover arbitrary string bytes, arrow
157 // evaluation, disk faults, or a future level scheduler.
158 template <class P, class Blob> struct query_cursor {
159 using policy_type = P;
160 using blob_type = Blob;
161 using pair_type = std::shared_ptr<blob_type const>;
162 using match_type = query_match<P, Blob>;
163
164 explicit query_cursor(query_root<P, Blob> const & root, bit_view query)
165 : current_(root.head()), context_(query) {
166 if (!current_) error_detail::raise<std::invalid_argument>("query root has no head");
167 if (!current_->virtual_size()) current_.reset();
168 }
169 bool done() const noexcept { return !current_ && !pending_; }
170 bool has_match() const noexcept { return pending_.has_value(); }
171 bool failed() const noexcept { return failed_; }
172
173 std::uint64_t step(std::uint64_t catalog_budget = 1) {
174 if (failed_) error_detail::raise<std::logic_error>("query cursor has failed");
175 if (pending_) return 0;
176 std::uint64_t visited = 0;
177 try {
178 while (current_ && visited != catalog_budget) {
179 auto result = current_->search_window(group_, context_);
180 auto target = current_->target();
181 if (result.borrowed_predecessor) {
182 auto const & next = *result.borrowed_predecessor;
183 if (!target || next.target_ordinal % P::group_size ||
184 next.target_ordinal >= target->virtual_size())
185 error_detail::raise<std::invalid_argument>("query descent has no valid target context");
186 }
187 if (result.native)
188 pending_.emplace(match_type{current_, result.native->ordinal, std::move(result.native->value)});
189 if (result.borrowed_predecessor) {
190 auto & next = *result.borrowed_predecessor;
191 group_ = next.target_ordinal / P::group_size;
192 context_ = std::move(next.comparison);
193 current_ = std::move(target);
194 } else {
195 // The first borrowed key samples the target's minimum. No borrowed
196 // predecessor means no downstream native key can equal this query.
197 current_.reset();
198 }
199 ++visited;
200 if (pending_) break;
201 }
202 } catch (...) {
203 failed_ = true;
204 throw;
205 }
206 return visited;
207 }
208 match_type take_match() {
209 if (failed_) error_detail::raise<std::logic_error>("query cursor has failed");
210 if (!pending_) error_detail::raise<std::logic_error>("query cursor has no pending match");
211 auto result = std::move(*pending_);
212 pending_.reset();
213 return result;
214 }
215
216 private:
217 pair_type current_;
218 profile_query_context<P> context_;
219 std::uint64_t group_ = 0;
220 std::optional<match_type> pending_;
221 bool failed_ = false;
222 };
