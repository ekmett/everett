59 // Shape validation visits the existing chain once without decoding its keys.
60 // A head already of size <= K keeps its exact identity and needs no sampler.
61 // Larger heads acquire empty-native routing catalogs through one pipeline:
62 // a source scan and successively ceil(size/K) samples, with no full-key array.
63 // step() inherits the pipeline's entry-work budget; bytes are not bounded by
64 // that budget, and finish() separately finalizes compact metadata.
65 template <class P> struct query_root_builder {
66 using policy_type = P;
67 using blob_type = profile_blob<P>;
68 using pair_type = std::shared_ptr<blob_type const>;
69
70 explicit query_root_builder(pair_type source) : head_(std::move(source)) {
71 validate(head_);
72 auto count = head_->virtual_size();
73 std::size_t levels = 0;
74 while (count > P::group_size) {
75 count = count / P::group_size + (count % P::group_size != 0);
76 ++levels;
77 }
78 if (levels) {
79 auto empty = std::make_shared<blob_type const>(blob_type::build({}));
80 pipeline_.emplace(head_, std::vector<pair_type>(levels, std::move(empty)));
81 }
82 }
83 query_root_builder(query_root_builder const &) = delete;
84 query_root_builder & operator=(query_root_builder const &) = delete;
85 query_root_builder(query_root_builder &&) = default;
86 query_root_builder & operator=(query_root_builder &&) = default;

88 bool done() const noexcept { return !pipeline_ || pipeline_->done(); }
89 bool finished() const noexcept { return finished_; }
90 std::uint64_t step(std::uint64_t quanta) {
91 if (!head_) throw std::logic_error("query root preparation has no source");
92 if (finished_) throw std::logic_error("query root preparation is finished");
93 return pipeline_ ? pipeline_->step(quanta) : 0;
94 }
95 query_root<P> finish() {
96 if (!head_) throw std::logic_error("query root preparation has no source");
97 if (!done()) throw std::logic_error("query root preparation still has input");
98 if (!finished_) {
99 if (pipeline_) head_ = pipeline_->finish();
100 finished_ = true;
101 }
102 return query_root<P>(head_);
103 }

106 pair_type head_;

110 static void validate(pair_type const & source) {
111 static_assert(P::group_size >= 3);
112 if (!source) throw std::invalid_argument("null query root");
113 std::unordered_set<blob_type const *> seen;
114 for (auto current = source; current; current = current->target()) {
115 if (!seen.insert(current.get()).second) throw std::invalid_argument("cyclic query chain");
116 auto native = current->native().size(), borrowed = current->borrowed().size();
117 if (native > std::numeric_limits<std::uint64_t>::max() - borrowed ||
118 native + borrowed != current->virtual_size())
119 throw std::invalid_argument("query catalog size mismatch");
120 auto target = current->target();
121 if (borrowed != (target ? target->group_count() : 0))
122 throw std::invalid_argument("query chain sample count or target mismatch");
123 if (current->cut_lcps().size() != current->group_count())
124 throw std::invalid_argument("query chain cut LCP count mismatch");
125 }
126 }

129 // The cursor owns its query and current target pin. Each successful step
130 // visits at most catalog_budget catalogs and stops at its first native match.
131 // Take that owned match before stepping again. Equal native keys do not stop
132 // descent: the downstream route is retained for the next call. Visited pairs
133 // can be released unless a pending/returned match owns them. Copying a cursor
134 // shares its immutable query and copies its context and pending value; copies advance independently
135 // while sharing the same immutable suffix and source pins.
136 //
137 // Search costs O(K+W) entry/header work per visited catalog plus literal
138 // comparison and value copying. Root preparation adds O(log_K A) catalogs for a head
139 // of A entries. These bounds do not cover arbitrary string bytes, arrow
140 // evaluation, disk faults, or a future level scheduler.
141 template <class P> struct query_cursor {
142 using policy_type = P;
143 using blob_type = profile_blob<P>;
144 using pair_type = std::shared_ptr<blob_type const>;
145 using match_type = query_match<P>;
146
147 explicit query_cursor(query_root<P> const & root, bit_view query)
148 : current_(root.head()), context_(query) {
149 if (!current_) throw std::invalid_argument("query root has no head");
150 if (!current_->virtual_size()) current_.reset();
151 }
152 bool done() const noexcept { return !current_ && !pending_; }
153 bool has_match() const noexcept { return pending_.has_value(); }
154 bool failed() const noexcept { return failed_; }
155
156 std::uint64_t step(std::uint64_t catalog_budget = 1) {
157 if (failed_) throw std::logic_error("query cursor has failed");
158 if (pending_) return 0;
159 std::uint64_t visited = 0;
160 try {
161 while (current_ && visited != catalog_budget) {
162 auto result = current_->search_window(group_, context_);
163 auto target = current_->target();
164 if (result.borrowed_predecessor) {
165 auto const & next = *result.borrowed_predecessor;
166 if (!target || next.target_ordinal % P::group_size ||
167 next.target_ordinal >= target->virtual_size())
168 throw std::invalid_argument("query descent has no valid target context");
169 }
170 if (result.native)
171 pending_.emplace(match_type{current_, result.native->ordinal, std::move(result.native->value)});
172 if (result.borrowed_predecessor) {
173 auto & next = *result.borrowed_predecessor;
174 group_ = next.target_ordinal / P::group_size;
175 context_ = std::move(next.comparison);
176 current_ = std::move(target);
177 } else {
178 // The first borrowed key samples the target's minimum. No borrowed
179 // predecessor means no downstream native key can equal this query.
180 current_.reset();
181 }
182 ++visited;
183 if (pending_) break;
184 }
185 } catch (...) {
186 failed_ = true;
187 throw;
188 }
189 return visited;
190 }
191 match_type take_match() {
192 if (failed_) throw std::logic_error("query cursor has failed");
193 if (!pending_) throw std::logic_error("query cursor has no pending match");
194 auto result = std::move(*pending_);
195 pending_.reset();
196 return result;
197 }
198
199 private:
200 pair_type current_;
201 profile_query_context<P> context_;
202 std::uint64_t group_ = 0;
203 std::optional<match_type> pending_;
204 bool failed_ = false;
205 };
