38 inline runtime_checkpoint checkpoint(std::span<std::byte const> encoded) {
39 if (encoded.size() < magic.size() || !std::equal(magic.begin(), magic.end(), encoded.begin()))
40 throw std::invalid_argument("unsupported Everett runtime checkpoint");
41 catalog_detail::outcome_reader input{encoded.subspan(magic.size())};
42 auto count = input.number();
43 if (count > 65 || count > input.data.size() / 16)
44 throw std::invalid_argument("invalid Everett runtime interval count");
45 runtime_checkpoint result;
46 result.intervals.reserve(static_cast<std::size_t>(count));
47 for (std::uint64_t i = 0; i != count; ++i) {
48 auto first = input.number(), last = input.number();
49 result.intervals.push_back({first, last});
