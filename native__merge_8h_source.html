33 // Merge two chronologically ordered native streams. Equal keys call
34 // compose(key, older_value, newer_value); default composition is replacement.
35 // Each input must have unique sorted keys. Input owners stay pinned,
36 // including after a decoding/composition failure.
37 // Callback keys borrow cursor scratch. A policy retaining a value view must
38 // retain its source owner too; reassignment can release earlier source pins.
39 // Construction decodes the first record from each nonempty source before
40 // any step budget is charged.
41 // One step unit handles one distinct key and at most two input records.
42 // Key bytes, composition work, output allocation and final EF work are extra.
43 template <class P, class Native = profile_array<P>, class Compose = replace_native_value>
44 struct native_merge_builder {
45 static_assert(std::is_same_v<typename Native::policy_type, P>);
46 using policy_type = P;
47 using source_type = Native;
48 using source_pointer = std::shared_ptr<Native const>;

83 native_merge_progress step(std::uint64_t key_budget = 1) {
84 require_active();
85 native_merge_progress work;
86 try {
87 while (work.keys < key_budget && !done()) {
88 auto comparison = compare_heads();
89 auto order = comparison.order;
90 auto consumed = order == 0 ? 2u : 1u;
91 auto total_keys = profile_detail::add(progress_.keys, 1);
92 auto total_records = profile_detail::add(progress_.input_records, consumed);
93 if (order < 0) {
94 auto item = older_cursor_.peek();
95 append(item.key.prefix, item.value, older_prefix_);
96 newer_prefix_ = comparison.common_bits >> P::unit_shift;
97 older_prefix_ = advance(older_cursor_);
98 } else if (order > 0) {
99 auto item = newer_cursor_.peek();
100 append(item.key.prefix, item.value, newer_prefix_);
101 older_prefix_ = comparison.common_bits >> P::unit_shift;
102 newer_prefix_ = advance(newer_cursor_);
103 } else {
104 auto older = older_cursor_.peek(), newer = newer_cursor_.peek();
105 auto value = std::invoke(compose_, older.key.prefix, older.value, newer.value);
106 if constexpr (std::is_same_v<decltype(value), bit_view>)
107 append(older.key.prefix, value, older_prefix_);
108 else append(older.key.prefix, value.view(), older_prefix_);
109 older_prefix_ = advance(older_cursor_);
110 newer_prefix_ = advance(newer_cursor_);
111 }
112 progress_ = {total_keys, total_records};
113 ++work.keys;
114 work.input_records += consumed;
115 }
116 } catch (...) {
117 failed_ = true;
118 throw;
119 }
120 return work;
121 }

130 void append(bit_view key, bit_view value, std::uint64_t retained) {
131 auto first = retained * P::bits_per_unit;
132 writer_.append(retained, key.subview(first, key.size() - first), value);
133 }
134 // Both heads follow the last emitted key p. The head sharing more of p
135 // sorts first. Equal LCPs need only a suffix comparison from that boundary.
136 bit_comparison compare_heads() const {
137 if (older_cursor_.done()) return {0, 1};
138 if (newer_cursor_.done()) return {0, -1};
139 if (older_prefix_ != newer_prefix_)
140 return {std::min(older_prefix_, newer_prefix_) * P::bits_per_unit,
141 older_prefix_ > newer_prefix_ ? -1 : 1};
142 auto older = older_cursor_.peek().key.prefix;
143 auto newer = newer_cursor_.peek().key.prefix;
144 auto start = older_prefix_ * P::bits_per_unit;
145 auto result = compare_common_bits(older.subview(start, older.size() - start),
146 newer.subview(start, newer.size() - start));
147 result.common_bits += start;
148 return result;
149 }
150 static std::uint64_t advance(profile_cursor<P, stream_role::native> & cursor) {
151 auto comparison = cursor.advance_comparison();
152 if (!comparison) return 0;
153 if (comparison->order >= 0)
154 throw std::invalid_argument("native merge inputs must have unique sorted keys");
155 return comparison->common_bits >> P::unit_shift;
156 }
