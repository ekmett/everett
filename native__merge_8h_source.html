35 // Keep the predecessor as immutable literal spans, without copying its
36 // inherited prefix. A new retained boundary can precede the immediately
37 // preceding literal, so checking that one literal alone is insufficient.
38 // Each record adds at most one span; truncation removes spans permanently.
39 // Deep proper-prefix chains can retain one span per key unit.
40 template <class P> struct encoded_source {
41 explicit encoded_source(profile_view<P> view)
42 : data_(view.bytes(), view.metadata().extent << P::unit_shift), cursor_(view) {
43 if (!done()) retain(cursor_.peek());
44 }
45 bool done() const noexcept { return cursor_.done(); }
46 profile_encoded_record const & peek() const & { return cursor_.peek(); }
47 profile_encoded_record const & peek() const && = delete;

49 std::optional<bit_comparison> advance_comparison() {
50 cursor_.advance();
51 if (done()) return std::nullopt;
52 auto const & next = cursor_.peek();
53 auto result = compare_successor(next.retained, next.suffix);
54 if (result.order >= 0)
55 throw std::invalid_argument("native merge inputs must have unique sorted keys");
56 retain(next);
57 return result;
58 }

61 struct span {
62 std::uint64_t source_bit_offset;
63 std::uint64_t end_units;
64 };
65 static_assert(sizeof(span) == 16);
66 bit_comparison compare_successor(std::uint64_t retained, bit_view suffix) const {
67 auto first = spans_.size();
68 // Every crossed span beyond this boundary will be removed by retain;
69 // walking back is amortized over the one span added per input record.
70 while (first && spans_[first - 1].end_units > retained) --first;
71 auto begin = first ? spans_[first - 1].end_units : 0;
72 // Ordinary FC differs in the first remaining policy unit. Handle it
73 // directly; redundant controls fall through to the full fragment walk.
74 if (first != spans_.size() && !suffix.empty()) {
75 auto offset = spans_[first].source_bit_offset + ((retained - begin) << P::unit_shift);
76 auto before = profile_detail::load_bits(data_, offset, P::bits_per_unit);
77 auto after = profile_detail::load_bits(suffix, 0, P::bits_per_unit);
78 if (before != after) {
79 auto common = unsigned(std::countl_zero(before ^ after)) - (64 - P::bits_per_unit);
80 return {(retained << P::unit_shift) + common, before < after ? -1 : 1};
81 }
82 }
83 auto position = retained;
84 std::uint64_t compared = 0;
85 while (first != spans_.size() && compared != suffix.size()) {
86 auto const & fragment = spans_[first];
87 auto offset = fragment.source_bit_offset + ((position - begin) << P::unit_shift);
88 auto count = std::min((fragment.end_units - position) << P::unit_shift,
89 suffix.size() - compared);
90 auto result = compare_common_bits(data_.subview(offset, count), suffix.subview(compared, count));
91 if (result.order) {
92 result.common_bits += (retained << P::unit_shift) + compared;
93 return result;
94 }
95 compared += count;
96 position = fragment.end_units;
97 begin = position;
98 ++first;
99 }
100 auto previous_units = spans_.empty() ? 0 : spans_.back().end_units;
101 auto previous_remaining = (previous_units - retained) << P::unit_shift;
102 return {(retained << P::unit_shift) + compared,
103 previous_remaining < suffix.size() ? -1 : previous_remaining > suffix.size() ? 1 : 0};
104 }
105 void retain(profile_encoded_record const & record) {
106 auto retained = record.retained;
107 while (!spans_.empty()) {
108 auto begin = spans_.size() > 1 ? spans_[spans_.size() - 2].end_units : 0;
109 if (begin >= retained) spans_.pop_back();
110 else {
111 if (spans_.back().end_units > retained) spans_.back().end_units = retained;
112 break;
113 }
114 }
115 // Every frame uses this one immutable backing. A span's logical start
116 // is the preceding endpoint, so truncation needs no copied length/view.
117 if (!record.suffix.empty()) spans_.push_back({record.suffix.offset(), record.key_units});
118 }
119 bit_view data_;
120 profile_encoded_cursor<P> cursor_;
121 std::vector<span> spans_;
122 };

130 // Merge two chronologically ordered native streams. Equal keys call
131 // compose(key, older_value, newer_value), or compose(older_value, newer_value)
132 // for value-only policies. Default composition is replacement. Key-aware
133 // policies retain materialized cursors; the other cases forward encoded
134 // literals and validate strict input order through immutable prefix spans.
135 // A custom policy accepting both forms uses the key-aware form.
136 // Each input must have unique sorted keys. Input owners stay pinned,
137 // including after a decoding/composition failure.
138 // Callback keys borrow cursor scratch. A policy retaining a value view must
139 // retain its source owner too; reassignment can release earlier source pins.
140 // Construction parses the first record from each nonempty source before any
141 // step budget is charged; key-aware policies also reconstruct those keys.
142 // One step unit handles one distinct key and at most two input records.
143 // Key bytes, composition work, output allocation and final EF work are extra.
144 // An alternate Output consumes append(retained,literal,value) synchronously,
145 // encodes policy P, and supplies size/common_value_width/finished/failed/finish.
146 // It starts empty and active; finished() and failed() are noexcept. The caller
147 // grants exclusive sink use to this builder, and keeps any referenced sink
148 // alive. A finish failure is retryable only when the sink reports !failed().
149 template <class P, class Native = profile_array<P>, class Compose = replace_native_value,
150 class Output = profile_detail::native_output<P>>
151 struct native_merge_builder {
152 static_assert(std::is_same_v<typename Native::policy_type, P>);
153 static_assert(std::is_same_v<typename Output::policy_type, P>);
154 using policy_type = P;
155 using source_type = Native;
156 using output_type = Output;
157 static_assert(noexcept(std::declval<Output const &>().finished()));
158 static_assert(noexcept(std::declval<Output const &>().failed()));
159 using source_pointer = std::shared_ptr<Native const>;
160 static constexpr bool encoded_keys = std::is_same_v<Compose, replace_native_value> ||

163 static_assert(encoded_keys || std::is_invocable_v<Compose &, bit_view, bit_view, bit_view>,

202 native_merge_progress step(std::uint64_t key_budget = 1) {
203 require_active();
204 native_merge_progress work;
205 try {
206 while (work.keys < key_budget && !done()) {
207 auto comparison = compare_heads();
208 auto order = comparison.order;
209 auto consumed = order == 0 ? 2u : 1u;
210 auto total_keys = profile_detail::add(progress_.keys, 1);
211 auto total_records = profile_detail::add(progress_.input_records, consumed);
212 if (order < 0) {
213 auto item = older_cursor_.peek();
214 append(item, item.value, older_prefix_);
215 newer_prefix_ = comparison.common_bits >> P::unit_shift;
216 older_prefix_ = advance(older_cursor_);
217 } else if (order > 0) {
218 auto item = newer_cursor_.peek();
219 append(item, item.value, newer_prefix_);
220 older_prefix_ = comparison.common_bits >> P::unit_shift;
221 newer_prefix_ = advance(newer_cursor_);
222 } else {
223 auto older = older_cursor_.peek(), newer = newer_cursor_.peek();
224 auto value = [&] {
225 if constexpr (encoded_keys) return std::invoke(compose_, older.value, newer.value);
226 else return std::invoke(compose_, older.key.prefix, older.value, newer.value);
227 }();
228 if constexpr (std::is_same_v<decltype(value), bit_view>)
229 append(older, value, older_prefix_);
230 else append(older, value.view(), older_prefix_);
231 older_prefix_ = advance(older_cursor_);
232 newer_prefix_ = advance(newer_cursor_);
233 }
234 progress_ = {total_keys, total_records};
235 ++work.keys;
236 work.input_records += consumed;
237 }
238 } catch (...) {
239 failed_ = true;
240 throw;
241 }
242 return work;
243 }

246 require_active();
247 if (!done()) throw std::logic_error("native merge has unconsumed inputs");
248 try { return writer_.finish(); }
249 catch (...) {
250 // In-memory EF allocation failure is retryable. A sink that has
251 // partially written final output must report its irreversible failure.
252 failed_ = writer_.failed();
253 throw;
254 }

260 static bit_view suffix(auto const & item, std::uint64_t retained) {
261 if constexpr (encoded_keys) {
262 if (retained < item.retained)
263 throw std::invalid_argument("native merge output prefix precedes input prefix");
264 auto first = (retained - item.retained) << P::unit_shift;
265 return item.suffix.subview(first, item.suffix.size() - first);
266 } else {
267 auto first = retained << P::unit_shift;
268 return item.key.prefix.subview(first, item.key.prefix.size() - first);
269 }
270 }
271 void append(auto const & item, bit_view value, std::uint64_t retained) {
272 writer_.append(retained, suffix(item, retained), value);
273 }
274 // Both heads follow the last emitted key p. The head sharing more of p
275 // sorts first. Equal LCPs need only a suffix comparison from that boundary.
276 bit_comparison compare_heads() const {
277 if (older_cursor_.done()) return {0, 1};
278 if (newer_cursor_.done()) return {0, -1};
279 if (older_prefix_ != newer_prefix_)
280 return {std::min(older_prefix_, newer_prefix_) << P::unit_shift,
281 older_prefix_ > newer_prefix_ ? -1 : 1};
282 auto start = older_prefix_ << P::unit_shift;
283 auto result = compare_common_bits(suffix(older_cursor_.peek(), older_prefix_),
284 suffix(newer_cursor_.peek(), newer_prefix_));
285 result.common_bits += start;
286 return result;
287 }
288 static std::uint64_t advance(cursor_type & cursor) {
289 auto comparison = cursor.advance_comparison();
290 if (!comparison) return 0;
291 if (comparison->order >= 0)
292 throw std::invalid_argument("native merge inputs must have unique sorted keys");
293 return comparison->common_bits >> P::unit_shift;
294 }
