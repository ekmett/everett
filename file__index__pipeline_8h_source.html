36 // Stages are ordered nearest the existing target first, new head last.
37 // Sample the exact mapped target once and pass front-coded samples directly
38 // between stages. All native objects are retained unchanged. Each stage has
39 // bounded payload buffering, current key contexts and compact navigation.
40 //
41 // Construction opens private attempts under caller-reserved identities.
42 // step budgets the same entry events as index_pipeline. After it completes,
43 // seal_next finalizes one stage, target outward. It does not publish a catalog
44 // root: record each receipt and register the dependencies before publication.
45 // Failures preserve earlier receipts and surviving names for reconciliation.
46 // A supplied Ops owner must outlive this nonmovable pipeline.

70 // Returns true after sealing one object, false when all are already sealed.
71 // The returned receipt is appended to completed_receipts before success.
72 bool seal_next() {
73 if (this->failed_) error_detail::raise<std::logic_error>("file index pipeline has failed");
74 if (!this->done()) error_detail::raise<std::logic_error>("file index pipeline still has input");
75 auto i = receipts_.size();
76 if (i == this->stages_.size()) { this->finished_ = true; return false; }
77 try {
78 auto count = i ? this->stages_[i - 1]->size() : this->target_->virtual_size();
79 receipts_.push_back(this->stages_[i]->finish(count));
80 this->finished_ = receipts_.size() == this->stages_.size();
81 return true;
82 } catch (...) {
83 this->failed_ = true;
84 throw;
85 }
86 }

119 template <bool ExternalOps>
120 static std::vector<std::unique_ptr<builder_type>> make_stages(std::filesystem::path const & root,
121 pair_type const & target, std::span<stage_type const> stages, Ops * ops) {
122 if (!target) error_detail::raise<std::invalid_argument>("null file index pipeline target");
123 std::unordered_set<std::string> ids;
124 ids.insert(target->identity().index.hex());
125 for (auto const & stage : stages) {
126 if (!stage.source || stage.identity.native.hex().size() != 32 ||
127 stage.identity.index.hex().size() != 32 || stage.attempt.hex().size() != 32)
128 error_detail::raise<std::invalid_argument>("invalid file index pipeline stage");
129 if (!ids.insert(stage.identity.index.hex()).second)
130 error_detail::raise<std::invalid_argument>("file index pipeline reuses an index identity");
131 }
132 std::vector<std::unique_ptr<builder_type>> result;
133 result.reserve(stages.size());
134 auto downstream = target->identity();
135 for (auto const & stage : stages) {
136 if constexpr (ExternalOps) result.push_back(std::make_unique<builder_type>(root, stage.identity.index, stage.attempt,
137 stage.source, stage.identity.native, downstream, *ops));
138 else result.push_back(std::make_unique<builder_type>(root, stage.identity.index, stage.attempt,
139 stage.source, stage.identity.native, downstream));
140 downstream = stage.identity;
141 }
142 return result;
143 }
