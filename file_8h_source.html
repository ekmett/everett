44 // Version 1, explicit little-endian integers; never a raw C++ struct:
45 // 0 magic[8] 8 version:u16 10 header_bytes:u16
46 // 12 flags:u32 16 unit:u8 17 checksum_kind:u8 (=1)
47 // 18 backspace:u8 19 reserved:u8 20 codec_W:u32
48 // 24 group_K:u64
49 // 32 policy_value_width:u64
50 // 40 common_width:u64 48 body_extent:u64
51 // 56 record_count:u64 64 body_crc32c:u32
52 // 68 header_crc32c:u32 72 body_offset:u64 (=96)
53 // 80 total_file_bytes:u64 88 backspace_parameter:u64
54 // Flags: bit0 fixed policy values; bit1 common width present. Header CRC
55 // covers all 96 bytes with bytes68..71 zero. Padding and reserved bits zero.
56 // Backspace: 0 exponential-Golomb (order 0..63), 1 Golomb (modulus >=1).
57 // Byte profiles require descriptor (0,0). Codec width W is independent of
58 // the virtual cascade stride K and is part of the exact stored policy.
59 inline std::uint64_t get(std::span<std::byte const> bytes, std::size_t at, unsigned width) noexcept {
60 std::uint64_t value = 0;
61 for (unsigned i = 0; i < width; ++i) value |= std::uint64_t(std::to_integer<unsigned>(bytes[at + i])) << (i << 3);
62 return value;
63 }
64 inline void put(std::span<std::byte> bytes, std::size_t at, unsigned width, std::uint64_t value) noexcept {
65 for (unsigned i = 0; i < width; ++i) bytes[at + i] = std::byte((value >> (i << 3)) & 255u);
66 }

78 template <class P> std::uint64_t total_bytes(std::uint64_t extent) {
79 if constexpr (P::unit == profile_unit::bit)
80 if (extent > std::numeric_limits<std::uint64_t>::max() - 7)
81 throw std::overflow_error("Diet file extent overflow");
82 auto bytes = body_bytes<P>(extent);
83 if (bytes > std::numeric_limits<std::uint64_t>::max() - header_bytes)
84 throw std::overflow_error("Diet file extent overflow");
85 return bytes + header_bytes;

87 template <class P> void validate_metadata(file_header<P> const & header) {
88 (void)magic(header.kind);
89 if (header.kind == file_kind::fractional_index) {
90 if (header.common_value_width != std::optional<std::uint64_t>{0})
91 throw std::invalid_argument("fractional index values must have width zero");
92 } else if (header.kind == file_kind::native_blob) {
93 if constexpr (P::fixed_width)
94 if (header.common_value_width && header.common_value_width != P::value_width)
95 throw std::invalid_argument("native common width disagrees with fixed policy");
96 auto width = header.common_value_width;
97 if constexpr (P::fixed_width) width = P::value_width;
98 if (width && *width && header.record_count > header.extent / *width)
99 throw std::invalid_argument("fixed value slots exceed body extent");
100 }
101 (void)total_bytes<P>(header.extent);
102 }
103 template <class P> void validate_body(file_header<P> const & header, std::span<std::byte const> body) {
104 if (body_bytes<P>(header.extent) != body.size())
105 throw std::invalid_argument("Diet body extent mismatch");
106 if constexpr (P::unit == profile_unit::bit) {
107 if (header.extent & 7) {
108 unsigned unused = 8 - unsigned(header.extent & 7);
109 if (std::to_integer<unsigned>(body.back()) & ((1u << unused) - 1))
110 throw std::invalid_argument("noncanonical bit-profile tail padding");
111 }
112 }

116 for (std::size_t i = 0; i < copy.size(); ++i) copy[i] = bytes[i];
117 put(copy, 68, 4, 0);

122 template <class P> file_header<P> decode_file_header(std::span<std::byte const> bytes) {
123 if (bytes.size() < file_detail::header_bytes) throw std::invalid_argument("truncated Diet header");
124 file_header<P> header;
125 bool recognized = false;
126 for (auto kind : {file_kind::native_blob, file_kind::fractional_index}) {
127 auto magic = file_detail::magic(kind);
128 bool match = true;
129 for (std::size_t i = 0; i < 8; ++i)
130 if (std::to_integer<unsigned char>(bytes[i]) != static_cast<unsigned char>(magic[i])) match = false;
131 if (match) { header.kind = kind; recognized = true; break; }
132 }
133 if (!recognized) throw std::invalid_argument("unrecognized Diet magic");
134 if (file_detail::get(bytes, 8, 2) != file_detail::version ||
135 file_detail::get(bytes, 10, 2) != file_detail::header_bytes)
136 throw std::invalid_argument("unsupported Diet file version or header length");
137 if (file_detail::get(bytes, 68, 4) != file_detail::header_checksum(bytes))
138 throw std::invalid_argument("Diet header CRC32C mismatch");
139 auto flags = file_detail::get(bytes, 12, 4);
140 if (flags & ~std::uint64_t{3}) throw std::invalid_argument("unsupported Diet header flags");
141 if (file_detail::get(bytes, 16, 1) > 1 ||
142 file_detail::get(bytes, 16, 1) != static_cast<unsigned>(P::unit))
143 throw std::invalid_argument("Diet address unit disagrees with policy");
144 if (file_detail::get(bytes, 17, 1) != 1) throw std::invalid_argument("unsupported Diet checksum kind");
145 if (file_detail::get(bytes, 19, 1))
146 throw std::invalid_argument("nonzero Diet reserved header bytes");
147 auto backspace = file_detail::get(bytes, 18, 1);
148 auto backspace_parameter = file_detail::get(bytes, 88, 8);
149 if (backspace > static_cast<unsigned>(bit_backspace_code::golomb) ||
150 (backspace == static_cast<unsigned>(bit_backspace_code::exponential_golomb) && backspace_parameter > 63) ||
151 (backspace == static_cast<unsigned>(bit_backspace_code::golomb) && !backspace_parameter) ||
152 (P::unit == profile_unit::byte && (backspace || backspace_parameter)))
153 throw std::invalid_argument("unsupported Diet backspace descriptor");
154 if (file_detail::get(bytes, 20, 4) != P::codec_block_size ||
155 file_detail::get(bytes, 24, 8) != P::group_size ||
156 bool(flags & 1) != P::fixed_width || file_detail::get(bytes, 32, 8) != P::value_width.value_or(0) ||
157 backspace != static_cast<unsigned>(P::backspace_code) || backspace_parameter != P::backspace_parameter)
158 throw std::invalid_argument("Diet stored policy descriptor mismatch");
159 if (flags & 2) header.common_value_width = file_detail::get(bytes, 40, 8);
160 else if (file_detail::get(bytes, 40, 8)) throw std::invalid_argument("absent common width must encode zero");
161 header.extent = file_detail::get(bytes, 48, 8);
162 header.record_count = file_detail::get(bytes, 56, 8);
163 file_detail::validate_metadata(header);
164 if (file_detail::get(bytes, 72, 8) != file_detail::header_bytes ||
165 file_detail::get(bytes, 80, 8) != file_detail::total_bytes<P>(header.extent))
166 throw std::invalid_argument("noncanonical Diet body offset or physical extent");
167 return header;
168 }

170 template <class P> file_header<P> validate_file(std::span<std::byte const> bytes) {
171 auto header = decode_file_header<P>(bytes);
172 if (file_detail::total_bytes<P>(header.extent) != bytes.size())
173 throw std::invalid_argument("truncated or trailing Diet object bytes");
174 auto body = bytes.subspan(file_detail::header_bytes);
175 file_detail::validate_body(header, body);
176 if (file_detail::get(bytes, 64, 4) != crc32c(body))
177 throw std::invalid_argument("Diet body CRC32C mismatch");
178 return header;
179 }

181 // Serialize only the canonical envelope. The caller supplies the finalized
182 // CRC of the physical body bytes and remains responsible for checking the
183 // body's extent and bit padding before publication. No body is read here.
184 template <class P> std::array<std::byte, file_detail::header_bytes>
185 encode_file_header(file_header<P> const & header, std::uint32_t body_crc) {
186 file_detail::validate_metadata(header);
187 auto total = file_detail::total_bytes<P>(header.extent);
188 std::array<std::byte, file_detail::header_bytes> result{};
189 auto magic = file_detail::magic(header.kind);
190 for (std::size_t i = 0; i < 8; ++i) result[i] = std::byte(static_cast<unsigned char>(magic[i]));
191 file_detail::put(result, 8, 2, file_detail::version);
192 file_detail::put(result, 10, 2, file_detail::header_bytes);
193 file_detail::put(result, 12, 4, (P::fixed_width ? 1u : 0u) | (header.common_value_width ? 2u : 0u));
194 file_detail::put(result, 16, 1, static_cast<unsigned>(P::unit));
195 file_detail::put(result, 17, 1, 1);
196 file_detail::put(result, 18, 1, static_cast<unsigned>(P::backspace_code));
197 file_detail::put(result, 20, 4, P::codec_block_size);
198 file_detail::put(result, 24, 8, P::group_size);
199 file_detail::put(result, 32, 8, P::value_width.value_or(0));
200 file_detail::put(result, 40, 8, header.common_value_width.value_or(0));
201 file_detail::put(result, 48, 8, header.extent);
202 file_detail::put(result, 56, 8, header.record_count);
203 file_detail::put(result, 64, 4, body_crc);
204 file_detail::put(result, 72, 8, file_detail::header_bytes);
205 file_detail::put(result, 80, 8, total);
206 file_detail::put(result, 88, 8, P::backspace_parameter);
207 file_detail::put(result, 68, 4, file_detail::header_checksum(result));
208 return result;
209 }

211 // Pure serialization of an already encoded body; no file writes or durability.
212 template <class P> std::vector<std::byte> encode_file(file_header<P> const & header,
213 std::span<std::byte const> body) {
214 file_detail::validate_metadata(header);
215 file_detail::validate_body(header, body);
216 auto total = file_detail::total_bytes<P>(header.extent);
217 if (total > std::numeric_limits<std::size_t>::max()) throw std::length_error("Diet file is too large");
218 auto prefix = encode_file_header(header, crc32c(body));
219 std::vector<std::byte> result(static_cast<std::size_t>(total));
220 for (std::size_t i = 0; i < prefix.size(); ++i) result[i] = prefix[i];
221 for (std::size_t i = 0; i < body.size(); ++i) result[file_detail::header_bytes + i] = body[i];
222 return result;
223 }

225 // Typed single-object reader. Checked opening validates the header and exact
226 // file extent without reading payload bytes. Trusted opening reads no mapped
227 // bytes: the caller vouches for the immutable object's format and policy.
228 // Both modes retain a mapping and require at least the fixed 96-byte header.
229 // body() takes the physical remainder without decoding metadata. header()
230 // returns a value, checking trusted metadata on explicit access without a
231 // mutable lazy cache. scan() always validates the entire object, including
232 // its header: O(physical body bytes). It does not validate the external name.
233 template <class P> struct file {
234 using policy_type = P;
235 static file from_slice(mapped_slice bytes, file_open_mode mode = file_open_mode::checked) {
236 switch (mode) {
237 case file_open_mode::checked: {
238 auto header = checked_header(bytes);
239 return {std::move(bytes), std::move(header)};
240 }
241 case file_open_mode::trusted:
242 if (bytes.size() < file_detail::header_bytes)
243 throw std::invalid_argument("truncated Diet header");
244 return {std::move(bytes), std::nullopt};
245 }
246 throw std::invalid_argument("unsupported Diet file open mode");
247 }
248 static file open(std::filesystem::path const & path, file_open_mode mode = file_open_mode::checked) {
249 auto mapping = mapped_file::open(path);
250 // All mapped-byte access is in the same helper exercised by from_slice.
251 auto result = from_slice(mapping.slice(0, mapping.size()), mode);
252 // Trusted callers also vouch for the filename; checking its agreement
253 // would require decoding the header that this mode deliberately skips.
254 if (mode == file_open_mode::checked &&
255 path.extension().generic_string() != file_extension(result.header_->kind))
256 throw std::invalid_argument("Diet extension disagrees with file magic");
257 return result;
258 }
259 file_header<P> header() const { return header_ ? *header_ : checked_header(bytes_); }
260 mapped_slice body() const { return bytes_.slice(file_detail::header_bytes, bytes_.size() - file_detail::header_bytes); }
261 void scan() const { (void)validate_file<P>(bytes_.bytes()); }
262 private:
263 static file_header<P> checked_header(mapped_slice const & bytes) {
264 auto header = decode_file_header<P>(bytes.bytes());
265 if (file_detail::total_bytes<P>(header.extent) != bytes.size())
266 throw std::invalid_argument("truncated or trailing Diet object bytes");
267 return header;
268 }
269 file(mapped_slice bytes, std::optional<file_header<P>> header) : bytes_(std::move(bytes)), header_(std::move(header)) {}
270 mapped_slice bytes_;
271 std::optional<file_header<P>> header_;
272 };
